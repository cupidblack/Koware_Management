<?php
/**
 * Plugin Name: Buzzjuice Connections Sync
 * Description: Canonical relationship and moderation synchronization across Buzzjuice WordPress/BuddyBoss, WoWonder Streams, and QuickDate Socials.
 * Version: 3.0.0
 * Author: Buzzjuice
 * License: GPL-2.0-or-later
 *
 * ARCHITECTURE:
 * - WordPress/BuddyBoss is the canonical authority
 * - Durable event ledger serves as audit trail, queue, and retry mechanism
 * - Four states: NONE, REQUESTED, CONNECTED, BLOCKED
 * - BLOCKED is absolute and dominant
 * - Unblocking transitions only to NONE (previous relationship not restored)
 * - Blocking removes all existing connections before setting BLOCKED state
 * - Internal context prevents recursive hook loops during block cleanup
 * - External platforms notify via REST API after successful local state change
 * - Event UUIDs are idempotent to prevent duplicate processing
 */

defined( 'ABSPATH' ) || exit;

// ==================== CONSTANTS ====================

define( 'BZJ_CONNECTIONS_SYNC_VERSION', '3.0.0' );

// Canonical relationship states
define( 'BZJ_CONN_NONE',      'none' );
define( 'BZJ_CONN_REQUESTED', 'requested' );
define( 'BZJ_CONN_CONNECTED', 'connected' );
define( 'BZJ_CONN_BLOCKED',   'blocked' );

// Platform origins
define( 'BZJ_CONN_ORIGIN_WP',      'wordpress' );
define( 'BZJ_CONN_ORIGIN_STREAMS', 'streams' );
define( 'BZJ_CONN_ORIGIN_SOCIALS', 'socials' );

// Log file location
define(
    'BZJ_CONNECTIONS_SYNC_LOG',
    WP_CONTENT_DIR . '/bzj-connections-sync.log'
);

// ==================== BOOT ====================

add_action( 'init', 'bzj_connections_sync_boot', 1 );

function bzj_connections_sync_boot() {
    bzj_connections_sync_install();
    bzj_connections_sync_register_hooks();
    bzj_connections_sync_register_rest_endpoints();
    bzj_connections_sync_register_cron_jobs();
}

// ==================== LOGGING ====================

function bzj_connections_sync_log( $message, $context = [] ) {

    $timestamp = gmdate( 'Y-m-d H:i:s' );
    $line = "[{$timestamp}] {$message}";

    if ( ! empty( $context ) ) {
        $line .= ' ' . wp_json_encode(
            $context,
            JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE
        );
    }

    $line .= PHP_EOL;

    @error_log( $line, 3, BZJ_CONNECTIONS_SYNC_LOG );
}

function bzj_connections_sync_install() {

    global $wpdb;

    $current_version = get_option( 'bzj_connections_sync_schema_version', '' );

    if ( $current_version === BZJ_CONNECTIONS_SYNC_VERSION ) {
        return;
    }

    require_once ABSPATH . 'wp-admin/includes/upgrade.php';

    $charset = $wpdb->get_charset_collate();
    $rel_table = bzj_connections_sync_relationships_table();
    $evt_table = bzj_connections_sync_events_table();

    $sql = "

    CREATE TABLE IF NOT EXISTS {$rel_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        user_a BIGINT UNSIGNED NOT NULL,
        user_b BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL DEFAULT 'none',
        requested_by BIGINT UNSIGNED NULL,
        blocked_by BIGINT UNSIGNED NULL,
        created_at DATETIME NOT NULL,
        updated_at DATETIME NOT NULL,
        last_event_uuid CHAR(36) NULL,
        PRIMARY KEY (id),
        UNIQUE KEY relationship_pair (user_a, user_b),
        KEY state (state),
        KEY requested_by (requested_by),
        KEY blocked_by (blocked_by),
        KEY updated_at (updated_at)
    ) {$charset};

    CREATE TABLE IF NOT EXISTS {$evt_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        event_uuid CHAR(36) NOT NULL,
        origin VARCHAR(20) NOT NULL,
        event_type VARCHAR(60) NOT NULL,
        actor_wp_id BIGINT UNSIGNED NOT NULL,
        target_wp_id BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL,
        payload LONGTEXT NULL,
        status VARCHAR(20) NOT NULL DEFAULT 'pending',
        attempts INT UNSIGNED NOT NULL DEFAULT 0,
        next_attempt_at DATETIME NOT NULL,
        last_error TEXT NULL,
        created_at DATETIME NOT NULL,

        UNIQUE KEY event_uuid (event_uuid),
        KEY retry_lookup (status, next_attempt_at),
        KEY pair_lookup (actor_wp_id, target_wp_id),
        KEY state_lookup (state),
        KEY created_at (created_at)

// ==================== UTILITIES ====================

function bzj_connections_sync_uuid() {
    return wp_generate_uuid4();
}

/**
 * Normalize a user pair so that user_a < user_b.
 * Prevents (A,B) and (B,A) from being stored as separate records.
 */
function bzj_connections_sync_pair( $user_a, $user_b ) {

    $user_a = absint( $user_a );
    $user_b = absint( $user_b );

    if ( ! $user_a || ! $user_b || $user_a === $user_b ) {
        return false;
    }

    if ( $user_a > $user_b ) {
        [ $user_a, $user_b ] = [ $user_b, $user_a ];
    }

    return [
        'user_a' => $user_a,
        'user_b' => $user_b,
    ];
}

/**
 * Acquire a MySQL named lock for a user pair.
 * Prevents race conditions during state transitions.
 */
function bzj_connections_sync_lock( $user_a, $user_b ) {

    global $wpdb;

    $pair = bzj_connections_sync_pair( $user_a, $user_b );

    if ( ! $pair ) {
        return false;
    }

    $lock_name = sprintf( 'bzj_conn_%d_%d', $pair['user_a'], $pair['user_b'] );

    $result = (int) $wpdb->get_var(
        $wpdb->prepare( 'SELECT GET_LOCK( %s, 5 )', $lock_name )
    );

    return 1 === $result ? $lock_name : false;
}

function bzj_connections_sync_unlock( $lock_name ) {

    global $wpdb;

    if ( ! $lock_name ) {
        return;
    }

    $wpdb->get_var(
        $wpdb->prepare( 'SELECT RELEASE_LOCK( %s )', $lock_name )
    );
}

/**
 * Set internal synchronization context to prevent recursive hooks.
 * Used during block cleanup to avoid nested state change events.
 */
function bzj_connections_sync_set_internal_context( $context = null ) {
    static $value = null;
    if ( $context !== null ) {
        $value = $context;
    }
    return $value;
}

function bzj_connections_sync_is_internal_context( $context = null ) {
    $current = bzj_connections_sync_set_internal_context();
    if ( $context === null ) {
        return ! empty( $current );
    }
    return $current === $context;
}

// ==================== CANONICAL STATE READER ====================

function bzj_connections_sync_get_state( $user_a, $user_b ) {

    global $wpdb;

    $pair = bzj_connections_sync_pair( $user_a, $user_b );

    if ( ! $pair ) {
        return false;
    }

    return $wpdb->get_row(
        $wpdb->prepare(
            'SELECT * FROM ' . bzj_connections_sync_relationships_table() . '
            WHERE user_a = %d AND user_b = %d LIMIT 1',
            $pair['user_a'],
            $pair['user_b']
        )
    );
}

function bzj_connections_sync_is_blocked( $user_a, $user_b ) {

    $state = bzj_connections_sync_get_state( $user_a, $user_b );

    return $state && $state->state === BZJ_CONN_BLOCKED;
}

// ==================== EVENT CREATION ====================

function bzj_connections_sync_create_event(
    $actor_wp_id,
    $target_wp_id,
    $state,
    $event_type,
    $payload = [],
    $origin = BZJ_CONN_ORIGIN_WP
) {

    global $wpdb;

    $event_uuid = bzj_connections_sync_uuid();
    $now = current_time( 'mysql', true );

    $result = $wpdb->insert(
        bzj_connections_sync_events_table(),
        [
            'event_uuid'       => $event_uuid,
            'origin'           => sanitize_key( $origin ),
            'event_type'       => sanitize_key( $event_type ),
            'actor_wp_id'      => absint( $actor_wp_id ),
            'target_wp_id'     => absint( $target_wp_id ),
            'state'            => sanitize_key( $state ),
            'payload'          => ! empty( $payload ) ? wp_json_encode( $payload ) : null,
            'status'           => 'pending',
            'attempts'         => 0,
            'next_attempt_at'  => $now,
            'created_at'       => $now,
        ],
        [ '%s', '%s', '%s', '%d', '%d', '%s', '%s', '%s', '%d', '%s', '%s' ]
    );

    if ( ! $result ) {
        bzj_connections_sync_log(
            'Event creation failed',
            [
                'error'  => $wpdb->last_error,
                'actor'  => $actor_wp_id,
                'target' => $target_wp_id,
                'type'   => $event_type,
            ]
        );
        return false;
    }

    return $event_uuid;
}

// ==================== CANONICAL STATE WRITER ====================

function bzj_connections_sync_set_state(
    $user_a,
    $user_b,
    $state,
    $requested_by = 0,
    $blocked_by = 0,
    $origin = BZJ_CONN_ORIGIN_WP,
    $event_type = 'state_changed',
    $payload = []
) {

    global $wpdb;

    $pair = bzj_connections_sync_pair( $user_a, $user_b );

    if ( ! $pair ) {
        return false;
    }

    $allowed_states = [
        BZJ_CONN_NONE,
        BZJ_CONN_REQUESTED,
        BZJ_CONN_CONNECTED,
        BZJ_CONN_BLOCKED,
    ];

    if ( ! in_array( $state, $allowed_states, true ) ) {
        return false;
    }

    $lock = bzj_connections_sync_lock( $pair['user_a'], $pair['user_b'] );

    if ( ! $lock ) {
        bzj_connections_sync_log(
            'Unable to acquire relationship pair lock',
            $pair
        );
        return false;
    }

    try {

        $existing = bzj_connections_sync_get_state( $pair['user_a'], $pair['user_b'] );
        $old_state = $existing ? $existing->state : BZJ_CONN_NONE;

        /*
         * BLOCK DOMINANCE RULE:
         * Once blocked, cannot transition to any other state except NONE.
         */
        if ( $old_state === BZJ_CONN_BLOCKED && $state !== BZJ_CONN_NONE ) {
            bzj_connections_sync_log(
                'Blocked state is dominant; transition rejected',
                [ 'pair' => $pair, 'attempted' => $state ]
            );
            return true;
        }

        /*
         * Avoid unnecessary events for identical states.
         */
        if ( $old_state === $state && ! $event_type ) {
            return true;
        }

        $now = current_time( 'mysql', true );
        $event_uuid = bzj_connections_sync_uuid();

        $update_data = [
            'state'            => $state,
            'requested_by'     => $requested_by ? absint( $requested_by ) : null,
            'blocked_by'       => $blocked_by ? absint( $blocked_by ) : null,
            'updated_at'       => $now,
            'last_event_uuid'  => $event_uuid,
        ];

        if ( $existing ) {

            $updated = $wpdb->update(
                bzj_connections_sync_relationships_table(),
                $update_data,
                [ 'user_a' => $pair['user_a'], 'user_b' => $pair['user_b'] ],
                [ '%s', '%d', '%d', '%s', '%s' ],
                [ '%d', '%d' ]
            );

        } else {

            $insert_data = array_merge(
                [ 'user_a' => $pair['user_a'], 'user_b' => $pair['user_b'], 'created_at' => $now ],
                $update_data
            );

            $updated = $wpdb->insert(
                bzj_connections_sync_relationships_table(),
                $insert_data,
                [ '%d', '%d', '%s', '%d', '%d', '%s', '%s' ]
            );
        }

        if ( $updated === false ) {
            bzj_connections_sync_log(
                'Canonical relationship update failed',
                [
                    'error'  => $wpdb->last_error,
                    'pair'   => $pair,
                    'state'  => $state,
                ]
            );
            return false;
        }

        $event_payload = array_merge(
            (array) $payload,
            [
                'previous_state'    => $old_state,
                'canonical_state'   => $state,
                'event_uuid'        => $event_uuid,
            ]
        );

        $event = bzj_connections_sync_create_event(
            $pair['user_a'],
            $pair['user_b'],
            $state,
            $event_type,
            $event_payload,
            $origin
        );

        bzj_connections_sync_log(
            'Canonical relationship state changed',
            [
                'pair'       => $pair,
                'from'       => $old_state,
                'to'         => $state,
                'event_uuid' => $event,
                'origin'     => $origin,
            ]
        );

        return $event;

    } finally {
        bzj_connections_sync_unlock( $lock );
    }
}

// ==================== BUDDYBOSS RELATIONSHIP CLEANUP ====================

/**
 * Remove existing BuddyBoss connections before blocking.
 * Must be done inside an internal context to prevent recursive events.
 */
function bzj_connections_sync_remove_wp_relationships( $user_a, $user_b ) {

    $user_a = absint( $user_a );
    $user_b = absint( $user_b );

    if ( ! $user_a || ! $user_b || $user_a === $user_b ) {
        return;
    }

    // Prevent recursive hook processing
    bzj_connections_sync_set_internal_context( 'block_cleanup' );

    try {

        // Remove friendship in both directions
        if ( function_exists( 'friends_check_friendship_status' ) &&
             function_exists( 'friends_remove_friend' ) ) {

            $status_a_to_b = friends_check_friendship_status( $user_a, $user_b );
            if ( $status_a_to_b === 'is_friend' ) {
                friends_remove_friend( $user_a, $user_b );
            }

            $status_b_to_a = friends_check_friendship_status( $user_b, $user_a );
            if ( $status_b_to_a === 'is_friend' ) {
                friends_remove_friend( $user_b, $user_a );
            }
        }

        // Reject pending requests in both directions
        if ( function_exists( 'friends_get_friendship_id' ) &&
             function_exists( 'friends_reject_friendship' ) ) {

            $friendship_id_a_b = friends_get_friendship_id( $user_a, $user_b );
            if ( $friendship_id_a_b ) {
                friends_reject_friendship( $friendship_id_a_b );
            }

            $friendship_id_b_a = friends_get_friendship_id( $user_b, $user_a );
            if ( $friendship_id_b_a ) {
                friends_reject_friendship( $friendship_id_b_a );
            }
        }

        bzj_connections_sync_log(
            'Removed BuddyBoss relationships before blocking',
            [ 'user_a' => $user_a, 'user_b' => $user_b ]
        );

    } finally {
        bzj_connections_sync_set_internal_context( null );
    }
}

function bzj_connections_sync_on_friendship_requested(
    $friendship_id,
    $initiator_id,
    $friend_id,
    $friendship
) {

    // Do not process internally-generated events
    if ( bzj_connections_sync_is_internal_context() ) {
        return;
    }

    $initiator_id = absint( $initiator_id );
    $friend_id = absint( $friend_id );

    if ( ! $initiator_id || ! $friend_id || $initiator_id === $friend_id ) {
        return;
    }

    // Blocked users cannot request connections
    if ( bzj_connections_sync_is_blocked( $initiator_id, $friend_id ) ) {
        bzj_connections_sync_log(
            'Friendship requested but users are blocked; skipping sync',
            [ 'initiator' => $initiator_id, 'friend' => $friend_id ]
        );
        return;
    }

    bzj_connections_sync_set_state(
        $initiator_id,
        $friend_id,
        BZJ_CONN_REQUESTED,
        $initiator_id,
        0,
        BZJ_CONN_ORIGIN_WP,
        'connection_requested'
    );
}

// ==================== MODERATION HANDLERS ====================

/**
 * Block handler.
 * Block is absolute and dominant.
 * Removes all existing connections before setting BLOCKED state.
 */
function bzj_connections_sync_on_block_created( $block_id, $block_object ) {

    if ( ! is_object( $block_object ) ||
         empty( $block_object->user_id ) ||
         empty( $block_object->item_id ) ) {
        return;
    }

    $blocker_id = absint( $block_object->user_id );
    $blocked_id = absint( $block_object->item_id );

    if ( ! $blocker_id || ! $blocked_id || $blocker_id === $blocked_id ) {
        return;
    }

    // Remove all existing BuddyBoss connections
    bzj_connections_sync_remove_wp_relationships( $blocker_id, $blocked_id );

    // Set canonical state to BLOCKED
    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        BZJ_CONN_BLOCKED,
        0,
        $blocker_id,
        BZJ_CONN_ORIGIN_WP,
        'block_created'
    );
}

/**
 * Unblock handler.
 * Unblocking only transitions to NONE.
 * Previous relationship is NOT automatically restored.
 */
function bzj_connections_sync_on_block_deleted( $block_id, $block_object ) {

    if ( ! is_object( $block_object ) ||
         empty( $block_object->user_id ) ||
         empty( $block_object->item_id ) ) {
        return;
    }

    $blocker_id = absint( $block_object->user_id );
    $blocked_id = absint( $block_object->item_id );

    if ( ! $blocker_id || ! $blocked_id || $blocker_id === $blocked_id ) {
        return;
    }

    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        BZJ_CONN_NONE,
        0,
        0,
        BZJ_CONN_ORIGIN_WP,
        'block_deleted'
    );
}

// ==================== IDENTITY RESOLUTION ====================

/**
 * Get cross-platform IDs for a WordPress user.
 */
function bzj_connections_sync_get_cross_platform_ids( $wp_user_id ) {

    $wp_user_id = absint( $wp_user_id );

    if ( ! $wp_user_id ) {
        return false;
    }

    $wo_id = absint( get_user_meta( $wp_user_id, 'wo_user_id', true ) );
    $qd_id = absint( get_user_meta( $wp_user_id, 'qd_user_id', true ) );

    return [
        'wp' => $wp_user_id,
        'wo' => $wo_id,
        'qd' => $qd_id,
    ];
}

/**
 * Resolve a WordPress user ID from an external platform ID.
 */
function bzj_connections_sync_resolve_wp_user( $external_id, $platform ) {

    $external_id = absint( $external_id );
    $platform = sanitize_key( $platform );

    if ( ! $external_id || ! in_array( $platform, [ 'streams', 'socials' ], true ) ) {
        return 0;
    }

    $meta_key = ( $platform === 'streams' ) ? 'wo_user_id' : 'qd_user_id';

    $users = get_users( [
        'meta_key'   => $meta_key,
        'meta_value' => (string) $external_id,
        'number'     => 1,
        'fields'     => 'ids',
    ] );

    return ! empty( $users ) ? absint( $users[0] ) : 0;
}

/**
 * Verify authorization for REST endpoint.
 * Requires Bearer token matching BUZZ_SSO_SECRET.
 */
function bzj_connections_sync_rest_verify_authorization() {

    $auth_header = isset( $_SERVER['HTTP_AUTHORIZATION'] )
        ? sanitize_text_field( $_SERVER['HTTP_AUTHORIZATION'] )
        : '';

    if ( empty( $auth_header ) || ! preg_match( '/^Bearer\s+(.+)$/i', $auth_header, $matches ) ) {
        return new WP_Error(
            'bzj_missing_auth',
            'Missing or invalid Authorization header',
            [ 'status' => 401 ]
        );
    }

    $provided_token = trim( $matches[1] );
    $expected_token = bzj_get_sso_secret();

    if ( ! hash_equals( $provided_token, $expected_token ) ) {
        return new WP_Error(
            'bzj_invalid_auth',
            'Invalid authorization token',
            [ 'status' => 403 ]
        );
    }

    return true;
}

/**
 * Handle relationship notifications from external platforms.
 */
function bzj_connections_sync_rest_handle_relationship( $request ) {

    $origin = $request->get_param( 'origin' );
    $origin_user_id = $request->get_param( 'origin_user_id' );
    $target_user_id = $request->get_param( 'target_user_id' );
    $operation = $request->get_param( 'operation' );
    $event_uuid = $request->get_param( 'event_uuid' );

    // Validate required parameters
    if ( ! in_array( $origin, [ 'streams', 'socials' ], true ) ) {
        return new WP_Error(
            'bzj_invalid_origin',
            'Invalid origin platform',
            [ 'status' => 400 ]
        );
    }

    // Resolve external user IDs to WordPress IDs
    $wp_actor = bzj_connections_sync_resolve_wp_user( $origin_user_id, $origin );
    $wp_target = bzj_connections_sync_resolve_wp_user( $target_user_id, $origin );

    if ( ! $wp_actor || ! $wp_target ) {
        bzj_connections_sync_log(
            'Cannot resolve external users to WordPress IDs',
            [
                'origin'           => $origin,
                'origin_user_id'   => $origin_user_id,
                'target_user_id'   => $target_user_id,
            ]
        );

        return new WP_Error(
            'bzj_unresolved_users',
            'Cannot map external users to WordPress',
            [ 'status' => 404 ]
        );
    }

    // Check for idempotency: if event_uuid already exists, don't reprocess
    if ( ! empty( $event_uuid ) ) {
        $existing_event = bzj_connections_sync_find_event_by_uuid( $event_uuid );
        if ( $existing_event ) {
            // Idempotent: return success without reprocessing
            bzj_connections_sync_log(
                'Idempotent event UUID already processed',
                [ 'event_uuid' => $event_uuid ]
            );

            return rest_ensure_response( [
                'success'  => true,
                'message'  => 'Event already processed (idempotent)',
                'event_id' => $existing_event->id,
            ] );
        }
    }

    // Canonicalize the user pair
    $pair = bzj_connections_sync_pair( $wp_actor, $wp_target );

    if ( ! $pair ) {
        return new WP_Error(
            'bzj_invalid_pair',
            'Invalid user pair',
            [ 'status' => 400 ]
        );
    }

    bzj_connections_sync_log(
        'REST endpoint: relationship notification received',
        [
            'origin'    => $origin,
            'operation' => $operation,
            'pair'      => $pair,
        ]
    );

    // Process the operation
    $result = bzj_connections_sync_handle_external_operation(
        $wp_actor,
        $wp_target,
        $operation,
        $origin,
        $event_uuid
    );

    if ( ! $result ) {
        return new WP_Error(
            'bzj_operation_failed',
            'Operation failed or was rejected',
            [ 'status' => 409 ]
        );
    }

    return rest_ensure_response( [
        'success' => true,
        'message' => 'Relationship operation processed',
    ] );
}

/**
 * Handle external platform operations.
 * External platform notifies AFTER successful local change.
 * WordPress validates and applies eligibility rules via BuddyBoss.
 */
function bzj_connections_sync_handle_external_operation(
    $wp_actor,
    $wp_target,
    $operation,
    $origin,
    $event_uuid = ''
) {

    $wp_actor = absint( $wp_actor );
    $wp_target = absint( $wp_target );

    if ( ! $wp_actor || ! $wp_target || $wp_actor === $wp_target ) {
        return false;
    }

    // Current canonical state
    $current_state = bzj_connections_sync_get_state( $wp_actor, $wp_target );
    $canonical_state = $current_state ? $current_state->state : BZJ_CONN_NONE;

    bzj_connections_sync_log(
        'Processing external operation',
        [
            'actor'     => $wp_actor,
            'target'    => $wp_target,
            'operation' => $operation,
            'origin'    => $origin,
            'state'     => $canonical_state,
        ]
    );

    switch ( $operation ) {

        case 'request':
            // External: user requested connection
            // WordPress: use BuddyBoss to initiate request (respects eligibility)
            return bzj_connections_sync_request_via_buddypress(
                $wp_actor,
                $wp_target,
                $origin,
                $event_uuid
            );

        case 'accept':
            // External: user accepted pending request
            // WordPress: call BuddyBoss friendship accept
            return bzj_connections_sync_accept_via_buddypress(
                $wp_actor,
                $wp_target,
                $origin,
                $event_uuid
            );

        case 'remove':
            // External: user removed connection
            // WordPress: transition to NONE
            return bzj_connections_sync_set_state(
                $wp_actor,
                $wp_target,
                BZJ_CONN_NONE,
                0,
                0,
                $origin,
                'external_remove',
                [ 'event_uuid' => $event_uuid ]
            );

        case 'block':
            // External: user blocked another user
            // WordPress: enforce block dominance and remove connections
            bzj_connections_sync_remove_wp_relationships( $wp_actor, $wp_target );
            return bzj_connections_sync_set_state(
                $wp_actor,
                $wp_target,
                BZJ_CONN_BLOCKED,
                0,
                $wp_actor,
                $origin,
                'external_block',
                [ 'event_uuid' => $event_uuid ]
            );

        case 'unblock':
            // External: user unblocked another user
            // WordPress: transition BLOCKED → NONE
            return bzj_connections_sync_set_state(
                $wp_actor,
                $wp_target,
                BZJ_CONN_NONE,
                0,
                0,
                $origin,
                'external_unblock',
                [ 'event_uuid' => $event_uuid ]
            );

        default:
            bzj_connections_sync_log(
                'Unknown external operation',
                [ 'operation' => $operation ]
            );
            return false;
    }
}

/**
 * Request connection via BuddyBoss.
 * BuddyBoss applies eligibility rules. Hook fires if accepted.
 */
function bzj_connections_sync_request_via_buddypress( $wp_actor, $wp_target, $origin, $event_uuid ) {

    // Check eligibility via BuddyBoss
    if ( function_exists( 'friends_check_friendship_status' ) ) {

        $status = friends_check_friendship_status( $wp_actor, $wp_target );

        // Already friends or request pending?
        if ( in_array( $status, [ 'is_friend', 'pending' ], true ) ) {
            bzj_connections_sync_log(
                'Cannot request: already connected or pending',
                [ 'actor' => $wp_actor, 'target' => $wp_target, 'status' => $status ]
            );
            return false;
        }

        // Check if blocked
        if ( bzj_connections_sync_is_blocked( $wp_actor, $wp_target ) ) {
            bzj_connections_sync_log(
                'Cannot request: users are blocked',
                [ 'actor' => $wp_actor, 'target' => $wp_target ]
            );
            return false;
        }

        // Attempt to add connection
        if ( function_exists( 'friends_add_friend' ) ) {
            $result = friends_add_friend( $wp_actor, $wp_target );
            if ( $result ) {
                bzj_connections_sync_log(
                    'Connection request initiated via BuddyBoss',
                    [ 'actor' => $wp_actor, 'target' => $wp_target, 'origin' => $origin ]
                );
                return true;
            }
        }
    }

    return false;
}

/**
 * Accept connection via BuddyBoss.
 */
function bzj_connections_sync_accept_via_buddypress( $wp_actor, $wp_target, $origin, $event_uuid ) {

    if ( function_exists( 'friends_accept_friendship' ) &&
         function_exists( 'friends_get_friendship_id' ) ) {

        $friendship_id = friends_get_friendship_id( $wp_target, $wp_actor );

        if ( $friendship_id ) {
            $result = friends_accept_friendship( $friendship_id );
            if ( $result ) {
                bzj_connections_sync_log(
                    'Connection accepted via BuddyBoss',
                    [ 'actor' => $wp_actor, 'target' => $wp_target, 'origin' => $origin ]
                );
                return true;
            }
        }
    }

    return false;
}

/**
 * Process pending events.
 * Attempts to deliver each event to both Streams and Socials.
 * Tracks success/failure separately.
 */
function bzj_connections_sync_process_pending_events() {

    global $wpdb;

    $events_table = bzj_connections_sync_events_table();
    $now = current_time( 'mysql', true );

    // Get pending events ready to retry
    $pending = $wpdb->get_results(
        $wpdb->prepare(
            "SELECT * FROM {$events_table}
            WHERE status = 'pending'
              AND next_attempt_at <= %s
            LIMIT 25",
            $now
        )
    );

    if ( empty( $pending ) ) {
        return;
    }

    foreach ( $pending as $event ) {

        if ( ! bzj_connections_sync_deliver_event( $event ) ) {
            // Increment retry counter and reschedule
            bzj_connections_sync_reschedule_event( $event->id );
        }
    }
}

/**
 * Deliver event to external platforms.
 */
function bzj_connections_sync_deliver_event( $event ) {

    $actor_ids = bzj_connections_sync_get_cross_platform_ids( $event->actor_wp_id );
    $target_ids = bzj_connections_sync_get_cross_platform_ids( $event->target_wp_id );

    if ( ! $actor_ids || ! $target_ids ) {
        bzj_connections_sync_log(
            'Cannot deliver event: missing cross-platform IDs',
            [ 'event_id' => $event->id ]
        );
        return false;
    }

    $payload = ! empty( $event->payload )
        ? json_decode( $event->payload, true )
        : [];

    $success = true;

    // Deliver to Streams (WoWonder)
    if ( ! empty( $actor_ids['wo'] ) && ! empty( $target_ids['wo'] ) ) {
        $delivered = bzj_connections_sync_deliver_to_platform(
            'streams',
            $event,
            $actor_ids['wo'],
            $target_ids['wo'],
            $payload
        );
        if ( ! $delivered ) {
            $success = false;
        }
    }

    // Deliver to Socials (QuickDate)
    if ( ! empty( $actor_ids['qd'] ) && ! empty( $target_ids['qd'] ) ) {
        $delivered = bzj_connections_sync_deliver_to_platform(
            'socials',
            $event,
            $actor_ids['qd'],
            $target_ids['qd'],
            $payload
        );
        if ( ! $delivered ) {
            $success = false;
        }
    }

    if ( $success ) {
        bzj_connections_sync_mark_event_processed( $event->id );
    }

    return $success;
}

/**
 * Deliver event to a specific platform via its webhook/callback.
 * This is a stub; integrate with actual Streams/Socials APIs.
 */
function bzj_connections_sync_deliver_to_platform(
    $platform,
    $event,
    $actor_id,
    $target_id,
    $payload
) {

    $endpoint = bzj_connections_sync_get_platform_webhook_url( $platform );

    if ( ! $endpoint ) {
        bzj_connections_sync_log(
            'No webhook URL configured for platform',
            [ 'platform' => $platform ]
        );
        return false;
    }

    $body = [
        'event_uuid'  => $event->event_uuid,
        'origin'      => 'wordpress',
        'actor_id'    => $actor_id,
        'target_id'   => $target_id,
        'event_type'  => $event->event_type,
        'state'       => $event->state,
        'payload'     => $payload,
    ];

    $response = wp_remote_post( $endpoint, [
        'timeout'     => 10,
        'redirection' => 0,
        'sslverify'   => true,
        'headers'     => [
            'Authorization' => 'Bearer ' . bzj_get_sso_secret(),
            'Content-Type'  => 'application/json',
        ],
        'body'        => wp_json_encode( $body ),
    ] );

    if ( is_wp_error( $response ) ) {
        bzj_connections_sync_log(
            'Webhook delivery failed',
            [
                'platform' => $platform,
                'event_id' => $event->id,
                'error'    => $response->get_error_message(),
            ]
        );
        return false;
    }

    $status_code = wp_remote_retrieve_response_code( $response );

    if ( $status_code < 200 || $status_code >= 300 ) {
        bzj_connections_sync_log(
            'Webhook delivery returned error status',
            [
                'platform'  => $platform,
                'event_id'  => $event->id,
                'status'    => $status_code,
            ]
        );
        return false;
    }

    bzj_connections_sync_log(
        'Webhook delivered successfully',
        [
            'platform' => $platform,
            'event_id' => $event->id,
        ]
    );

    return true;
}

function bzj_connections_sync_reschedule_event( $event_id ) {

    global $wpdb;

    $event = $wpdb->get_row(
        $wpdb->prepare(
            'SELECT * FROM ' . bzj_connections_sync_events_table() . ' WHERE id = %d',
            absint( $event_id )
        )
    );

    if ( ! $event ) {
        return;
    }

    $attempts = absint( $event->attempts ) + 1;
    $max_attempts = 10;

    if ( $attempts >= $max_attempts ) {

        $wpdb->update(
            bzj_connections_sync_events_table(),
            [
                'status'     => 'failed',
                'attempts'   => $attempts,
                'last_error' => 'Max retry attempts reached',
            ],
            [ 'id' => absint( $event_id ) ],
            [ '%s', '%d', '%s' ],
            [ '%d' ]
        );

        bzj_connections_sync_log(
            'Event max retries reached; marked failed',
            [ 'event_id' => $event_id, 'attempts' => $attempts ]
        );

        return;
    }

    // Exponential backoff: 2^attempts minutes
    $backoff_minutes = pow( 2, $attempts );
    $next_attempt = new DateTime( $event->next_attempt_at );
    $next_attempt->add( new DateInterval( "PT{$backoff_minutes}M" ) );

    $wpdb->update(
        bzj_connections_sync_events_table(),
        [
            'attempts'         => $attempts,
            'next_attempt_at'  => $next_attempt->format( 'Y-m-d H:i:s' ),
        ],
        [ 'id' => absint( $event_id ) ],
        [ '%d', '%s' ],
        [ '%d' ]
    );

    bzj_connections_sync_log(
        'Event rescheduled',
        [
            'event_id'        => $event_id,
            'attempts'        => $attempts,
            'next_attempt_at' => $next_attempt->format( 'Y-m-d H:i:s' ),
        ]
    );
}

// ==================== SSO SECRET HELPER ====================

/**
 * Get the Buzzjuice SSO secret for authorization.
 * Cascades: ENV → Constant → Fallback
 */
function bzj_get_sso_secret() {

    // 1. Environment variable
    $env_secret = getenv( 'BUZZ_SSO_SECRET' );
    if ( ! empty( $env_secret ) ) {
        return $env_secret;
    }

    // 2. WordPress constant
    if ( defined( 'BUZZ_SSO_SECRET' ) && BUZZ_SSO_SECRET ) {
        return BUZZ_SSO_SECRET;
    }

    // 3. Fallback (should not use in production)
    bzj_connections_sync_log(
        'WARNING: SSO secret not configured; using fallback',
        []
    );

    return 'bzj_fallback_' . hash( 'sha256', ABSPATH );
}

    // Check if exists
    $existing = $conn->query(
        "SELECT * FROM followers WHERE following_id = {$following_id} AND follower_id = {$follower_id} LIMIT 1"
    );

    if ( $existing && $existing->num_rows > 0 ) {
        // Update
        $conn->query(
            "UPDATE followers SET active = {$active}, timestamp = NOW()
            WHERE following_id = {$following_id} AND follower_id = {$follower_id}"
        );
    } else {
        // Insert
        $conn->query(
            "INSERT INTO followers (following_id, follower_id, active, timestamp)
            VALUES ({$following_id}, {$follower_id}, {$active}, NOW())"
        );
    }

    if ( $conn->error ) {
        wwqd_debug( 'Streams follow sync failed', [
            'error' => $conn->error,
            'following' => $following_id,
            'follower'  => $follower_id,
        ] );
        return false;
    }

    return true;
}

/**
 * Remove a follow relationship from WoWonder Streams.
 */
function wwqd_sync_streams_unfollow( $conn, $following_id, $follower_id ) {

    if ( ! $conn ) {
        return false;
    }

    $following_id = absint( $following_id );
    $follower_id = absint( $follower_id );

    $conn->query(
        "DELETE FROM followers WHERE following_id = {$following_id} AND follower_id = {$follower_id}"
    );

    return ! $conn->error;
}

/**
 * Create a block in WoWonder Streams.
 * 
 * Streams table: T_BLOCKS(blocker_id, blocked_id, timestamp)
 */
function wwqd_sync_streams_block( $conn, $blocker_id, $blocked_id ) {

    if ( ! $conn ) {
        return false;
    }

    $blocker_id = absint( $blocker_id );
    $blocked_id = absint( $blocked_id );

    if ( ! $blocker_id || ! $blocked_id || $blocker_id === $blocked_id ) {
        return false;
    }

    // Remove follow in both directions first
    $conn->query(
        "DELETE FROM followers WHERE
        (following_id = {$blocker_id} AND follower_id = {$blocked_id}) OR
        (following_id = {$blocked_id} AND follower_id = {$blocker_id})"
    );

    // Check if block already exists
    $existing = $conn->query(
        "SELECT * FROM T_BLOCKS WHERE blocker_id = {$blocker_id} AND blocked_id = {$blocked_id} LIMIT 1"
    );

    if ( ! $existing || $existing->num_rows === 0 ) {
        $conn->query(
            "INSERT INTO T_BLOCKS (blocker_id, blocked_id, timestamp)
            VALUES ({$blocker_id}, {$blocked_id}, NOW())"
        );
    }

    if ( $conn->error ) {
        wwqd_debug( 'Streams block sync failed', [
            'error'    => $conn->error,
            'blocker'  => $blocker_id,
            'blocked'  => $blocked_id,
        ] );
        return false;
    }

    return true;
}

## PART 4: KEY DESIGN DECISIONS EXPLAINED

| Decision | Rationale |
|----------|-----------|
| **Single MU Plugin** | Consolidates all sync logic; loads before other plugins |
| **Canonical State Model** | One source of truth prevents divergence across platforms |
| **MySQL Named Locks** | Per-pair serialization prevents race conditions |
| **Internal Context Flag** | Prevents recursive hook loops during block cleanup |
| **Event Ledger** | Durable, auditable, retry mechanism; not ephemeral queue |
| **Idempotent Event UUIDs** | External platforms can safely retry without duplication |
| **BuddyBoss Eligibility Preserved** | Sync engine doesn't duplicate role/profile restrictions |
| **Block Dominance** | Blocks cannot be overridden by lower operations |
| **Directional Blocking** | A blocks B ≠ B blocks A (correct model) |
| **Cascading Backoff** | Exponential retry (2, 4, 8, 16, 32 min) with max 10 attempts |
| **Log File Only** | No audit table; keeps schema minimal |
