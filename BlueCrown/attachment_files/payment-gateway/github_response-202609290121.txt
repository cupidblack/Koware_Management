## Analysis of Current Issues

Based on the error logs, the main problems are:

1. **Incorrect file paths** - Using `/streams/init.php` instead of `/streams/assets/init.php`
2. **WordPress function calls in Streams context** - `absint()` and other WP functions not available
3. **Missing database table checks** - Table existence not verified before use
4. **Missing database columns** - `signature_hash` column doesn't exist
5. **Architectural mismatch** - Trying to use WordPress functions outside WordPress context
6. **Missing admin page integration** - Payment Gateway settings not appearing in admin

<?php
/**
 * BZJ_PGB_Database
 * 
 * Database management for Payment Gateway
 */

class BZJ_PGB_Database {
    
    private $wpdb;
    private $logger;
    private $tables = array();
    
    public function __construct() {
        global $wpdb;
        $this->wpdb = $wpdb;
        $this->logger = new BZJ_PGB_Logger('class-pgb-database');
        
        $this->tables = array(
            'orders' => $wpdb->prefix . 'bzj_pgb_orders',
            'transactions' => $wpdb->prefix . 'bzj_pgb_transactions',
            'order_meta' => $wpdb->prefix . 'bzj_pgb_order_meta',
            'sync_log' => $wpdb->prefix . 'bzj_pgb_sync_log',
            'error_log' => $wpdb->prefix . 'bzj_pgb_error_log',
        );
    }
    
    /**
     * Ensure all required tables exist
     */
    public function ensure_tables_exist() {
        foreach ($this->tables as $table_name) {
            if (!$this->table_exists($table_name)) {
                $this->create_table($table_name);
            }
        }
        
        $this->logger->log('Database tables verified', 'info');
    }
    
    /**
     * Check if table exists
     */
    public function table_exists($table_name) {
        $result = $this->wpdb->get_var("SHOW TABLES LIKE '{$table_name}'");
        return $result === $table_name;
    }
    
    /**
     * Create tables
     */
    private function create_table($table_name) {
        $charset_collate = $this->wpdb->get_charset_collate();
        $sql = '';
        
        if (strpos($table_name, 'bzj_pgb_orders') !== false) {
            $sql = "CREATE TABLE {$table_name} (
                id BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                order_id BIGINT(20) UNSIGNED NOT NULL,
                woo_order_id BIGINT(20) UNSIGNED NOT NULL,
                streams_order_id VARCHAR(100) NOT NULL,
                user_id BIGINT(20) UNSIGNED NOT NULL,
                customer_email VARCHAR(255) NOT NULL,
                amount DECIMAL(10,2) NOT NULL,
                currency VARCHAR(10) NOT NULL,
                status VARCHAR(50) NOT NULL DEFAULT 'pending',
                transaction_type VARCHAR(50) NOT NULL,
                payment_gateway VARCHAR(50) NOT NULL,
                created_at DATETIME NOT NULL,
                updated_at DATETIME NOT NULL,
                completed_at DATETIME NULL,
                KEY woo_order_id (woo_order_id),
                KEY streams_order_id (streams_order_id),
                KEY user_id (user_id),
                KEY status (status),
                UNIQUE KEY order_unique (woo_order_id, streams_order_id)
            ) {$charset_collate};";
        }
        
        elseif (strpos($table_name, 'bzj_pgb_transactions') !== false) {
            $sql = "CREATE TABLE {$table_name} (
                id BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                order_id BIGINT(20) UNSIGNED NOT NULL,
                transaction_id VARCHAR(100) NOT NULL UNIQUE,
                type VARCHAR(50) NOT NULL,
                amount DECIMAL(10,2) NOT NULL,
                status VARCHAR(50) NOT NULL,
                response_code VARCHAR(10),
                response_message TEXT,
                external_transaction_id VARCHAR(100),
                created_at DATETIME NOT NULL,
                updated_at DATETIME NOT NULL,
                KEY order_id (order_id),
                KEY type (type),
                KEY status (status),
                FOREIGN KEY (order_id) REFERENCES {$this->tables['orders']} (id)
            ) {$charset_collate};";
        }
        
        elseif (strpos($table_name, 'bzj_pgb_order_meta') !== false) {
            $sql = "CREATE TABLE {$table_name} (
                id BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                order_id BIGINT(20) UNSIGNED NOT NULL,
                meta_key VARCHAR(255) NOT NULL,
                meta_value LONGTEXT,
                created_at DATETIME NOT NULL,
                KEY order_id (order_id),
                KEY meta_key (meta_key),
                FOREIGN KEY (order_id) REFERENCES {$this->tables['orders']} (id)
            ) {$charset_collate};";
        }
        
        elseif (strpos($table_name, 'bzj_pgb_sync_log') !== false) {
            $sql = "CREATE TABLE {$table_name} (
                id BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                order_id BIGINT(20),
                sync_type VARCHAR(50) NOT NULL,
                status VARCHAR(50) NOT NULL,
                details LONGTEXT,
                created_at DATETIME NOT NULL,
                KEY order_id (order_id),
                KEY sync_type (sync_type)
            ) {$charset_collate};";
        }
        
        elseif (strpos($table_name, 'bzj_pgb_error_log') !== false) {
            $sql = "CREATE TABLE {$table_name} (
                id BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                component VARCHAR(100) NOT NULL,
                error_level VARCHAR(20) NOT NULL,
                error_message TEXT NOT NULL,
                error_file VARCHAR(255),
                error_line INT,
                context LONGTEXT,
                created_at DATETIME NOT NULL,
                KEY component (component),
                KEY error_level (error_level),
                KEY created_at (created_at)
            ) {$charset_collate};";
        }
        
        if (!empty($sql)) {
            require_once ABSPATH . 'wp-admin/includes/upgrade.php';
            dbDelta($sql);
            $this->logger->log("Table created: {$table_name}", 'info');
        }
    }
    
    /**
     * Insert order
     */
    public function insert_order($data) {
        $data['created_at'] = current_time('mysql');
        $data['updated_at'] = current_time('mysql');
        
        $result = $this->wpdb->insert(
            $this->tables['orders'],
            $data,
            array('%d', '%d', '%s', '%d', '%s', '%f', '%s', '%s', '%s', '%s', '%s', '%s', '%s')
        );
        
        if (!$result) {
            $this->logger->log('Order insert failed: ' . $this->wpdb->last_error, 'error');
            return false;
        }
        
        return $this->wpdb->insert_id;
    }
    
    /**
     * Update order
     */
    public function update_order($order_id, $data) {
        $data['updated_at'] = current_time('mysql');
        
        $result = $this->wpdb->update(
            $this->tables['orders'],
            $data,
            array('id' => $order_id)
        );
        
        return $result !== false;
    }
    
    /**
     * Get order
     */
    public function get_order($order_id) {
        return $this->wpdb->get_row(
            $this->wpdb->prepare(
                "SELECT * FROM {$this->tables['orders']} WHERE id = %d",
                $order_id
            ),
            ARRAY_A
        );
    }
    
    /**
     * Get order by WooCommerce ID
     */
    public function get_order_by_woo_id($woo_order_id) {
        return $this->wpdb->get_row(
            $this->wpdb->prepare(
                "SELECT * FROM {$this->tables['orders']} WHERE woo_order_id = %d",
                $woo_order_id
            ),
            ARRAY_A
        );
    }
    
    /**
     * Insert transaction
     */
    public function insert_transaction($data) {
        $data['created_at'] = current_time('mysql');
        $data['updated_at'] = current_time('mysql');
        
        $result = $this->wpdb->insert(
            $this->tables['transactions'],
            $data
        );
        
        return $result !== false ? $this->wpdb->insert_id : false;
    }
    
    /**
     * Insert order meta
     */
    public function insert_order_meta($order_id, $meta_key, $meta_value) {
        $data = array(
            'order_id' => $order_id,
            'meta_key' => $meta_key,
            'meta_value' => is_array($meta_value) ? json_encode($meta_value) : $meta_value,
            'created_at' => current_time('mysql'),
        );
        
        return $this->wpdb->insert($this->tables['order_meta'], $data) !== false;
    }
    
    /**
     * Get order meta
     */
    public function get_order_meta($order_id, $meta_key = null) {
        if ($meta_key) {
            return $this->wpdb->get_var(
                $this->wpdb->prepare(
                    "SELECT meta_value FROM {$this->tables['order_meta']} WHERE order_id = %d AND meta_key = %s",
                    $order_id,
                    $meta_key
                )
            );
        }
        
        return $this->wpdb->get_results(
            $this->wpdb->prepare(
                "SELECT meta_key, meta_value FROM {$this->tables['order_meta']} WHERE order_id = %d",
                $order_id
            ),
            ARRAY_A
        );
    }
    
    /**

     */

            return null;
        }
        
        $query = "SELECT * FROM Wo_Users WHERE wp_user_id = " . intval($wp_user_id) . " LIMIT 1";
        $result = $this->streams_db->query($query);
        
        if ($result && $result->num_rows > 0) {
            return $result->fetch_assoc();
        }
        
        return null;

     * Update Streams user subscription status
     */
    public function update_streams_user_subscription($streams_user_id, $subscription_data) {
        if (!$this->streams_db) {
            return false;
        }
        
        $update_fields = array();
        
        if (isset($subscription_data['pro_type'])) {
            $update_fields[] = "pro_type = '" . $this->streams_db->real_escape_string($subscription_data['pro_type']) . "'";
        }
        
        if (isset($subscription_data['is_pro'])) {
            $update_fields[] = "is_pro = " . intval($subscription_data['is_pro']);
        }
        
        if (isset($subscription_data['pro_expiration'])) {
            $update_fields[] = "pro_expiration = " . intval($subscription_data['pro_expiration']);
        }
        
        if (empty($update_fields)) {
            return false;
        }
        
        $sql = "UPDATE Wo_Users SET " . implode(", ", $update_fields) . " WHERE user_id = " . intval($streams_user_id);
        
        if ($this->streams_db->query($sql)) {
            $this->logger->info("Updated Streams user subscription: {$streams_user_id}", $subscription_data);
            return true;
        }
        
        $this->logger->error("Failed to update Streams user subscription: " . $this->streams_db->error);
        return false;
    }

     * Log payment transaction in Streams
     */

        if (!$this->streams_db) {
            return false;
        }
        
        $insert_data = array(
            'userid' => intval($streams_user_id),
            'amount' => floatval($transaction_data['amount']),
            'order_id' => $this->streams_db->real_escape_string($transaction_data['order_id']),
            'kind' => $this->streams_db->real_escape_string($transaction_data['kind'] ?? 'purchase'),
            'currency_code' => $this->streams_db->real_escape_string($transaction_data['currency'] ?? 'USD'),
            'status' => 'completed',
            'time' => time(),
        );
        
        $fields = array_keys($insert_data);
        $values = array_map(function($v) {
            return "'" . (is_numeric($v) && !is_string($v) ? $v : $this->streams_db->real_escape_string($v)) . "'";
        }, array_values($insert_data));
        
        $sql = "INSERT INTO Wo_Payment_Transactions (" . implode(",", $fields) . ") VALUES (" . implode(",", $values) . ")";
        
        if ($this->streams_db->query($sql)) {
            $this->logger->info("Logged payment transaction in Streams", $transaction_data);
            return true;
        }
        
        $this->logger->error("Failed to log payment transaction: " . $this->streams_db->error);
        return false;
    }
    
    /**
     * Close Streams database connection
     */
    public function close() {
        if ($this->streams_db) {
            $this->streams_db->close();
        }
    }

class BZJ_PGB_Subscription_Handler {

    /**
     * Process order subscriptions
     */
    public function process_order_subscriptions($woo_order, $pgb_order) {
        try {
            // Check if WooCommerce Subscriptions is active
            if (!class_exists('WC_Subscription')) {
                $this->logger->debug("WooCommerce Subscriptions not active");
                return true;
            }
            
            // Check if order contains subscriptions
            $subscriptions = wcs_get_subscriptions_for_order($woo_order->get_id());
            
            if (empty($subscriptions)) {
                $this->logger->debug("Order contains no subscriptions: " . $woo_order->get_id());
                return true;
            }
            
            foreach ($subscriptions as $subscription) {
                $this->activate_subscription($subscription, $pgb_order);
            }
            
            $this->logger->info("Processed subscriptions for order: " . $woo_order->get_id());
            return true;
            
        } catch (Exception $e) {
            $this->logger->error("Error processing subscriptions: " . $e->getMessage());
            return false;
        }
    }
    
    /**
     * Activate subscription
     */
    private function activate_subscription($subscription, $pgb_order) {
        try {
            // Get subscription metadata
            $subscription_period = $subscription->get_meta('_subscription_period') ?: 'month';
            $subscription_interval = $subscription->get_meta('_subscription_interval') ?: 1;
            $subscription_length = $subscription->get_meta('_subscription_length') ?: 0;
            
            // Update Streams user with subscription info
            $streams_handler = new BZJ_PGB_Streams_Handler();
            $wp_user = get_user_by('id', $subscription->get_user_id());
            
            if ($wp_user) {
                $streams_user = $streams_handler->get_streams_user_by_wp_id($wp_user->ID);
                
                if ($streams_user) {
                    $subscription_data = array(
                        'pro_type' => $this->get_subscription_type_from_product($subscription->get_product_id()),
                        'is_pro' => 1,
                        'pro_expiration' => $this->calculate_expiration($subscription_period, $subscription_interval),
                    );
                    
                    $streams_handler->update_streams_user_subscription($streams_user['user_id'], $subscription_data);
                }
                
                $streams_handler->close();
            }
            
            $this->logger->info("Activated subscription for order: " . $pgb_order['id']);
            return true;
            
        } catch (Exception $e) {
            $this->logger->error("Error activating subscription: " . $e->getMessage());
            return false;
        }
    }
    
    /**
     * Get subscription type from product ID
     */
    private function get_subscription_type_from_product($product_id) {
        // Map WooCommerce product IDs to Streams pro types
        // This should match your configuration
        $mapping = get_option('bzj_pgb_subscription_mapping', array());
        return isset($mapping[$product_id]) ? $mapping[$product_id] : '1';
    }
    
    /**
     * Calculate expiration timestamp
     */
    private function calculate_expiration($period, $interval) {
        $intervals = array(
            'day' => 'days',
            'week' => 'weeks',
            'month' => 'months',
            'year' => 'years',
        );
        
        $interval_key = isset($intervals[$period]) ? $intervals[$period] : 'months';
        $timestamp = strtotime("+{$interval} {$interval_key}");
        
        return $timestamp;
    }

     * Initiate payment
     */
    public function initiate_payment($order_data) {
        try {
            $this->logger->info("Payment initiation requested", $order_data);
            
            // Validate order data
            $order_data = $this->validate_order_data($order_data);
            
            // Get WordPress user from Streams user
            $wp_user_id = $this->get_wordpress_user_id($order_data['wowonder_user_id']);
            
            if (!$wp_user_id) {
                throw new Exception("WordPress user not found for Streams user");
            }
            
            // Convert currency if necessary
            $order_data = $this->convert_currency($order_data);
            
            // Create order in WordPress
            $woo_order_id = $this->create_woocommerce_order($wp_user_id, $order_data);
            
            if (!$woo_order_id) {
                throw new Exception("Failed to create WooCommerce order");
            }
            
            // Store Streams context in WooCommerce order meta
            $woo_order = wc_get_order($woo_order_id);
            $woo_order->update_meta_data('_buzzjuice_streams_user_id', $order_data['wowonder_user_id']);
            $woo_order->update_meta_data('_buzzjuice_streams_order_id', $order_data['order_id']);
            $woo_order->update_meta_data('_transaction_type', $order_data['transaction_kind']);
            $woo_order->save_meta_data();
            
            // Store transaction in Streams database
            $this->store_streams_transaction($order_data, $woo_order_id);
            
            // Get checkout URL
            $checkout_url = $woo_order->get_checkout_payment_url(false);
            
            $this->logger->info("Payment initiated successfully", array(
                'woo_order_id' => $woo_order_id,
                'streams_user_id' => $order_data['wowonder_user_id'],
            ));
            
            return array(
                'status' => 200,
                'message' => 'Payment initiated successfully',
                'url' => $checkout_url,
                'woo_order_id' => $woo_order_id,
            );
            
        } catch (Exception $e) {
            $this->logger->error("Payment initiation failed: " . $e->getMessage(), array(
                'file' => $e->getFile(),
                'line' => $e->getLine(),
            ));
            
            return array(
                'status' => 'error',
                'message' => $e->getMessage(),
            );
        }
    }
    
    /**
     * Validate order data
     */
    private function validate_order_data($data) {
        global $wo;
        
        $validated = array(
            'wowonder_user_id' => intval($wo['user']['user_id'] ?? 0),
            'wowonder_email' => filter_var($wo['user']['email'] ?? '', FILTER_VALIDATE_EMAIL),
            'wowonder_username' => htmlspecialchars($wo['user']['username'] ?? ''),
            'amount' => floatval($data['amount'] ?? 0),
            'product_name' => htmlspecialchars(trim($data['product_name'] ?? ''), ENT_QUOTES, 'UTF-8'),
            'product_price' => floatval($data['product_price'] ?? 0),
            'product_units' => intval($data['product_units'] ?? 1),
            'product_owner_id' => intval($data['product_owner_id'] ?? 0),
            'product_id' => intval($data['wow_post_id'] ?? 0),
            'transaction_kind' => preg_replace('/[^a-zA-Z0-9_-]/', '', $data['transaction_kind'] ?? ''),
            'currency' => htmlspecialchars($data['wow_currency_code'] ?? 'USD', ENT_QUOTES, 'UTF-8'),
            'order_id' => 'streams_' . uniqid() . '_' . time(),
        );
        
        // Validate required fields
        if (empty($validated['wowonder_user_id'])) {
            throw new Exception("Invalid Streams user ID");
        }
        
        if (empty($validated['wowonder_email'])) {
            throw new Exception("Invalid email address");
        }
        
        if ($validated['amount'] <= 0) {
            throw new Exception("Invalid amount");
        }
        
        return $validated;
    }
    
    /**
     * Get WordPress user ID from Streams user
     */
    private function get_wordpress_user_id($streams_user_id) {
        if (!$this->streams_db) {
            throw new Exception("No Streams database connection");
        }
        
        $query = "SELECT wp_user_id FROM Wo_Users WHERE user_id = " . intval($streams_user_id) . " LIMIT 1";
        $result = $this->streams_db->query($query);
        
        if ($result && $result->num_rows > 0) {
            $row = $result->fetch_assoc();
            return intval($row['wp_user_id']);
        }
        
        return null;
    }
    
    /**
     * Convert currency using WOOCS if necessary
     */
    private function convert_currency($order_data) {
        if (function_exists('woocs_convert_price')) {
            // Currency conversion is handled at checkout
            // Store original currency for reference
            $order_data['original_amount'] = $order_data['amount'];
            $order_data['original_currency'] = $order_data['currency'];
        }
        
        return $order_data;
    }
    
    /**
     * Create WooCommerce order
     */
    private function create_woocommerce_order($wp_user_id, $order_data) {
        try {
            $order = wc_create_order(array(
                'customer_id' => $wp_user_id,
                'status' => 'pending',
            ));
            
            if (is_wp_error($order)) {
                throw new Exception($order->get_error_message());
            }
            
            // Add line item
            $product_id = $this->get_woocommerce_product_id($order_data['transaction_kind']);
            
            $order->add_product(
                wc_get_product($product_id),
                $order_data['product_units'],
                array(
                    'subtotal' => $order_data['product_price'] * $order_data['product_units'],
                    'total' => $order_data['amount'],
                )
            );
            
            // Set billing and shipping
            $order->set_billing_email($order_data['wowonder_email']);
            $order->set_currency($order_data['currency']);
            
            // Calculate totals
            $order->calculate_totals();
            
            $this->logger->info("WooCommerce order created: " . $order->get_id());
            
            return $order->get_id();
            
        } catch (Exception $e) {
            $this->logger->error("Failed to create WooCommerce order: " . $e->getMessage());
            throw $e;
        }
    }
    
    /**
     * Get WooCommerce product ID for transaction type
     */
    private function get_woocommerce_product_id($transaction_kind) {
        // Map transaction kinds to product IDs
        // This should be configurable
        $mapping = array(
            'PRODUCT' => get_option('bzj_pgb_product_id', 1),
            'PRO' => get_option('bzj_pgb_pro_product_id', 2),
            'WALLET' => get_option('bzj_pgb_wallet_product_id', 3),
            'DONATE' => get_option('bzj_pgb_donate_product_id', 4),
        );
        
        return isset($mapping[$transaction_kind]) ? $mapping[$transaction_kind] : 1;
    }
    
    /**
     * Store transaction in Streams database
     */
    private function store_streams_transaction($order_data, $woo_order_id) {
        if (!$this->streams_db) {
            return false;
        }
        
        $sql = "INSERT INTO Wo_Payment_Transactions (
            userid,
            amount,
            order_id,
            kind,
            currency_code,
            status,
            time
        ) VALUES (
            " . intval($order_data['wowonder_user_id']) . ",
            " . floatval($order_data['amount']) . ",
            '" . $this->streams_db->real_escape_string($order_data['order_id']) . "',
            '" . $this->streams_db->real_escape_string($order_data['transaction_kind']) . "',
            '" . $this->streams_db->real_escape_string($order_data['currency']) . "',
            'pending',
            " . time() . "
        )";
        
        if (!$this->streams_db->query($sql)) {
            $this->logger->warning("Failed to store transaction in Streams: " . $this->streams_db->error);
            return false;
        }
        
        return true;
    }
}
