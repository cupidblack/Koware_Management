```php
 * This function creates pending orders in a special table that the MU plugin

function pgb_create_pending_order($wow_user_id, $order_data) {

    // Create table if it doesn't exist

    $wpdb->query("CREATE TABLE IF NOT EXISTS `{$table}` (
        id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
        wow_user_id BIGINT UNSIGNED NOT NULL,
        wow_order_id VARCHAR(255) NOT NULL UNIQUE,
        woo_order_id BIGINT UNSIGNED,
        order_data LONGTEXT NOT NULL,
        status VARCHAR(50) DEFAULT 'pending',
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        processed_at TIMESTAMP NULL,
        INDEX idx_wow_user (wow_user_id),
        INDEX idx_status (status)
    )");
    
    $result = $wpdb->insert(
        $table,
        [
            'wow_user_id' => $wow_user_id,
            'wow_order_id' => $order_data['wow_order_id'] ?? uniqid('wow_'),
            'order_data' => json_encode($order_data),
            'status' => 'pending'
        ],
        ['%d', '%s', '%s', '%d']
    );
    
    return $result ? $wpdb->insert_id : false;
}

            "SELECT * FROM wp_bzj_pgb_pending_orders WHERE wow_order_id = %s AND status = 'pending' LIMIT 1",

        ARRAY_A
    );
```

```php
<?php
/**
 * Payment Gateway Bridge - Logger
 * Centralized logging with component-based file separation
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted');
}

class PGB_Logger {
    const LOG_OFF = 0;
    const LOG_CRITICAL = 1;
    const LOG_ERROR = 2;
    const LOG_WARNING = 3;
    const LOG_INFO = 4;
    const LOG_DEBUG = 5;
    
    private static $instance = null;
    private $log_dir = '';
    private $enabled = true;
    private $level = self::LOG_INFO;
    private $file_handles = [];
    
    public static function instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->log_dir = BZJ_PGB_LOG;
        $this->enabled = get_option('pgb_logging_enabled', true);
        $this->level = $this->level_from_string(get_option('pgb_log_level', 'info'));
    }
    
    /**
     * Convert string level to constant
     */
    private function level_from_string($level) {
        $levels = [
            'off' => self::LOG_OFF,
            'critical' => self::LOG_CRITICAL,
            'error' => self::LOG_ERROR,
            'warning' => self::LOG_WARNING,
            'info' => self::LOG_INFO,
            'debug' => self::LOG_DEBUG,
        ];
        return $levels[strtolower($level)] ?? self::LOG_INFO;
    }
    
    /**
     * Main logging function
     */
    public function log($message, $level = self::LOG_INFO, $context = [], $component = 'core') {
        if (!$this->enabled || $level > $this->level) {
            return;
        }
        
        $timestamp = date('Y-m-d H:i:s');
        $level_name = $this->level_name($level);
        
        // Redact sensitive data
        $message = $this->redact_sensitive($message);
        
        // Format with context
        $formatted = "[{$timestamp}] [{$level_name}] {$message}";
        if (!empty($context)) {
            $formatted .= " | " . json_encode($this->redact_sensitive($context));
        }
        $formatted .= "\n";
        
        // Write to component log file
        $this->write_log_file($component, $formatted);
        
        // Also log to WordPress error log for critical errors
        if ($level <= self::LOG_ERROR) {
            error_log("[PGB-{$component}] {$message}");
        }
    }
    
    /**
     * Write to log file
     */
    private function write_log_file($component, $message) {
        $file_path = $this->log_dir . '/' . $component . '.log';
        
        // Rotate if file is too large (10MB)
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }
        
        // Write line
        $handle = fopen($file_path, 'a');
        if ($handle) {
            fwrite($handle, $message);
            fclose($handle);
        }
    }
    
    /**
     * Redact sensitive data
     */
    private function redact_sensitive($data) {
        if (is_string($data)) {
            $data = preg_replace('/api[_-]?key["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i', 'api_key=***REDACTED***', $data);
            $data = preg_replace('/api[_-]?secret["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i', 'api_secret=***REDACTED***', $data);
            $data = preg_replace('/password["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i', 'password=***REDACTED***', $data);
            $data = preg_replace('/token["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i', 'token=***REDACTED***', $data);
        } elseif (is_array($data)) {
            foreach ($data as $key => &$value) {
                if (stripos($key, 'secret') !== false || stripos($key, 'key') !== false || stripos($key, 'password') !== false || stripos($key, 'token') !== false) {
                    $value = '***REDACTED***';
                }
            }
        }
        return $data;
    }
    
    /**
     * Get level name
     */
    private function level_name($level) {
        $names = [
            self::LOG_OFF => 'OFF',
            self::LOG_CRITICAL => 'CRITICAL',
            self::LOG_ERROR => 'ERROR',
            self::LOG_WARNING => 'WARNING',
            self::LOG_INFO => 'INFO',
            self::LOG_DEBUG => 'DEBUG',
        ];
        return $names[$level] ?? 'UNKNOWN';
    }
    
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}
```

```php
class PGB_Validator {
    /**
     * Validate email
     */
    public static function email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }
    
    /**
     * Sanitize email
     */
    public static function sanitize_email($email) {
        return sanitize_email($email);
    }
    
    /**
     * Validate amount
     */
    public static function amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }
    
    /**
     * Validate user ID
     */
    public static function user_id($user_id) {
        return intval($user_id) > 0;
    }
    
    /**
     * Validate transaction type
     */
    public static function transaction_type($type) {
        $allowed = ['PRODUCT', 'PRO', 'WALLET', 'DONATE', 'PURCHASE', 'SALE'];
        return in_array(strtoupper($type), $allowed);
    }
    
    /**
     * Validate currency code
     */
    public static function currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;
    }
    
    /**
     * Validate WooCommerce order data
     */
    public static function woocommerce_order($order_data) {
        $errors = [];
        
        if (empty($order_data['customer_id']) || !self::user_id($order_data['customer_id'])) {
            $errors[] = 'Invalid customer_id';
        }
        
        if (empty($order_data['line_items']) || !is_array($order_data['line_items'])) {
            $errors[] = 'No line items provided';
        }
        
        if (!empty($order_data['line_items'])) {
            foreach ($order_data['line_items'] as $idx => $item) {
                if (empty($item['product_id'])) {
                    $errors[] = "Line item {$idx}: missing product_id";
                }
                if (empty($item['quantity']) || $item['quantity'] <= 0) {
                    $errors[] = "Line item {$idx}: invalid quantity";
                }
                if (!isset($item['total']) || $item['total'] < 0) {
                    $errors[] = "Line item {$idx}: invalid total";
                }
            }
        }
        
        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }
}
```

```php
<?php
/**
 * Payment Gateway Bridge - Order Manager
 * Handles order creation, state management, and fulfillment
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted');
}

class PGB_Order_Manager {
    private $logger = null;
    private $config = null;
    private $wpdb = null;
    private $wow_db = null;
    
    public function __construct() {
        $this->logger = PGB_Logger::instance();
        $this->config = PGB_Config::instance();
        $this->wpdb = $GLOBALS['wpdb'] ?? null;
        $this->wow_db = get_wowonder_db();
    }
    
    /**
     * Process pending order from database
     * This is called via AJAX after Streams redirects to WordPress
     */
    public function process_pending_order($pending) {
        $this->logger->log('Processing pending order: ' . $pending['wow_order_id'], PGB_Logger::LOG_INFO, [], 'order-manager');
        
        $order_data = json_decode($pending['order_data'], true);
        if (!$order_data) {
            return [
                'success' => false,
                'error' => 'Invalid order data',
            ];
        }
        
        // Validate
        $validation = PGB_Validator::woocommerce_order($order_data);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                $validation['errors'],
                'order-manager'
            );
            return [
                'success' => false,
                'error' => 'Order validation failed: ' . implode(', ', $validation['errors']),
            ];
        }
        
        // Create WooCommerce order
        try {
            $woo_order = $this->create_woocommerce_order($order_data);
            if (!$woo_order) {
                return [
                    'success' => false,
                    'error' => 'Failed to create WooCommerce order',
                ];
            }
            
            // Update pending order
            pgb_update_pending_order($pending['wow_order_id'], 'processed', $woo_order->get_id());
            
            // Get checkout URL
            $checkout_url = $woo_order->get_checkout_payment_url();
            
            $this->logger->log(
                'Order processed: ' . $woo_order->get_id(),
                PGB_Logger::LOG_INFO,
                ['checkout_url' => $checkout_url],
                'order-manager'
            );
            
            return [
                'success' => true,
                'checkout_url' => $checkout_url,
                'order_id' => $woo_order->get_id(),
            ];
        } catch (Exception $e) {
            $this->logger->log(
                'Exception: ' . $e->getMessage(),
                PGB_Logger::LOG_CRITICAL,
                ['trace' => $e->getTraceAsString()],
                'order-manager'
            );
            return [
                'success' => false,
                'error' => 'Internal error',
            ];
        }
    }
    
    /**
     * Create WooCommerce order
     */
    private function create_woocommerce_order($order_data) {
        $order = wc_create_order();
        
        // Set customer
        if (!empty($order_data['customer_id'])) {
            $order->set_customer_id($order_data['customer_id']);
        }
        
        // Add line items
        foreach ($order_data['line_items'] as $item_data) {
            $product = wc_get_product($item_data['product_id']);
            if ($product) {
                $order->add_product(
                    $product,
                    $item_data['quantity'] ?? 1,
                    [
                        'subtotal' => $item_data['subtotal'] ?? $item_data['total'],
                        'total' => $item_data['total'],
                    ]
                );
            }
        }
        
        // Set currency
        if (!empty($order_data['currency'])) {
            $order->set_currency($order_data['currency']);
        }
        
        // Add metadata
        if (!empty($order_data['meta_data'])) {
            foreach ($order_data['meta_data'] as $meta) {
                $order->update_meta_data($meta['key'], $meta['value']);
            }
        }
        
        // Add billing address if provided
        if (!empty($order_data['billing'])) {
            $order->set_address($order_data['billing'], 'billing');
        }
        
        // Add shipping address if provided
        if (!empty($order_data['shipping'])) {
            $order->set_address($order_data['shipping'], 'shipping');
        }
        
        // Calculate totals
        $order->calculate_totals();
        
        // Save order
        $order->save();
        
        // Fire hooks
        PGB_Hooks::do_action(PGB_Hooks::AFTER_ORDER_CREATE, $order);
        
        return $order;
    }
    
    /**
     * Called when payment is completed
     */
    public function on_payment_complete($order_id) {
        $order = wc_get_order($order_id);
        if (!$order) return;
```

```php
        return;

    $affiliate_id = 0;
```

```php
/**
 * Convert currency if needed
 */
add_filter('pgb_before_order_create', function($order_data) {
    $logger = PGB_Logger::instance();
    
    if (empty($order_data['currency']) || empty($order_data['line_items'])) {
        return $order_data;
    }
    
    $currency = $order_data['currency'];
    $store_currency = get_woocommerce_currency();
    
    if ($currency === $store_currency) {
        return $order_data; // No conversion needed
    }
    
    $logger->log(
        'Currency conversion needed',
        PGB_Logger::LOG_DEBUG,
        ['from' => $currency, 'to' => $store_currency],
        'addon-currency'
    );
    
    // Apply WOOCS conversion if available
    $woocs = apply_filters('woocs_get_currencies', []);
    
    if (isset($woocs[$currency]) && isset($woocs[$store_currency])) {
        $rate = floatval($woocs[$currency]['rate'] ?? 1.0) / floatval($woocs[$store_currency]['rate'] ?? 1.0);
        
        // Convert all prices
        foreach ($order_data['line_items'] as &$item) {
            $item['total'] = round(floatval($item['total']) * $rate, 2);
            $item['subtotal'] = round(floatval($item['subtotal']) * $rate, 2);
        }
        
        $order_data['currency'] = $store_currency;
        
        $logger->log(
            'Currency converted',
            PGB_Logger::LOG_DEBUG,
            ['rate' => $rate],
            'addon-currency'
        );
    }
    
    return $order_data;
```

```php
add_action('pgb_after_order_complete', function($order) {
    $logger = PGB_Logger::instance();
    
    // Get WooCommerce order ID
    $woo_order_id = $order->get_id();
    $wow_user_id = intval($order->get_meta('wow_user_id') ?? 0);
    
    if (!$wow_user_id) {
        return;
    }
    
    $logger->log(
        'Processing Jewel rebate',
        PGB_Logger::LOG_INFO,
        ['woo_order_id' => $woo_order_id, 'wow_user_id' => $wow_user_id],
        'int-jewel-affiliate'
    );
    
    // Check if already processed
    if ($order->get_meta('_bzj_jewel_processed')) {
        $logger->log(
            'Jewel already processed',
            PGB_Logger::LOG_DEBUG,
            ['woo_order_id' => $woo_order_id],
            'int-jewel-affiliate'
        );
        return;
    }
    
    // Load mapping from WP options
    $mapping_json = get_option('bz_rebate_mapping_json', '');
    $mapping = json_decode($mapping_json, true) ?: [];
    
    if (empty($mapping)) {
        $logger->log(
            'No Jewel mapping configured',
            PGB_Logger::LOG_WARNING,
            [],
            'int-jewel-affiliate'
        );
        return;
    }
    
    // Check order items against mapping
    $wow_db = get_wowonder_db();
    if (!$wow_db) {
        return;
    }
    
    $items = $order->get_items();
    foreach ($items as $item) {
        $product_id = $item->get_product_id();
        $variation_id = $item->get_variation_id() ?: $product_id;
        
        // Check if this variation is in mapping
        $matched = false;
        foreach ($mapping as $m) {
            if (intval($m['variation_id'] ?? 0) === $variation_id) {
                $matched = true;
                $pro_type = intval($m['wow_pro_type'] ?? 0);
                $rebate_cents = intval($m['rebate_amount'] ?? 0);
                
                // Activate pro membership in Streams
                if ($pro_type > 0) {
                    $sql = "UPDATE Wo_Users SET pro_type = ?, is_pro = '1', pro_time = ? WHERE user_id = ? LIMIT 1";
                    $stmt = $wow_db->prepare($sql);
                    if ($stmt) {
                        $now = time();
                        $stmt->bind_param('iii', $pro_type, $now, $wow_user_id);
                        $stmt->execute();
                        $stmt->close();
                        
                        $logger->log(
                            'Pro activated in Streams',
                            PGB_Logger::LOG_INFO,
                            ['pro_type' => $pro_type, 'wow_user_id' => $wow_user_id],
                            'int-jewel-affiliate'
                        );
                    }
                }
                
                // Apply rebate credit
                if ($rebate_cents > 0) {
                    $rebate_amount = $rebate_cents / 100;
                    $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ? LIMIT 1";
                    $stmt = $wow_db->prepare($sql);
                    if ($stmt) {
                        $stmt->bind_param('di', $rebate_amount, $wow_user_id);
                        $stmt->execute();
                        $stmt->close();
                        
                        $logger->log(
                            'Rebate applied',
                            PGB_Logger::LOG_INFO,
                            ['amount' => $rebate_amount, 'wow_user_id' => $wow_user_id],
                            'int-jewel-affiliate'
                        );
                    }
                }
                
                break;
            }
        }
    }
    
    // Mark as processed
    $order->update_meta_data('_bzj_jewel_processed', true);
    $order->save();
    
}, 30, 1);

?>
```

## **PART 10: TROUBLESHOOTING**

### **Orders Not Being Created**

**Check:**
1. Log file: `shared/payment-gateway/log/order-manager.log`
2. Verify WP user exists: `SELECT * FROM wp_users WHERE ID = {wp_user_id}`
3. Verify WooCommerce products exist: `SELECT * FROM wp_posts WHERE post_type = 'product'`

### **AffiliateWP Not Processing**

**Check:**
1. `addon-affiliate-wp.log` for errors
2. Affiliate exists: `SELECT * FROM wp_affiliate_wp_affiliates`
3. AffiliateWP plugin is active

### **Subscriptions Not Creating**

**Check:**
1. `addon-subscriptions.log` for errors
2. WC Subscriptions plugin is active
3. Order meta contains `transaction_kind = 'PRO'`
