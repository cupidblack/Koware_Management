/**
 * Handle REST order creation
 */
function bzj_pgb_rest_order_handler(WP_REST_Request $request) {
    $logger = PGB_Bootstrap::instance()->get_logger();
    $manager = PGB_Bootstrap::instance()->get_manager();
    
    $logger->log(
        'REST order endpoint called',
        PGB_Logger::LOG_DEBUG,
        ['method' => $request->get_method()],
        __FILE__
    );
    
    // Validate signature
    $headers = getallheaders();
    $timestamp = $headers['X-Bzj-Pgb-Timestamp'] ?? '';
    $request_id = $headers['X-Bzj-Pgb-Request-Id'] ?? '';
    $signature = $headers['X-Bzj-Pgb-Signature'] ?? '';
    
    if (!$timestamp || !$request_id || !$signature) {
        $logger->log(
            'Missing required headers',
            PGB_Logger::LOG_WARNING,
            [],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Missing required headers',
        ], 400);
    }
    
    // Verify timestamp (within 5 minutes)
    $now = time();
    if (abs($now - intval($timestamp)) > 300) {
        $logger->log(
            'Timestamp verification failed',
            PGB_Logger::LOG_WARNING,
            ['timestamp' => $timestamp, 'now' => $now],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Request timestamp expired',
        ], 401);
    }
    
    // Get body for signature verification
    $body = $request->get_body();
    $secret = defined('BZJ_PGB_SECRET') ? BZJ_PGB_SECRET : getenv('BZJ_PGB_SECRET');
    
    if (empty($secret)) {
        $logger->log(
            'BZJ_PGB_SECRET not configured',
            PGB_Logger::LOG_CRITICAL,
            [],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Server configuration error',
        ], 500);
    }
    
    // Verify HMAC-SHA256 signature
    $expected_signature = base64_encode(hash_hmac(
        'sha256',
        $timestamp . "\n" . $request_id . "\n" . $body,
        $secret,
        true
    ));
    
    if (!hash_equals($signature, $expected_signature)) {
        $logger->log(
            'Signature verification failed',
            PGB_Logger::LOG_WARNING,
            ['provided' => $signature, 'expected' => $expected_signature],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Invalid signature',
        ], 401);
    }
    
    // Parse JSON body
    $data = json_decode($body, true);
    if (!$data) {
        $logger->log(
            'Invalid JSON payload',
            PGB_Logger::LOG_WARNING,
            [],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Invalid JSON',
        ], 400);
    }
    
    // Extract and validate data
    $wow_user_id = intval($data['wow_user_id'] ?? 0);
    $wow_order_id = trim($data['wow_order_id'] ?? '');
    $line_items = $data['line_items'] ?? [];
    $meta_data = $data['meta_data'] ?? [];
    $billing = $data['billing'] ?? [];
    $shipping = $data['shipping'] ?? [];
    
    if (!$wow_user_id || !$wow_order_id || empty($line_items)) {
        $logger->log(
            'Missing required order data',
            PGB_Logger::LOG_WARNING,
            ['wow_user_id' => $wow_user_id, 'wow_order_id' => $wow_order_id],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'Missing required fields',
        ], 400);
    }
    
    // Get WordPress user ID from Streams user ID mapping
    $wp_user_id = bzj_pgb_get_wp_user_id($wow_user_id);
    if (!$wp_user_id) {
        // Try to create or link user
        $wp_user_id = bzj_pgb_create_or_link_user($data);
    }
    
    if (!$wp_user_id) {
        $logger->log(
            'Failed to resolve WordPress user',
            PGB_Logger::LOG_ERROR,
            ['wow_user_id' => $wow_user_id],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => 'User not found or could not be created',
        ], 400);
    }
    
    // Prepare WooCommerce order
    $order_data = [
        'customer_id'  => $wp_user_id,
        'set_paid'     => false,
        'currency'     => $data['currency'] ?? 'GHS',
        'line_items'   => $line_items,
        'meta_data'    => array_merge($meta_data, [
            ['key' => '_bzj_wow_user_id', 'value' => $wow_user_id],
            ['key' => '_bzj_wow_order_id', 'value' => $wow_order_id],
            ['key' => '_bzj_request_id', 'value' => $request_id],
        ]),
    ];
    
    if (!empty($billing)) {
        $order_data['billing'] = $billing;
    }
    if (!empty($shipping)) {
        $order_data['shipping'] = $shipping;
    }
    
    try {
        // Create WooCommerce order
        $order = wc_create_order();
        
        if (is_wp_error($order)) {
            throw new Exception('Failed to create order: ' . $order->get_error_message());
        }
        
        // Set customer
        $order->set_customer_id($wp_user_id);
        
        // Add line items
        foreach ($line_items as $item) {
            $product = wc_get_product(intval($item['product_id'] ?? 0));
            if (!$product) {
                throw new Exception('Product not found: ' . $item['product_id']);
            }
            
            $order->add_product(
                $product,
                intval($item['quantity'] ?? 1),
                [
                    'subtotal' => floatval($item['subtotal'] ?? 0),
                    'total'    => floatval($item['total'] ?? 0),
                ]
            );
        }
        
        // Add meta data
        foreach ($order_data['meta_data'] as $meta) {
            $order->update_meta_data($meta['key'], $meta['value']);
        }
        
        // Set addresses
        if (!empty($order_data['billing'])) {
            $order->set_address($order_data['billing'], 'billing');
        }
        if (!empty($order_data['shipping'])) {
            $order->set_address($order_data['shipping'], 'shipping');
        }
        
        // Set currency
        if (!empty($order_data['currency'])) {
            $order->set_currency($order_data['currency']);
        }
        
        // Save order
        $order->save();
        $order_id = $order->get_id();
        
        $logger->log(
            'Order created successfully',
            PGB_Logger::LOG_INFO,
            ['order_id' => $order_id, 'wow_order_id' => $wow_order_id],
            __FILE__
        );
        
        // Generate checkout URL
        $checkout_url = $order->get_checkout_payment_url();
        
        return new WP_REST_Response([
            'status'        => 'success',
            'order_id'      => $order_id,
            'wow_order_id'  => $wow_order_id,
            'checkout_url'  => $checkout_url,
        ], 201);
        
    } catch (Exception $e) {
        $logger->log(
            'Order creation failed: ' . $e->getMessage(),
            PGB_Logger::LOG_CRITICAL,
            ['exception' => $e->getMessage()],
            __FILE__
        );
        return new WP_REST_Response([
            'status'  => 'error',
            'message' => $e->getMessage(),
        ], 500);
    }
}

/**
 * Extract payment data from WooCommerce order
 */
function bzj_pgb_extract_payment_data($order) {
    $payment_data = [
        'order_id'       => $order->get_id(),
        'wow_user_id'    => intval($order->get_meta('_bzj_wow_user_id')),
        'wow_order_id'   => $order->get_meta('_bzj_wow_order_id'),
        'amount'         => floatval($order->get_total()),
        'currency'       => $order->get_currency(),
        'status'         => $order->get_status(),
        'transaction_kind' => 'UNKNOWN',
        'line_items'     => [],
    ];
    
    foreach ($order->get_items() as $item) {
        $payment_data['line_items'][] = [
            'product_id'   => $item->get_product_id(),
            'quantity'     => $item->get_quantity(),
            'total'        => $item->get_total(),
        ];
    }
    
    return $payment_data;
}

/**
 * Get WordPress user ID for WoWonder user
 */
function bzj_pgb_get_wp_user_id($wow_user_id) {
    if (empty($wow_user_id)) return 0;
    
    // Load db_helpers
    $helpers = dirname(ABSPATH) . '/buzzjuice.net/shared/db_helpers.php';
    if (!file_exists($helpers)) return 0;
    
    require_once $helpers;
    
    $wow_db = get_wowonder_db();
    if (!$wow_db) return 0;
    
    $sql = "SELECT wp_user_id FROM Wo_Users WHERE user_id = ? LIMIT 1";
    $stmt = $wow_db->prepare($sql);
    if (!$stmt) return 0;
    
    $stmt->bind_param('i', $wow_user_id);
    $stmt->execute();
    $stmt->bind_result($wp_user_id);
    $found = $stmt->fetch();
    $stmt->close();
    
    return $found ? intval($wp_user_id) : 0;
}

/**
 * Create or link WordPress user for WoWonder user
 */
function bzj_pgb_create_or_link_user($data) {
    $wow_user_id = intval($data['wow_user_id'] ?? 0);
    $email = trim($data['email'] ?? '');
    $username = sanitize_user($data['username'] ?? 'user_' . $wow_user_id);
    
    if (!$wow_user_id || !$email) return 0;
    
    // Try to find existing user by email
    $user = get_user_by('email', $email);
    if ($user) {
        $wp_user_id = $user->ID;
        
        // Link to Streams
        $helpers = dirname(ABSPATH) . '/buzzjuice.net/shared/db_helpers.php';
        if (file_exists($helpers)) {
            require_once $helpers;
            $wow_db = get_wowonder_db();
            if ($wow_db) {
                $sql = "UPDATE Wo_Users SET wp_user_id = ? WHERE user_id = ? LIMIT 1";
                $stmt = $wow_db->prepare($sql);
                if ($stmt) {
                    $stmt->bind_param('ii', $wp_user_id, $wow_user_id);
                    $stmt->execute();
                    $stmt->close();
                }
            }
        }
        
        return $wp_user_id;
    }
    
    // Create new WordPress user
    $user_id = wp_create_user($username, wp_generate_password(), $email);
    if (is_wp_error($user_id)) {
        return 0;
    }
    
    // Link to Streams
    $helpers = dirname(ABSPATH) . '/buzzjuice.net/shared/db_helpers.php';
    if (file_exists($helpers)) {
        require_once $helpers;
        $wow_db = get_wowonder_db();
        if ($wow_db) {
            $sql = "UPDATE Wo_Users SET wp_user_id = ? WHERE user_id = ? LIMIT 1";
            $stmt = $wow_db->prepare($sql);
            if ($stmt) {
                $stmt->bind_param('ii', $user_id, $wow_user_id);
                $stmt->execute();
                $stmt->close();
            }
        }
    }
    
    return $user_id;
}

<?php
/**
 * buzzjuice.net/shared/payment-gateway/core/class-pgb-logger.php
 * 
 * Centralized logging with component-level files and redaction
 */

if (!defined('ABSPATH')) {
    // Allow outside WordPress
}

class PGB_Logger {
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_DEBUG;
    private $file_handles = [];
    private $sensitive_keys = [
        'api_key', 'api_secret', 'secret', 'password', 'token',
        'consumer_key', 'consumer_secret', 'webhook_secret',
        'credit_card', 'cvv', 'ssn', 'social_security',
    ];

    public function __construct($config = []) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = $config['logging_enabled'] ?? true;
        $this->min_level = $config['log_level'] ?? self::LOG_DEBUG;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Log a message
     */
    public function log($message, $level = self::LOG_INFO, $context = [], $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        $source = $source_file ? basename($source_file) : 'unknown';

        // Redact sensitive data from context
        $context = $this->redact_sensitive($context);

        // Format message
        $formatted_msg = $message;
        if (!empty($context)) {
            $formatted_msg .= ' | ' . json_encode($context);
        }

        $log_line = "[{$timestamp}] [{$level_upper}] [{$source}] {$formatted_msg}\n";

        // Write to component log file
        $this->write_to_file($source, $log_line);

        // Write critical errors to WordPress error log
        if (in_array($level, [self::LOG_ERROR, self::LOG_CRITICAL])) {
            error_log($log_line);
        }
    }

    /**
     * Redact sensitive data from arrays
     */
    private function redact_sensitive($data) {
        if (!is_array($data)) {
            return $data;
        }

        $redacted = [];
        foreach ($data as $key => $value) {
            if ($this->is_sensitive_key($key)) {
                $redacted[$key] = '***REDACTED***';
            } elseif (is_array($value)) {
                $redacted[$key] = $this->redact_sensitive($value);
            } else {
                $redacted[$key] = $value;
            }
        }
        return $redacted;
    }

    /**
     * Check if key is sensitive
     */
    private function is_sensitive_key($key) {
        $key_lower = strtolower($key);
        foreach ($this->sensitive_keys as $sensitive) {
            if (stripos($key_lower, $sensitive) !== false) {
                return true;
            }
        }
        return false;
    }

    /**
     * Check if message should be logged
     */
    private function should_log($level) {
        $levels = [
            self::LOG_DEBUG => 0,
            self::LOG_INFO => 1,
            self::LOG_WARNING => 2,
            self::LOG_ERROR => 3,
            self::LOG_CRITICAL => 4,
        ];
        return ($levels[$level] ?? 0) >= ($levels[$this->min_level] ?? 0);
    }

    /**
     * Write to file
     */
    private function write_to_file($source, $message) {
        $log_file = str_replace('.php', '', $source);
        $file_path = $this->log_dir . $log_file . '.log';

        // Rotate if > 10MB
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        // Open/create file handle
        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    /**
     * Cleanup
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}

    /**
     * Process payment completion
     */
    public function process_payment_completion($woo_order_id, $payment_data) {
        $this->logger->log(
            'Processing payment completion',
            PGB_Logger::LOG_INFO,
            ['order_id' => $woo_order_id],
            __FILE__
        );

        // Extract transaction data
        $transaction_kind = $payment_data['transaction_kind'] ?? 'UNKNOWN';
        $wow_user_id = intval($payment_data['wow_user_id'] ?? 0);
        $wow_order_id = $payment_data['wow_order_id'] ?? '';

        if (!$wow_user_id || !$wow_order_id) {
            $this->logger->log(
                'Missing Streams user or order ID',
                PGB_Logger::LOG_ERROR,
                $payment_data,
                __FILE__
            );
            return false;
        }

        // Fire before processing hook
        $this->hooks->do_action(
            PGB_Hooks::BEFORE_PAYMENT_PROCESS,
            $woo_order_id,
            $transaction_kind
        );

        // Handle by transaction type
        switch (strtoupper($transaction_kind)) {
            case 'PRO':
                $this->handle_pro_subscription($payment_data, $woo_order_id);
                break;
            case 'WALLET':
                $this->handle_wallet_topup($payment_data, $woo_order_id);
                break;
            case 'DONATE':
                $this->handle_donation($payment_data, $woo_order_id);
                break;
            case 'PRODUCT':
            case 'PURCHASE':
                $this->handle_product_purchase($payment_data, $woo_order_id);
                break;
        }

        // Fire after completion hook
        $this->hooks->do_action(
            PGB_Hooks::AFTER_PAYMENT_COMPLETE,
            $woo_order_id,
            $payment_data
        );

        // Fire addon hooks
        $this->hooks->do_action(
            PGB_Hooks::AFFILIATE_PROCESS,
            $wow_user_id,
            $woo_order_id,
            $payment_data['amount'] ?? 0
        );

        $this->hooks->do_action(
            PGB_Hooks::JEWEL_PROCESS,
            $wow_user_id,
            $woo_order_id,
            $payment_data['line_items'] ?? []
        );

        $this->logger->log(
            'Payment completion processed successfully',
            PGB_Logger::LOG_INFO,
            ['order_id' => $woo_order_id],
            __FILE__
        );

        return true;
    }

    /**
     * Handle PRO subscription
     */
    private function handle_pro_subscription($payment_data, $woo_order_id) {
        $wow_user_id = intval($payment_data['wow_user_id'] ?? 0);
        
        if (!$this->wow_db || !$wow_user_id) {
            return;
        }

        $now = time();
        $pro_type = $this->extract_pro_type($payment_data['line_items'] ?? []);

        if ($pro_type <= 0) {
            return;

        $sql = "UPDATE Wo_Users SET is_pro = '1', pro_type = ?, pro_time = ? WHERE user_id = ? LIMIT 1";

                'Pro subscription activated',
                PGB_Logger::LOG_INFO,
                ['user_id' => $wow_user_id, 'pro_type' => $pro_type],

            $this->hooks->do_action(
                PGB_Hooks::SUBSCRIPTION_ACTIVATED,
                $wow_user_id,
                $woo_order_id,
                ['pro_type' => $pro_type]
            );
        }

    private function handle_wallet_topup($payment_data, $woo_order_id) {

        $amount = floatval($payment_data['amount'] ?? 0);

        if (!$this->wow_db || !$wow_user_id || $amount <= 0) {
            return;

        // Update wallet
        $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ? LIMIT 1";

                'Wallet topup processed',
                PGB_Logger::LOG_INFO,
                ['user_id' => $wow_user_id, 'amount' => $amount],
                __FILE__
            );
        }
    }

     * Handle donation

        // Mark as processed in WoWonder if needed
        $wow_order_id = $payment_data['wow_order_id'] ?? '';
        
        if ($this->wow_db && $wow_order_id) {
            $status = 'paid';

            if ($stmt) {

/**
 * Validate Streams input
 */
function pgb_validate_streams_input($post_data, $wo) {
    $errors = [];
    $required = [
        'amount', 'product_name', 'product_price', 'product_units',
        'product_id', 'transaction_kind', 'wow_currency_code'
    ];
    
    foreach ($required as $field) {
        if (empty($post_data[$field])) {
            $errors[] = "Missing field: {$field}";
        }
    }
    
    $amount = floatval($post_data['amount'] ?? 0);
    if ($amount <= 0) {
        $errors[] = "Amount must be > 0";
    }
    
    $units = intval($post_data['product_units'] ?? 0);
    if ($units <= 0) {
        $errors[] = "Units must be > 0";
    }
    
    $kind = trim($post_data['transaction_kind'] ?? '');
    if (!in_array(strtoupper($kind), ['PRODUCT', 'PRO', 'WALLET', 'DONATE'])) {
        $errors[] = "Invalid transaction kind";
    }
    
    if (!empty($errors)) {
        return [
            'valid'   => false,
            'errors'  => $errors,
            'message' => implode('; ', $errors),
        ];
    }
    
    return [
        'valid' => true,
        'data'  => [
            'amount'             => $amount,
            'product_name'       => sanitize_text_field($post_data['product_name']),
            'product_price'      => floatval($post_data['product_price']),
            'product_units'      => $units,
            'product_id'         => intval($post_data['product_id']),
            'transaction_kind'   => strtoupper($kind),
            'wow_currency_code'  => trim($post_data['wow_currency_code']),
            'wow_post_id'        => intval($post_data['wow_post_id'] ?? 0),
            'address_id'         => intval($post_data['address_id'] ?? 0),
        ],
    ];
}

/**
 * Capture AffiliateWP context snapshot
 */
function pgb_capture_affwp_snapshot($wp_user_id, $email) {
    $snapshot = [
        'wp_user_id'   => $wp_user_id,
        'email'        => $email,
        'timestamp'    => gmdate('c'),
        'http_referer' => $_SERVER['HTTP_REFERER'] ?? '',
    ];
    
    $cookie_keys = [
        'affwp_ref', 'affwp_visit', 'affwp_visit_id',
        'affwp_affiliate_id', 'affwp_campaign', 'affwp_coupon'
    ];
    
    foreach ($cookie_keys as $key) {
        if (!empty($_COOKIE[$key])) {
            $snapshot[$key] = (string)$_COOKIE[$key];
        }
    }
    
    return $snapshot;
}

/**
 * Send cURL request
 */
function pgb_send_curl_request($url, $method = 'GET', $data = null, $headers = []) {
    $curl = curl_init();
    
    $options = [
        CURLOPT_URL            => $url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_CUSTOMREQUEST  => strtoupper($method),
        CURLOPT_HTTPHEADER     => $headers,
        CURLOPT_TIMEOUT        => 30,
        CURLOPT_SSL_VERIFYHOST => 0,
        CURLOPT_SSL_VERIFYPEER => 0,
    ];
    
    if ($data) {
        $options[CURLOPT_POSTFIELDS] = $data;
    }
    
    curl_setopt_array($curl, $options);
    
    $response = curl_exec($curl);
    $error = curl_error($curl);
    
    curl_close($curl);
    
    if ($error) {
        error_log('[PGB-Streams] cURL error: ' . $error);
        return null;
    }
    
    return $response;
}

<?php
/**
 * buzzjuice.net/shared/payment-gateway/addons/pgb-addon-affiliate-wp.php
 * 
 * AffiliateWP integration addon
 * Processes referrals and commissions after payment
 */

if (!defined('ABSPATH')) {
    // Allow outside WordPress

if (function_exists('add_action')) {
    add_action('pgb_affiliate_process', 'pgb_addon_process_affiliatewp', 10, 3);

/**
 * Process AffiliateWP rewards
 * 
 * @param int $wow_user_id WoWonder user ID
 * @param int $woo_order_id WooCommerce order ID
 * @param float $amount Order amount
 */
function pgb_addon_process_affiliatewp($wow_user_id, $woo_order_id, $amount) {
    if (!class_exists('Affiliate_WP')) {
        return;
    }

    $order = wc_get_order($woo_order_id);
    if (!$order) {
        return;
    }

    error_log("[PGB-AffiliateWP] Processing commission for order {$woo_order_id}, user {$wow_user_id}");

    // Get snapshot from order meta
    $snapshot = $order->get_meta('_bzj_affwp_snapshot');
    if (!empty($snapshot) && is_string($snapshot)) {
        $snapshot = json_decode($snapshot, true);
    } else {
        $snapshot = [];
    }

    // Attempt to get affiliate ID from snapshot
    $affiliate_id = intval($snapshot['affwp_affiliate_id'] ?? 0);

    // If no affiliate ID from snapshot, try customer resolution
    if ($affiliate_id <= 0) {
        $customer_email = $order->get_billing_email();
        if ($customer_email) {
            $affiliate_id = affwp_get_affiliate_id_by_user_email($customer_email);
        }
    }

    if ($affiliate_id > 0) {
        // Check if referral already exists (idempotency)
        $existing = affwp_get_referral([
            'reference'      => $woo_order_id,
            'affiliate_id'   => $affiliate_id,
            'status'         => 'paid',
        ]);

        if ($existing) {
            error_log("[PGB-AffiliateWP] Referral already exists for order {$woo_order_id}");
            $order->update_meta_data('_bzj_affiliate_processed', true);
            $order->save();
            return;
        }

        // Create referral
        $referral_id = affwp_add_referral([
            'affiliate_id' => $affiliate_id,
            'reference'    => $woo_order_id,
            'amount'       => $amount,
            'status'       => 'paid',
        ]);

        if ($referral_id) {
            error_log("[PGB-AffiliateWP] Referral created: {$referral_id} for order {$woo_order_id}");
            $order->update_meta_data('_bzj_affiliate_processed', true);
            $order->update_meta_data('_bzj_affiliate_id', $affiliate_id);
            $order->update_meta_data('_bzj_referral_id', $referral_id);
            $order->save();
        }
    }
}

/**
 * Get affiliate ID by customer email
 */
function affwp_get_affiliate_id_by_user_email($email) {
    $affiliate_data = affwp_get_affiliate_by('user_email', $email);
    if ($affiliate_data) {
        return intval($affiliate_data->affiliate_id);
    }
    return 0;
}

/**
 * Process Jewel Affiliate rewards
 * 
 * @param int $wow_user_id WoWonder user ID
 * @param int $woo_order_id WooCommerce order ID
 * @param array $line_items Order line items
 */
function pgb_int_process_jewel($wow_user_id, $woo_order_id, $line_items) {
    if (!$wow_user_id || !$woo_order_id) {
        return;
    }

    error_log("[PGB-Jewel] Processing Jewel order {$woo_order_id} for user {$wow_user_id}");

    // Load WoWonder DB
    $helpers = dirname(__FILE__, 2) . '/../db_helpers.php';
    if (!file_exists($helpers)) {
        error_log("[PGB-Jewel] db_helpers not found");
        return;
    }

    require_once $helpers;
    $wow_db = get_wowonder_db();
    if (!$wow_db) {
        error_log("[PGB-Jewel] WoWonder DB connection failed");
        return;
    }

    // Load mapping from WP options
    $mapping = pgb_jewel_load_mapping();
    if (empty($mapping)) {
        error_log("[PGB-Jewel] No mapping configured");
        return;
    }

    // Process line items
    foreach ($line_items as $item) {
        $product_id = intval($item['product_id'] ?? 0);
        $variation_id = intval($item['variation_id'] ?? $product_id);

        if (!isset($mapping[$variation_id])) {
            continue;
        }

        $map = $mapping[$variation_id];
        $pro_type = intval($map['wow_pro_type'] ?? 0);
        $rebate_cents = intval($map['rebate_cents'] ?? 0);
        $wp_role = trim($map['wp_role'] ?? '');

        // Activate Pro
        if ($pro_type > 0) {
            $now = time();
            $sql = "UPDATE Wo_Users SET is_pro = '1', pro_type = ?, pro_time = ? WHERE user_id = ? LIMIT 1";
            $stmt = $wow_db->prepare($sql);
            if ($stmt) {
                $stmt->bind_param('iii', $pro_type, $now, $wow_user_id);
                $stmt->execute();
                $stmt->close();
                error_log("[PGB-Jewel] Activated pro_type {$pro_type} for user {$wow_user_id}");
            }
        }

        // Apply rebate
        if ($rebate_cents > 0) {
            pgb_jewel_apply_rebate($wow_db, $wow_user_id, $rebate_cents, $woo_order_id);
        }

        // Sync WP roles
        if (!empty($wp_role)) {
            pgb_jewel_sync_wp_roles($wow_user_id, $wp_role);
        }
    }
}

/**
 * Load Jewel mapping from WP options
 */
function pgb_jewel_load_mapping() {
    $helpers = dirname(__FILE__, 2) . '/../db_helpers.php';
    if (!file_exists($helpers)) {
        return [];
    }

    require_once $helpers;
    $wpdb = get_wp_db_conn();
    if (!$wpdb) {
        return [];
    }

    $option_name = 'bzj_jewel_mapping';
    $prefix = defined('WP_TABLE_PREFIX') ? WP_TABLE_PREFIX : 'wp_';
    $table = $prefix . 'options';

    $sql = "SELECT option_value FROM `{$table}` WHERE option_name = ? LIMIT 1";
    $stmt = $wpdb->prepare($sql);
    if (!$stmt) {
        return [];
    }

    $stmt->bind_param('s', $option_name);
    if (!$stmt->execute()) {
        $stmt->close();
        return [];
    }

    $stmt->bind_result($json);
    $stmt->fetch();
    $stmt->close();

    if (empty($json)) {
        return [];
    }

    $mapping = json_decode($json, true);
    return is_array($mapping) ? $mapping : [];
}

/**
 * Apply rebate credit (idempotent)
 */
function pgb_jewel_apply_rebate($wow_db, $wow_user_id, $rebate_cents, $woo_order_id) {
    $amount = round($rebate_cents / 100.0, 2);

    // Check if already applied
    $sql = "SELECT id FROM Wo_Payment_Transactions WHERE userid = ? AND notes LIKE ? LIMIT 1";
    $stmt = $wow_db->prepare($sql);
    if ($stmt) {
        $like = '%Jewel Rebate%' . $woo_order_id . '%';
        $stmt->bind_param('is', $wow_user_id, $like);
        $stmt->execute();
        if ($stmt->num_rows > 0) {
            $stmt->close();
            error_log("[PGB-Jewel] Rebate already applied for order {$woo_order_id}");
            return;
        }
        $stmt->close();
    }

    // Begin transaction
    try {
        $wow_db->begin_transaction();

        // Update wallet
        $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ? LIMIT 1";
        $stmt = $wow_db->prepare($sql);
        if (!$stmt) throw new Exception("Prepare update wallet failed");
        $stmt->bind_param('di', $amount, $wow_user_id);
        if (!$stmt->execute()) throw new Exception("Execute update wallet failed");
        $stmt->close();

        // Insert transaction record
        $note = "Jewel Rebate - Order {$woo_order_id}";
        $kind = 'JEWEL_REBATE';
        $status = 'completed';
        $sql = "INSERT INTO Wo_Payment_Transactions (userid, kind, amount, notes, payment_status, woo_order_id, transaction_dt) 
                VALUES (?, ?, ?, ?, ?, ?, NOW())";
        $stmt = $wow_db->prepare($sql);
        if (!$stmt) throw new Exception("Prepare insert transaction failed");
        $stmt->bind_param('isdssi', $wow_user_id, $kind, $amount, $note, $status, $woo_order_id);
        if (!$stmt->execute()) throw new Exception("Execute insert transaction failed");
        $stmt->close();

        $wow_db->commit();
        error_log("[PGB-Jewel] Rebate {$amount} applied for user {$wow_user_id}");
    } catch (Exception $e) {
        $wow_db->rollback();
        error_log("[PGB-Jewel] Rebate failed: " . $e->getMessage());
    }
}

/**
 * Sync WordPress roles
 */
function pgb_jewel_sync_wp_roles($wow_user_id, $wp_role) {
    $helpers = dirname(__FILE__, 2) . '/../db_helpers.php';
    if (!file_exists($helpers)) {
        return;
    }

    require_once $helpers;

    // Get WP user ID
    $wow_db = get_wowonder_db();
    if (!$wow_db) {
        return;
    }

    $sql = "SELECT wp_user_id FROM Wo_Users WHERE user_id = ? LIMIT 1";
    $stmt = $wow_db->prepare($sql);
    if (!$stmt) return;

    $stmt->bind_param('i', $wow_user_id);
    $stmt->execute();
    $stmt->bind_result($wp_user_id);
    if (!$stmt->fetch()) {
        $stmt->close();
        return;
    }
    $stmt->close();

    if (!$wp_user_id || !function_exists('get_user_by')) {
        return;
    }

    $user = get_user_by('ID', $wp_user_id);
    if (!$user) {
        return;
    }

    // Add role
    if (function_exists('wp_get_current_user') && function_exists('$user->add_role')) {
        $user->add_role($wp_role);
        error_log("[PGB-Jewel] Added role '{$wp_role}' for WP user {$wp_user_id}");
    }
}
