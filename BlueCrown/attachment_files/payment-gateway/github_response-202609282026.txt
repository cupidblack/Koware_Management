### **FILE 2: core/class-pgb-fulfillment.php (THE CRITICAL FILE - FIXES YOUR ISSUE)**

```php
<?php
/**
 * Payment Gateway Bridge - Fulfillment Engine
 * 
 * Processes all post-payment operations:
 * - Subscriptions activation
 * - Affiliate commission calculation
 * - Jewel rebate processing
 * - Streams database synchronization
 * - Role assignment
 * 
 * Location: buzzjuice.net/shared/payment-gateway/core/class-pgb-fulfillment.php
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Fulfillment {
    private $logger = null;
    private $wow_db = null;
    private $wp_db = null;

    public function __construct($logger) {
        $this->logger = $logger;
        $this->init_databases();
    }

    /**
     * Initialize database connections
     */
    private function init_databases() {
        if (!function_exists('get_wowonder_db')) {
            $helpers_file = dirname(__FILE__, 3) . '/db_helpers.php';
            if (file_exists($helpers_file)) {
                require_once $helpers_file;
            }
        }

        if (function_exists('get_wowonder_db')) {
            $this->wow_db = get_wowonder_db();
        }
        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * MAIN ENTRY POINT: Process complete order fulfillment
     * Called when thank you page loads and order is completed
     */
    public function process_order_fulfillment($order) {
        $order_id = $order->get_id();
        
        $this->logger->log(
            'Starting fulfillment for order',
            PGB_Logger::LOG_INFO,
            ['order_id' => $order_id],
            __FILE__
        );

        try {
            // 1. Extract metadata
            $order_meta = $this->extract_order_metadata($order);
            if (!$order_meta) {
                $this->logger->log(
                    'Could not extract order metadata',
                    PGB_Logger::LOG_WARNING,
                    ['order_id' => $order_id],
                    __FILE__
                );
                return;
            }

            $wow_user_id = $order_meta['userid'] ?? null;
            $wow_order_id = $order_meta['wow_order_id'] ?? null;
            $transaction_kind = $order_meta['transaction_kind'] ?? 'UNKNOWN';

            if (!$wow_user_id || !$wow_order_id) {
                $this->logger->log(
                    'Missing critical metadata',
                    PGB_Logger::LOG_WARNING,
                    ['order_id' => $order_id, 'metadata' => $order_meta],
                    __FILE__
                );
                return;
            }

            // 2. Check idempotency (prevent double processing)
            if ($this->order_already_fulfilled($wow_order_id)) {
                $this->logger->log(
                    'Order already fulfilled',
                    PGB_Logger::LOG_DEBUG,
                    ['wow_order_id' => $wow_order_id],
                    __FILE__
                );
                return;
            }

            // 3. Get Streams user data
            $streams_user = $this->get_streams_user($wow_user_id);
            if (!$streams_user) {
                $this->logger->log(
                    'Streams user not found',
                    PGB_Logger::LOG_ERROR,
                    ['wow_user_id' => $wow_user_id],
                    __FILE__
                );
                return;
            }

            // 4. Handle transaction type
            $this->process_transaction_kind($transaction_kind, $order, $order_meta, $streams_user);

            // 5. Process subscriptions (if any)
            $this->process_subscriptions($order, $wow_user_id);

            // 6. Process affiliate rewards
            $this->process_affiliate_rewards($order, $streams_user);

            // 7. Process Jewel rebates
            $this->process_jewel_rebates($order, $streams_user);

            // 8. Update Streams fulfillment status
            $this->mark_order_fulfilled($wow_order_id, $order_id);

            $this->logger->log(
                'Order fulfillment completed successfully',
                PGB_Logger::LOG_INFO,
                ['order_id' => $order_id, 'wow_order_id' => $wow_order_id],
                __FILE__
            );

        } catch (Exception $e) {
            $this->logger->log(
                'Fulfillment error: ' . $e->getMessage(),
                PGB_Logger::LOG_CRITICAL,
                ['trace' => $e->getTraceAsString()],
                __FILE__
            );
        }
    }

    /**
     * Extract metadata from WooCommerce order
     */
    private function extract_order_metadata($order) {
        $metadata = [];
        
        foreach ($order->get_meta_data() as $meta) {
            $metadata[$meta->key] = $meta->value;
        }

        // For PRO packages, detect transaction kind from variation
        if (empty($metadata['transaction_kind'])) {
            $items = $order->get_items();
            if (!empty($items)) {
                $item = reset($items);
                $product_id = $item->get_product_id();
                $variation_id = $item->get_variation_id();
                
                // Check if it's a PRO package
                $pro_ids = [
                    get_option('pgb_wow_pro_package_id'),
                    get_option('pgb_wow_pro_package_id_2'),
                    get_option('pgb_wow_pro_package_id_3'),
                    get_option('pgb_wow_pro_package_id_4'),
                ];
                
                if (in_array($variation_id ?: $product_id, $pro_ids)) {
                    $metadata['transaction_kind'] = 'PRO';
                }
            }
        }

        return $metadata;
    }

    /**
     * Check if order already fulfilled (idempotency)
     */
    private function order_already_fulfilled($wow_order_id) {
        if (!$this->wow_db) return false;

        $sql = "SELECT fulfillment_status FROM Wo_Payment_Transactions 
                WHERE order_id = ? LIMIT 1";
        $stmt = $this->wow_db->prepare($sql);
        if (!$stmt) return false;

        $stmt->bind_param('s', $wow_order_id);
        $stmt->execute();
        $result = $stmt->get_result();
        
        if ($row = $result->fetch_assoc()) {
            return $row['fulfillment_status'] === 'fulfilled';
        }
        return false;
    }

    /**
     * Get Streams user data
     */
    private function get_streams_user($user_id) {
        if (!$this->wow_db) return null;

        $sql = "SELECT * FROM Wo_Users WHERE user_id = ? LIMIT 1";
        $stmt = $this->wow_db->prepare($sql);
        if (!$stmt) return null;

        $stmt->bind_param('i', $user_id);
        $stmt->execute();
        return $stmt->get_result()->fetch_assoc();
    }

    /**
     * Process based on transaction kind
     */
    private function process_transaction_kind($kind, $order, $metadata, $user) {
        switch ($kind) {
            case 'PRO':
                $this->activate_pro_membership($order, $metadata, $user);
                break;

            case 'WALLET':
                $this->add_wallet_topup($order, $metadata, $user);
                break;

            case 'DONATE':
            case 'CROWDFUND':
                $this->process_crowdfund($order, $metadata, $user);
                break;

            case 'PURCHASE':
            case 'PRODUCT':
                $this->process_product_purchase($order, $metadata, $user);
                break;

            default:
                $this->logger->log(
                    'Unknown transaction kind: ' . $kind,
                    PGB_Logger::LOG_WARNING,
                    [],
                    __FILE__
                );
        }
    }

    /**
     * Activate PRO membership in Streams
     */
    private function activate_pro_membership($order, $metadata, $user) {
        if (!$this->wow_db) return;

        $user_id = $user['user_id'];
        $pro_type = $metadata['wow_post_id'] ?? 1; // Default to Classic
        $pro_time = time();
        $is_pro = 1;

        $sql = "UPDATE Wo_Users SET pro_time = ?, is_pro = ?, pro_type = ? WHERE user_id = ?";
        $stmt = $this->wow_db->prepare($sql);
        
        if ($stmt) {
            $stmt->bind_param('isii', $pro_time, $is_pro, $pro_type, $user_id);
            $stmt->execute();

            $this->logger->log(
                'Pro membership activated',
                PGB_Logger::LOG_INFO,
                ['user_id' => $user_id, 'pro_type' => $pro_type],
                __FILE__
            );
        }
    }

    /**
     * Process wallet topup
     */
    private function add_wallet_topup($order, $metadata, $user) {
        if (!$this->wow_db) return;

        $user_id = $user['user_id'];
        $amount = floatval($order->get_total());

        $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ?";
        $stmt = $this->wow_db->prepare($sql);
        
        if ($stmt) {
            $stmt->bind_param('di', $amount, $user_id);
            $stmt->execute();

            $this->logger->log(
                'Wallet topped up',
                PGB_Logger::LOG_INFO,
                ['user_id' => $user_id, 'amount' => $amount],
                __FILE__
            );
        }
    }

    /**
     * Process crowdfunding contribution
     */
    private function process_crowdfund($order, $metadata, $user) {
        if (!$this->wow_db) return;

        $user_id = $user['user_id'];
        $funding_id = $metadata['wow_post_id'] ?? 0;
        $amount = floatval($order->get_total());
        $time = time();

        if ($funding_id > 0) {
            $sql = "INSERT INTO Wo_Funding_Raise (funding_id, user_id, amount, time) 
                    VALUES (?, ?, ?, ?)";
            $stmt = $this->wow_db->prepare($sql);
            
            if ($stmt) {
                $stmt->bind_param('iidi', $funding_id, $user_id, $amount, $time);
                $stmt->execute();

                $this->logger->log(
                    'Crowdfund contribution recorded',
                    PGB_Logger::LOG_INFO,
                    ['user_id' => $user_id, 'funding_id' => $funding_id, 'amount' => $amount],
                    __FILE__
                );
            }
        }
    }

    /**
     * Process product purchase
     */
    private function process_product_purchase($order, $metadata, $user) {
        if (!$this->wow_db) return;

        // Your existing product purchase logic
        $this->logger->log(
            'Product purchase processed',
            PGB_Logger::LOG_DEBUG,
            ['user_id' => $user['user_id']],
            __FILE__
        );
    }

    /**
     * Process WooCommerce subscriptions
     */
    private function process_subscriptions($order, $wow_user_id) {
        if (!function_exists('wcs_order_contains_subscription')) {
            return; // WC Subscriptions not active
        }

        if (!wcs_order_contains_subscription($order->get_id())) {
            return; // No subscriptions in order
        }

        try {
            $subscriptions = wcs_get_subscriptions_for_order($order->get_id(), ['order_type' => 'parent']);
            
            foreach ($subscriptions as $subscription) {
                $this->logger->log(
                    'Subscription found and activated',
                    PGB_Logger::LOG_INFO,
                    [
                        'subscription_id' => $subscription->get_id(),
                        'wow_user_id' => $wow_user_id
                    ],
                    __FILE__
                );

                // Subscription is auto-active via WooCommerce, just log
                do_action('pgb_subscription_activated', $wow_user_id, $subscription);
            }

        } catch (Exception $e) {
            $this->logger->log(
                'Error processing subscriptions: ' . $e->getMessage(),
                PGB_Logger::LOG_ERROR,
                [],
                __FILE__
            );
        }
    }

    /**
     * Process AffiliateWP commission
     */
    private function process_affiliate_rewards($order, $streams_user) {
        if (!class_exists('Affiliate_WP')) {
            return; // AffiliateWP not active
        }

        $this->logger->log(
            'Processing affiliate rewards',
            PGB_Logger::LOG_INFO,
            ['user_id' => $streams_user['user_id']],
            __FILE__
        );

        // Check if user is linked to affiliate
        $customer_email = $order->get_billing_email();
        $affiliate = affwp_get_affiliate_by('email', $customer_email);
        
        if (!$affiliate) {
            $this->logger->log(
                'No affiliate found for customer',
                PGB_Logger::LOG_DEBUG,
                ['email' => $customer_email],
                __FILE__
            );
            return;
        }

        // Create referral if not exists
        $order_id = $order->get_id();
        $existing_referral = affiliate_wp()->referrals->get_by('reference', $order_id, 'woocommerce_product_purchase');
        
        if (!$existing_referral) {
            $amount = floatval($order->get_total());
            
            $referral_data = [
                'affiliate_id' => $affiliate->ID,
                'amount' => $amount,
                'reference' => $order_id,
                'description' => 'Order #' . $order_id . ' from Streams',
                'type' => 'woocommerce_product_purchase',
                'status' => 'pending',
                'date' => gmdate('Y-m-d H:i:s')
            ];

            $referral_id = affiliate_wp()->referrals->add($referral_data);

            $this->logger->log(
                'Affiliate referral created',
                PGB_Logger::LOG_INFO,
                ['referral_id' => $referral_id, 'affiliate_id' => $affiliate->ID],
                __FILE__
            );

            // Update referral status if immediate payment
            if (get_option('pgb_affiliate_rebate_enabled')) {
                affiliate_wp()->referrals->update($referral_id, ['status' => 'approved']);
            }
        }
    }
```

```php
        ]
    ];

    $order_data = [
        'customer_id' => $wp_user_id,
        'set_paid' => false,
        'currency' => $input['wow_currency_code'],

        'billing' => [

            'email' => $email,
            'phone' => $wo['user']['phone_number'] ?? '',

            ['key' => 'transaction_kind', 'value' => $input['transaction_kind']],
            ['key' => '_buzzjuice_origin', 'value' => 'streams'],
        ]
    ];

    if ($input['transaction_kind'] === 'PRO') {

        if (!empty($wo['pro_packages'][$pro_id])) {
            $package = $wo['pro_packages'][$pro_id];
            $order_data['meta_data'][] = ['key' => '_subscription_period', 'value' => $package['time'] ?? 'month'];
            $order_data['meta_data'][] = ['key' => '_subscription_interval', 'value' => (int)($package['count'] ?? 1)];
        }
    }

    return $order_data;

/**
 * Create WooCommerce order via REST API
 */
function pgb_create_wc_order($order_data) {
    global $wo;

    $api_url = rtrim($wo['config']['wow_api_url'] ?? '', '/') . '/wc/v3/orders';
    $api_key = $wo['config']['wow_api_key'] ?? '';
    $api_secret = $wo['config']['wow_api_secret'] ?? '';

    if (empty($api_url) || empty($api_key) || empty($api_secret)) {
        return [
            'success' => false,
            'error' => 'WooCommerce API credentials not configured'
        ];
    }

    $curl = curl_init();
    curl_setopt_array($curl, [
        CURLOPT_URL => $api_url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
        CURLOPT_CUSTOMREQUEST => 'POST',
        CURLOPT_POSTFIELDS => json_encode($order_data),
        CURLOPT_USERPWD => "$api_key:$api_secret",
        CURLOPT_SSL_VERIFYHOST => 0,
        CURLOPT_SSL_VERIFYPEER => 0,
    ]);

    $response = curl_exec($curl);
    $http_code = curl_getinfo($curl, CURLINFO_HTTP_CODE);
    $curl_error = curl_error($curl);
    curl_close($curl);

    if ($curl_error) {
        return [
            'success' => false,
            'error' => $curl_error
        ];
    }

    if ($http_code !== 201) {
        return [
            'success' => false,
            'error' => "HTTP $http_code: " . $response
        ];
    }

    $order_response = json_decode($response, true);

    if (empty($order_response['id']) || empty($order_response['checkout_payment_url'])) {
        return [
            'success' => false,
            'error' => 'Invalid WooCommerce response'
        ];
    }

    return [
        'success' => true,
        'woo_order_id' => $order_response['id'],
        'checkout_url' => $order_response['checkout_payment_url']
    ];
}
```

```php
class PGB_Logger {
    const LOG_DEBUG = 'DEBUG';
    const LOG_INFO = 'INFO';
    const LOG_WARNING = 'WARNING';
    const LOG_ERROR = 'ERROR';
    const LOG_CRITICAL = 'CRITICAL';

    private $log_dir = '';
    private $enabled = true;
    private $level_map = [
        'DEBUG' => 0,
        'INFO' => 1,
        'WARNING' => 2,
        'ERROR' => 3,
        'CRITICAL' => 4,
    ];
    private $current_level = 0;
    private $file_handles = [];

    public function __construct($config = []) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = $config['logging_enabled'] ?? true;
        $current_level_name = $config['log_level'] ?? 'DEBUG';
        $this->current_level = $this->level_map[$current_level_name] ?? 0;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Main logging method
     */
    public function log($message, $level = 'INFO', $context = [], $source_file = null) {
        if (!$this->enabled) return;

        $level_value = $this->level_map[$level] ?? 0;
        if ($level_value < $this->current_level) return;

        $timestamp = date('Y-m-d H:i:s');
        $source = $source_file ? basename($source_file, '.php') : 'unknown';

        // Format message
        $log_entry = "[$timestamp] [$level] [$source] $message";
        if (!empty($context)) {
            $log_entry .= " | Context: " . json_encode($context);
        }

        $this->write_to_file($source, $log_entry);

        // Also log errors to WordPress error log
        if (in_array($level, ['ERROR', 'CRITICAL']) && function_exists('error_log')) {
            error_log($log_entry);
        }
    }

    /**
     * Write to component-specific log file
     */
    private function write_to_file($component, $entry) {
        $file_path = $this->log_dir . $component . '.log';

        // Rotate log if too large (10MB)
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $entry . "\n");
            fflush($this->file_handles[$file_path]);
        }
    }

    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                @fclose($handle);
            }
        }
    }
}
```

```php
/**
 * Process affiliate commission on payment completion
 */
add_action('pgb_thankyou_page_loaded', function($order_id) {
    if (!class_exists('Affiliate_WP')) {
        return; // AffiliateWP not active
    }

    $logger = PGB_Bootstrap::instance()->get_logger();
    $order = wc_get_order($order_id);

    if (!$order) {
        return;
    }

    $logger->log(
        'Processing AffiliateWP commission',
        PGB_Logger::LOG_DEBUG,
        ['order_id' => $order_id],
        __FILE__
    );

    try {
        // Get customer info
        $customer_email = $order->get_billing_email();
        $customer_id = $order->get_customer_id();

        // Step 1: Try to find affiliate by email
        $affiliate = affwp_get_affiliate_by('email', $customer_email);
        
        // Step 2: If no affiliate, check metadata snapshot from Streams
        if (!$affiliate) {
            $affwp_snapshot = $order->get_meta('_buzzjuice_affwp_context_snapshot');
            if (!empty($affwp_snapshot)) {
                $snapshot = json_decode($affwp_snapshot, true);
                if (!empty($snapshot['affwp_affiliate_id'])) {
                    $affiliate = affwp_get_affiliate($snapshot['affwp_affiliate_id']);
                }
            }
        }

        if (!$affiliate) {
            $logger->log(
                'No affiliate found for order',
                PGB_Logger::LOG_DEBUG,
                ['customer_email' => $customer_email],
                __FILE__
            );
            return;
        }

        // Step 3: Check if referral already exists (idempotency)
        $existing_referral = affiliate_wp()->referrals->get_by(
            'reference',
            $order_id,
            'woocommerce_product_purchase'
        );

        if ($existing_referral) {
            $logger->log(
                'Referral already exists',
                PGB_Logger::LOG_DEBUG,
                ['referral_id' => $existing_referral->ID],
                __FILE__
            );
            return;
        }

        // Step 4: Calculate commission
        $order_total = floatval($order->get_total());
        
        // Get affiliate commission rate (default 20%)
        $commission_rate = (float) affiliate_wp()->settings->get('referral_rate', 0.2);
        $commission_amount = $order_total * $commission_rate;

        // Step 5: Create referral
        $referral_id = affiliate_wp()->referrals->add([
            'affiliate_id' => $affiliate->ID,
            'reference' => $order_id,
            'description' => 'Order #' . $order_id . ' from Streams',
            'amount' => $commission_amount,
            'type' => 'woocommerce_product_purchase',
            'status' => 'pending',
            'date' => gmdate('Y-m-d H:i:s')
        ]);

        if (!$referral_id) {
            $logger->log(
                'Failed to create referral',
                PGB_Logger::LOG_ERROR,
                ['affiliate_id' => $affiliate->ID, 'order_id' => $order_id],
                __FILE__
            );
            return;
        }

        $logger->log(
            'Affiliate referral created',
            PGB_Logger::LOG_INFO,
            [
                'referral_id' => $referral_id,
                'affiliate_id' => $affiliate->ID,
                'commission' => $commission_amount
            ],
            __FILE__
        );

        // Step 6: Update order meta
        $order->update_meta_data('_bzj_affwp_referral_id', $referral_id);
        $order->update_meta_data('_bzj_affwp_affiliate_id', $affiliate->ID);
        $order->save();

        // Step 7: Process lifetime commissions (if plugin is active)
        if (class_exists('Affiliate_WP_Lifetime_Commissions')) {
            affwp_lifetime_commissions()->process_commission($referral_id, $affiliate->ID);
            
            $logger->log(
                'Lifetime commissions processed',
                PGB_Logger::LOG_DEBUG,
                ['referral_id' => $referral_id],
                __FILE__
            );
        }

    } catch (Exception $e) {
        $logger->log(
            'AffiliateWP processing error: ' . $e->getMessage(),
            PGB_Logger::LOG_ERROR,
            ['trace' => $e->getTraceAsString()],
            __FILE__
        );
    }

}, 15);
```
