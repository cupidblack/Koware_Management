### **Key Architectural Decisions**

| Decision | Rationale | Implementation |
|----------|-----------|-----------------|
| **No cURL REST calls from Streams** | API rate limits + domain resolution failures | Database pending orders table |
| **WooCommerce order creation in WordPress** | Uses native WC_Order API + generates correct checkout URL | `wc_create_order()` in manager |
| **Payment confirmation = WooCommerce payment_complete** | Reliable, hooks-based, prevents unpaid orders from fulfilling | Listens to `woocommerce_payment_complete` |
| **Streams updates AFTER payment** | Prevents dropped orders from being fulfillment records prematurely | Addon runs post-payment hook |
| **Signed handoff validation** | Prevents order tampering between Streams and WordPress | HMAC-SHA256 signature verification |
| **Per-component logging** | Easier debugging and audit trails | Separate `.log` files per component |
| **Addons load dynamically** | Easy to enable/disable features without code changes | Glob pattern scanning in mu-plugin |

```php
/**
 * Handle checkout creation via REST endpoint
 */
function bzj_pgb_checkout_handler($request) {
    $manager = PGB_Manager::instance();
    $logger = PGB_Logger::instance();
    
    try {
        $params = $request->get_json_params();
        $wow_order_id = sanitize_text_field($params['wow_order_id'] ?? '');
        
        if (empty($wow_order_id)) {
            return new \WP_REST_Response([
                'success' => false,
                'message' => 'Invalid order ID',
            ], 400);
        }
        
        // Retrieve pending order from database
        $pending = bzj_pgb_get_pending_order($wow_order_id);
        if (!$pending) {
            $logger->log(
                'Pending order not found: ' . $wow_order_id,
                PGB_Logger::LOG_WARNING,
                [],
                'rest-checkout'
            );
            return new \WP_REST_Response([
                'success' => false,
                'message' => 'Order not found',
            ], 404);
        }
        
        // Process pending order
        $result = $manager->process_pending_order($pending);
        
        if ($result['success']) {
            return new \WP_REST_Response([
                'success' => true,
                'checkout_url' => $result['checkout_url'],
                'order_id' => $result['order_id'] ?? 0,
            ], 200);
        } else {
            return new \WP_REST_Response([
                'success' => false,
                'message' => $result['error'] ?? 'Failed to process order',
            ], 500);
        }
    } catch (Exception $e) {
        $logger = PGB_Logger::instance();
        $logger->log(
            'REST endpoint exception: ' . $e->getMessage(),
            PGB_Logger::LOG_CRITICAL,
            ['trace' => $e->getTraceAsString()],
            'rest-checkout'
        );
        return new \WP_REST_Response([
            'success' => false,
            'message' => 'Internal server error',
        ], 500);
    }
}

        ARRAY_A
    );
```

```php
<?php
/**
 * Payment Gateway Bridge - Logger
 * Component-based logging with sensitivity redaction
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    // Log levels
    const LOG_OFF      = 0;
    const LOG_CRITICAL = 1;
    const LOG_ERROR    = 2;
    const LOG_WARNING  = 3;
    const LOG_INFO     = 4;
    const LOG_DEBUG    = 5;

    private static $instance = null;
    private $log_dir = '';
    private $enabled = true;
    private $level = self::LOG_INFO;
    private $file_handles = [];

    /**
     * Get singleton instance
     */
    public static function instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    /**
     * Private constructor (singleton)
     */
    private function __construct() {
        $this->log_dir = defined('BZJ_PGB_LOG_DIR') ? BZJ_PGB_LOG_DIR : sys_get_temp_dir() . '/bzj-pgb-log';
        $this->enabled = get_option('pgb_logging_enabled', true);
        $this->level = $this->level_from_string(get_option('pgb_log_level', 'info'));

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Convert string level name to constant
     */
    private function level_from_string($level_name) {
        $levels = [
            'off'      => self::LOG_OFF,
            'critical' => self::LOG_CRITICAL,
            'error'    => self::LOG_ERROR,
            'warning'  => self::LOG_WARNING,
            'info'     => self::LOG_INFO,
            'debug'    => self::LOG_DEBUG,
        ];
        return $levels[strtolower($level_name)] ?? self::LOG_INFO;
    }

    /**
     * Get level name from constant
     */
    private function level_name($level) {
        $names = [
            self::LOG_OFF      => 'OFF',
            self::LOG_CRITICAL => 'CRITICAL',
            self::LOG_ERROR    => 'ERROR',
            self::LOG_WARNING  => 'WARNING',
            self::LOG_INFO     => 'INFO',
            self::LOG_DEBUG    => 'DEBUG',
        ];
        return $names[$level] ?? 'UNKNOWN';
    }

    /**
     * Main logging function
     */
    public function log($message, $level = self::LOG_INFO, $context = [], $component = 'core') {
        if (!$this->enabled || $level > $this->level) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_name = $this->level_name($level);
        $remote_ip = $_SERVER['REMOTE_ADDR'] ?? 'CLI';

        // Redact sensitive data from message and context
        $message = $this->redact_sensitive($message);
        if (!empty($context)) {
            $context = $this->redact_sensitive($context);
        }

        // Format log entry
        $formatted = "[{$timestamp}] [{$level_name}] [{$component}] IP:{$remote_ip} {$message}";
        if (!empty($context)) {
            $formatted .= " | " . (is_array($context) ? json_encode($context) : $context);
        }
        $formatted .= "\n";

        // Write to component-specific log file
        $this->write_log_file($component, $formatted);

        // Also write to WordPress error log for critical errors
        if ($level <= self::LOG_ERROR) {
            error_log("[PGB-{$component}] {$message}");
        }
    }

    /**
     * Write message to log file
     */
    private function write_log_file($component, $message) {
        $safe_component = preg_replace('/[^a-z0-9-_]/i', '', $component);
        $file_path = $this->log_dir . '/' . $safe_component . '.log';

        // Rotate file if too large (10MB)
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        // Open and write
        $handle = @fopen($file_path, 'a');
        if ($handle) {
            fwrite($handle, $message);
            fflush($handle);
            fclose($handle);
        }
    }

    /**
     * Redact sensitive data from string or array
     */
    private function redact_sensitive($data) {
        if (is_string($data)) {
            // Redact common sensitive patterns
            $patterns = [
                '/api[_-]?key["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i' => 'api_key=***REDACTED***',
                '/api[_-]?secret["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i' => 'api_secret=***REDACTED***',
                '/password["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i' => 'password=***REDACTED***',
                '/token["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i' => 'token=***REDACTED***',
                '/secret["\']?\s*[:=]\s*["\']?([^"\']+)["\']?/i' => 'secret=***REDACTED***',
            ];

            foreach ($patterns as $pattern => $replacement) {
                $data = preg_replace($pattern, $replacement, $data);
            }
        } elseif (is_array($data)) {
            foreach ($data as $key => &$value) {
                $key_lower = strtolower($key);
                if (stripos($key_lower, 'secret') !== false ||
                    stripos($key_lower, 'key') !== false ||
                    stripos($key_lower, 'password') !== false ||
                    stripos($key_lower, 'token') !== false) {
                    $value = '***REDACTED***';
                }
            }
        }
        return $data;
    }

    /**
     * Destructor - close file handles
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                @fclose($handle);
            }
        }
    }
}
```

```php
    /**
     * Sanitize email address

    }

    /**
     * Validate amount (must be positive number)
     */
    public static function amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }

    /**
     * Validate user ID (must be positive integer)
     */
    public static function user_id($user_id) {
        return intval($user_id) > 0;
    }

    /**
     * Validate transaction type
     */
    public static function transaction_type($type) {
        $allowed = ['PRODUCT', 'PRO', 'WALLET', 'DONATE', 'PURCHASE', 'SALE'];
        return in_array(strtoupper($type), $allowed);
    }

    /**
     * Validate currency code (ISO 4217)
     */
    public static function currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;
    }

    /**
     * Validate WooCommerce order data structure
     */
    public static function woocommerce_order($order_data) {
        $errors = [];

        if (empty($order_data['customer_id']) || !self::user_id($order_data['customer_id'])) {
            $errors[] = 'Invalid customer_id (must be positive integer)';
        }

        if (empty($order_data['line_items']) || !is_array($order_data['line_items'])) {
            $errors[] = 'No line items provided or invalid format';
        }

        if (!empty($order_data['line_items'])) {
            foreach ($order_data['line_items'] as $idx => $item) {
                if (empty($item['product_id'])) {
                    $errors[] = "Line item {$idx}: missing product_id";
                }
                if (empty($item['quantity']) || intval($item['quantity']) <= 0) {
                    $errors[] = "Line item {$idx}: invalid quantity";
                }
                if (!isset($item['total']) || floatval($item['total']) < 0) {
                    $errors[] = "Line item {$idx}: invalid total";
                }
            }
        }

        return [
            'valid'  => empty($errors),
            'errors' => $errors,
        ];
    }

    /**
     * Validate handoff signature
     */
    public static function signature($wow_order_id, $timestamp, $nonce, $signature, $secret) {
        if (empty($secret)) {
            return false;
        }

        $data = "{$wow_order_id}|{$timestamp}|{$nonce}";
        $expected = hash_hmac('sha256', $data, $secret);

        return hash_equals($expected, $signature);
    }

    /**
     * Validate handoff freshness (must be within 15 minutes)
     */
    public static function handoff_fresh($timestamp, $max_age_seconds = 900) {
        $now = time();
        $age = $now - intval($timestamp);
        return $age >= 0 && $age <= $max_age_seconds;
    }
}
```

```php
    /**
     * Process pending order from database
     * Called after signed handoff validation
     */
    public function process_pending_order($pending) {
        $this->logger->log(
            'Processing pending order',
            PGB_Logger::LOG_INFO,
            ['wow_order_id' => $pending['wow_order_id']],
            'pgb-manager'
        );

        $order_data = json_decode($pending['order_data'], true);
        if (!$order_data) {
            $this->logger->log(
                'Invalid JSON in pending order data',
                PGB_Logger::LOG_ERROR,
                ['wow_order_id' => $pending['wow_order_id']],
                'pgb-manager'
            );
            return [
                'success' => false,
                'error'   => 'Invalid order data',
            ];
        }

        // Validate order data
        $validation = PGB_Validator::woocommerce_order($order_data);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                ['errors' => $validation['errors']],
                'pgb-manager'
            );
            return [
                'success' => false,
                'error'   => 'Order validation failed',
            ];
        }

        try {
            // Create WooCommerce order
            $wc_order = $this->create_woocommerce_order($order_data);
            if (!$wc_order) {
                return [
                    'success' => false,
                    'error'   => 'Failed to create WooCommerce order',
                ];
            }

            // Update pending order with WooCommerce ID
            bzj_pgb_update_pending_order($pending['wow_order_id'], 'processed', $wc_order->get_id());

            // Get checkout URL
            $checkout_url = $wc_order->get_checkout_payment_url();

            $this->logger->log(
                'Pending order processed successfully',
                PGB_Logger::LOG_INFO,
                [
                    'wow_order_id'   => $pending['wow_order_id'],
                    'woo_order_id'   => $wc_order->get_id(),
                    'checkout_url'   => $checkout_url,
                ],
                'pgb-manager'
            );

            return [
                'success'      => true,
                'checkout_url' => $checkout_url,
                'order_id'     => $wc_order->get_id(),
            ];
        } catch (Exception $e) {
            $this->logger->log(
                'Exception processing pending order: ' . $e->getMessage(),
                PGB_Logger::LOG_CRITICAL,
                ['trace' => $e->getTraceAsString()],
                'pgb-manager'
            );
            return [
                'success' => false,
                'error'   => 'Internal server error',
            ];
        }
    }

    /**
     * Create WooCommerce order from order data
     */
    private function create_woocommerce_order($order_data) {
        // Create order
        $order = wc_create_order();

        // Set customer
        if (!empty($order_data['customer_id'])) {
            $order->set_customer_id($order_data['customer_id']);
        }

        // Add line items
        if (!empty($order_data['line_items']) && is_array($order_data['line_items'])) {
            foreach ($order_data['line_items'] as $item_data) {
                if (!empty($item_data['product_id'])) {
                    $product = wc_get_product($item_data['product_id']);
                    if ($product) {
                        $order->add_product(
                            $product,
                            $item_data['quantity'] ?? 1,
                            [
                                'subtotal' => $item_data['subtotal'] ?? $item_data['total'],
                                'total'    => $item_data['total'],
                            ]
                        );
                    }
                }
            }
        }

        // Set currency
        if (!empty($order_data['currency'])) {
            $order->set_currency($order_data['currency']);
        }

        // Add metadata
        if (!empty($order_data['meta_data']) && is_array($order_data['meta_data'])) {
            foreach ($order_data['meta_data'] as $meta) {
                if (!empty($meta['key'])) {
                    $order->update_meta_data($meta['key'], $meta['value'] ?? '');
                }
            }
        }

        // Add billing address
        if (!empty($order_data['billing'])) {
            $order->set_address($order_data['billing'], 'billing');
        }

        // Add shipping address
        if (!empty($order_data['shipping'])) {
            $order->set_address($order_data['shipping'], 'shipping');
        }

        // Calculate and save
        $order->calculate_totals();
        $order->save();

        // Fire hooks
        PGB_Hooks::do_action(PGB_Hooks::AFTER_ORDER_CREATE, $order);

        return $order;
    }

    /**
     * Handle payment completion
     * This is the PRIMARY hook that triggers fulfillment

    public function on_payment_complete($order_id) {
        $order = wc_get_order($order_id);
        if (!$order) {
            return;

            'WooCommerce payment complete triggered',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $order_id],
            'pgb-manager'
        );

        PGB_Hooks::do_action(PGB_Hooks::BEFORE_PAYMENT, $order);

        $this->update_streams_database($order);

        // Fire after-payment hook (addons listen here)
        PGB_Hooks::do_action(PGB_Hooks::AFTER_PAYMENT_SUCCESS, $order);

            'Payment complete processing finished',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $order_id],
            'pgb-manager'

     * Handle order completion (secondary path)

    public function on_order_completed($order_id) {

            return;
        }

            'WooCommerce order status changed to completed',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $order_id],
            'pgb-manager'

        // Fire completion hook
        PGB_Hooks::do_action(PGB_Hooks::AFTER_ORDER_COMPLETE, $order);
    }

    /**
     * Update Streams database with payment information
     */
    private function update_streams_database($order) {
        if (!$this->wow_db) {
            $this->logger->log(
                'WoWonder database not available',
                PGB_Logger::LOG_WARNING,
                [],
                'pgb-manager'
            );
            return;
        }

        // Get wow_order_id from order metadata
        $wow_order_id = $order->get_meta('wow_order_id');
        if (!$wow_order_id) {
            return; // Not a Buzzjuice order
        }

        try {
            // Update Wo_Payment_Transactions
            $sql = "UPDATE Wo_Payment_Transactions 
                    SET payment_status = 'completed', 
                        woo_order_id = ?, 
                        transaction_dt = NOW()
                    WHERE order_id = ?
                    LIMIT 1";
            
            $stmt = $this->wow_db->prepare($sql);

                $woo_id = $order->get_id();

                $stmt->close();

                    'Updated Streams transaction record',
                    PGB_Logger::LOG_DEBUG,
                    ['wow_order_id' => $wow_order_id, 'woo_order_id' => $woo_id],
                    'pgb-manager'
                );
            }
        } catch (Exception $e) {
            $this->logger->log(
                'Error updating Streams database: ' . $e->getMessage(),
                PGB_Logger::LOG_ERROR,
                ['wow_order_id' => $wow_order_id],
                'pgb-manager'
            );
        }
    }
```

```php
function pgb_generate_handoff_signature($wow_order_id, $timestamp, $nonce, $secret) {
    $data = "{$wow_order_id}|{$timestamp}|{$nonce}";
    return hash_hmac('sha256', $data, $secret);
}

function pgb_streams_init_payment() {
    header('Content-Type: application/json; charset=utf-8');
    
    // Ensure we're in Streams/WoWonder context
    if (!isset($GLOBALS['wo']) || !is_array($GLOBALS['wo'])) {
        http_response_code(500);
        return json_encode(['status' => 'error', 'message' => 'WoWonder context not available']);
    }

    $wo = $GLOBALS['wo'];
    $sqlConnect = $GLOBALS['sqlConnect'] ?? null;

    // ========================================
    // 1. AUTHENTICATION CHECK
    // ========================================

    if (empty($wo['loggedin']) || $wo['loggedin'] !== true) {
        http_response_code(401);
        return json_encode(['status' => 'error', 'message' => 'Not authenticated']);
    }

    // ========================================
    // 2. VALIDATE WOWONDER USER DATA
    // ========================================

    $wowonder_user = $wo['user'] ?? [];
    $wow_user_id = intval($wowonder_user['user_id'] ?? 0);
    $wow_user_email = trim($wowonder_user['email'] ?? '');
    $wow_username = htmlspecialchars($wowonder_user['username'] ?? '');

    if (!$wow_user_id || !filter_var($wow_user_email, FILTER_VALIDATE_EMAIL) || !$wow_username) {
        http_response_code(400);
        return json_encode(['status' => 'error', 'message' => 'Invalid WoWonder user data']);
    }

    // ========================================
    // 3. VALIDATE REQUEST PARAMETERS
    // ========================================

    $required_params = ['amount', 'product_name', 'product_price', 'product_units', 'product_owner_id', 'transaction_kind', 'wow_currency_code'];
    foreach ($required_params as $param) {
        if (empty($_POST[$param])) {
            http_response_code(400);
            return json_encode(['status' => 'error', 'message' => "Missing required parameter: {$param}"]);
        }
    }

    // ========================================
    // 4. SANITIZE INPUT
    // ========================================

    $amount = floatval($_POST['amount']);
    $product_name = htmlspecialchars(trim($_POST['product_name']), ENT_QUOTES, 'UTF-8');
    $product_price = floatval($_POST['product_price']);
    $product_units = intval($_POST['product_units']);
    $product_owner_id = intval($_POST['product_owner_id']);
    $transaction_kind = strtoupper(preg_replace('/[^a-zA-Z0-9_-]/', '', $_POST['transaction_kind']));
    $wow_currency_code = strtoupper(htmlspecialchars(trim($_POST['wow_currency_code']), ENT_QUOTES, 'UTF-8'));
    $wow_post_id = isset($_POST['wow_post_id']) ? intval($_POST['wow_post_id']) : 0;
    $address_id = isset($_POST['address_id']) ? intval($_POST['address_id']) : 0;

    // Validate amounts
    if ($amount <= 0 || $product_price < 0) {
        http_response_code(400);
        return json_encode(['status' => 'error', 'message' => 'Invalid amount or price']);
    }

    // ========================================
    // 5. CREATE DURABLE WO_PAYMENT_TRANSACTIONS RECORD
    // ========================================

    if (!$sqlConnect) {
        http_response_code(500);
        return json_encode(['status' => 'error', 'message' => 'Database connection unavailable']);
    }

    $wow_order_id = 'wow_' . uniqid('', true);
    $timestamp = time();

    $sql = "INSERT INTO Wo_Payment_Transactions 
            (userid, amount, order_id, kind, currency_code, notes, created_at) 
            VALUES (?, ?, ?, ?, ?, ?, NOW())";
    
    $stmt = $sqlConnect->prepare($sql);
    if (!$stmt) {
        http_response_code(500);
        return json_encode(['status' => 'error', 'message' => 'Database error: prepare failed']);
    }

    // Store full order data as JSON in notes field
    $order_payload = [
        'wow_user_id' => $wow_user_id,
        'wow_username' => $wow_username,
        'wow_email' => $wow_user_email,
        'product_name' => $product_name,
        'product_price' => $product_price,
        'product_units' => $product_units,
        'product_owner_id' => $product_owner_id,
        'amount' => $amount,
        'currency_code' => $wow_currency_code,
        'wow_post_id' => $wow_post_id,
        'address_id' => $address_id,
        'timestamp' => $timestamp,
    ];

    // Capture AffiliateWP context snapshot (if available)
    if (!empty($_COOKIE['affwp_ref']) || !empty($_COOKIE['affwp_visit_id']) || !empty($_COOKIE['affwp_affiliate_id'])) {
        $order_payload['affwp_snapshot'] = [
            'affwp_ref' => $_COOKIE['affwp_ref'] ?? '',
            'affwp_visit_id' => $_COOKIE['affwp_visit_id'] ?? '',
            'affwp_affiliate_id' => $_COOKIE['affwp_affiliate_id'] ?? '',
            'affwp_campaign' => $_COOKIE['affwp_campaign'] ?? '',
            'http_referer' => $_SERVER['HTTP_REFERER'] ?? '',
        ];
    }

    $notes_json = json_encode($order_payload);

    $stmt->bind_param(
        'idsss',
        $wow_user_id,
        $amount,
        $wow_order_id,
        $transaction_kind,
        $wow_currency_code
    );

    // Note: We can't bind $notes_json directly with bind_param, so we use a second query
    $stmt->execute();
    $stmt->close();

    // Update with notes field
    $sql_update = "UPDATE Wo_Payment_Transactions SET notes = ? WHERE order_id = ? LIMIT 1";
    $stmt_update = $sqlConnect->prepare($sql_update);
    if ($stmt_update) {
        $stmt_update->bind_param('ss', $notes_json, $wow_order_id);
        $stmt_update->execute();
        $stmt_update->close();
    }

    // ========================================
    // 6. VERIFY RECORD WAS STORED
    // ========================================

    $verify_sql = "SELECT order_id FROM Wo_Payment_Transactions WHERE order_id = ? LIMIT 1";
    $verify_stmt = $sqlConnect->prepare($verify_sql);
    if ($verify_stmt) {
        $verify_stmt->bind_param('s', $wow_order_id);
        $verify_stmt->execute();
        $result = $verify_stmt->get_result();
        
        if ($result->num_rows === 0) {
            http_response_code(500);
            return json_encode(['status' => 'error', 'message' => 'Failed to store transaction record']);
        }
        $verify_stmt->close();
    }

    // ========================================
    // 7. GENERATE SIGNED HANDOFF
    // ========================================

    // Get handoff secret from environment or database
    $handoff_secret = getenv('BZJ_PGB_HANDOFF_SECRET') ?: 'fallback-secret-change-this';
    
    if ($handoff_secret === 'fallback-secret-change-this') {
        error_log('WARNING: BZJ_PGB_HANDOFF_SECRET not configured, using fallback secret');
    }

    $nonce = bin2hex(random_bytes(16));
    $signature = pgb_generate_handoff_signature($wow_order_id, $timestamp, $nonce, $handoff_secret);

    // ========================================
    // 8. BUILD REDIRECT URL
    // ========================================

    $checkout_redirect_params = [
        'wow_order_id' => $wow_order_id,
        'wow_user_id' => $wow_user_id,
        'ts' => $timestamp,
        'nonce' => $nonce,
        'sig' => $signature,
    ];

    $wordpress_base_url = $wo['config']['wow_store_url'] ?? 'https://buzzjuice.net';
    $wordpress_base_url = rtrim($wordpress_base_url, '/');
    
    // Build redirect URL
    $redirect_url = $wordpress_base_url . '/wp-json/bzj-pgb/v1/checkout?' . http_build_query($checkout_redirect_params);

    // ========================================
    // 9. RETURN SUCCESS RESPONSE
    // ========================================

    http_response_code(200);
    return json_encode([
        'status' => 'success',
        'message' => 'Payment initiated',
        'wow_order_id' => $wow_order_id,
        'redirect_url' => $redirect_url,
        'timestamp' => $timestamp,
    ]);
}

// ============================================================================
// EXECUTE IF THIS FILE WAS CALLED DIRECTLY
// ============================================================================

if (basename(__FILE__) === basename($_SERVER['PHP_SELF'] ?? '')) {
    echo pgb_streams_init_payment();
}

?>
```

```php
/**
 * Payment Gateway Bridge - WoWonder Addon

if (!defined('ABSPATH')) {

class PGB_Addon_WoWonder {
    private static $instance = null;
    private $logger = null;
    private $wow_db = null;

    /**
     * Get singleton instance
     */
    public static function instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    /**
     * Private constructor
     */
    private function __construct() {
        $this->logger = PGB_Logger::instance();
        $this->wow_db = get_wowonder_db();
        
        // Register hooks
        $this->register_hooks();
        
        $this->logger->log(
            'WoWonder addon initialized',
            PGB_Logger::LOG_DEBUG,
            [],
            'pgb-addon-wowonder'
        );
    }

    /**
     * Register hooks
     */
    private function register_hooks() {
        // Listen to after payment success
        PGB_Hooks::add_action(PGB_Hooks::AFTER_PAYMENT_SUCCESS, [$this, 'on_payment_success'], 10, 1);
    }

    /**
     * Handle payment success - update WoWonder database
     */
    public function on_payment_success($wc_order) {
        if (!$this->wow_db) {
            $this->logger->log(
                'WoWonder database not available',
                PGB_Logger::LOG_WARNING,
                [],
                'pgb-addon-wowonder'
            );
            return;
        }

        try {
            $wow_order_id = $wc_order->get_meta('wow_order_id');
            $wow_user_id = $wc_order->get_meta('wow_user_id');
            
            if (!$wow_order_id) {
                return; // Not a WoWonder order
            }

            $this->logger->log(
                'Processing WoWonder fulfillment',
                PGB_Logger::LOG_INFO,
                ['wow_order_id' => $wow_order_id, 'woo_order_id' => $wc_order->get_id()],
                'pgb-addon-wowonder'
            );

            // Update Wo_Payment_Transactions with payment status
            $this->update_payment_transaction($wc_order, $wow_order_id);

            // Handle wallet top-up
            $this->handle_wallet_topup($wc_order, $wow_user_id);

            // Handle crowdfunding
            $this->handle_crowdfunding($wc_order, $wow_user_id);

            // Handle PRO membership
            $this->handle_pro_membership($wc_order, $wow_user_id);

            // Handle product purchases
            $this->handle_product_purchases($wc_order, $wow_user_id);

        } catch (Exception $e) {
            $this->logger->log(
                'Exception in WoWonder fulfillment: ' . $e->getMessage(),
                PGB_Logger::LOG_ERROR,
                ['trace' => $e->getTraceAsString()],
                'pgb-addon-wowonder'
            );
        }
    }

    /**
     * Update payment transaction record
     */
    private function update_payment_transaction($wc_order, $wow_order_id) {
        $payment_method = $wc_order->get_payment_method() ?? 'unknown';
        $total = floatval($wc_order->get_total());
        $currency = $wc_order->get_currency();
        
        $sql = "UPDATE Wo_Payment_Transactions 
                SET payment_status = 'completed', 
                    payment_method = ?,
                    woo_order_id = ?,
                    amount = ?,
                    currency_code = ?,
                    transaction_dt = NOW()
                WHERE order_id = ?
                LIMIT 1";
        
        $stmt = $this->wow_db->prepare($sql);
        if ($stmt) {
            $woo_id = $wc_order->get_id();
            $stmt->bind_param('ssdss', $payment_method, $woo_id, $total, $currency, $wow_order_id);
            $stmt->execute();
            $stmt->close();
        }
    }

    /**
     * Handle wallet top-up
     */
    private function handle_wallet_topup($wc_order, $wow_user_id) {
        if (!$wow_user_id) return;

        $config = PGB_Config::instance();
        $wallet_product_id = $config->get('wallet_topup_id');
        
        if (!$wallet_product_id) return;

        // Check if order contains wallet product
        $line_items = $wc_order->get_items();
        foreach ($line_items as $item) {
            $product_id = $item->get_product_id();
            if ($product_id == $wallet_product_id) {
                $amount = floatval($item->get_total());
                
                // Update user wallet
                $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ? LIMIT 1";
                $stmt = $this->wow_db->prepare($sql);
                if ($stmt) {
                    $stmt->bind_param('di', $amount, $wow_user_id);
                    $stmt->execute();
                    $stmt->close();
                    
                    $this->logger->log(
                        'Wallet updated',
                        PGB_Logger::LOG_INFO,
                        ['user_id' => $wow_user_id, 'amount' => $amount],
                        'pgb-addon-wowonder'
                    );
                }
                break;
            }
        }
    }

    /**
     * Handle crowdfunding
     */
    private function handle_crowdfunding($wc_order, $wow_user_id) {
        if (!$wow_user_id) return;

        $config = PGB_Config::instance();
        $crowdfund_product_id = $config->get('crowdfund_id');
        
        if (!$crowdfund_product_id) return;

        $wow_post_id = $wc_order->get_meta('wow_post_id');
        if (!$wow_post_id) return;

        $line_items = $wc_order->get_items();
        foreach ($line_items as $item) {
            $product_id = $item->get_product_id();
            if ($product_id == $crowdfund_product_id) {
                $amount = floatval($item->get_total());
                
                // Insert funding contribution
                $sql = "INSERT INTO Wo_Funding_Raise (funding_id, user_id, amount, time) VALUES (?, ?, ?, ?)";
                $stmt = $this->wow_db->prepare($sql);
                if ($stmt) {
                    $time = time();
                    $stmt->bind_param('iidi', $wow_post_id, $wow_user_id, $amount, $time);
                    $stmt->execute();
                    $stmt->close();
                    
                    $this->logger->log(
                        'Crowdfunding contribution recorded',
                        PGB_Logger::LOG_INFO,
                        ['funding_id' => $wow_post_id, 'user_id' => $wow_user_id, 'amount' => $amount],
                        'pgb-addon-wowonder'
                    );
                }
                break;
            }
        }
    }

    /**
     * Handle PRO membership activation
     */
    private function handle_pro_membership($wc_order, $wow_user_id) {
        if (!$wow_user_id) return;

        $config = PGB_Config::instance();
        $pro_packages = $config->get('pro_packages');

        if (!$wow_post_id || $wow_post_id < 1 || $wow_post_id > 4) return;

            if (in_array($product_id, array_values($pro_packages))) {
                // Activate PRO
                $pro_time = time();
                $is_pro = '1';

                $sql = "UPDATE Wo_Users SET pro_time = ?, is_pro = ?, pro_type = ? WHERE user_id = ? LIMIT 1";
                $stmt = $this->wow_db->prepare($sql);
                if ($stmt) {

                    $stmt->execute();
                    $stmt->close();
                    
                    $this->logger->log(
                        'PRO membership activated',
                        PGB_Logger::LOG_INFO,
                        ['user_id' => $wow_user_id, 'pro_type' => $pro_type],
                        'pgb-addon-wowonder'
                    );
```
