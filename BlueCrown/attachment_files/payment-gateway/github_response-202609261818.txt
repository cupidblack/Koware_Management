define('PGB_VERSION', '4.0.0');
define('PGB_PATH', dirname(__FILE__) . '/../../shared/payment-gateway');
define('PGB_URL', WP_CONTENT_URL . '/mu-plugins/../../../shared/payment-gateway');
define('PGB_DEBUG', defined('WP_DEBUG') && WP_DEBUG);

/**
 * Primary bootstrap class
 */
class PGB_Bootstrap {
    private static $instance = null;
    private $logger = null;
    private $client = null;
    private $fulfillment = null;
    private $admin = null;

    public static function instance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    private function __construct() {
        // Load components
        $this->load_components();

    }

    private function load_components() {
        // Logger (initialize first, all others depend on it)
        require_once PGB_PATH . '/core/class-pgb-logger.php';
        $this->logger = PGB_Logger::instance();

        // Order management
        require_once PGB_PATH . '/core/class-pgb-order.php';

        // Idempotency handler
        require_once PGB_PATH . '/core/class-pgb-idempotency.php';

        // Core API client
        require_once PGB_PATH . '/core/class-pgb-client.php';
        $this->client = PGB_Client::instance($this->logger);

        // Fulfillment engine
        require_once PGB_PATH . '/core/class-pgb-fulfillment.php';
        $this->fulfillment = PGB_Fulfillment::instance($this->logger);

        // Admin (only in wp-admin)
        if (is_admin()) {
            require_once PGB_PATH . '/admin/pgb-admin.php';
            $this->admin = PGB_Admin::instance($this->logger);
        }

        $this->logger->info('PGB components loaded', __FILE__, __LINE__);
    }

    private function init_hooks() {
        // REST API endpoint for Streams to submit payment requests
        add_action('rest_api_init', [$this, 'register_rest_endpoints']);

        // WooCommerce lifecycle hooks (payment completion)
        add_action('woocommerce_payment_complete', [$this, 'on_payment_complete'], 10, 1);
        add_action('woocommerce_order_status_changed', [$this, 'on_order_status_changed'], 10, 3);

        // Scheduled retry action
        add_action('pgb_process_retry', [$this, 'process_retry_queue']);
        if (!wp_next_scheduled('pgb_process_retry')) {
            wp_schedule_event(time(), 'hourly', 'pgb_process_retry');
        }

        $this->logger->info('PGB hooks registered', __FILE__, __LINE__);
    }

    /**
     * REST API endpoints for Streams
     */
    public function register_rest_endpoints() {
        register_rest_route('bzj-pgb/v1', '/order', [
            'methods'             => 'POST',
            'callback'            => [$this, 'handle_order_request'],
            'permission_callback' => [$this, 'verify_request_signature'],
            'args'                => [
                'timestamp'  => ['required' => true, 'type' => 'string'],
                'request_id' => ['required' => true, 'type' => 'string'],
                'data'       => ['required' => true, 'type' => 'object'],
            ],
        ]);
    }

    /**
     * Verify HMAC-SHA256 signature from Streams
     */
    public function verify_request_signature(\WP_REST_Request $request) {
        $timestamp = $request->get_param('timestamp');
        $signature = $request->get_header('X-BZJ-PGB-Signature');
        
        if (!$timestamp || !$signature) {
            return new \WP_Error('missing_auth', 'Missing auth headers', ['status' => 401]);
        }

        // Validate timestamp is within 5 minutes
        $now = time();
        $time = intval($timestamp);
        if (abs($now - $time) > 300) {
            return new \WP_Error('timestamp_expired', 'Request timestamp too old', ['status' => 401]);
        }

        // Get shared secret from environment
        $secret = getenv('BZJ_PGB_SECRET');
        if (empty($secret)) {
            $this->logger->error('BZJ_PGB_SECRET not configured', __FILE__, __LINE__);
            return new \WP_Error('config_error', 'Server configuration error', ['status' => 500]);
        }

        // Reconstruct signature
        $body = file_get_contents('php://input');
        $expected = hash_hmac('sha256', $timestamp . "\n" . $body, $secret);

        if (!hash_equals($expected, $signature)) {
            $this->logger->warning('Signature verification failed', __FILE__, __LINE__);
            return new \WP_Error('invalid_signature', 'Signature verification failed', ['status' => 403]);
        }

        return true;
    }

    /**
     * Handle incoming payment order from Streams
     */
    public function handle_order_request(\WP_REST_Request $request) {
        try {
            $timestamp = $request->get_param('timestamp');
            $request_id = $request->get_param('request_id');
            $data = $request->get_param('data');

            $this->logger->info("Order request received: $request_id", __FILE__, __LINE__);

            // Delegate to client
            $result = $this->client->process_order($request_id, $timestamp, (array)$data);

            return new \WP_REST_Response($result, 200);
        } catch (Exception $e) {
            $this->logger->error('Order request exception: ' . $e->getMessage(), __FILE__, __LINE__);
            return new \WP_REST_Response(
                ['error' => $e->getMessage()],
                500
            );
        }
    }

    /**
     * WooCommerce payment complete hook
     */
    public function on_payment_complete($order_id) {
        $this->logger->info("Payment complete: WC order $order_id", __FILE__, __LINE__);
        
        // Trigger fulfillment
        $this->fulfillment->process_woocommerce_order($order_id);
    }

    /**
     * WooCommerce order status change hook
     */
    public function on_order_status_changed($order_id, $old_status, $new_status) {
        $this->logger->info("Order status changed: $order_id ($old_status → $new_status)", __FILE__, __LINE__);
        
        if (in_array($new_status, ['completed', 'processing'])) {
            $this->fulfillment->process_woocommerce_order($order_id);
        }
    }

    /**
     * Process retry queue (WP-Cron)
     */
    public function process_retry_queue() {
        $this->fulfillment->retry_failed_orders();
    }
}

// shared/payment-gateway/core/class-pgb-logger.php
<?php
/**
 * PGB Logger
 * 
 * Component-level logging with:
 * - Multiple log files (one per component)
 * - Configurable log level
 * - Secret redaction
 * - Enable/disable toggle
 */

class PGB_Logger {
    const DEBUG = 1;
    const INFO = 2;
    const WARNING = 3;
    const ERROR = 4;

    private static $instance = null;
    private $log_dir = null;
    private $log_level = self::INFO;
    private $enabled = true;
    private $secrets = [];

    private static $level_names = [
        self::DEBUG => 'DEBUG',
        self::INFO => 'INFO',
        self::WARNING => 'WARNING',
        self::ERROR => 'ERROR',
    ];

    public static function instance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    private function __construct() {
        $this->log_dir = dirname(__FILE__) . '/../../log';
        $this->ensure_log_dir();

        // Load settings from WP options
        $this->enabled = get_option('pgb_logging_enabled', true);
        $level = get_option('pgb_log_level', 'info');
        $this->log_level = $this->level_from_string($level);

        // Register secrets for redaction
        $this->register_secrets();
    }

    private function ensure_log_dir() {
        if (!file_exists($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
        
        // Protect from web access
        if (!file_exists($this->log_dir . '/.htaccess')) {
            @file_put_contents($this->log_dir . '/.htaccess', "Deny from all\n");
        }
    }

    private function register_secrets() {
        // Register secrets that should be redacted from logs
        $secret = getenv('BZJ_PGB_SECRET');
        if (!empty($secret)) {
            $this->secrets[] = $secret;
        }

        // Add other sensitive patterns
        $this->secrets[] = getenv('WORDPRESS_DB_PASS');
        $this->secrets[] = getenv('WOWONDER_DB_PASS');
    }

    private function level_from_string($level) {
        $map = [
            'debug' => self::DEBUG,
            'info' => self::INFO,
            'warning' => self::WARNING,
            'error' => self::ERROR,
        ];
        return $map[strtolower($level)] ?? self::INFO;
    }

    /**
     * Log message at specified level
     */
    public function log($level, $message, $file = '', $line = 0) {
        if (!$this->enabled || $level < $this->log_level) {
            return;
        }

        $level_name = self::$level_names[$level] ?? 'UNKNOWN';
        $timestamp = gmdate('Y-m-d H:i:s');
        
        // Redact secrets
        $message = $this->redact_secrets($message);
        $file = basename($file);

        // Determine log file based on calling file
        $log_file = $this->get_log_file($file);

        // Format entry
        $entry = sprintf(
            "[%s] %s — %s:%s\n",
            $timestamp,
            $level_name,
            $file,
            $line
        ) . $message . "\n\n";

        // Write to file
        @file_put_contents($log_file, $entry, FILE_APPEND | LOCK_EX);
    }

    public function debug($message, $file = '', $line = 0) {
        $this->log(self::DEBUG, $message, $file, $line);
    }

    public function info($message, $file = '', $line = 0) {
        $this->log(self::INFO, $message, $file, $line);
    }

    public function warning($message, $file = '', $line = 0) {
        $this->log(self::WARNING, $message, $file, $line);
    }

    public function error($message, $file = '', $line = 0) {
        $this->log(self::ERROR, $message, $file, $line);
    }

    private function get_log_file($calling_file) {
        // Extract component name from file
        $name = preg_replace('/\.php$/', '', $calling_file);
        $log_file = $this->log_dir . '/' . $name . '.log';
        return $log_file;
    }

    private function redact_secrets($message) {
        foreach ($this->secrets as $secret) {
            if (!empty($secret)) {
                $message = str_replace(
                    $secret,
                    '[REDACTED]',
                    $message
                );
            }
        }
        return $message;
    }

    /**
     * Get recent log entries
     */
    public function get_recent_logs($component = '', $limit = 50) {
        if ($component) {
            $log_file = $this->log_dir . '/' . $component . '.log';
            if (file_exists($log_file)) {
                $lines = explode("\n", trim(file_get_contents($log_file)));
                return array_slice($lines, -($limit * 3), $limit * 3);
            }
        }
        return [];
    }

    /**
     * Clear logs
     */
    public function clear_logs($component = '') {
        if ($component) {
            $log_file = $this->log_dir . '/' . $component . '.log';
            @unlink($log_file);
        } else {
            $files = glob($this->log_dir . '/*.log');
            foreach ($files as $file) {
                @unlink($file);
            }
        }
    }
}

/**

class PGB_Order {

    private $logger = null;

    public function __construct(PGB_Logger $logger) {

        $this->table = $wpdb->prefix . 'bzj_pgb_transactions';
        $this->logger = $logger;
        $this->ensure_table_exists();
    }

    /**
     * Ensure PGB transaction table exists
     */
    private function ensure_table_exists() {
        global $wpdb;
        
        $sql = "CREATE TABLE IF NOT EXISTS $this->table (
            id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
            pgb_transaction_id VARCHAR(64) UNIQUE NOT NULL,
            idempotency_key VARCHAR(255) UNIQUE,
            request_id VARCHAR(64),
            wow_order_id VARCHAR(64),
            wow_user_id BIGINT UNSIGNED,
            wp_user_id BIGINT UNSIGNED,
            woo_order_id BIGINT UNSIGNED,
            transaction_type VARCHAR(32),
            wow_post_id BIGINT UNSIGNED,
            woo_product_id BIGINT UNSIGNED,
            currency VARCHAR(10),
            requested_amount DECIMAL(10, 2),
            final_amount DECIMAL(10, 2),
            payment_status VARCHAR(32) DEFAULT 'pending',
            fulfillment_status VARCHAR(32) DEFAULT 'pending',
            affiliate_status VARCHAR(32) DEFAULT 'pending',
            retry_count INT DEFAULT 0,
            next_retry_at DATETIME,
            last_error TEXT,
            original_request LONGTEXT,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
            INDEX idx_wow_order (wow_order_id),
            INDEX idx_woo_order (woo_order_id),
            INDEX idx_wow_user (wow_user_id),
            INDEX idx_status (payment_status, fulfillment_status),
            INDEX idx_retry (retry_count, next_retry_at)
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;";

        $wpdb->query($sql);
        $this->logger->info('PGB table verified', __FILE__, __LINE__);
    }

    /**
     * Create new PGB transaction
     */
    public function create($data) {
        global $wpdb;

        $pgb_id = 'pgb_' . uniqid(bin2hex(random_bytes(4)));

        $insert = [
            'pgb_transaction_id' => $pgb_id,
            'idempotency_key' => $data['idempotency_key'] ?? null,
            'request_id' => $data['request_id'] ?? null,
            'wow_order_id' => $data['wow_order_id'] ?? null,
            'wow_user_id' => $data['wow_user_id'] ?? null,
            'wp_user_id' => $data['wp_user_id'] ?? null,
            'woo_order_id' => $data['woo_order_id'] ?? null,
            'transaction_type' => $data['transaction_type'] ?? 'PURCHASE',
            'wow_post_id' => $data['wow_post_id'] ?? null,
            'woo_product_id' => $data['woo_product_id'] ?? null,
            'currency' => $data['currency'] ?? 'GHS',
            'requested_amount' => $data['requested_amount'] ?? 0,
            'final_amount' => $data['final_amount'] ?? 0,
            'payment_status' => 'pending',
            'fulfillment_status' => 'pending',
            'original_request' => json_encode($data['original_request'] ?? []),
        ];

        $wpdb->insert($this->table, $insert);

        if ($wpdb->last_error) {
            $this->logger->error('Failed to create PGB order: ' . $wpdb->last_error, __FILE__, __LINE__);
            return false;
        }

        $this->logger->info("Created PGB order: $pgb_id", __FILE__, __LINE__);
        return $pgb_id;
    }

    /**
     * Find by idempotency key
     */
    public function find_by_idempotency_key($key) {
        global $wpdb;

        $result = $wpdb->get_row(
            $wpdb->prepare("SELECT * FROM $this->table WHERE idempotency_key = %s LIMIT 1", $key),
            ARRAY_A
        );

        return $result;
    }

    /**
     * Find by wow_order_id
     */
    public function find_by_wow_order_id($wow_order_id) {
        global $wpdb;

        $result = $wpdb->get_row(
            $wpdb->prepare("SELECT * FROM $this->table WHERE wow_order_id = %s LIMIT 1", $wow_order_id),
            ARRAY_A
        );

        return $result;
    }

    /**
     * Update status
     */
    public function update_status($pgb_id, $payment_status = null, $fulfillment_status = null) {
        global $wpdb;

        $update = [];
        if ($payment_status !== null) {
            $update['payment_status'] = $payment_status;
        }
        if ($fulfillment_status !== null) {
            $update['fulfillment_status'] = $fulfillment_status;
        }
        $update['updated_at'] = current_time('mysql');

        $wpdb->update($this->table, $update, ['pgb_transaction_id' => $pgb_id]);

        if ($wpdb->last_error) {
            $this->logger->error('Failed to update status: ' . $wpdb->last_error, __FILE__, __LINE__);

        }

        return true;
    }

    /**

     */
    public function mark_for_retry($pgb_id, $error_message = '', $retry_after_seconds = 3600) {
        global $wpdb;

        $next_retry = date('Y-m-d H:i:s', current_time('timestamp') + $retry_after_seconds);

        $wpdb->query($wpdb->prepare(
            "UPDATE $this->table SET 
             retry_count = retry_count + 1,
             next_retry_at = %s,
             last_error = %s,
             fulfillment_status = 'retry'
             WHERE pgb_transaction_id = %s",
            $next_retry,
            $error_message,

        ));

        $this->logger->warning("Marked for retry: $pgb_id — $error_message", __FILE__, __LINE__);
    }

    /**

        global $wpdb;

        $result = $wpdb->get_results(

             LIMIT $limit",

        );

        return $result ?: [];

// shared/payment-gateway/core/class-pgb-idempotency.php
<?php
/**
 * PGB Idempotency Handler
 * 
 * Ensures duplicate requests (browser retries, double-clicks, timeouts)
 * don't create multiple orders
 */

class PGB_Idempotency {
    private $logger = null;

    public function __construct(PGB_Logger $logger) {
        $this->logger = $logger;
    }

    /**
     * Generate idempotency key from request parameters
     * 
     * Key = HMAC-SHA256(wow_user_id + wow_order_id + amount + currency)
     * This ensures same user + same order = same idempotency key
     */
    public function generate_key($wow_user_id, $wow_order_id, $amount, $currency) {
        $payload = sprintf('%s:%s:%s:%s', $wow_user_id, $wow_order_id, $amount, $currency);
        return hash('sha256', $payload);
    }

    /**
     * Check if request is duplicate
     * Returns existing order if found
     */
    public function check_duplicate($idempotency_key, PGB_Order $order_manager) {
        $existing = $order_manager->find_by_idempotency_key($idempotency_key);

        if ($existing) {
            $this->logger->info(
                "Idempotency match: duplicate request detected for $idempotency_key",
                __FILE__,
                __LINE__
            );
            return $existing;
        }

        return null;
    }
}

 * Handles:
 * - Receiving order requests from Streams
 * - Validating request data
 * - Creating PGB transactions
 * - Creating WooCommerce orders
 */

class PGB_Client {
    private static $instance = null;
    private $logger = null;
    private $order_mgr = null;
    private $idempotency = null;
    private $wow_db = null;
    private $retry_limit = 5;

    public static function instance(PGB_Logger $logger) {
        if (self::$instance === null) {
            self::$instance = new self($logger);
        }
        return self::$instance;
    }

    private function __construct(PGB_Logger $logger) {
        $this->logger = $logger;
        $this->order_mgr = new PGB_Order($logger);
        $this->idempotency = new PGB_Idempotency($logger);
        $this->wow_db = get_wowonder_db();

        if (!$this->wow_db) {
            $this->logger->error('Failed to connect to Streams DB', __FILE__, __LINE__);
        }
    }

    /**
     * Main entry point: Process order request from Streams
     */
    public function process_order($request_id, $timestamp, $data) {
        try {
            $this->logger->info("Processing order: $request_id", __FILE__, __LINE__);

            // Step 1: Validate request structure
            $validation = $this->validate_request($data);
            if ($validation !== true) {
                return [
                    'status' => 'error',
                    'message' => $validation,
                ];
            }

            // Step 2: Extract and sanitize data
            $wow_user_id = intval($data['wow_user_id'] ?? 0);
            $wow_order_id = $data['wow_order_id'] ?? '';
            $transaction_type = strtoupper($data['transaction_type'] ?? 'PURCHASE');
            $amount = floatval($data['amount'] ?? 0);
            $currency = strtoupper($data['currency'] ?? 'GHS');
            $wow_post_id = intval($data['wow_post_id'] ?? 0);

            // Step 3: Generate idempotency key
            $idempotency_key = $this->idempotency->generate_key(
                $wow_user_id,
                $wow_order_id,
                $amount,
                $currency
            );

            // Step 4: Check for duplicate
            $existing_pgb = $this->idempotency->check_duplicate($idempotency_key, $this->order_mgr);
            if ($existing_pgb) {
                // Return existing Woo order link
                if (!empty($existing_pgb['woo_order_id'])) {
                    $woo_order = wc_get_order($existing_pgb['woo_order_id']);
                    if ($woo_order) {
                        return [
                            'status' => 'success',
                            'message' => 'Order already exists (idempotent)',
                            'woo_order_id' => $existing_pgb['woo_order_id'],
                            'payment_url' => $woo_order->get_checkout_payment_url(),
                        ];
                    }
                }
            }

            // Step 5: Resolve WP user from WoWonder user
            $wp_user_id = $this->get_wp_user_id($wow_user_id);
            if (!$wp_user_id) {
                return [
                    'status' => 'error',
                    'message' => 'Unable to resolve WordPress user',
                ];
            }

            // Step 6: Get product mapping
            $product_mapping = $this->get_product_mapping($transaction_type, $wow_post_id);
            if (!$product_mapping) {
                return [
                    'status' => 'error',
                    'message' => 'Product not mapped to WooCommerce',
                ];
            }

            // Step 7: Apply business rules (prices, limits)
            $final_amount = $this->apply_business_rules($transaction_type, $amount, $currency);
            if ($final_amount <= 0) {
                return [
                    'status' => 'error',
                    'message' => 'Invalid amount after business rule application',
                ];
            }

            // Step 8: Create WooCommerce order
            $woo_order_id = $this->create_woocommerce_order(
                $wp_user_id,
                $product_mapping['woo_product_id'],
                $final_amount,
                $currency,
                [
                    'wow_user_id' => $wow_user_id,
                    'wow_order_id' => $wow_order_id,
                    'wow_post_id' => $wow_post_id,
                    'transaction_type' => $transaction_type,
                ]
            );

            if (!$woo_order_id) {
                return [
                    'status' => 'error',
                    'message' => 'Failed to create WooCommerce order',
                ];
            }

            // Step 9: Create PGB transaction record
            $pgb_id = $this->order_mgr->create([
                'idempotency_key' => $idempotency_key,
                'request_id' => $request_id,
                'wow_order_id' => $wow_order_id,
                'wow_user_id' => $wow_user_id,
                'wp_user_id' => $wp_user_id,
                'woo_order_id' => $woo_order_id,
                'transaction_type' => $transaction_type,
                'wow_post_id' => $wow_post_id,
                'woo_product_id' => $product_mapping['woo_product_id'],
                'currency' => $currency,
                'requested_amount' => $amount,
                'final_amount' => $final_amount,
                'original_request' => $data,
            ]);

            if (!$pgb_id) {
                return [
                    'status' => 'error',
                    'message' => 'Failed to create PGB transaction',
                ];
            }

            // Step 10: Get payment URL
            $woo_order = wc_get_order($woo_order_id);
            $payment_url = $woo_order ? $woo_order->get_checkout_payment_url() : '';

            $this->logger->info(
                "Order processed successfully: PGB=$pgb_id WC=$woo_order_id",
                __FILE__,
                __LINE__
            );

            return [
                'status' => 'success',
                'pgb_transaction_id' => $pgb_id,
                'woo_order_id' => $woo_order_id,
                'payment_url' => $payment_url,
                'wow_order_id' => $wow_order_id,
            ];

        } catch (Exception $e) {
            $this->logger->error('Order processing exception: ' . $e->getMessage(), __FILE__, __LINE__);
            return [
                'status' => 'error',
                'message' => $e->getMessage(),
            ];
        }
    }

    /**
     * Validate request structure
     */
    private function validate_request($data) {
        $required = [
            'wow_user_id',
            'wow_order_id',
            'transaction_type',
            'amount',
            'currency',
        ];

        foreach ($required as $field) {
            if (empty($data[$field])) {
                return "Missing required field: $field";
            }
        }

        $valid_types = ['PRO', 'PRODUCT', 'WALLET', 'DONATE'];
        if (!in_array(strtoupper($data['transaction_type'] ?? ''), $valid_types)) {
            return "Invalid transaction_type";
        }

        $amount = floatval($data['amount'] ?? 0);
        if ($amount <= 0) {
            return "Invalid amount";
        }

        return true;
    }

    /**
     * Get WordPress user ID from WoWonder user ID
     */
    private function get_wp_user_id($wow_user_id) {
        if (!$this->wow_db) return false;

        $result = $this->wow_db->query(
            "SELECT wp_user_id FROM Wo_Users WHERE user_id = " . intval($wow_user_id) . " LIMIT 1"
        );

        if ($result && $row = $result->fetch_assoc()) {
            return intval($row['wp_user_id']);
        }

        return false;
    }

    /**
     * Get WooCommerce product mapping based on transaction type
     */
    private function get_product_mapping($transaction_type, $wow_post_id = 0) {
        if (!$this->wow_db) return false;

        $config_result = $this->wow_db->query("SELECT data FROM Wo_Config LIMIT 1");
        if (!$config_result || !$row = $config_result->fetch_assoc()) {
            return false;
        }

        $config = json_decode($row['data'] ?? '{}', true);

        $mappings = [
            'PRO' => $config['wow_pro_package_id'] ?? 0,
            'PRODUCT' => $config['wow_market_id'] ?? 0,
            'WALLET' => $config['wow_wallet_topup_id'] ?? 0,
            'DONATE' => $config['wow_crowdfund_id'] ?? 0,
        ];

        $woo_product_id = $mappings[$transaction_type] ?? 0;

        if ($woo_product_id <= 0) {
            return false;
        }

        return [
            'woo_product_id' => $woo_product_id,
            'transaction_type' => $transaction_type,
        ];
    }

    /**
     * Apply business rules to determine final amount
     */
    private function apply_business_rules($transaction_type, $amount, $currency) {
        // For fixed products (PRO, PRODUCT), get price from WooCommerce
        if (in_array($transaction_type, ['PRO', 'PRODUCT'])) {
            // Price is authoritative from WooCommerce
            // This is handled when creating the order
            return $amount;
        }

        // For dynamic (WALLET, DONATE), apply limits
        $min_limit = get_option('pgb_dynamic_minimum', 1.00);
        $max_limit = get_option('pgb_dynamic_maximum', 5000.00);

        if ($amount < $min_limit || $amount > $max_limit) {
            $amount = max($min_limit, min($amount, $max_limit));
        }

        return round($amount, 2);
    }

    /**
     * Create WooCommerce order
     */
    private function create_woocommerce_order($wp_user_id, $product_id, $amount, $currency, $meta) {
        try {
            $wp_user = get_user_by('id', $wp_user_id);
            if (!$wp_user) {
                return false;
            }

            $order = wc_create_order([
                'customer_id' => $wp_user_id,
                'customer_note' => 'Order from Buzzjuice Streams',
                'created_via' => 'buzjuice-pgb-v4',
            ]);

            if (is_wp_error($order)) {
                return false;
            }

            // Add line item
            $product = wc_get_product($product_id);
            if (!$product) {
                $order->delete();
                return false;
            }

            $order->add_product($product, 1, [
                'subtotal' => $amount,
                'total' => $amount,
            ]);

            // Set currency
            $order->set_currency($currency);

            // Add PGB metadata
            foreach ($meta as $key => $value) {
                $order->add_meta_data('_pgb_' . $key, $value, true);
            }

            // Save
            $order->save();

            $this->logger->info(
                "Created WC order: " . $order->get_id(),
                __FILE__,
                __LINE__
            );

            return $order->get_id();

        } catch (Exception $e) {
            $this->logger->error('WC order creation failed: ' . $e->getMessage(), __FILE__, __LINE__);
            return false;
        }
    }
}

// shared/payment-gateway/core/class-pgb-fulfillment.php
<?php
/**
 * PGB Fulfillment Engine
 * 
 * Handles payment fulfillment:
 * - Updates Streams database
 * - Processes addons (AffiliateWP)
 * - Processes integrations (Jewel)
 * - Manages retries
 */

class PGB_Fulfillment {
    private static $instance = null;
    private $logger = null;
    private $order_mgr = null;
    private $wow_db = null;

    public static function instance(PGB_Logger $logger) {
        if (self::$instance === null) {
            self::$instance = new self($logger);
        }
        return self::$instance;
    }

    private function __construct(PGB_Logger $logger) {
        $this->logger = $logger;
        $this->order_mgr = new PGB_Order($logger);
        $this->wow_db = get_wowonder_db();
    }

    /**
     * Process WooCommerce order (called when payment completes)
     */
    public function process_woocommerce_order($woo_order_id) {
        try {
            $this->logger->info("Fulfillment: WC order $woo_order_id", __FILE__, __LINE__);

            $woo_order = wc_get_order($woo_order_id);
            if (!$woo_order) {
                $this->logger->error("WC order not found: $woo_order_id", __FILE__, __LINE__);
                return;
            }

            // Find matching PGB transaction
            $pgb_order = $this->order_mgr->find_by_woocommerce_id($woo_order_id);
            if (!$pgb_order) {
                $this->logger->warning(
                    "No PGB transaction found for WC order $woo_order_id",
                    __FILE__,
                    __LINE__
                );
                return;
            }

            $pgb_id = $pgb_order['pgb_transaction_id'];

            // Update payment status
            $this->order_mgr->update_status($pgb_id, 'paid');

            // Step 1: Fulfill in Streams
            $streams_ok = $this->fulfill_streams($pgb_order);
            if (!$streams_ok) {
                $this->order_mgr->mark_for_retry(
                    $pgb_id,
                    'Streams fulfillment failed',
                    3600
                );
                return;
            }

            // Step 2: Process addons
            $this->process_addons($pgb_order, $woo_order);

            // Step 3: Process integrations
            $this->process_integrations($pgb_order, $woo_order);

            // Step 4: Mark complete
            $this->order_mgr->update_status($pgb_id, 'paid', 'completed');

            $this->logger->info(
                "Fulfillment complete: $pgb_id",
                __FILE__,
                __LINE__
            );

        } catch (Exception $e) {
            $this->logger->error(
                'Fulfillment exception: ' . $e->getMessage(),
                __FILE__,
                __LINE__
            );
        }
    }

    /**
     * Fulfill in Streams database
     */
    private function fulfill_streams($pgb_order) {
        if (!$this->wow_db) {
            $this->logger->error('No Streams DB connection', __FILE__, __LINE__);
            return false;
        }

        try {
            $transaction_type = $pgb_order['transaction_type'];
            $wow_user_id = $pgb_order['wow_user_id'];
            $wow_order_id = $pgb_order['wow_order_id'];
            $woo_order_id = $pgb_order['woo_order_id'];
            $amount = $pgb_order['final_amount'];

            // Update Wo_Payment_Transactions
            $stmt = $this->wow_db->prepare(
                "UPDATE Wo_Payment_Transactions SET 
                 payment_status = ?, 
                 woo_order_id = ?,
                 transaction_dt = NOW()
                 WHERE order_id = ?"
            );

            if (!$stmt) {
                $this->logger->error('prepare failed: ' . $this->wow_db->error, __FILE__, __LINE__);
                return false;
            }

            $status = 'completed';
            $stmt->bind_param('sss', $status, $woo_order_id, $wow_order_id);
            if (!$stmt->execute()) {
                $this->logger->error('execute failed: ' . $stmt->error, __FILE__, __LINE__);
                $stmt->close();
                return false;
            }
            $stmt->close();

            // Handle transaction-specific fulfillment
            switch ($transaction_type) {
                case 'PRO':
                    $this->fulfill_pro_subscription($wow_user_id);
                    break;
                case 'PRODUCT':
                    $this->fulfill_product_purchase($pgb_order);
                    break;
                case 'WALLET':
                    $this->fulfill_wallet_topup($wow_user_id, $amount);
                    break;
                case 'DONATE':
                    $this->fulfill_donation($pgb_order);
                    break;
            }

            return true;

        } catch (Exception $e) {
            $this->logger->error('Streams fulfillment: ' . $e->getMessage(), __FILE__, __LINE__);
            return false;
        }
    }

    private function fulfill_wallet_topup($wow_user_id, $amount) {
        $stmt = $this->wow_db->prepare(
            "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ?"
        );
        if ($stmt) {
            $stmt->bind_param('di', $amount, $wow_user_id);

            $stmt->bind_param('iidi', $funding_id, $wow_user_id, $amount, $now);
            if ($stmt->execute()) {

            }

    }

    /**
     * Process retry queue (called by WP-Cron)
     */
    public function retry_failed_orders() {
        $retry_queue = $this->order_mgr->get_retry_queue();

        if (empty($retry_queue)) {
            return;
        }

        $this->logger->info("Processing retry queue: " . count($retry_queue) . " orders", __FILE__, __LINE__);

        foreach ($retry_queue as $pgb_order) {
            if ($pgb_order['retry_count'] >= 5) {
                $this->order_mgr->update_status(
                    $pgb_order['pgb_transaction_id'],
                    null,
                    'failed'
                );
                $this->logger->error(
                    "Order abandoned after max retries: " . $pgb_order['pgb_transaction_id'],
                    __FILE__,
                    __LINE__
                );
                continue;
            }

            // Retry fulfillment
            $this->fulfill_streams($pgb_order);
        }
    }

/**

 * This addon only runs AFTER payment is confirmed in WooCommerce

    function pgb_addon_affiliate_wp_process($pgb_order, $woo_order, PGB_Logger $logger, $wow_db = null) {
        if (!class_exists('Affiliate_WP')) {
            $logger->warning('AffiliateWP not installed', __FILE__, __LINE__);
            return;
        }

        try {
            $logger->info(
                "AffiliateWP addon: processing PGB " . $pgb_order['pgb_transaction_id'],
                __FILE__,
                __LINE__
            );

            $woo_order_id = $woo_order->get_id();
            $wow_user_id = $pgb_order['wow_user_id'];

            // Check if referral already exists (idempotency)
            $existing = affwp_get_referrals([
                'reference' => $woo_order_id,
                'context' => 'woocommerce',
            ]);

            if (!empty($existing)) {
                $logger->info("Referral already exists for order $woo_order_id", __FILE__, __LINE__);
                return;
            }

            // Determine affiliate from cookie or tracking
            $affiliate_id = affwp_get_affiliate_id();
            if (empty($affiliate_id)) {
                $logger->warning("No affiliate identified for order $woo_order_id", __FILE__, __LINE__);
                return;
            }

            // Calculate commission based on WooCommerce product
            $commission = affwp_calculate_referral_amount(
                $woo_order->get_total(),
                $affiliate_id,
                $woo_order_id
            );

            if ($commission <= 0) {
                $logger->warning(
                    "No commission calculated for affiliate $affiliate_id",
                    __FILE__,
                    __LINE__
                );
                return;
            }

            // Create referral
            $referral_id = affwp_add_referral([
                'affiliate_id' => $affiliate_id,
                'amount' => $commission,
                'reference' => $woo_order_id,
                'context' => 'woocommerce',
                'custom_referral_id' => 'pgb_' . $pgb_order['pgb_transaction_id'],
            ]);

            if ($referral_id) {
                $logger->info(
                    "Created AffiliateWP referral: $referral_id for affiliate $affiliate_id",
                    __FILE__,
                    __LINE__
                );
            } else {
                $logger->error("Failed to create referral", __FILE__, __LINE__);
            }

        } catch (Exception $e) {
            $logger->error("AffiliateWP addon exception: " . $e->getMessage(), __FILE__, __LINE__);
        }
    }

<?php
/**
 * PGB Integration: Jewel Affiliate
 * 
 * Handles:
 * - Pro subscription activation
 * - Rebate credit application
 * - Role synchronization
 * - WP role mapping
 */

if (!function_exists('pgb_int_jewel_affiliate_process')) {

    function pgb_int_jewel_affiliate_process($pgb_order, $woo_order, PGB_Logger $logger, $wow_db = null) {
        if (!$wow_db) {
            $logger->error('No Streams DB connection for Jewel integration', __FILE__, __LINE__);
            return;
        }

        try {
            $logger->info(
                "Jewel integration: processing PGB " . $pgb_order['pgb_transaction_id'],
                __FILE__,
                __LINE__
            );

            $wow_user_id = $pgb_order['wow_user_id'];
            $wp_user_id = $pgb_order['wp_user_id'];
            $transaction_type = $pgb_order['transaction_type'];

            // Load mapping from WP options
            $mapping = jewel_load_mapping();
            if (empty($mapping)) {
                $logger->debug('No Jewel mapping configured', __FILE__, __LINE__);
                return;
            }

            // Process line items for Jewel variations
            foreach ($woo_order->get_items() as $item) {
                $variation_id = $item->get_variation_id() ?: $item->get_product_id();

                if (!isset($mapping[$variation_id])) {
                    continue;
                }

                $map = $mapping[$variation_id];
                $pro_type = intval($map['wow_pro_type'] ?? 0);

                if ($pro_type > 0) {
                    jewel_activate_pro($wow_db, $wow_user_id, $pro_type, $logger);
                }

                // Apply rebate if configured
                if (!empty($map['rebate_cents'])) {
                    jewel_apply_rebate(
                        $wow_db,
                        $wow_user_id,
                        intval($map['rebate_cents']),
                        $woo_order->get_id(),
                        $logger
                    );
                }

                // Sync WP roles if configured
                if (!empty($map['wp_role'])) {
                    jewel_sync_wp_roles($wp_user_id, $map['wp_role'], $logger);
                }
            }

        } catch (Exception $e) {
            $logger->error("Jewel integration exception: " . $e->getMessage(), __FILE__, __LINE__);
        }
    }

    function jewel_load_mapping() {
        $option = get_option('pgb_jewel_mapping', '');
        if (empty($option)) {
            return [];
        }
        $data = json_decode($option, true);
        $map = [];
        if (is_array($data)) {
            foreach ($data as $row) {
                $vid = intval($row['variation_id'] ?? 0);
                if ($vid > 0) {
                    $map[$vid] = $row;
                }
            }
        }
        return $map;
    }

    function jewel_activate_pro($wow_db, $wow_user_id, $pro_type, PGB_Logger $logger) {
        $now = time();
        $stmt = $wow_db->prepare(
            "UPDATE Wo_Users SET is_pro = 1, pro_type = ?, pro_time = ? WHERE user_id = ?"
        );
        if ($stmt) {
            $stmt->bind_param('iii', $pro_type, $now, $wow_user_id);
            if ($stmt->execute()) {
                $logger->info(
                    "Jewel: Pro type $pro_type activated for user $wow_user_id",
                    __FILE__,
                    __LINE__
                );
            } else {
                $logger->error("Jewel: Failed to activate pro: " . $stmt->error, __FILE__, __LINE__);
            }
            $stmt->close();
        }
    }

    function jewel_apply_rebate($wow_db, $wow_user_id, $rebate_cents, $woo_order_id, PGB_Logger $logger) {
        $amount = round($rebate_cents / 100.0, 2);

        // Idempotency check
        $result = $wow_db->query(
            "SELECT id FROM Wo_Payment_Transactions 
             WHERE userid = $wow_user_id 
             AND woo_order_id = $woo_order_id 
             AND notes LIKE '%Jewel%Rebate%' 
             LIMIT 1"
        );

        if ($result && $result->num_rows > 0) {
            $logger->info("Jewel: Rebate already applied", __FILE__, __LINE__);
            return;
        }

        // Apply rebate
        $stmt = $wow_db->prepare(
            "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ?"
        );
        if ($stmt) {
            $stmt->bind_param('di', $amount, $wow_user_id);
            if ($stmt->execute()) {
                // Log transaction
                $note = "Jewel Affiliate Rebate - WC Order $woo_order_id";
                $insert = $wow_db->prepare(
                    "INSERT INTO Wo_Payment_Transactions 
                    (userid, kind, amount, notes, payment_status, woo_order_id, transaction_dt)
                    VALUES (?, 'RECEIVED', ?, ?, 'completed', ?, NOW())"
                );
                if ($insert) {
                    $kind = 'REBATE';
                    $insert->bind_param('idsi', $wow_user_id, $amount, $note, $woo_order_id);
                    $insert->execute();
                    $insert->close();
                }
                
                $logger->info(
                    "Jewel: Rebate of $amount applied to user $wow_user_id",
                    __FILE__,
                    __LINE__
                );
            }
            $stmt->close();
        }
    }

    function jewel_sync_wp_roles($wp_user_id, $wp_role, PGB_Logger $logger) {

// shared/payment-gateway/js/pgb-js-payment-handler.js
/**
 * PGB Payment Handler - Client-side
 * 
 * Responsibilities:
 * - Handle payment button clicks
 * - Submit order to PGB endpoint
 * - Manage redirect to WooCommerce
 * - Show loading and error states
 */

(function() {
    'use strict';

    window.PGB = {
        config: {
            endpoint: '/wp-json/bzj-pgb/v1/order',
            secret: null  // Never include secret in JS
        },

        /**
         * Initialize payment handler
         */
        init: function(config) {
            Object.assign(this.config, config || {});
            this.bindEvents();
        },

        /**
         * Bind button click events
         */
        bindEvents: function() {
            var self = this;
            
            document.addEventListener('click', function(e) {
                if (e.target.classList.contains('pgb-payment-btn')) {
                    e.preventDefault();
                    self.handlePayment(e.target);
                }
            });
        },

        /**
         * Main payment handler
         */
        handlePayment: function(button) {
            var self = this;

            // Disable button to prevent double-click
            button.disabled = true;
            button.classList.add('loading');

            // Gather order data from data attributes
            var data = {
                wow_user_id: parseInt(button.dataset.userId || 0),
                wow_order_id: button.dataset.orderId || 'order_' + Date.now(),
                transaction_type: (button.dataset.type || 'PURCHASE').toUpperCase(),
                amount: parseFloat(button.dataset.amount || 0),
                currency: button.dataset.currency || 'GHS',
                wow_post_id: parseInt(button.dataset.postId || 0),
            };

            // Validate
            if (!data.wow_user_id || !data.amount) {
                alert('Invalid order data');
                button.disabled = false;
                button.classList.remove('loading');
                return;
            }

            // Submit to PGB
            this.submitOrder(data, button);
        },

        /**
         * Submit order to PGB API
         */
        submitOrder: function(data, button) {
            var self = this;
            var timestamp = Math.floor(Date.now() / 1000).toString();

            // Create a request ID
            var requestId = 'req_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);

            // Prepare payload
            var payload = {
                timestamp: timestamp,
                request_id: requestId,
                data: data
            };

            // Note: In a real implementation, you would compute HMAC-SHA256 on the server,
            // not in the browser. This is just a placeholder fetch.

            fetch(this.config.endpoint, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-BZJ-PGB-Timestamp': timestamp,
                    'X-BZJ-PGB-Request-ID': requestId,
                    // Signature would be computed server-side
                },
                body: JSON.stringify(payload)
            })
            .then(response => response.json())
            .then(result => {
                if (result.status === 'success' && result.payment_url) {
                    // Redirect to WooCommerce checkout
                    window.location.href = result.payment_url;
                } else {
                    alert('Error: ' + (result.message || 'Unknown error'));
                    button.disabled = false;
                    button.classList.remove('loading');
                }
            })
            .catch(error => {
                console.error('Payment submission error:', error);
                alert('Payment submission failed: ' + error.message);
                button.disabled = false;
                button.classList.remove('loading');
            });
        }
    };

    // Auto-initialize on DOM ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', function() {
            window.PGB.init();
        });
    } else {
        window.PGB.init();
    }
})();

// streams/assets/wow-pgb/wow-pgb_init.php (REPLACEMENT)
<?php
/**
 * Streams PGB Client v4
 * 
 * THIN CLIENT - Only responsibilities:
 * - Receive payment request from frontend
 * - Validate user authentication
 * - Extract order parameters
 * - Send to WordPress PGB API
 * - Return payment URL
 */

require_once dirname(__DIR__, 2) . '/config.php';
require_once dirname(__DIR__) . '/init.php';

header('Content-Type: application/json');

try {
    // 1. Verify user is authenticated
    if ($wo['loggedin'] == false) {
        http_response_code(401);
        echo json_encode(['status' => 'error', 'message' => 'Not authenticated']);
        exit();
    }

    // 2. Verify request method
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        http_response_code(405);
        echo json_encode(['status' => 'error', 'message' => 'Method not allowed']);
        exit();
    }

    // 3. Extract parameters
    $wow_user_id = intval($wo['user']['user_id'] ?? 0);
    $transaction_type = strtoupper($_POST['transaction_type'] ?? 'PURCHASE');
    $amount = floatval($_POST['amount'] ?? 0);
    $currency = strtoupper($_POST['currency'] ?? 'GHS');
    $wow_post_id = intval($_POST['wow_post_id'] ?? 0);

    // Generate WoWonder order ID
    $wow_order_id = 'wow_' . uniqid();

    // 4. Build request to WordPress PGB
    $request_data = [
        'wow_user_id' => $wow_user_id,
        'wow_order_id' => $wow_order_id,
        'transaction_type' => $transaction_type,
        'amount' => $amount,
        'currency' => $currency,
        'wow_post_id' => $wow_post_id,
    ];

    $request_id = 'req_' . uniqid();
    $timestamp = time();

    // Compute HMAC-SHA256
    $secret = getenv('BZJ_PGB_SECRET');
    if (empty($secret)) {
        http_response_code(500);
        echo json_encode(['status' => 'error', 'message' => 'Server configuration error']);
        exit();
    }

    $request_body = json_encode($request_data);
    $signature = hash_hmac('sha256', $timestamp . "\n" . $request_body, $secret);

    // 5. Call WordPress PGB endpoint
    $wp_url = getenv('WP_BASE_SITE_URL') ?: 'https://buzzjuice.net';
    $api_url = rtrim($wp_url, '/') . '/wp-json/bzj-pgb/v1/order';

    $response = wp_remote_post($api_url, [
        'method' => 'POST',
        'headers' => [
            'Content-Type' => 'application/json',
            'X-BZJ-PGB-Timestamp' => $timestamp,
            'X-BZJ-PGB-Request-ID' => $request_id,
            'X-BZJ-PGB-Signature' => $signature,
        ],
        'body' => $request_body,
        'timeout' => 30,
    ]);

    if (is_wp_error($response)) {
        http_response_code(500);
        echo json_encode(['status' => 'error', 'message' => $response->get_error_message()]);
        exit();
    }

    $body = wp_remote_retrieve_body($response);
    $result = json_decode($body, true);

    if (!is_array($result)) {
        http_response_code(500);
        echo json_encode(['status' => 'error', 'message' => 'Invalid response from gateway']);
        exit();
    }

    // 6. Log in Streams database
    if (!empty($result['woo_order_id'])) {
        global $sqlConnect;
        $stmt = $sqlConnect->prepare(
            "INSERT INTO Wo_Payment_Transactions 
            (userid, amount, order_id, kind, currency_code, payment_status, woo_order_id, transaction_dt)
            VALUES (?, ?, ?, ?, ?, ?, ?, NOW())"
        );
        if ($stmt) {
            $kind = 'INITIATED';
            $status = 'pending';
            $stmt->bind_param('idssssS', $wow_user_id, $amount, $wow_order_id, $kind, $currency, $status, $result['woo_order_id']);
            $stmt->execute();
            $stmt->close();
        }
    }

    // 7. Return payment URL
    http_response_code(200);
    echo json_encode($result);

} catch (Exception $e) {
    http_response_code(500);
    echo json_encode(['status' => 'error', 'message' => $e->getMessage()]);
}

## **KEY ARCHITECTURAL IMPROVEMENTS**

| Issue | Old Implementation | v4 Solution |
|-------|-------------------|------------|
| **Payment Completion Depends On** | Browser returning to site | WooCommerce server-side events |
| **Order State Authority** | Scattered across Streams, WC, browser | Centralized in PGB ledger |
| **Duplicate Orders** | Possible (double-click, retries) | Prevented by idempotency key |
| **Configuration Sources** | Multiple competing sources | Single source: Streams admin reads by WordPress |
| **Failure Recovery** | Manual intervention | Automatic retry with backoff |
| **Addon Integration** | Direct manipulation of Streams DB | Modular addon API |
| **Logging** | Single large file | Component-level with redaction |
| **API Credentials Exposure** | Stored in Streams HTML/JS | Never exposed to client |
