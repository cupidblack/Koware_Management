```php
<?php
/**
 * Plugin Name: Buzzjuice Payment Gateway Bridge Engine
 * Plugin URI: https://buzzjuice.net
 * Description: Central orchestrator for WooCommerce & Streams payment integration
 * Version: 3.0.0
 * Author: Buzzjuice Development Team
 * License: Proprietary
 * Text Domain: pgb-engine
 *
 * This must-use plugin serves as the core engine orchestrating all payment
 * gateway interactions between Streams (WoWonder) and WooCommerce, with support
 * for addons and integrations.
 */

// Prevent direct access
if (!defined('ABSPATH')) {
    exit('Direct access prohibited');
}

// Define PGB constants
if (!defined('PGB_VERSION')) define('PGB_VERSION', '3.0.0');
if (!defined('PGB_PATH')) define('PGB_PATH', dirname(__FILE__) . '/../../../shared/payment-gateway');
if (!defined('PGB_ADDONS_PATH')) define('PGB_ADDONS_PATH', PGB_PATH . '/addons');
if (!defined('PGB_INTEGRATIONS_PATH')) define('PGB_INTEGRATIONS_PATH', PGB_PATH . '/integrations');
if (!defined('PGB_LOG_PATH')) define('PGB_LOG_PATH', PGB_PATH . '/log');

// Prevent double-loading
if (class_exists('PGB_Engine')) {
    return;
}

/**
 * Main PGB Engine class
 */
class PGB_Engine {
    
    private static $instance = null;
    private $client = null;
    private $logger = null;
    private $config = null;
    private $addons = [];
    private $integrations = [];
    private $enabled = true;
    
    /**
     * Get singleton instance
     */
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    /**
     * Initialize engine
     */
    private function __construct() {
        $this->init_paths();
        $this->init_files();
        $this->setup_hooks();
    }
    
    /**
     * Verify required paths exist
     */
    private function init_paths() {
        if (!is_dir(PGB_PATH)) {
            wp_die('PGB: Core path not found: ' . PGB_PATH);
        }
        // Create log directory if missing
        if (!is_dir(PGB_LOG_PATH)) {
            wp_mkdir_p(PGB_LOG_PATH);
        }
    }
    
    /**
     * Load core PGB files
     */
    private function init_files() {
        $required_files = [
            'pgb-config.php',
            'pgb-logger.php',
            'pgb-validator.php',
            'pgb-processor.php',
            'pgb-state-manager.php',
            'pgb-hooks.php',
            'pgb-client.php',
            'pgb-client-streams.php',
        ];
        
        foreach ($required_files as $file) {
            $path = PGB_PATH . '/' . $file;
            if (!file_exists($path)) {
                error_log("PGB: Required file missing: {$file}");
                $this->enabled = false;
                return;
            }
            require_once $path;
        }
        
        // Initialize core components
        $this->config = PGB_Config::get_instance();
        $this->logger = PGB_Logger::get_instance();
        $this->client = PGB_Client::get_instance();
        
        $this->logger->info('PGB Engine initialized', 'pgb-engine');
    }
    
    /**
     * Setup WordPress hooks
     */
    private function setup_hooks() {
        if (!$this->enabled) return;
        
        // WooCommerce Order Completed hook
        add_action('woocommerce_order_status_completed', [$this, 'on_order_completed'], 10, 1);
        
        // WordPress init - load addons & integrations
        add_action('wp_loaded', [$this, 'load_addons_and_integrations'], 5);
        
        // Streams integration hooks (if available)
        if (function_exists('do_action')) {
            add_action('pgb_order_initialized', [$this, 'log_order_event']);
            add_action('pgb_payment_completed', [$this, 'finalize_payment']);
            add_action('pgb_payment_failed', [$this, 'handle_payment_failure']);
        }
        
        // Admin menu (if user is admin)
        if (is_admin()) {
            add_action('admin_menu', [$this, 'register_admin_menu'], 25);
        }
    }
    
    /**
     * Handler for WooCommerce order completion
     */
    public function on_order_completed($order_id) {
        if (!$this->enabled || !$this->client) {
            return;
        }
        
        try {
            $order = wc_get_order($order_id);
            if (!$order) {
                return;
            }
            
            // Delegate to PGB Client
            $this->client->process_woocommerce_order($order);
            
        } catch (Exception $e) {
            $this->logger->error('Order completion hook failed: ' . $e->getMessage(), 'pgb-engine');
        }
    }
    
    /**
     * Load addon and integration files
     */
    public function load_addons_and_integrations() {
        if (!$this->enabled) return;
        
        // Load enabled addons
        $this->load_directory(PGB_ADDONS_PATH, 'addon');
        
        // Load enabled integrations
        $this->load_directory(PGB_INTEGRATIONS_PATH, 'integration');
        
        do_action('pgb_addons_loaded');
    }
    
    /**
     * Scan and load PHP files from directory
     */
    private function load_directory($dir, $type) {
        if (!is_dir($dir)) {
            return;
        }
        
        $files = glob($dir . '/*.php');
        foreach ($files as $file) {
            // Skip if filename starts with underscore (disabled marker)
            if (basename($file)[0] === '_') {
                continue;
            }
            
            try {
                require_once $file;
                $this->logger->debug("Loaded {$type}: " . basename($file), 'pgb-engine');
            } catch (Exception $e) {
                $this->logger->error("Failed to load {$type} " . basename($file) . ': ' . $e->getMessage(), 'pgb-engine');
            }
        }
    }
    
    /**
     * Register admin menu (Blue Crown integration)
     */
    public function register_admin_menu() {
        // Check if Blue Crown parent menu exists
        $has_blue_crown = false;
        global $submenu;
        foreach ($submenu as $parent => $items) {
            if (strpos($parent, 'blue-crown') !== false || strpos($parent, 'Blue Crown') !== false) {
                $has_blue_crown = true;
                break;
            }
        }
        
        if ($has_blue_crown) {
            add_submenu_page(
                'blue-crown',
                'Payment Gateway',
                'Payment Gateway',
                'manage_options',
                'pgb-dashboard',
                [$this, 'render_admin_page']
            );
        }
    }
    
    /**
     * Render admin dashboard page
     */
    public function render_admin_page() {
        require_once PGB_PATH . '/admin/pgb-dashboard.php';
    }
    
    /**
     * Log order event for auditing
     */
    public function log_order_event($data) {
        $this->logger->info('Order event: ' . json_encode($data), 'pgb-order-events');
    }
    
    /**
     * Finalize payment processing
     */
    public function finalize_payment($data) {
        // Hook for payment finalization
        do_action('pgb_payment_finalize_internal', $data);
    }
    
    /**
     * Handle payment failures
     */
    public function handle_payment_failure($data) {
        $this->logger->warning('Payment failed: ' . json_encode($data), 'pgb-payment-failures');
        do_action('pgb_payment_failure_internal', $data);
    }
    
    /**
     * Get enabled status
     */
    public function is_enabled() {
        return $this->enabled;
    }
    
    /**
     * Get logger instance
     */
    public function get_logger() {
        return $this->logger;
    }
    
    /**
     * Get client instance
     */
    public function get_client() {
        return $this->client;
    }
}

// Initialize engine on WordPress plugins_loaded hook
add_action('plugins_loaded', function() {
    PGB_Engine::get_instance();
}, 1);

 * Global helper to access PGB Engine
 */

    return PGB_Engine::get_instance();
}

 * Global helper to access logger
 */
function pgb_log($message, $level = 'info', $file = 'pgb-global') {
    $engine = pgb_get_engine();
    if ($engine && $engine->is_enabled()) {
        $logger = $engine->get_logger();
        if ($logger) {
            $logger->log($message, $level, $file);
        }
    }
}
```

```php
<?php
/**
 * shared/payment-gateway/pgb-config.php
 * 
 * Configuration management for Payment Gateway Bridge.
 * Loads settings from WP options, Streams admin config, environment variables.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Config {
    
    private static $instance = null;
    private $config = [];
    private $streams_config = null;
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->load_config();
    }
    
    /**
     * Load configuration from multiple sources with priority:
     * 1. Streams admin settings (highest priority)
     * 2. WP options
     * 3. Environment variables
     */
    private function load_config() {
        // Load Streams config if available
        $this->load_streams_config();
        
        // Load WP options
        $this->load_wp_options();
        
        // Load environment variables
        $this->load_env_variables();
        
        // Set defaults
        $this->set_defaults();
    }
    
    /**
     * Load Streams (WoWonder) configuration
     */
    private function load_streams_config() {
        // Try to load from Streams config file
        $streams_config_path = defined('ABSPATH') ? 
            trailingslashit(dirname(ABSPATH, 2)) . 'streams/config.php' : 
            null;
        
        if ($streams_config_path && file_exists($streams_config_path)) {
            require_once $streams_config_path;
            // After including, $wo global should be available
            if (isset($GLOBALS['wo']) && is_array($GLOBALS['wo'])) {
                $this->streams_config = $GLOBALS['wo'];
                $this->config['streams'] = $this->streams_config;
            }
        }
    }
    
    /**
     * Load WP options
     */
    private function load_wp_options() {
        if (!function_exists('get_option')) {
            return;
        }
        
        $pgb_options = get_option('pgb_settings', []);
        if (is_array($pgb_options)) {
            $this->config['wp_options'] = $pgb_options;
        }
        
        // WooCommerce API settings
        $this->config['woo_api_key'] = get_option('woocommerce_api_key');
        $this->config['woo_api_secret'] = get_option('woocommerce_api_secret');
    }
    
    /**
     * Load environment variables
     */
    private function load_env_variables() {
        $env_keys = [
            'PGB_ENABLED',
            'PGB_DEBUG_MODE',
            'PGB_LOG_LEVEL',
            'WOO_API_URL',
            'WOO_API_KEY',
            'WOO_API_SECRET',
            'AFFILIATE_WP_ENABLED',
        ];
        
        foreach ($env_keys as $key) {
            $value = getenv($key);
            if ($value !== false) {
                $this->config['env_' . strtolower($key)] = $value;
            }
        }
    }
    
    /**
     * Set configuration defaults
     */
    private function set_defaults() {
        $defaults = [
            'pgb_enabled' => true,
            'debug_mode' => defined('WP_DEBUG') && WP_DEBUG,
            'log_level' => 'info', // debug, info, warning, error
            'log_retention_days' => 30,
            'webhook_signature_verify' => true,
            'idempotency_check' => true,
            'max_retries' => 3,
            'retry_delay_seconds' => 5,
        ];
        
        foreach ($defaults as $key => $value) {
            if (!isset($this->config[$key])) {
                $this->config[$key] = $value;
            }
        }
    }
    
    /**
     * Get configuration value by dot notation
     * Example: get('streams.wow_wallet_topup_id')
     */
    public function get($key, $default = null) {
        $keys = explode('.', $key);
        $value = $this->config;
        
        foreach ($keys as $k) {
            if (is_array($value) && isset($value[$k])) {
                $value = $value[$k];
            } else {
                return $default;
            }
        }
        
        return $value;
    }
    
    /**
     * Set configuration value
     */
    public function set($key, $value) {
        $keys = explode('.', $key);
        $config = &$this->config;
        
        foreach ($keys as $k) {
            if (!isset($config[$k])) {
                $config[$k] = [];
            }
            $config = &$config[$k];
        }
        
        $config = $value;
    }
    
    /**
     * Get all configuration
     */
    public function get_all() {
        return $this->config;
    }
    
    /**
     * Check if PGB is enabled
     */
    public function is_enabled() {
        return (bool) $this->get('pgb_enabled', true);
    }
    
    /**
     * Get debug mode status
     */
    public function debug_mode() {
        return (bool) $this->get('debug_mode', false);
    }
    
    /**
     * Get log level
     */
    public function log_level() {
        return strtolower($this->get('log_level', 'info'));
    }
}
```

## **3. LOGGING: pgb-logger.php**

```php
<?php
/**
 * shared/payment-gateway/pgb-logger.php
 * 
 * Comprehensive logging system with file rotation and level management.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Logger {
    
    private static $instance = null;
    private $config = null;
    private $log_levels = [
        'debug' => 0,
        'info' => 1,
        'warning' => 2,
        'error' => 3,
    ];
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->config = PGB_Config::get_instance();
    }
    
    /**
     * Main logging method
     */
    public function log($message, $level = 'info', $file = 'pgb-general') {
        $level = strtolower($level);
        
        // Check if this level should be logged
        $config_level = $this->log_levels[$this->config->log_level()] ?? 1;
        $message_level = $this->log_levels[$level] ?? 1;
        
        if ($message_level < $config_level) {
            return false;
        }
        
        $timestamp = current_time('mysql');
        $log_entry = sprintf(
            "[%s] [%s] %s\n",
            $timestamp,
            strtoupper($level),
            $this->sanitize_message($message)
        );
        
        return $this->write_log($log_entry, $file);
    }
    
    /**
     * Log debug message
     */
    public function debug($message, $file = 'pgb-general') {
        return $this->log($message, 'debug', $file);
    }
    
    /**
     * Log info message
     */
    public function info($message, $file = 'pgb-general') {
        return $this->log($message, 'info', $file);
    }
    
    /**
     * Log warning message
     */
    public function warning($message, $file = 'pgb-general') {
        return $this->log($message, 'warning', $file);
    }
    
    /**
     * Log error message
     */
    public function error($message, $file = 'pgb-general') {
        return $this->log($message, 'error', $file);
    }
    
    /**
     * Sanitize log message (remove sensitive data)
     */
    private function sanitize_message($message) {
        // Remove API keys, secrets, tokens
        $patterns = [
            '/api[_-]?key[\'"]?\s*[:=]\s*[\'"]?[a-zA-Z0-9_\-]+/i',
            '/secret[\'"]?\s*[:=]\s*[\'"]?[a-zA-Z0-9_\-]+/i',
            '/password[\'"]?\s*[:=]\s*[\'"]?[^"\']+/i',
            '/token[\'"]?\s*[:=]\s*[\'"]?[a-zA-Z0-9_\-]+/i',
        ];
        
        $message = preg_replace($patterns, '***REDACTED***', $message);
        
        return $message;
    }
    
    /**
     * Write log entry to file
     */
    private function write_log($entry, $file = 'pgb-general') {
        $log_dir = PGB_LOG_PATH;
        if (!is_dir($log_dir)) {
            wp_mkdir_p($log_dir);
        }
        
        // Ensure file name is safe
        $file = preg_replace('/[^a-z0-9_\-]/', '', strtolower($file));
        $file = $file ?: 'pgb-general';
        
        $log_file = $log_dir . '/' . $file . '.log';
        
        // Check file size and rotate if needed
        if (file_exists($log_file) && filesize($log_file) > 10 * 1024 * 1024) { // 10MB
            $this->rotate_log($log_file);
        }
        
        return error_log($entry, 3, $log_file);
    }
    
    /**
     * Rotate log file when size exceeds limit
     */
    private function rotate_log($file) {
        $timestamp = date('Y-m-d_H-i-s');
        $new_name = $file . '.' . $timestamp;
        
        if (rename($file, $new_name)) {
            // Optionally compress old log
            if (function_exists('gzcompress')) {
                $content = file_get_contents($new_name);
                if ($content !== false) {
                    file_put_contents($new_name . '.gz', gzcompress($content, 9));
                    unlink($new_name);
                }
            }
            
            // Clean old logs (older than retention period)
            $this->cleanup_old_logs(dirname($file));
        }
    }
    
    /**
     * Remove logs older than retention period
     */
    private function cleanup_old_logs($dir) {
        $retention_days = $this->config->get('log_retention_days', 30);
        $cutoff_time = time() - ($retention_days * 24 * 60 * 60);
        
        $files = glob($dir . '/*.log*');
        foreach ($files as $file) {
            if (filemtime($file) < $cutoff_time) {
                @unlink($file);
            }
        }
    }
    
    /**
     * Get all log files
     */
    public function get_logs($pattern = '*.log') {
        $files = glob(PGB_LOG_PATH . '/' . $pattern);
        return array_map('basename', $files);
    }
    
    /**
     * Read log file contents
     */
    public function read_log($file, $lines = 100) {
        $log_file = PGB_LOG_PATH . '/' . basename($file);
        if (!file_exists($log_file)) {
            return '';
        }
        
        $handle = fopen($log_file, 'r');
        if (!$handle) {
            return '';
        }
        
        // Get last N lines
        $file_size = filesize($log_file);
        $buffer_size = 8192;
        $output = '';
        $line_count = 0;
        
        fseek($handle, 0, SEEK_END);
        while ($line_count < $lines && ftell($handle) > 0) {
            $pos = ftell($handle);
            $read_size = min($buffer_size, $pos);
            $pos -= $read_size;
            fseek($handle, $pos);
            $chunk = fread($handle, $read_size);
            $output = $chunk . $output;
            $line_count = substr_count($output, "\n");
        }
        
        fclose($handle);
        
        $lines_array = explode("\n", $output);
        return implode("\n", array_slice($lines_array, -$lines));
    }
}
```

```php
<?php
/**
 * shared/payment-gateway/pgb-validator.php
 * 
 * Input validation and sanitization for orders and transactions.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Validator {
    
    /**
     * Validate email address
     */
    public static function validate_email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL);
    }
    
    /**
     * Validate and sanitize email
     */
    public static function sanitize_email($email) {
        $email = trim($email);
        if (self::validate_email($email)) {
            return sanitize_email($email);
        }
        return '';
    }
    
    /**
     * Validate numeric value (price, amount, etc.)
     */
    public static function validate_numeric($value, $min = 0, $max = PHP_FLOAT_MAX) {
        $value = floatval($value);
        return $value >= $min && $value <= $max;
    }
    
    /**
     * Validate integer
     */
    public static function validate_integer($value, $min = 0, $max = PHP_INT_MAX) {
        $value = intval($value);
        return $value >= $min && $value <= $max;
    }
    
    /**
     * Validate URL
     */
    public static function validate_url($url) {
        return filter_var($url, FILTER_VALIDATE_URL) !== false;
    }
    
    /**
     * Validate WooCommerce order object
     */
    public static function validate_woo_order($order) {
        if (!is_object($order) || !method_exists($order, 'get_id')) {
            return false;
        }
        return $order->get_id() > 0;
    }
    
    /**
     * Sanitize string for database/output
     */
    public static function sanitize_string($value, $allow_html = false) {
        $value = trim($value);
        if ($allow_html) {
            return wp_kses_post($value);
        }
        return sanitize_text_field($value);
    }
    
    /**
     * Validate WooCommerce webhook signature
     */
    public static function verify_webhook_signature($payload, $signature, $secret) {
        $expected = base64_encode(hash_hmac('sha256', $payload, $secret, true));
        return hash_equals($signature, $expected);
    }
    
    /**
     * Validate required fields in array
     */
    public static function validate_required_fields(array $data, array $required_fields) {
        $missing = [];
        foreach ($required_fields as $field) {
            if (empty($data[$field])) {
                $missing[] = $field;
            }
        }
        return empty($missing) ? true : $missing;
    }
    
    /**
     * Build validation result object
     */
    public static function result($is_valid, $message = '', $data = []) {
        return [
            'valid' => (bool) $is_valid,
            'message' => $message,
            'data' => $data,
        ];
    }
}
```

```php
if (!defined('ABSPATH') && !defined('PGB_PATH')) {

}

class PGB_State_Manager {
    
    private static $instance = null;
    private $db = null;
    private $logger = null;
    private $table_name = 'pgb_transaction_states';
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        global $wpdb;
        $this->db = $wpdb;
        $this->logger = PGB_Logger::get_instance();

    /**
     * Ensure state tracking table exists
     */
    private function ensure_table() {
        $table = $this->db->prefix . $this->table_name;
        
        if ($this->db->get_var("SHOW TABLES LIKE '$table'") !== $table) {
            $charset_collate = $this->db->get_charset_collate();
            
            $sql = "CREATE TABLE {$table} (
                id bigint(20) NOT NULL AUTO_INCREMENT PRIMARY KEY,
                transaction_type varchar(50) NOT NULL,
                order_id bigint(20),
                woo_order_id bigint(20),
                wow_user_id bigint(20),
                wow_order_id varchar(100),
                state varchar(50) NOT NULL DEFAULT 'pending',
                idempotency_key varchar(255) UNIQUE,
                data longtext,
                created_at datetime DEFAULT CURRENT_TIMESTAMP,
                updated_at datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
                KEY (transaction_type, state),
                KEY (woo_order_id),
                KEY (wow_user_id),
                KEY (created_at)
            ) $charset_collate;";
            
            require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
            dbDelta($sql);
            
            $this->logger->info('State tracking table created', 'pgb-state-manager');
        }
    }
    
    /**
     * Store transaction state
     */
    public function store_state($type, $state, $data = [], $idempotency_key = null) {
        $table = $this->db->prefix . $this->table_name;
        
        $insert = [
            'transaction_type' => $type,
            'state' => $state,
            'data' => json_encode($data),
            'idempotency_key' => $idempotency_key,
        ];
        
        $result = $this->db->insert($table, $insert);
        
        if ($result) {
            return $this->db->insert_id;
        }
        
        $this->logger->error("Failed to store state for type={$type}: " . $this->db->last_error, 'pgb-state-manager');
        return false;
    }
    
    /**
     * Update transaction state
     */
    public function update_state($id, $state, $data = []) {
        $table = $this->db->prefix . $this->table_name;
        
        $update = [
            'state' => $state,
        ];
        
        if (!empty($data)) {
            $update['data'] = json_encode($data);
        }
        
        $result = $this->db->update($table, $update, ['id' => $id]);
        
        if ($result === false) {
            $this->logger->error("Failed to update state id={$id}: " . $this->db->last_error, 'pgb-state-manager');
            return false;
        }
        
        return true;
    }
    
    /**
     * Get state by idempotency key
     */
    public function get_by_idempotency_key($key) {
        $table = $this->db->prefix . $this->table_name;
        $result = $this->db->get_row(
            $this->db->prepare("SELECT * FROM {$table} WHERE idempotency_key = %s", $key)
        );
        
        return $result;
    }
    
    /**
     * Get state by transaction ID
     */
    public function get_by_id($id) {
        $table = $this->db->prefix . $this->table_name;
        $result = $this->db->get_row(
            $this->db->prepare("SELECT * FROM {$table} WHERE id = %d", $id)
        );
        
        if ($result && !empty($result->data)) {
            $result->data = json_decode($result->data, true);
        }
        
        return $result;
    }
    
    /**
     * Check if transaction already processed (idempotency)
     */
    public function is_idempotent($key) {
        $state = $this->get_by_idempotency_key($key);
        return $state && in_array($state->state, ['completed', 'failed']);
    }
}
```

```php
<?php
/**
 * shared/payment-gateway/pgb-processor.php
 * 
 * Core order processing logic and WooCommerce API integration.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Processor {
    
    private static $instance = null;
    private $logger = null;
    private $config = null;
    private $validator = null;
    private $state_manager = null;
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->logger = PGB_Logger::get_instance();
        $this->config = PGB_Config::get_instance();
        $this->validator = PGB_Validator::class;
        $this->state_manager = PGB_State_Manager::get_instance();
    }
    
    /**
     * Process WooCommerce order and sync with Streams
     */
    public function process_order($order) {
        try {
            // Validate order
            if (!call_user_func([$this->validator, 'validate_woo_order'], $order)) {
                throw new Exception('Invalid WooCommerce order object');
            }
            
            $order_id = $order->get_id();
            $this->logger->info("Processing WooCommerce order {$order_id}", 'pgb-processor');
            
            // Generate idempotency key
            $idempotency_key = $this->generate_idempotency_key($order);
            
            // Check if already processed
            if ($this->state_manager->is_idempotent($idempotency_key)) {
                $this->logger->warning("Duplicate order processing detected: {$order_id}", 'pgb-processor');
                return ['success' => false, 'message' => 'Order already processed'];
            }
            
            // Store initial state
            $state_id = $this->state_manager->store_state(
                'woo_order',
                'processing',
                ['order_id' => $order_id],
                $idempotency_key
            );
            
            // Extract order data
            $order_data = $this->extract_order_data($order);
            
            // Process with Streams
            $result = apply_filters('pgb_process_order_data', $order_data, $order);
            
            // Update state based on result
            if ($result && isset($result['success']) && $result['success']) {
                $this->state_manager->update_state($state_id, 'completed', $result);
                $this->logger->info("Order {$order_id} processed successfully", 'pgb-processor');
                return ['success' => true, 'data' => $result];
            } else {
                $this->state_manager->update_state($state_id, 'failed', $result);
                $this->logger->error("Order {$order_id} processing failed", 'pgb-processor');
                return ['success' => false, 'message' => $result['message'] ?? 'Processing failed'];
            }
            
        } catch (Exception $e) {
            $this->logger->error("Exception processing order: " . $e->getMessage(), 'pgb-processor');
            return ['success' => false, 'message' => $e->getMessage()];
        }
    }
    
    /**
     * Extract relevant data from WooCommerce order
     */
    private function extract_order_data($order) {
        $data = [
            'woo_order_id' => $order->get_id(),
            'order_status' => $order->get_status(),
            'customer_email' => $order->get_billing_email(),
            'customer_id' => $order->get_customer_id(),
            'total_amount' => $order->get_total(),
            'currency' => $order->get_currency(),
            'payment_method' => $order->get_payment_method(),
            'line_items' => [],
            'meta_data' => [],
        ];
        
        // Extract line items
        foreach ($order->get_items() as $item) {
            $data['line_items'][] = [
                'product_id' => $item->get_product_id(),
                'variation_id' => $item->get_variation_id(),
                'name' => $item->get_name(),
                'quantity' => $item->get_quantity(),
                'total' => $item->get_total(),
                'meta_data' => $item->get_meta_data(),
            ];
        }
        
        // Extract order meta
        foreach ($order->get_meta_data() as $meta) {
            $data['meta_data'][] = [
                'key' => $meta->key,
                'value' => $meta->value,
            ];
        }
        
        return $data;
    }
    
    /**
     * Generate idempotency key for order
     */
    private function generate_idempotency_key($order) {
        return hash(
            'sha256',
            'pgb_order_' . $order->get_id() . '_' . $order->get_date_created()->getTimestamp()
        );
    }
    
    /**
     * Sync completed order with Streams database
     */
    public function sync_order_to_streams($order_data) {
        try {
            // Get Streams database connection
            if (!function_exists('get_wowonder_db')) {
                require_once dirname(__DIR__) . '/shared/db_helpers.php';
            }
            
            $streams_db = call_user_func('get_wowonder_db');
            if (!$streams_db) {
                throw new Exception('Cannot connect to Streams database');
            }
            
            // Find or create Streams user
            $wow_user_id = $this->resolve_streams_user($order_data, $streams_db);
            if (!$wow_user_id) {
                throw new Exception('Cannot resolve Streams user');
            }
            
            $order_data['wow_user_id'] = $wow_user_id;
            
            // Update Streams payment transactions
            $this->update_streams_payment_transaction($order_data, $streams_db);
            
            // Trigger order-specific processing (product purchase, wallet topup, etc.)
            do_action('pgb_streams_order_synced', $order_data, $streams_db);
            
            return true;
            
        } catch (Exception $e) {
            $this->logger->error('Streams sync failed: ' . $e->getMessage(), 'pgb-processor');
            return false;
        }
    }
    
    /**
     * Find WoWonder user from order
     */
    private function resolve_streams_user($order_data, $db) {
        $email = $order_data['customer_email'] ?? '';
        if (empty($email)) {
            return null;
        }
        
        $stmt = $db->prepare("SELECT user_id FROM Wo_Users WHERE email = ? LIMIT 1");
        if (!$stmt) {
            $this->logger->error('prepare failed: ' . $db->error, 'pgb-processor');
            return null;
        }
        
        $stmt->bind_param('s', $email);
        if (!$stmt->execute()) {
            $this->logger->error('execute failed: ' . $stmt->error, 'pgb-processor');
            $stmt->close();
            return null;
        }
        
        $stmt->bind_result($user_id);
        $result = $stmt->fetch();
        $stmt->close();
        
        return $result ? $user_id : null;
    }
    
    /**
     * Update Streams payment transaction record
     */
    private function update_streams_payment_transaction($order_data, $db) {
        $wow_user_id = $order_data['wow_user_id'];
        $woo_order_id = $order_data['woo_order_id'];
        $amount = $order_data['total_amount'];
        
        // Update existing or insert new transaction
        $sql = "INSERT INTO Wo_Payment_Transactions 
                (userid, woo_order_id, amount, payment_status, payment_method, transaction_dt)
                VALUES (?, ?, ?, ?, ?, NOW())
                ON DUPLICATE KEY UPDATE
                payment_status = VALUES(payment_status),
                payment_method = VALUES(payment_method),
                transaction_dt = NOW()";
        
        $stmt = $db->prepare($sql);
        if (!$stmt) {
            throw new Exception('prepare failed: ' . $db->error);
        }
        
        $status = 'completed';
        $method = $order_data['payment_method'] ?? 'woocommerce';
        
        $stmt->bind_param('idiss', $wow_user_id, $woo_order_id, $amount, $status, $method);
        
        if (!$stmt->execute()) {
            throw new Exception('execute failed: ' . $stmt->error);
        }
        
        $stmt->close();
    }
}
```

```php
<?php
/**
 * shared/payment-gateway/pgb-client.php
 * 
 * Main PGB Client - orchestrates payment processing flow.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Client {
    
    private static $instance = null;
    private $logger = null;
    private $config = null;
    private $processor = null;
    private $validator = null;
    private $state_manager = null;
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->logger = PGB_Logger::get_instance();
        $this->config = PGB_Config::get_instance();
        $this->processor = PGB_Processor::get_instance();
        $this->validator = PGB_Validator::class;
        $this->state_manager = PGB_State_Manager::get_instance();
    }
    
    /**
     * Main entry point: Process WooCommerce order
     */
    public function process_woocommerce_order($order) {
        try {
            if (!$this->config->is_enabled()) {
                $this->logger->warning('PGB is disabled', 'pgb-client');
                return false;
            }
            
            $order_id = $order->get_id();
            $this->logger->info("Client: Processing WooCommerce order {$order_id}", 'pgb-client');
            
            // Validate order
            if (!apply_filters('pgb_validate_order', true, $order)) {
                throw new Exception('Order failed custom validation');
            }
            
            // Process order through processor
            $result = $this->processor->process_order($order);
            
            if ($result['success']) {
                // Sync to Streams
                $order_data = $this->processor->extract_order_data($order);
                $this->processor->sync_order_to_streams($order_data);
                
                // Fire completion hook
                do_action('pgb_payment_completed', $result, $order_id);
                
                return true;
            } else {
                do_action('pgb_payment_failed', $result, $order_id);
                return false;
            }
            
        } catch (Exception $e) {
            $this->logger->error('Client processing error: ' . $e->getMessage(), 'pgb-client');
            return false;
        }
    }
    
    /**
     * Process Streams order initialization (called from Streams frontend)
     */
    public function init_streams_order($streams_order_data) {
        try {
            $this->logger->info('Client: Initializing Streams order', 'pgb-client-streams');
            
            // Validate input
            $validation = call_user_func(
                [$this->validator, 'validate_required_fields'],
                $streams_order_data,
                ['amount', 'product_name', 'product_price', 'wow_currency_code']
            );
            
            if ($validation !== true) {
                throw new Exception('Missing required fields: ' . implode(', ', $validation));
            }
            
            // Sanitize input
            $sanitized = $this->sanitize_streams_order($streams_order_data);
            
            // Store state
            $state_id = $this->state_manager->store_state(
                'streams_order',
                'initialized',
                $sanitized
            );
            
            // Fire initialization hook
            do_action('pgb_order_initialized', $sanitized);
            
            // Prepare WooCommerce order
            $woo_order_data = $this->prepare_woocommerce_order($sanitized);
            
            return [
                'success' => true,
                'state_id' => $state_id,
                'woo_order_data' => $woo_order_data,
            ];
            
        } catch (Exception $e) {
            $this->logger->error('Streams order init failed: ' . $e->getMessage(), 'pgb-client-streams');
            return ['success' => false, 'message' => $e->getMessage()];
        }
    }
    
    /**
     * Sanitize Streams order data
     */
    private function sanitize_streams_order($data) {
        return [
            'amount' => floatval($data['amount'] ?? 0),
            'product_name' => call_user_func([$this->validator, 'sanitize_string'], $data['product_name'] ?? ''),
            'product_price' => floatval($data['product_price'] ?? 0),
            'product_units' => intval($data['product_units'] ?? 1),
            'product_owner_id' => intval($data['product_owner_id'] ?? 0),
            'wow_post_id' => intval($data['wow_post_id'] ?? 0),
            'wow_currency_code' => call_user_func([$this->validator, 'sanitize_string'], $data['wow_currency_code'] ?? 'GHS'),
            'transaction_kind' => preg_replace('/[^A-Z_]/', '', strtoupper($data['transaction_kind'] ?? 'PRODUCT')),
        ];
    }
    
    /**
     * Prepare WooCommerce order from Streams data
     */
    private function prepare_woocommerce_order($streams_data) {
        $config = $this->config->get('streams', []);
        
        // Build line items
        $line_items = [];
        
        if ($streams_data['transaction_kind'] === 'PRODUCT') {
            $product_id = $config['wow_product_id'] ?? 0;
        } elseif ($streams_data['transaction_kind'] === 'PRO') {
            $product_id = $config['wow_pro_package_id'] ?? 0;
        } elseif ($streams_data['transaction_kind'] === 'WALLET') {
            $product_id = $config['wow_wallet_topup_id'] ?? 0;
        } elseif ($streams_data['transaction_kind'] === 'DONATE') {
            $product_id = $config['wow_crowdfund_id'] ?? 0;
        } else {
            $product_id = 0;
        }
        
        $line_items[] = [
            'product_id' => $product_id,
            'quantity' => $streams_data['product_units'],
            'total' => $streams_data['amount'],
        ];
        
        return [
            'currency' => $streams_data['wow_currency_code'],
            'line_items' => $line_items,
            'meta_data' => [
                ['key' => 'wow_order_id', 'value' => uniqid('wow_')],
                ['key' => 'wow_post_id', 'value' => $streams_data['wow_post_id']],
                ['key' => 'transaction_kind', 'value' => $streams_data['transaction_kind']],
                ['key' => '_buzzjuice_origin', 'value' => 'streams'],
            ],
        ];
    }
    
    /**
     * Handle webhook payload from WooCommerce
     */
    public function process_webhook($payload, $signature = null) {
        try {
            $config_secret = $this->config->get('streams.webhook_secret', '');
            
            if ($this->config->get('webhook_signature_verify', true) && !empty($config_secret)) {
                if (!call_user_func(
                    [$this->validator, 'verify_webhook_signature'],
                    $payload,
                    $signature,
                    $config_secret
                )) {
                    throw new Exception('Invalid webhook signature');
                }
            }
            
            $data = json_decode($payload, true);
            if (!$data) {
                throw new Exception('Invalid JSON payload');
            }
            
            $this->logger.info('Webhook received for order ' . ($data['id'] ?? 'unknown'), 'pgb-client');
            
            // Process through processor
            // This will be handled by addon or hook system
            do_action('pgb_webhook_received', $data);
            
            return ['success' => true];
            
        } catch (Exception $e) {
            $this->logger->error('Webhook processing failed: ' . $e->getMessage(), 'pgb-client');
            return ['success' => false, 'message' => $e->getMessage()];
        }
    }
}
```

```php
<?php
/**
 * shared/payment-gateway/pgb-client-streams.php
 * 
 * Streams (WoWonder) specific integration layer and API endpoints.
 */

if (!defined('ABSPATH') && !defined('PGB_PATH')) {
    exit('Direct access prohibited');
}

class PGB_Client_Streams {
    
    private static $instance = null;
    private $logger = null;
    private $client = null;
    private $config = null;
    
    public static function get_instance() {
        if (null === self::$instance) {
            self::$instance = new self();
        }
        return self::$instance;
    }
    
    private function __construct() {
        $this->logger = PGB_Logger::get_instance();
        $this->client = PGB_Client::get_instance();
        $this->config = PGB_Config::get_instance();
    }
    
    /**
     * Initialize payment order from Streams frontend
     * Called from: streams/assets/wow-pgb/wow-pgb_init.php
     */
    public function init_payment_order($post_data) {
        try {
            $this->logger->info('Streams: Initializing payment order', 'pgb-client-streams');
            
            // Call PGB Client to initialize
            $result = $this->client->init_streams_order($post_data);
            
            if (!$result['success']) {
                return [
                    'status' => 400,
                    'error' => $result['message'],
                ];
            }
            
            // Prepare WooCommerce order
            $woo_order_data = $result['woo_order_data'];
            
            // Generate order on WooCommerce
            $woo_order_id = $this->create_woocommerce_order($woo_order_data, $post_data);
            
            if (!$woo_order_id) {
                return [
                    'status' => 500,
                    'error' => 'Failed to create WooCommerce order',
                ];
            }
            
            // Get order details
            $order = wc_get_order($woo_order_id);
            $payment_url = $order->get_checkout_payment_url();
            
            $this->logger->info("Streams order initialized: WooCommerce Order {$woo_order_id}", 'pgb-client-streams');
            
            return [
                'status' => 200,
                'woo_order_id' => $woo_order_id,
                'url' => $payment_url,
            ];
            
        } catch (Exception $e) {
            $this->logger->error('Payment order init failed: ' . $e->getMessage(), 'pgb-client-streams');
            return [
                'status' => 500,
                'error' => $e->getMessage(),
            ];
        }
    }
    
    /**
     * Create WooCommerce order from Streams data
     */
    private function create_woocommerce_order($woo_order_data, $streams_data) {
        try {
            // Create order
            $order = wc_create_order();
            
            if (!$order || is_wp_error($order)) {
                throw new Exception('Failed to create WooCommerce order: ' . ($order->get_error_message() ?? 'Unknown error'));
            }
            
            // Set customer
            $customer_email = PGB_Validator::sanitize_email($streams_data['customer_email'] ?? '');
            if (!empty($customer_email)) {
                $order->set_billing_email($customer_email);
            }
            
            // Set billing & shipping
            if (!empty($streams_data['billing_address'])) {
                $order->set_address($streams_data['billing_address'], 'billing');
            }
            if (!empty($streams_data['shipping_address'])) {
                $order->set_address($streams_data['shipping_address'], 'shipping');
            }
            
            // Add line items
            foreach ($woo_order_data['line_items'] as $item) {
                $product = wc_get_product($item['product_id']);
                if ($product) {
                    $order->add_product($product, $item['quantity']);
                }
            }
            
            // Add meta data
            foreach ($woo_order_data['meta_data'] as $meta) {
                $order->update_meta_data($meta['key'], $meta['value']);
            }
            
            // Set currency
            if (!empty($woo_order_data['currency'])) {
                update_post_meta($order->get_id(), '_order_currency', $woo_order_data['currency']);
            }
            
            // Recalculate totals
            $order->calculate_totals();
            
            // Save
            $order->save();
            
            return $order->get_id();
            
        } catch (Exception $e) {
            $this->logger->error('WooCommerce order creation failed: ' . $e->getMessage(), 'pgb-client-streams');
            return false;
        }
    }
    
    /**
     * Get AffiliateWP context if available
     */
    public function capture_affiliate_context() {
        $context = [
            'timestamp' => current_time('c'),
            'snapshot_version' => '2',
        ];
        
        // Capture AffiliateWP cookies if present
        $affwp_keys = [
            'affwp_ref',
            'affwp_visit',
            'affwp_visit_id',
            'affwp_affiliate_id',
            'affwp_campaign',
            'affwp_coupon',
            'affwp_referrer',
        ];
        
        foreach ($affwp_keys as $key) {
            if (isset($_COOKIE[$key])) {
                $context[$key] = sanitize_text_field($_COOKIE[$key]);
            }
        }
        
        // HTTP referrer
        if (!empty($_SERVER['HTTP_REFERER'])) {
            $context['http_referer'] = esc_url_raw($_SERVER['HTTP_REFERER']);
        }
        
        return $context;
    }
    
    /**
     * Get Streams user context
     */
    public function capture_streams_user_context($wow_user_id) {
        try {
            if (!function_exists('get_wowonder_db')) {
                require_once dirname(__DIR__) . '/shared/db_helpers.php';
            }
            
            $db = call_user_func('get_wowonder_db');
            if (!$db) {
                return null;
            }
            
            $stmt = $db->prepare("SELECT user_id, username, email, wp_user_id FROM Wo_Users WHERE user_id = ? LIMIT 1");
            if (!$stmt) {
                return null;
            }
            
            $stmt->bind_param('i', $wow_user_id);
            $stmt->execute();
            $stmt->bind_result($uid, $username, $email, $wp_user_id);
            $found = $stmt->fetch();
            $stmt->close();
            
            if (!$found) {
                return null;
            }
            
            return [
                'wow_user_id' => $uid,
                'username' => $username,
                'email' => $email,
                'wp_user_id' => $wp_user_id,
            ];
            
        } catch (Exception $e) {
            $this->logger->error('Failed to capture Streams user context: ' . $e->getMessage(), 'pgb-client-streams');
            return null;
        }
    }
```

```php
class PGB_Int_WoW_User_Sync {

     * Sync Streams user with WordPress
     */
    public function sync_user_data($order_data, $streams_db) {
        try {
            if (!$streams_db || empty($order_data['wow_user_id'])) {
                return;
            }
            
            $wow_user_id = $order_data['wow_user_id'];
            
            // Get Streams user
            $stmt = $streams_db->prepare("SELECT wp_user_id, email FROM Wo_Users WHERE user_id = ? LIMIT 1");
            if (!$stmt) {
                $this->logger->error('prepare failed: ' . $streams_db->error, 'pgb-int-wow-user-sync');
                return;
            }
            
            $stmt->bind_param('i', $wow_user_id);
            $stmt->execute();
            $stmt->bind_result($wp_user_id, $email);
            $stmt->fetch();
            $stmt->close();
            
            if (empty($wp_user_id)) {
                $this->logger->debug("No WP user mapped for Streams user {$wow_user_id}", 'pgb-int-wow-user-sync');
                return;
            }
            
            // Sync user meta with WordPress
            $this->sync_user_metadata($wp_user_id, $wow_user_id, $email);
            
        } catch (Exception $e) {
            $this->logger->error('User sync failed: ' . $e->getMessage(), 'pgb-int-wow-user-sync');
        }
    }
    
    /**
     * Sync user metadata between systems
     */
    private function sync_user_metadata($wp_user_id, $wow_user_id, $email) {
        if (!function_exists('update_user_meta')) {
            return;
        }
        
        // Update WordPress user meta with Streams data
        update_user_meta($wp_user_id, '_buzzjuice_wow_user_id', $wow_user_id);
        update_user_meta($wp_user_id, '_buzzjuice_wow_sync_timestamp', current_time('mysql'));
```

```php
        'product_units' => intval($_POST['product_units']),
        'product_owner_id' => intval($_POST['product_owner_id']),
        'transaction_kind' => preg_replace('/[^A-Z_]/', '', strtoupper($_POST['transaction_kind'])),
```

```php
<?php
/**
 * streams/wow-pgb_webhook.php
 * 
 * REFACTORED VERSION - Uses new Payment Gateway Bridge (v3.0)
 * Receives WooCommerce webhooks and processes order completion.
 */

// Load WordPress if not already loaded
if (!defined('ABSPATH')) {
    require_once dirname(__DIR__) . '/wp-load.php';
}

try {
    // 1. VALIDATE WEBHOOK REQUEST
    $payload = file_get_contents('php://input');
    if (empty($payload)) {
        throw new Exception('Empty webhook payload', 400);
    }
    
    $headers = getallheaders() ?? [];
    $signature = $headers['X-WC-Webhook-Signature'] ?? '';
    
    // 2. LOAD PGB ENGINE
    $engine = pgb_get_engine();
    if (!$engine || !$engine->is_enabled()) {
        throw new Exception('PGB Engine not available', 503);
    }
    
    // 3. PROCESS WEBHOOK
    $client = $engine->get_client();
    $result = $client->process_webhook($payload, $signature);
    
    if (!$result['success']) {
        throw new Exception($result['message'], 400);
    }
    
    // 4. RETURN SUCCESS
    http_response_code(200);
    echo json_encode([
        'success' => true,
        'message' => 'Webhook processed successfully',
    ]);
    
} catch (Exception $e) {
    $status = $e->getCode() ?: 500;
    http_response_code($status);
    error_log('Webhook error: ' . $e->getMessage());
    echo json_encode([
        'success' => false,

    ]);
}
```
