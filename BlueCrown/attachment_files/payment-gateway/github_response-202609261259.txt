<?php
/**
 * Payment Gateway Bridge Logger
 *
 * Provides centralized logging with file rotation and level-based filtering.
 */

class PGB_Logger {
    const LOG_DIR = BZ_PGB_DIR . 'log/';
    const LOG_LEVEL_DEBUG = 'debug';
    const LOG_LEVEL_INFO = 'info';
    const LOG_LEVEL_WARNING = 'warning';
    const LOG_LEVEL_ERROR = 'error';
    const LOG_LEVEL_CRITICAL = 'critical';

    private $enabled = true;
    private $min_level = self::LOG_LEVEL_DEBUG;
    private $log_levels = [
        'debug' => 0,
        'info' => 1,
        'warning' => 2,
        'error' => 3,
        'critical' => 4,
    ];

    public function __construct($enabled = true, $min_level = 'debug') {
        $this->enabled = defined('BZ_PGB_LOGGING_ENABLED') ? BZ_PGB_LOGGING_ENABLED : $enabled;
        $this->min_level = $min_level;

        if (!is_dir(self::LOG_DIR)) {
            wp_mkdir_p(self::LOG_DIR);
        }
    }

    /**
     * Log a message
     *
     * @param string $message Message to log
     * @param string $level Log level
     * @param string $file Source file
     * @param int $line Source line
     * @param int $max_size Maximum log file size (bytes)
     */
    public function log($message, $level = 'info', $file = '', $line = 0, $max_size = 5242880) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $level = strtolower($level);
        $log_file = $this->get_log_file($file);
        $timestamp = current_time('mysql', true);
        $formatted_msg = sprintf(
            "[%s] [%s] %s (File: %s, Line: %d)\n",
            $timestamp,
            strtoupper($level),
            $message,
            basename($file),
            $line
        );

        // Rotate log if too large
        if (file_exists($log_file) && filesize($log_file) > $max_size) {
            $this->rotate_log($log_file);
        }

        // Write to file
        error_log($formatted_msg, 3, $log_file);

        // Also log to WordPress error log if critical
        if ($level === 'critical') {
            error_log('[BZ-PGB] ' . $message);
        }
    }

    private function should_log($level) {
        $level = strtolower($level);
        $current_level = $this->log_levels[$level] ?? 0;
        $min_required = $this->log_levels[$this->min_level] ?? 0;
        return $current_level >= $min_required;
    }

    private function get_log_file($source_file) {
        if (empty($source_file)) {
            $source_file = 'general';
        }

        $base_name = basename($source_file, '.php');
        $log_file = self::LOG_DIR . $base_name . '.log';

        return $log_file;
    }

    private function rotate_log($log_file) {
        $timestamp = date('YmdHis');
        $rotated_file = $log_file . '.' . $timestamp;
        rename($log_file, $rotated_file);

        // Keep only last 5 rotated files
        $files = glob($log_file . '.*');
        if (count($files) > 5) {
            usort($files, function ($a, $b) {
                return filemtime($a) - filemtime($b);
            });
            unlink(array_shift($files));
        }
    }
}
?>

            'currency_switcher_enabled' => get_option('bz_pgb_currency_switcher_enabled', false),
            'test_mode' => get_option('bz_pgb_test_mode', false),
        ];

        if ($this->logger) {
            $this->logger->log('Configuration loaded', 'debug', __FILE__, __LINE__);
        }
    }

    public function get($key, $default = null) {
        return $this->config[$key] ?? $default;
    }

    public function set($key, $value) {
        $this->config[$key] = $value;
        update_option('bz_pgb_' . $key, $value);
    }

    public function is_valid() {
        $required = ['woocommerce_api_url', 'woocommerce_api_key', 'woocommerce_api_secret'];
        foreach ($required as $key) {
            if (empty($this->get($key))) {
                if ($this->logger) {
                    $this->logger->log("Missing required config: {$key}", 'error', __FILE__, __LINE__);
                }
                return false;
            }
        }
        return true;
    }
}

     * Get WoWonder user by email
     */
    public function get_wow_user_by_email($email) {
        if (!$this->wow_db) {
            return null;
        }

        $stmt = $this->wow_db->prepare("SELECT user_id, username, email, wp_user_id, wallet, is_pro, pro_type FROM Wo_Users WHERE email = ? LIMIT 1");
        if (!$stmt) {
            if ($this->logger) {
                $this->logger->log("Prepare failed: {$this->wow_db->error}", 'error', __FILE__, __LINE__);
            }
            return null;
        }

        $stmt->bind_param('s', $email);
        if (!$stmt->execute()) {
            if ($this->logger) {
                $this->logger->log("Execute failed: {$stmt->error}", 'error', __FILE__, __LINE__);
            }
            $stmt->close();
            return null;
        }

        $result = $stmt->get_result();
        $user = $result->fetch_assoc();
        $stmt->close();

        return $user;
    }

    /**
     * Update WoWonder payment transaction
     */
    public function update_payment_transaction($order_id, $data) {
        if (!$this->wow_db) {
            return false;
        }

        $allowed_fields = ['payment_status', 'payment_method', 'woo_order_id', 'amount', 'currency_code', 'notes'];
        $set_clauses = [];
        $values = [];
        $types = '';

        foreach ($data as $key => $value) {
            if (!in_array($key, $allowed_fields)) {
                continue;
            }
            $set_clauses[] = "$key = ?";
            $values[] = $value;
            $types .= is_int($value) ? 'i' : 's';
        }

        if (empty($set_clauses)) {
            return false;
        }

        $values[] = $order_id;
        $types .= 's';

        $sql = "UPDATE Wo_Payment_Transactions SET " . implode(', ', $set_clauses) . " WHERE order_id = ? LIMIT 1";
        $stmt = $this->wow_db->prepare($sql);

        if (!$stmt) {
            if ($this->logger) {
                $this->logger->log("Prepare failed: {$this->wow_db->error}", 'error', __FILE__, __LINE__);
            }
            return false;
        }

        $stmt->bind_param($types, ...$values);
        $result = $stmt->execute();
        $stmt->close();

        return $result;
    }

<?php
/**
 * Payment Gateway Bridge Bootstrap
 *
 * Orchestrates initialization and coordination of all payment gateway components.
 */

class PGB_Bootstrap {
    private $logger = null;
    private $config = null;
    private $db_manager = null;
    private $woo_api = null;
    private $client_streams = null;

    public function __construct(PGB_Logger $logger) {
        $this->logger = $logger;
    }

    public function initialize() {
        $this->logger->log('Bootstrap initialization started', 'info', __FILE__, __LINE__);

        try {
            // Load all dependencies
            require_once BZ_PGB_DIR . 'core/class-pgb-config.php';
            require_once BZ_PGB_DIR . 'core/class-pgb-db-manager.php';
            require_once BZ_PGB_DIR . 'client/class-pgb-woocommerce-api.php';
            require_once BZ_PGB_DIR . 'client/class-pgb-client-streams.php';

            // Initialize components
            $this->config = PGB_Config::get_instance($this->logger);
            $this->db_manager = PGB_DB_Manager::get_instance($this->logger);
            $this->woo_api = new PGB_WooCommerce_API($this->config, $this->logger);
            $this->client_streams = new PGB_Client_Streams($this->db_manager, $this->logger);

            // Validate configuration
            if (!$this->config->is_valid()) {
                $this->logger->log('Configuration validation failed', 'critical', __FILE__, __LINE__);
                return;
            }

            // Load integrations
            $this->load_integrations();

            // Register hooks
            $this->register_hooks();

            // Setup webhook endpoint
            $this->setup_webhook();

            $this->logger->log('Bootstrap initialization completed successfully', 'info', __FILE__, __LINE__);
        } catch (Exception $e) {
            $this->logger->log("Bootstrap initialization failed: {$e->getMessage()}", 'critical', __FILE__, __LINE__);
        }
    }

    private function load_integrations() {
        // Load integrations if enabled
        if (file_exists(BZ_PGB_DIR . 'integrations/pgb-int-affiliate-wp.php')) {
            require_once BZ_PGB_DIR . 'integrations/pgb-int-affiliate-wp.php';
        }

        if (file_exists(BZ_PGB_DIR . 'integrations/pgb-int-currency-switcher.php')) {
            require_once BZ_PGB_DIR . 'integrations/pgb-int-currency-switcher.php';
        }

        if (file_exists(BZ_PGB_DIR . 'integrations/pgb-int-streams-sync.php')) {
            require_once BZ_PGB_DIR . 'integrations/pgb-int-streams-sync.php';
        }

        // Load addons
        if (file_exists(BZ_PGB_DIR . 'addons/pgb-addon-webhook-handler.php')) {
            require_once BZ_PGB_DIR . 'addons/pgb-addon-webhook-handler.php';
            $handler = new PGB_Addon_Webhook_Handler($this->logger, $this->db_manager, $this->config);
            $handler->register_hooks();
        }
    }

    private function register_hooks() {
        add_action('woocommerce_order_status_completed', [$this, 'process_completed_order']);
        add_filter('bz_pgb_order_data', [$this, 'filter_order_data']);
        do_action('bz_pgb_bootstrap_loaded', $this);
    }

    private function setup_webhook() {
        add_rewrite_rule('^buzzjuice-webhook/?$', 'index.php?buzzjuice_webhook=1', 'top');

        add_action('parse_request', function ($wp) {
            if (isset($wp->query_vars['buzzjuice_webhook'])) {
                if (file_exists(BZ_PGB_DIR . 'addons/pgb-addon-webhook-handler.php')) {
                    require_once BZ_PGB_DIR . 'addons/pgb-addon-webhook-handler.php';
                    $handler = new PGB_Addon_Webhook_Handler($this->logger, $this->db_manager, $this->config);
                    $handler->handle_webhook();
                    exit;
                }
            }
        });

        flush_rewrite_rules();
    }

    public function process_completed_order($order_id) {
        $order = wc_get_order($order_id);
        if (!$order) {
            return;
        }

        $this->logger->log("Processing completed order {$order_id}", 'info', __FILE__, __LINE__);

        try {
            // Get order meta
            $wow_order_id = $order->get_meta('wow_order_id');
            $wow_user_id = $order->get_meta('wow_user_id');

            if (!$wow_order_id || !$wow_user_id) {
                $this->logger->log("Missing WoWonder metadata for order {$order_id}", 'warning', __FILE__, __LINE__);
                return;
            }

            // Process via Streams client
            $this->client_streams->process_order_completion($order, $wow_order_id, $wow_user_id);

            // Process affiliate integration
            do_action('bz_pgb_order_completed', $order, $wow_order_id, $wow_user_id);
        } catch (Exception $e) {
            $this->logger->log("Error processing order {$order_id}: {$e->getMessage()}", 'error', __FILE__, __LINE__);
        }
    }

    public function filter_order_data($data) {
        // Allow third-party modifications
        return $data;
    }

    public function get_config() {
        return $this->config;
    }

    public function get_db_manager() {
        return $this->db_manager;
    }

    public function get_woo_api() {
        return $this->woo_api;

        }
    }

    /**
     * Create order in WooCommerce
     */
    public function create_order($order_data) {
        if (!$this->is_configured()) {
            $this->logger->log('WooCommerce API not configured', 'error', __FILE__, __LINE__);
            return null;
        }

        $endpoint = $this->api_url . '/orders';
        $response = $this->send_request('POST', $endpoint, $order_data);

        if (!$response || $response['http_code'] !== 201) {
            $this->logger->log("Failed to create WooCommerce order: " . json_encode($response), 'error', __FILE__, __LINE__);
            return null;
        }

        $order = json_decode($response['response'], true);
        $this->logger->log("WooCommerce order created: {$order['id']}", 'info', __FILE__, __LINE__);

        return $order;
    }

    /**
     * Update order status
     */
    public function update_order_status($order_id, $status) {
        if (!$this->is_configured()) {
            return false;
        }

        $endpoint = $this->api_url . "/orders/{$order_id}";
        $data = ['status' => $status];
        $response = $this->send_request('POST', $endpoint, $data);

        if (!$response || $response['http_code'] !== 200) {
            $this->logger->log("Failed to update WooCommerce order {$order_id} status", 'error', __FILE__, __LINE__);
            return false;
        }

        return true;
    }

    /**
     * Send HTTP request to WooCommerce API
     */
    private function send_request($method, $endpoint, $data = null) {
        if (!filter_var($endpoint, FILTER_VALIDATE_URL)) {
            $this->logger->log("Invalid URL: {$endpoint}", 'error', __FILE__, __LINE__);
            return null;
        }

        $headers = [
            'Content-Type' => 'application/json',
            'Authorization' => 'Basic ' . base64_encode("{$this->api_key}:{$this->api_secret}"),
        ];

        $args = [
            'method' => strtoupper($method),
            'headers' => $headers,
            'timeout' => 30,
            'sslverify' => !$this->config->get('test_mode', false),
        ];

        if (!empty($data)) {
            $args['body'] = json_encode($data);
        }

        $response = wp_remote_request($endpoint, $args);

        if (is_wp_error($response)) {
            $this->logger->log("Request error: " . $response->get_error_message(), 'error', __FILE__, __LINE__);
            return null;
        }

        return [
            'response' => wp_remote_retrieve_body($response),
            'http_code' => wp_remote_retrieve_response_code($response),
            'headers' => wp_remote_retrieve_headers($response),
        ];
    }

    private function is_configured() {
        return !empty($this->api_url) && !empty($this->api_key) && !empty($this->api_secret);
    }

    public function handle_webhook() {
        try {
            // Get payload
            $payload = file_get_contents('php://input');
            $headers = $this->get_headers();

            if (empty($payload)) {
                $this->logger->log('Empty webhook payload', 'warning', __FILE__, __LINE__);
                http_response_code(400);
                exit('Invalid payload');
            }

            $data = json_decode($payload, true);
            if (!$data) {
                $this->logger->log('Invalid JSON webhook payload', 'warning', __FILE__, __LINE__);
                http_response_code(400);
                exit('Invalid JSON');
            }

            // Verify signature
            if (!$this->verify_signature($payload, $headers)) {
                $this->logger->log('Webhook signature verification failed', 'error', __FILE__, __LINE__);
                http_response_code(403);
                exit('Invalid signature');
            }

            $this->logger->log('Webhook received and verified', 'info', __FILE__, __LINE__);

            // Process webhook
            $this->process_webhook($data);

            http_response_code(200);
            exit('Webhook processed');
        } catch (Exception $e) {
            $this->logger->log("Webhook error: {$e->getMessage()}", 'error', __FILE__, __LINE__);
            http_response_code(500);
            exit('Error processing webhook');
        }
    }

    private function process_webhook($data) {
        $status = strtolower($data['status'] ?? '');

        if ($status !== 'completed') {
            $this->logger->log("Webhook status not completed: {$status}", 'info', __FILE__, __LINE__);
            return;
        }

        // Get order meta
        $woo_order_id = $data['id'] ?? null;
        $billing_email = $data['billing']['email'] ?? '';

        if (!$woo_order_id || !$billing_email) {
            $this->logger->log('Missing required webhook data', 'warning', __FILE__, __LINE__);
            return;
        }

        // Get WoWonder user
        $user = $this->db_manager->get_wow_user_by_email($billing_email);
        if (!$user) {
            $this->logger->log("User not found for email: {$billing_email}", 'warning', __FILE__, __LINE__);
            return;
        }

        // Get WoWonder order ID from metadata
        $wow_order_id = null;
        if (!empty($data['meta_data']) && is_array($data['meta_data'])) {
            foreach ($data['meta_data'] as $meta) {
                if (($meta['key'] ?? '') === 'wow_order_id') {
                    $wow_order_id = $meta['value'];
                    break;
                }
            }
        }

        if (!$wow_order_id) {
            $this->logger->log("WoWonder order ID not found in metadata for WooCommerce order {$woo_order_id}", 'warning', __FILE__, __LINE__);
            return;
        }

        // Trigger order completion via WordPress
        do_action('bz_pgb_webhook_order_completed', $woo_order_id, $user, $wow_order_id, $data);
    }

    private function verify_signature($payload, $headers) {
        $secret = $this->config->get('webhook_secret', '');
        if (empty($secret)) {
            $this->logger->log('Webhook secret not configured', 'warning', __FILE__, __LINE__);
            return false;
        }

        $signature = $headers['X-WC-Webhook-Signature'] ?? '';
        $expected_signature = base64_encode(hash_hmac('sha256', $payload, $secret, true));

        return hash_equals($signature, $expected_signature);
    }

    private function get_headers() {
        $headers = [];
        foreach ($_SERVER as $key => $value) {
            if (strpos($key, 'HTTP_') === 0) {
                $header_name = str_replace('HTTP_', '', $key);
                $header_name = str_replace('_', '-', $header_name);
                $headers[$header_name] = $value;
            }
        }
        return $headers;
    }
}

        $user_id = $this->wo['user']['user_id'] ?? 0;
        $user_email = filter_var($this->wo['user']['email'], FILTER_VALIDATE_EMAIL);

        if (!$user_id || !$user_email) {
            throw new Exception('Invalid user information');
        }

        // Generate order ID
        $wow_order_id = uniqid('wow_');

        // Insert transaction into WoWonder
        $transaction_data = [
            'userid' => $user_id,
            'amount' => $payment_data['amount'],
            'order_id' => $wow_order_id,
            'kind' => $payment_data['transaction_kind'],
            'currency_code' => $payment_data['wow_currency_code'],
            'payment_status' => 'pending',
            'payment_method' => 'woocommerce',
            'woo_order_id' => '',
            'notes' => $payment_data['product_name'],
        ];

        if (!$this->db_manager->insert_payment_transaction($transaction_data)) {
            throw new Exception('Failed to record transaction');
        }

        // Create WooCommerce order
        $woo_order_data = $this->prepare_woocommerce_order($user_id, $user_email, $payment_data, $wow_order_id);
        
        require_once dirname(__DIR__) . '/shared/payment-gateway/client/class-pgb-woocommerce-api.php';
        $woo_api = new PGB_WooCommerce_API($this->config, $this->logger);
        $woo_order = $woo_api->create_order($woo_order_data);

        if (!$woo_order || empty($woo_order['payment_url'])) {
            throw new Exception('Failed to create WooCommerce order');
        }

        // Update transaction with WooCommerce order ID
        $this->db_manager->update_payment_transaction($wow_order_id, [
            'woo_order_id' => $woo_order['id'],
        ]);

        $this->logger->log("Payment processed: wow_order={$wow_order_id}, woo_order={$woo_order['id']}", 'info', __FILE__, __LINE__);

        return [
            'status' => 200,
            'url' => $woo_order['payment_url'],
        ];
    }

    private function sanitize_payment_data($data) {
        return [
            'amount' => floatval($data['amount']),
            'product_name' => htmlspecialchars(trim($data['product_name']), ENT_QUOTES, 'UTF-8'),
            'product_price' => floatval($data['product_price']),
            'product_units' => intval($data['product_units']),
            'product_owner_id' => intval($data['product_owner_id']),
            'transaction_kind' => preg_replace('/[^a-zA-Z0-9_-]/', '', $data['transaction_kind']),
            'wow_currency_code' => htmlspecialchars($data['wow_currency_code'], ENT_QUOTES, 'UTF-8'),
            'wow_post_id' => isset($data['wow_post_id']) ? intval($data['wow_post_id']) : 0,
            'address_id' => isset($data['address_id']) ? intval($data['address_id']) : 0,
        ];
    }

    private function prepare_woocommerce_order($user_id, $user_email, $payment_data, $wow_order_id) {
        // Get WP user ID
        $wp_user_id = get_user_by('email', $user_email);
        $wp_user_id = $wp_user_id ? $wp_user_id->ID : 0;

        $line_item = [
            'name' => $payment_data['product_name'],
            'product_id' => 0,
            'quantity' => $payment_data['product_units'],
            'price' => $payment_data['product_price'],
            'total' => $payment_data['amount'],
            'meta_data' => [
                ['key' => 'wow_order_id', 'value' => $wow_order_id],
                ['key' => 'wow_user_id', 'value' => $user_id],
                ['key' => 'wow_product_owner_id', 'value' => $payment_data['product_owner_id']],
                ['key' => 'wow_post_id', 'value' => $payment_data['wow_post_id']],
                ['key' => 'wow_order_type', 'value' => $payment_data['transaction_kind']],
            ],
        ];

        return [
            'customer_id' => $wp_user_id,
            'currency' => $payment_data['wow_currency_code'],
            'set_paid' => false,
            'line_items' => [$line_item],
            'meta_data' => [
                ['key' => 'wow_order_id', 'value' => $wow_order_id],
                ['key' => 'wow_user_id', 'value' => $user_id],
                ['key' => '_buzzjuice_origin', 'value' => 'streams'],
            ],
        ];
    }

    public function get_logger() {
        return $this->logger;
    }

    public function get_config() {
        return $this->config;
    }

    public function get_db_manager() {
        return $this->db_manager;
    }
}
?>
