<?php
/**
 * PGB Order Processor - Handle all order operations
 */

if ( ! defined( 'ABSPATH' ) ) {
    exit;
}

class PGB_Order_Processor {

    public function create_order_from_streams( WP_REST_Request $request ) {
        $params = $request->get_json_params();

        bzj_pgb_log( 'Creating order from Streams', $params, 'info', __FILE__ );

        // Validate required parameters
        $required = [ 'user_id', 'amount', 'product_name', 'product_price', 'product_units' ];
        foreach ( $required as $field ) {
            if ( empty( $params[ $field ] ) ) {
                throw new Exception( "Missing required field: $field" );
            }
        }

        // Get or create WordPress user linked to Streams user
        $wp_user = $this->get_or_create_wp_user_from_streams( 
            sanitize_text_field( $params['user_id'] ),
            sanitize_text_field( $params['user_email'] ?? '' )
        );

        if ( ! $wp_user ) {
            throw new Exception( 'Could not create/retrieve WordPress user' );
        }

        // Get product ID from mapping
        $product_id = $this->get_wc_product_id( sanitize_text_field( $params['product_type'] ?? 'default' ) );

        if ( ! $product_id ) {
            throw new Exception( 'Product mapping not configured' );
        }

        // Get product and prepare order data
        $product = wc_get_product( $product_id );
        if ( ! $product ) {
            throw new Exception( 'Invalid product ID: ' . $product_id );
        }

        // Handle currency conversion
        $currency_handler = new PGB_Currency_Handler();
        $converted_price = $currency_handler->convert_price(
            floatval( $params['product_price'] ),
            sanitize_text_field( $params['product_currency'] ?? 'USD' )
        );

        // Create order
        $order = wc_create_order( [ 'customer_id' => $wp_user->ID ] );

        if ( is_wp_error( $order ) ) {
            throw new Exception( 'Could not create WooCommerce order: ' . $order->get_error_message() );
        }

        // Add product to order
        $order->add_product(
            $product,
            intval( $params['product_units'] ?? 1 ),
            [
                'subtotal' => $converted_price,
                'total'    => $converted_price,
            ]
        );

        // Add Streams metadata
        $order->update_meta_data( 'bzj_streams_order_id', sanitize_text_field( $params['order_id'] ?? '' ) );
        $order->update_meta_data( 'bzj_streams_user_id', sanitize_text_field( $params['user_id'] ) );
        $order->update_meta_data( 'bzj_streams_product_type', sanitize_text_field( $params['product_type'] ?? 'default' ) );

        if ( ! empty( $params['subscription_type'] ) ) {
            $order->update_meta_data( 'bzj_subscription_type', sanitize_text_field( $params['subscription_type'] ) );
        }

        $order->calculate_totals();
        $order->save();

        bzj_pgb_log( 
            'Order created successfully', 
            [ 'order_id' => $order->get_id(), 'wp_user_id' => $wp_user->ID ], 
            'info', 
            __FILE__ 
        );

        // Get checkout URL
        $checkout_url = $order->get_checkout_payment_url( false );

        return [
            'success'       => true,
            'order_id'      => $order->get_id(),
            'checkout_url'  => $checkout_url,
            'total'         => $order->get_total(),
        ];
    }

    public function handle_order_completion( $order_id ) {
        $order = wc_get_order( $order_id );

        if ( ! $order ) {
            throw new Exception( 'Invalid order ID: ' . $order_id );
        }

        bzj_pgb_log( 'Processing completed order', [ 'order_id' => $order_id ], 'info', __FILE__ );

        // Get Streams metadata
        $streams_user_id = $order->get_meta( 'bzj_streams_user_id' );
        $streams_order_id = $order->get_meta( 'bzj_streams_order_id' );
        $subscription_type = $order->get_meta( 'bzj_subscription_type' );

        if ( ! $streams_user_id ) {
            bzj_pgb_log( 'No Streams user ID found in order metadata', [ 'order_id' => $order_id ], 'warning', __FILE__ );
            return;
        }

        // Update Streams database if needed
        $this->update_streams_user( $streams_user_id, $order, $subscription_type );

        // Handle subscriptions
        if ( wcs_order_contains_subscription( $order_id ) ) {
            $subscription_handler = new PGB_Subscription_Handler();
            $subscription_handler->handle_subscription( $order_id, $streams_user_id );
        }

        // Handle affiliate commissions
        do_action( 'bzj_pgb_order_completed', $order_id, $streams_user_id, $order );

        bzj_pgb_log( 
            'Order processing complete', 
            [ 'order_id' => $order_id, 'streams_user_id' => $streams_user_id ], 
            'info', 
            __FILE__ 
        );
    }

    public function handle_order_status_change( $order_id, $old_status, $new_status, $order ) {
        bzj_pgb_log( 
            'Order status changed', 
            [ 'order_id' => $order_id, 'old' => $old_status, 'new' => $new_status ], 
            'info', 
            __FILE__ 
        );

        // Handle non-completion statuses if needed
        if ( in_array( $new_status, [ 'failed', 'cancelled' ], true ) ) {
            $streams_user_id = $order->get_meta( 'bzj_streams_user_id' );
            if ( $streams_user_id ) {
                do_action( 'bzj_pgb_order_failed', $order_id, $streams_user_id, $new_status );
            }
        }
    }

    private function get_or_create_wp_user_from_streams( $streams_user_id, $email = '' ) {
        // Check if user already linked
        $wp_users = get_users( [
            'meta_key'   => 'bzj_streams_user_id',
            'meta_value' => $streams_user_id,
        ] );

        if ( ! empty( $wp_users ) ) {
            return $wp_users[0];
        }

        // Create new user if email provided
        if ( ! empty( $email ) && is_email( $email ) ) {
            $username = 'streams_' . $streams_user_id;
            $user_id = wp_create_user( $username, wp_generate_password(), $email );

            if ( is_wp_error( $user_id ) ) {
                bzj_pgb_log( 
                    'Failed to create WordPress user', 
                    [ 'username' => $username, 'error' => $user_id->get_error_message() ], 
                    'error', 
                    __FILE__ 
                );
                return null;
            }

            $user = get_user_by( 'id', $user_id );
            update_user_meta( $user_id, 'bzj_streams_user_id', $streams_user_id );

            bzj_pgb_log( 
                'WordPress user created for Streams user', 
                [ 'wp_user_id' => $user_id, 'streams_user_id' => $streams_user_id ], 
                'info', 
                __FILE__ 
            );

            return $user;
        }

        return null;
    }

    private function get_wc_product_id( $product_type ) {
        $mapping = [
            'pro_weekly'   => (int) get_option( 'bzj_pgb_pro_weekly_product' ),
            'pro_monthly'  => (int) get_option( 'bzj_pgb_pro_monthly_product' ),
            'pro_yearly'   => (int) get_option( 'bzj_pgb_pro_yearly_product' ),
            'pro_lifetime' => (int) get_option( 'bzj_pgb_pro_lifetime_product' ),
            'wallet'       => (int) get_option( 'bzj_pgb_wallet_product' ),
            'crowdfund'    => (int) get_option( 'bzj_pgb_crowdfund_product' ),
            'default'      => (int) get_option( 'bzj_pgb_default_product' ),
        ];

        return $mapping[ $product_type ] ?? 0;
    }

    private function update_streams_user( $streams_user_id, $order, $subscription_type = '' ) {
        // Load Streams database
        $db_helpers = dirname( dirname( dirname( dirname( __DIR__ ) ) ) ) . '/shared/db_helpers.php';
        if ( ! file_exists( $db_helpers ) ) {
            bzj_pgb_log( 'Streams database helper not found', [], 'warning', __FILE__ );
            return;
        }

        require_once $db_helpers;
        $conn = get_palmier_db_conn();

        if ( ! $conn ) {
            bzj_pgb_log( 'Could not connect to Streams database', [], 'warning', __FILE__ );
            return;
        }

        // Update based on order type
        if ( ! empty( $subscription_type ) ) {
            // Handle pro subscription
            $type_map = [ 'weekly' => 1, 'monthly' => 2, 'yearly' => 3, 'lifetime' => 4 ];
            $pro_type = $type_map[ $subscription_type ] ?? 1;

            $query = "UPDATE Wo_Users SET is_pro = 1, pro_type = $pro_type, pro_time = " . time() . " WHERE user_id = " . intval( $streams_user_id );
            $conn->query( $query );

            bzj_pgb_log( 
                'Streams user pro status updated', 
                [ 'user_id' => $streams_user_id, 'pro_type' => $pro_type ], 
                'info', 
                __FILE__ 
            );
        }
    }
}

/**
 * PGB Currency Handler - Handle multi-currency conversions
 */

if ( ! defined( 'ABSPATH' ) ) {
    exit;
}

class PGB_Currency_Handler {

    public function convert_price( $amount, $from_currency = 'USD' ) {
        // Get WooCommerce currency
        $woo_currency = get_woocommerce_currency();

        if ( $from_currency === $woo_currency ) {
            return floatval( $amount );
        }

        // Try WOOCS currency switcher
        $converted = $this->convert_via_woocs( $amount, $from_currency, $woo_currency );
        if ( $converted !== false ) {
            return floatval( $converted );
        }

        // Fallback: Log and return original amount
        bzj_pgb_log( 
            'Currency conversion not available, using original amount', 
            [ 'from' => $from_currency, 'to' => $woo_currency, 'amount' => $amount ], 
            'warning', 
            __FILE__ 
        );

        return floatval( $amount );
    }

    private function convert_via_woocs( $amount, $from, $to ) {
        if ( ! function_exists( 'WOOCS' ) ) {
            return false;
        }

        try {
            $woocs = WOOCS();
            
            // Get exchange rates
            if ( ! isset( $woocs->rates ) || ! is_array( $woocs->rates ) ) {
                return false;
            }

            if ( ! isset( $woocs->rates[ $from ] ) || ! isset( $woocs->rates[ $to ] ) ) {
                return false;
            }

            $from_rate = floatval( $woocs->rates[ $from ] );
            $to_rate = floatval( $woocs->rates[ $to ] );

            if ( $from_rate === 0 ) {
                return false;
            }

            return ( $amount / $from_rate ) * $to_rate;
        } catch ( Throwable $e ) {
            bzj_pgb_log( 
                'WOOCS conversion error: ' . $e->getMessage(), 
                [ 'from' => $from, 'to' => $to ], 
                'warning', 
                __FILE__ 
            );
            return false;
        }
    }
}

 */

class PGB_Subscription_Handler {

    public function handle_subscription( $order_id, $streams_user_id ) {
        if ( ! function_exists( 'wcs_get_subscriptions_for_order' ) ) {
            bzj_pgb_log( 'WooCommerce Subscriptions not available', [], 'warning', __FILE__ );
            return;
        }

        $subscriptions = wcs_get_subscriptions_for_order( $order_id );

        if ( empty( $subscriptions ) ) {
            return;
        }

        bzj_pgb_log( 
            'Processing subscriptions for order', 
            [ 'order_id' => $order_id, 'count' => count( $subscriptions ) ], 
            'info', 
            __FILE__ 
        );

        foreach ( $subscriptions as $subscription ) {
            $this->activate_subscription( $subscription, $streams_user_id );
        }
    }

    private function activate_subscription( $subscription, $streams_user_id ) {
        $subscription_id = $subscription->get_id();

        // Set subscription as active
        if ( method_exists( $subscription, 'set_status' ) ) {
            $subscription->set_status( 'active' );
            $subscription->save();
        }

        bzj_pgb_log( 
            'Subscription activated', 
            [ 'subscription_id' => $subscription_id, 'streams_user_id' => $streams_user_id ], 
            'info', 
            __FILE__ 
        );

        // Apply to Streams if needed
        $this->apply_subscription_to_streams( $subscription, $streams_user_id );
    }

    private function apply_subscription_to_streams( $subscription, $streams_user_id ) {
        // Get subscription product details
        $items = $subscription->get_items();

        if ( empty( $items ) ) {
            return;
        }

        $item = reset( $items );
        $product_id = $item->get_product_id();
        $product = wc_get_product( $product_id );

        if ( ! $product ) {
            return;
        }

        // Map subscription to Streams pro type
        $sku = $product->get_sku();
        $subscription_type = $this->map_sku_to_pro_type( $sku );

        if ( ! $subscription_type ) {
            bzj_pgb_log( 
                'Could not map product SKU to pro type', 
                [ 'sku' => $sku ], 
                'warning', 
                __FILE__ 
            );
            return;
        }

        // Update Streams database
        $db_helpers = dirname( dirname( dirname( dirname( __DIR__ ) ) ) ) . '/shared/db_helpers.php';
        if ( ! file_exists( $db_helpers ) ) {
            return;
        }

        require_once $db_helpers;
        $conn = get_palmier_db_conn();

        if ( ! $conn ) {
            return;
        }

        $query = "UPDATE Wo_Users SET is_pro = 1, pro_type = " . intval( $subscription_type ) . ", pro_time = " . time() . " WHERE user_id = " . intval( $streams_user_id );
        $conn->query( $query );

        bzj_pgb_log( 
            'Subscription applied to Streams', 
            [ 'subscription_id' => $subscription->get_id(), 'streams_user_id' => $streams_user_id, 'pro_type' => $subscription_type ], 
            'info', 
            __FILE__ 
        );
    }

    private function map_sku_to_pro_type( $sku ) {
        $mapping = [
            'pro-weekly'   => 1,
            'pro-monthly'  => 2,
            'pro-yearly'   => 3,
            'pro-lifetime' => 4,
        ];

        return $mapping[ strtolower( $sku ) ] ?? null;
    }

        $this->connect_to_streams();

    private function connect_to_streams() {
        $db_helpers = dirname( dirname( dirname( dirname( __DIR__ ) ) ) ) . '/shared/db_helpers.php';

        if ( ! file_exists( $db_helpers ) ) {
            bzj_pgb_log( 'Streams DB helpers not found', [], 'warning', __FILE__ );
            return;
        }

        require_once $db_helpers;
        $this->db = get_palmier_db_conn();
    }

    public function get_streams_user( $user_id ) {
        if ( ! $this->db ) {
            return null;
        }

        $query = $this->db->prepare(
            "SELECT * FROM Wo_Users WHERE user_id = %d LIMIT 1",
            $user_id
        );

        $result = $this->db->query( $query );
        return $result ? $result->fetch_assoc() : null;
    }

    public function update_streams_wallet( $user_id, $amount ) {
        if ( ! $this->db ) {
            return false;
        }

        $query = $this->db->prepare(
            "UPDATE Wo_Users SET wallet = wallet + %f WHERE user_id = %d",
            floatval( $amount ),
            $user_id
        );

        return $this->db->query( $query );
    }

    public function record_transaction( $user_id, $amount, $type, $reference_id = null ) {
        if ( ! $this->db ) {
            return false;
        }

        $query = $this->db->prepare(
            "INSERT INTO Wo_Payment_Transactions (user_id, amount, payment_type, status, time, order_id) VALUES (%d, %f, %s, %s, %d, %s)",
            $user_id,
            floatval( $amount ),
            $type,
            'completed',
            time(),
            $reference_id ?? ''
        );

        return $this->db->query( $query );
    }
}

    public function handle_order_completion( $order_id, $streams_user_id, $order ) {
        $customer = $order->get_user();

        if ( ! $customer ) {
            bzj_pgb_log( 'No customer found for order', [ 'order_id' => $order_id ], 'warning', __FILE__ );
            return;
        }

        // Check if customer is an affiliate
        $affiliate = affwp_get_affiliate_by_user_id( $customer->ID );

        if ( ! $affiliate || ! affwp_is_active_affiliate( $affiliate->ID ) ) {
            bzj_pgb_log( 
                'Customer is not an active affiliate', 
                [ 'customer_id' => $customer->ID ], 
                'info', 
                __FILE__ 
            );
            return;
        }

        // Check for lifetime commissions plugin
        $commission_rate = $this->get_commission_rate( $order );

        if ( $commission_rate <= 0 ) {
            bzj_pgb_log( 
                'No commission rate found', 
                [ 'order_id' => $order_id, 'affiliate_id' => $affiliate->ID ], 
                'warning', 
                __FILE__ 
            );
            return;
        }

        // Calculate commission
        $commission = floatval( $order->get_total() ) * ( $commission_rate / 100 );

        // Create referral
        $referral_id = affwp_add_referral( [
            'affiliate_id'  => $affiliate->ID,
            'reference'     => $order_id,
            'amount'        => $commission,
            'status'        => 'unpaid',
            'currency'      => $order->get_currency(),
            'description'   => 'WooCommerce Order',
        ] );

        if ( ! $referral_id ) {
            throw new Exception( 'Failed to create referral' );
        }

        bzj_pgb_log( 
            'Affiliate referral created', 
            [ 'referral_id' => $referral_id, 'affiliate_id' => $affiliate->ID, 'commission' => $commission ], 
            'info', 
            __FILE__ 
        );
    }

    private function get_commission_rate( $order ) {
        // Check for lifetime commissions
        if ( function_exists( 'affwp_is_lifetime_commission_order' ) ) {
            if ( affwp_is_lifetime_commission_order( $order->get_id() ) ) {
                return apply_filters( 'bzj_pgb_lifetime_commission_rate', 5.0 );
            }
        }

        // Check for multi-tier
        if ( function_exists( 'affwp_get_multi_tier_commission' ) ) {
            $rate = affwp_get_multi_tier_commission( $order->get_id() );
            if ( $rate > 0 ) {
                return floatval( $rate );
            }
        }

        // Default commission rate
        return apply_filters( 'bzj_pgb_default_commission_rate', 10.0 );
    }

/**
 * Streams Payment Handler
 */
class PGB_Streams_Payment_Handler {

    public function process_payment( $post_data, $streams_user_id ) {
        // Validate required fields
        $required = [ 'amount', 'product_name', 'product_price', 'product_type' ];
        foreach ( $required as $field ) {
            if ( empty( $post_data[ $field ] ) ) {
                throw new Exception( "Missing required field: $field" );
            }
        }

        // Get Streams user data
        $connector = new PGB_Streams_Connector();
        $streams_user = $connector->get_streams_user( $streams_user_id );

        if ( ! $streams_user ) {
            throw new Exception( 'Streams user not found' );
        }

        // Prepare REST API request
        $request_body = [
            'user_id'           => $streams_user_id,
            'user_email'        => $streams_user['email'] ?? '',
            'amount'            => floatval( $post_data['amount'] ),
            'product_name'      => sanitize_text_field( $post_data['product_name'] ),
            'product_price'     => floatval( $post_data['product_price'] ),
            'product_units'     => intval( $post_data['product_units'] ?? 1 ),
            'product_type'      => sanitize_text_field( $post_data['product_type'] ),
            'product_currency'  => sanitize_text_field( $post_data['product_currency'] ?? 'USD' ),
            'order_id'          => uniqid( 'bzj_' ),
            'subscription_type' => sanitize_text_field( $post_data['subscription_type'] ?? '' ),
        ];

        // Call REST API to create order
        $response = wp_remote_post(
            rest_url( 'bzj-pgb/v1/create-order' ),
            [
                'method'  => 'POST',
                'headers' => [
                    'Content-Type' => 'application/json',
                ],
                'body'    => wp_json_encode( $request_body ),
            ]
        );

        if ( is_wp_error( $response ) ) {
            throw new Exception( 'API request failed: ' . $response->get_error_message() );
        }

        $body = json_decode( wp_remote_retrieve_body( $response ), true );

        if ( empty( $body['success'] ) || empty( $body['checkout_url'] ) ) {
            throw new Exception( $body['error'] ?? 'Order creation failed' );
        }

        bzj_pgb_log( 
            'Payment processing initiated from Streams', 
            [ 'streams_user_id' => $streams_user_id, 'order_id' => $body['order_id'] ?? '' ], 
            'info', 
            __FILE__ 
        );

        return [
            'success'      => true,
            'checkout_url' => $body['checkout_url'],
            'order_id'     => $body['order_id'] ?? '',
        ];
    }
}
