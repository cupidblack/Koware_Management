```php
 * REST endpoint handler: POST /wp-json/bzj-pgb/v1/order
 * 
 * REQUIRED HEADERS:
 * - X-BZJ-PGB-Timestamp: Unix timestamp of request
 * - X-BZJ-PGB-Request-ID: Unique request identifier
 * - X-BZJ-PGB-Signature: HMAC-SHA256 signature
 * 
 * REQUIRED BODY (JSON):
 * {
 *     "wow_user_id": 123,
 *     "wow_email": "user@streams.com",
 *     "wow_first_name": "John",
 *     "wow_last_name": "Doe",
 *     "transaction_kind": "PRO|PRODUCT|WALLET|DONATE",
 *     "line_items": [
 *         {"product_id": 456, "quantity": 1, "price": 99.99}
 *     ],
 *     "currency": "GHS",
 *     "amount": 99.99
 * }
 */
function pgb_rest_create_order(WP_REST_Request $request) {
    if (!class_exists('PGB_Bootstrap')) {
        return new WP_Error(
            'pgb_not_initialized',
            'Payment Gateway Bridge not initialized',
            array('status' => 503)
        );
    }
    
    $pgb = PGB_Bootstrap::instance();
    $logger = $pgb->get_logger();
    $manager = $pgb->get_manager();
    
    // Extract request components
    $timestamp = $request->get_header('X-BZJ-PGB-Timestamp');
    $request_id = $request->get_header('X-BZJ-PGB-Request-ID');
    $signature = $request->get_header('X-BZJ-PGB-Signature');
    $body = file_get_contents('php://input');
    $params = json_decode($body, true);
    
    $logger->log(
        'REST order endpoint called',
        PGB_Logger::LOG_DEBUG,
        array('request_id' => $request_id, 'method' => $_SERVER['REQUEST_METHOD']),
        __FILE__
    );
    
    // Validate JSON payload
    if (json_last_error() !== JSON_ERROR_NONE || !$params) {
        $logger->log(
            'Invalid JSON in request',
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id, 'json_error' => json_last_error_msg()),
            __FILE__
        );
        return new WP_Error('invalid_json', 'Invalid JSON payload', array('status' => 400));
    }
    
    // Verify HMAC signature
    if (!pgb_verify_signature($timestamp, $request_id, $body, $signature)) {
        $logger->log(
            'Signature verification failed',
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error(
            'invalid_signature',
            'HMAC signature verification failed',
            array('status' => 401)
        );
    }
    
    // Verify timestamp freshness (5 minute window)
    $ts_int = intval($timestamp);
    if (abs(time() - $ts_int) > 300) {
        $logger->log(
            'Timestamp outside acceptable window',
            PGB_Logger::LOG_WARNING,
            array(
                'request_id' => $request_id,
                'age_seconds' => abs(time() - $ts_int),
            ),
            __FILE__
        );
        return new WP_Error('stale_timestamp', 'Request timestamp too old', array('status' => 401));
    }
    
    // Check idempotency - prevent duplicate processing
    $dup_key = 'pgb_req_' . $request_id;
    $existing = get_transient($dup_key);
    if ($existing) {
        $logger->log(
            'Duplicate request detected - returning cached result',
            PGB_Logger::LOG_WARNING,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error('duplicate_request', 'Request already processed', array('status' => 409));
    }
    
    // Mark request as processed (24 hour expiration)
    set_transient($dup_key, time(), 86400);
    
    // Create order in WooCommerce
    $result = $manager->create_order($params, $request_id);
    
    if (!$result['success']) {
        $logger->log(
            'Order creation failed',
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id, 'error' => $result['error']),
            __FILE__
        );
        return new WP_Error('order_failed', $result['error'], array('status' => 500));
    }
    
    $logger->log(
        'Order created successfully',
        PGB_Logger::LOG_INFO,
        array(
            'request_id' => $request_id,
            'woo_order_id' => $result['order_id'],
        ),
        __FILE__
    );
    
    return array(
        'success' => true,
        'order_id' => $result['order_id'],
        'payment_url' => $result['payment_url'],
    );
}

/**
 * Verify HMAC-SHA256 signature
 * 
 * Format: HMAC-SHA256(timestamp + "\n" + request_id + "\n" + raw_body, secret)
 * Encoded: base64(hmac_result)
 * 
 * @param string $timestamp Request timestamp
 * @param string $request_id Request ID
 * @param string $body Raw request body
 * @param string $signature Provided signature header
 * @return bool True if signature is valid
 */
function pgb_verify_signature($timestamp, $request_id, $body, $signature) {
    // Get secret from server environment ONLY
    $secret = getenv('BZJ_PGB_SECRET');
    
    if (empty($secret)) {
        error_log('[PGB] ERROR: BZJ_PGB_SECRET environment variable not set');
        return false;
    }
    
    // Build message to sign
    $signed_msg = $timestamp . "\n" . $request_id . "\n" . $body;
    
    // Calculate expected signature
    $expected = base64_encode(hash_hmac('sha256', $signed_msg, $secret, true));
    
    // Constant-time comparison to prevent timing attacks
    return hash_equals($expected, (string)$signature);
}
```

```php
/**
 * CLASS: PGB_Logger
 * 
 * Comprehensive logging system with:
 * - Per-component log files (class-pgb-manager.log, pgb-addon-affiliate-wp.log, etc.)
 * - Log level filtering (debug, info, warning, error, critical)

 * - Sensitive data redaction (passwords, secrets, tokens, API keys)
 * - JSON context formatting for structured logs
 * 
 * Usage:
 *     $logger->log('Message', PGB_Logger::LOG_INFO, array('key' => 'value'), __FILE__);
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    // Log level constants
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_INFO;
    private $file_handles = array();
    
    // Level priority mapping for filtering
    private $level_priority = array(
        'debug' => 0,
        'info' => 1,
        'warning' => 2,
        'error' => 3,
        'critical' => 4,
    );

    /**
     * Constructor
     * 
     * @param array $config Configuration options
     */
    public function __construct($config = array()) {
        $this->log_dir = PGB_LOG_DIR;
        
        // Check if logging is enabled via environment
        $this->enabled = (bool)getenv('BZJ_PGB_DEBUG');
        if (isset($config['logging_enabled'])) {
            $this->enabled = (bool)$config['logging_enabled'];
        }
        
        // Set log level from environment or config
        $this->min_level = getenv('BZJ_PGB_LOG_LEVEL') ?: (isset($config['log_level']) ? $config['log_level'] : self::LOG_INFO);
        
        // Ensure log directory exists
        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Main logging method
     * 
     * @param string $message The log message
     * @param string $level Log level (debug, info, warning, error, critical)
     * @param array $context Additional context data (will be JSON encoded)
     * @param string $source_file The source file (__FILE__) for log file naming
     */
    public function log($message, $level = self::LOG_INFO, $context = array(), $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        // Format timestamp
        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        
        // Extract source filename
        $source = 'pgb-system';
        if ($source_file) {
            $source = str_replace('.php', '', basename($source_file));
        }

        // Build log entry
        $log_line = "[{$timestamp}] [{$level_upper}] [{$source}] {$message}";

        // Add context if provided
        if (!empty($context) && is_array($context)) {
            $context = $this->redact_sensitive_data($context);
            $log_line .= " | " . json_encode($context, JSON_UNESCAPED_SLASHES);
        }

        $log_line .= "\n";

        // Write to file
        $this->write_to_file($source, $log_line);

        // Also write critical/error to WordPress error_log
        if (in_array($level, array(self::LOG_ERROR, self::LOG_CRITICAL))) {
            error_log('[PGB-' . $level_upper . '] [' . $source . '] ' . $message);
        }
    }

    /**
     * Determine if a log level should be output
     * 
     * @param string $level Log level to check
     * @return bool True if this level should be logged
     */
    private function should_log($level) {
        $current_priority = $this->level_priority[$level] ?? 0;
        $minimum_priority = $this->level_priority[$this->min_level] ?? 0;
        return $current_priority >= $minimum_priority;
    }

    /**
     * Redact sensitive values from log context
     * Prevents passwords, tokens, and secrets from being written to logs
     * 
     * @param array $data Data array to redact
     * @return array Redacted data array
     */
    private function redact_sensitive_data($data) {
        if (!is_array($data)) {
            return $data;
        }

        $sensitive_keys = array(
            'password', 'secret', 'api_key', 'api_secret', 'token', 'signature',
            'credit_card', 'cvv', 'ssn', 'secret_key', 'private_key', 
            'consumer_secret', 'authorization', 'auth', 'bearer',
            'cookie', 'session', 'jwt', 'hmac'
        );

        foreach ($data as $key => &$value) {
            if (is_array($value)) {
                $value = $this->redact_sensitive_data($value);
            } else {
                $lower_key = strtolower($key);
                foreach ($sensitive_keys as $sensitive) {
                    if (strpos($lower_key, $sensitive) !== false) {
                        $value = '[REDACTED]';
                        break;
                    }
                }
            }
        }

        return $data;
    }

    /**
     * Write log message to file with automatic rotation
     * 
     * @param string $source Source/component name (becomes filename)
     * @param string $message Log message with formatting
     */
    private function write_to_file($source, $message) {
        $file_path = $this->log_dir . $source . '.log';

        // Rotate log if it exceeds 10MB
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            if (@rename($file_path, $archived)) {
                error_log('[PGB] Log rotated: ' . basename($file_path) . ' → ' . basename($archived));
            }
        }

        // Open or reuse file handle
        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        // Write to file
        if (is_resource($this->file_handles[$file_path])) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        } else {
            // Fallback to WordPress error log if file handle fails
            error_log('[PGB] Could not write to log file: ' . $file_path);
        }
    }

    /**
     * Get log file path for a component
     * 
     * @param string $component Component name or file path
     * @return string Full path to log file
     */
    public function get_log_file($component) {
        $name = str_replace('.php', '', basename($component));
        return $this->log_dir . $name . '.log';
    }

    /**
     * Cleanup - close all file handles on object destruction
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                @fclose($handle);
            }
        }
    }
}
```

```php
<?php
/**
 * CLASS: PGB_Manager
 * 
 * Central transaction manager orchestrating the payment flow:
 * 1. Create order from Streams REST request
 * 2. Validate order data
 * 3. Create WooCommerce order
 * 4. Store transaction ledger record
 * 5. Process payment completion
 * 6. Trigger fulfillment addons/integrations
 * 
 * State machine lifecycle:
 * initiated → order_created → paid → fulfilling → fulfilled
 * 
 * CRITICAL: This class bridges Streams ↔ WooCommerce
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Manager {
    private $logger = null;
    private $hooks = null;
    private $schema = null;
    private $idempotency = null;
    private $wpdb = null;

    public function __construct($logger, $hooks) {
        global $wpdb;
        $this->logger = $logger;
        $this->hooks = $hooks;
        $this->wpdb = $wpdb;
        
        require_once dirname(__FILE__) . '/class-pgb-schema.php';
        $this->schema = new PGB_Schema($logger);
        
        require_once dirname(__FILE__) . '/class-pgb-idempotency.php';
        $this->idempotency = new PGB_Idempotency($logger);
    }

    /**
     * Create order from Streams REST request
     * 
     * Called by: pgb_rest_create_order() in mu-plugin
     * 
     * Expected payload:
     * {
     *     "wow_user_id": 123,
     *     "wow_email": "user@streams.local",
     *     "wow_first_name": "John",
     *     "wow_last_name": "Doe",
     *     "transaction_kind": "PRO|PRODUCT|WALLET|DONATE",
     *     "line_items": [
     *         {"product_id": 456, "quantity": 1, "price": 99.99}
     *     ],
     *     "currency": "GHS",
     *     "amount": 99.99
     * }
     * 
     * @param array $params Request payload
     * @param string $request_id Unique request identifier
     * @return array Response ['success' => bool, 'order_id' => int, 'payment_url' => string, 'error' => string]
     */
    public function create_order($params, $request_id) {
        $this->logger->log(
            'create_order() called',
            PGB_Logger::LOG_INFO,
            array('request_id' => $request_id),
            __FILE__
        );
        
        // Step 1: Validate parameters
        $validation = $this->validate_order_params($params);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                array('request_id' => $request_id, 'errors' => $validation['errors']),
                __FILE__
            );
            return array(
                'success' => false,
                'error' => implode('; ', $validation['errors']),
            );
        }
        
        $params = $validation['data'];
        
        // Step 2: Check idempotency - return existing order if duplicate
        $existing = $this->idempotency->check_duplicate($request_id, $params);
        if ($existing && isset($existing['woo_order_id'])) {
            $this->logger->log(
                'Duplicate request - returning existing order',
                PGB_Logger::LOG_WARNING,
                array('request_id' => $request_id, 'woo_order_id' => $existing['woo_order_id']),
                __FILE__
            );
            
            // Get WooCommerce order and return payment URL
            $order = wc_get_order($existing['woo_order_id']);
            if ($order) {
                return array(
                    'success' => true,
                    'order_id' => $order->get_id(),
                    'payment_url' => $order->get_checkout_payment_url(),
                );
            }
        }
        
        // Step 3: Create WooCommerce order
        $woo_order_id = $this->create_woocommerce_order($params, $request_id);
        if (!$woo_order_id) {
            $this->logger->log(
                'WooCommerce order creation failed',
                PGB_Logger::LOG_ERROR,
                array('request_id' => $request_id),
                __FILE__
            );
            return array(
                'success' => false,
                'error' => 'Failed to create WooCommerce order',
            );
        }
        
        // Step 4: Get order and attach metadata
        $order = wc_get_order($woo_order_id);
        if (!$order) {
            $this->logger->log(
                'WooCommerce order not found after creation',
                PGB_Logger::LOG_ERROR,
                array('woo_order_id' => $woo_order_id),
                __FILE__
            );
            return array(
                'success' => false,
                'error' => 'Created order not accessible',
            );
        }
        
        // Attach Streams metadata
        $order->update_meta_data('wow_user_id', $params['wow_user_id']);
        $order->update_meta_data('wow_order_id', $params['wow_order_id'] ?? '');
        $order->update_meta_data('transaction_kind', $params['transaction_kind']);
        $order->update_meta_data('request_id', $request_id);
        $order->save();
        
        // Step 5: Create transaction ledger record
        $ledger_id = $this->schema->insert_transaction(array(
            'woo_order_id' => $woo_order_id,
            'wow_order_id' => $params['wow_order_id'] ?? ('wow_' . uniqid()),
            'request_id' => $request_id,
            'wow_user_id' => $params['wow_user_id'],
            'wp_user_id' => $order->get_user_id(),
            'transaction_kind' => $params['transaction_kind'],
            'amount' => floatval($params['amount'] ?? 0),
            'currency' => $params['currency'] ?? 'GHS',
            'status' => 'order_created',
            'fulfillment_status' => 'pending',
            'payload' => json_encode($params),
        ));
        
        if (!$ledger_id) {
            $this->logger->log(
                'Failed to create transaction ledger record',
                PGB_Logger::LOG_ERROR,
                array('woo_order_id' => $woo_order_id),
                __FILE__
            );
        }
        
        // Step 6: Return success response
        $payment_url = $order->get_checkout_payment_url();
        
        $this->logger->log(
            'Order created successfully',
            PGB_Logger::LOG_INFO,
            array(
                'request_id' => $request_id,
                'woo_order_id' => $woo_order_id,
                'wow_user_id' => $params['wow_user_id'],
            ),
            __FILE__
        );
        
        return array(
            'success' => true,
            'order_id' => $woo_order_id,
            'payment_url' => $payment_url,
        );
    }

    /**
     * Validate order parameters from Streams request
     * 
     * @param array $params Parameters to validate
     * @return array ['valid' => bool, 'errors' => array, 'data' => array]
     */
    private function validate_order_params($params) {
        $errors = array();
        
        require_once dirname(__FILE__) . '/class-pgb-validator.php';
        
        // Required fields
        if (empty($params['wow_user_id'])) {
            $errors[] = 'Missing wow_user_id';
        } elseif (!PGB_Validator::validate_user_id($params['wow_user_id'])) {
            $errors[] = 'Invalid wow_user_id (must be positive integer)';
        }
        
        if (empty($params['wow_email'])) {
            $errors[] = 'Missing wow_email';
        } elseif (!PGB_Validator::validate_email($params['wow_email'])) {
            $errors[] = 'Invalid wow_email format';
        }
        
        if (empty($params['transaction_kind'])) {
            $errors[] = 'Missing transaction_kind';
        } elseif (!PGB_Validator::validate_transaction_type($params['transaction_kind'])) {
            $errors[] = 'Invalid transaction_kind (must be PRO, PRODUCT, WALLET, or DONATE)';
        }
        
        if (empty($params['amount'])) {
            $errors[] = 'Missing amount';
        } elseif (!PGB_Validator::validate_amount($params['amount'])) {
            $errors[] = 'Invalid amount (must be positive number)';
        }
        
        if (empty($params['currency'])) {
            $params['currency'] = 'GHS';
        } elseif (!PGB_Validator::validate_currency($params['currency'])) {
            $errors[] = 'Invalid currency code (must be 3-letter ISO code)';
        }
        
        if (empty($params['line_items']) || !is_array($params['line_items'])) {
            $errors[] = 'Missing or invalid line_items array';
        } else {
            foreach ($params['line_items'] as $idx => $item) {
                if (empty($item['product_id'])) {
                    $errors[] = "Line item $idx: missing product_id";
                }
                if (empty($item['quantity']) || $item['quantity'] <= 0) {
                    $errors[] = "Line item $idx: invalid quantity";
                }
                if (empty($item['price']) || $item['price'] <= 0) {
                    $errors[] = "Line item $idx: invalid price";
                }
            }
        }
        
        // Generate Streams order ID if not provided
        if (empty($params['wow_order_id'])) {
            $params['wow_order_id'] = 'wow_' . uniqid();
        }
        
        return array(
            'valid' => empty($errors),
            'errors' => $errors,
            'data' => $params,
        );
    }

    /**
     * Create WooCommerce order from Streams data
     * 
     * @param array $params Order parameters
     * @param string $request_id Request ID for tracking
     * @return int|false WooCommerce order ID or false
     */
    private function create_woocommerce_order($params, $request_id) {
        try {
            // Step 1: Get or create WordPress user
            $wp_user_id = email_exists($params['wow_email']);
            if (!$wp_user_id) {
                // Create new WordPress user from Streams data
                $wp_user_id = wp_create_user(
                    sanitize_user($params['wow_email']),
                    wp_generate_password(),
                    $params['wow_email']
                );
                
                if (is_wp_error($wp_user_id)) {
                    $this->logger->log(
                        'Failed to create WordPress user',
                        PGB_Logger::LOG_ERROR,
                        array('error' => $wp_user_id->get_error_message()),
                        __FILE__
                    );
                    return false;
                }
                
                $this->logger->log(
                    'Created new WordPress user',
                    PGB_Logger::LOG_INFO,
                    array('wp_user_id' => $wp_user_id, 'email' => $params['wow_email']),
                    __FILE__
                );
            }
            
            // Step 2: Create WooCommerce order
            $order = wc_create_order(array('customer_id' => $wp_user_id));
            if (is_wp_error($order)) {
                $this->logger->log(
                    'Failed to create WooCommerce order',
                    PGB_Logger::LOG_ERROR,
                    array('error' => $order->get_error_message()),
                    __FILE__
                );
                return false;
            }
            
            // Step 3: Add line items
            foreach ($params['line_items'] as $item) {
                $product = wc_get_product(intval($item['product_id']));
                if (!$product) {
                    $this->logger->log(
                        'Product not found',
                        PGB_Logger::LOG_WARNING,
                        array('product_id' => $item['product_id']),
                        __FILE__
                    );
                    continue;
                }
                
                $order->add_product(
                    $product,
                    intval($item['quantity'] ?? 1),
                    array('subtotal' => floatval($item['price']), 'total' => floatval($item['price'])}
                );
            }
            
            // Step 4: Calculate totals
            $order->calculate_totals();
            
            // Step 5: Set order status to pending
            $order->set_status('pending');
            $order->save();
            
            $this->logger->log(
                'WooCommerce order created',
                PGB_Logger::LOG_INFO,
                array('woo_order_id' => $order->get_id(), 'wp_user_id' => $wp_user_id),
                __FILE__
            );
            
            return $order->get_id();
            
        } catch (Exception $e) {
            $this->logger->log(
                'Exception creating WooCommerce order',
                PGB_Logger::LOG_CRITICAL,
                array('error' => $e->getMessage(), 'trace' => $e->getTraceAsString()),
                __FILE__
            );
            return false;
        }
    }

    /**
     * Process payment completion from WooCommerce webhook
     * 
     * Called by: pgb_wc_payment_complete() hook handler
     * 
     * Triggers fulfillment addons/integrations:
     * - Streams fulfillment
     * - AffiliateWP reward processing
     * - Jewel Affiliate processing
     * - Wallet crediting
     * - etc.
     * 
     * @param int $woo_order_id WooCommerce order ID
     * @param array $payment_data Payment data from hook
     * @return bool True if successful
     */
    public function process_payment_completion($woo_order_id, $payment_data) {
        $this->logger->log(
            'process_payment_completion() started',
            PGB_Logger::LOG_INFO,
            array('woo_order_id' => $woo_order_id),
            __FILE__
        );
        
        // Step 1: Get transaction ledger record
        $txn = $this->schema->get_transaction_by_woo_order_id($woo_order_id);
        if (!$txn) {
            // Not a PGB order
            return false;
        }
        
        // Step 2: Check if already fulfilled
        if ($txn['status'] === 'fulfilled') {
            $this->logger->log(
                'Order already fulfilled - skipping',
                PGB_Logger::LOG_WARNING,
                array('woo_order_id' => $woo_order_id),
                __FILE__
            );
            return true; // Already done
        }
        
        // Step 3: Update ledger to "paid"
        $this->schema->update_transaction($woo_order_id, array(
            'status' => 'paid',
            'paid_at' => current_time('mysql'),
        ));
        
        // Step 4: Transition to fulfilling
        $this->schema->update_transaction($woo_order_id, array(
            'status' => 'fulfilling',
            'fulfillment_status' => 'processing',
        ));
        
        // Step 5: Fire fulfillment hooks for addons/integrations
        do_action('pgb_payment_confirmed', $woo_order_id, $payment_data, $txn);
        do_action('pgb_handle_' . strtolower($txn['transaction_kind']), $woo_order_id, $payment_data, $txn);
        
        // Step 6: Update ledger to "fulfilled"
        $this->schema->update_transaction($woo_order_id, array(
            'status' => 'fulfilled',
            'fulfillment_status' => 'completed',
            'fulfilled_at' => current_time('mysql'),
        ));
        
        $this->logger->log(
            'Payment completion processing finished',
            PGB_Logger::LOG_INFO,
            array('woo_order_id' => $woo_order_id),
            __FILE__
        );
        
        return true;
    }
}
```

```php
/**
 * CLASS: PGB_Idempotency
 * 
 * Idempotency checking to prevent duplicate order processing.
 * 
 * Strategy:
 * - Database: Check wp_bzj_pgb_transactions for UNIQUE constraints
 * - Transient: Check WordPress transient cache
 * - Result: Return cached result OR create new order
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Idempotency {
    private $logger = null;

    public function __construct($logger = null) {
        $this->logger = $logger;
    }

    /**
     * Check for duplicate request
     * 
     * Returns existing transaction if found by request_id or wow_order_id
     * 
     * @param string $request_id Request ID to check
     * @param array $params Order parameters (contains wow_order_id, etc.)
     * @return array|null Existing transaction or null
     */
    public function check_duplicate($request_id, $params) {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'bzj_pgb_transactions';
        
        // Check by request_id first (most reliable)
        $existing = $wpdb->get_row(
            $wpdb->prepare(
                "SELECT * FROM " . $table_name . " WHERE request_id = %s LIMIT 1",
                $request_id
            ),
            ARRAY_A
        );
        
        if ($existing) {
            if ($this->logger) {
                $this->logger->log(
                    'Duplicate detected by request_id',
                    PGB_Logger::LOG_WARNING,
                    array('request_id' => $request_id),
                    __FILE__
                );
            }
            return $existing;
        }
        
        // Check by wow_order_id if provided
        if (!empty($params['wow_order_id'])) {
            $existing = $wpdb->get_row(
                $wpdb->prepare(
                    "SELECT * FROM " . $table_name . " WHERE wow_order_id = %s LIMIT 1",
                    $params['wow_order_id']
                ),
                ARRAY_A
            );
            
            if ($existing) {
                if ($this->logger) {
                    $this->logger->log(
                        'Duplicate detected by wow_order_id',
                        PGB_Logger::LOG_WARNING,
                        array('wow_order_id' => $params['wow_order_id']),
                        __FILE__
                    );
                }
                return $existing;
            }
        }
        
        return null;
    }

    /**
     * Mark request as processed
     */
    public function mark_processed($request_id, $woo_order_id) {
        set_transient('pgb_proc_' . $request_id, $woo_order_id, 86400); // 24 hours
    }
}
```

### **FILE 10: shared/payment-gateway/pgb-client-streams.php**

```php
<?php
/**
 * FILE: pgb-client-streams.php
 * 
 * Buzzjuice Streams Payment Gateway Client
 * 
 * Invoked from: streams/assets/wow-pgb/wow-pgb_init.php
 * 
 * Purpose: 
 * - Accept Streams payment initiation
 * - Prepare order data
 * - Send HMAC-signed request to WordPress PGB endpoint
 * - Return payment URL to Streams
 * 
 * CRITICAL: This file runs in Streams context (NO WordPress functions available)
 * Use native PHP only: curl, json, basic functions
 */

// No WordPress loading - Streams is standalone

/**
 * Initialize payment request from Streams
 * 
 * Expected POST parameters from Streams:
 * - wow_user_id: WoWonder user ID
 * - wow_email: User email
 * - wow_first_name, wow_last_name: User name
 * - transaction_kind: PRO|PRODUCT|WALLET|DONATE
 * - amount: Total amount
 * - currency: ISO currency code
 * - line_items: JSON array of items
 * - product_id, product_price, product_units: Legacy compatibility
 * - wow_post_id, address_id: Optional metadata
 * 
 * @return array JSON response
 */
function pgb_streams_init_payment() {
    header('Content-Type: application/json; charset=utf-8');
    
    // Security: Verify we have authenticated Streams user
    if (!isset($GLOBALS['loggedin']) || !$GLOBALS['loggedin']) {
        http_response_code(401);
        return json_encode(array('status' => 'error', 'message' => 'Not authenticated'));
    }
    
    $user = $GLOBALS['user'];
    $user_id = intval($user['user_id'] ?? 0);
    $user_email = trim($user['email'] ?? '');
    
    if (!$user_id || !filter_var($user_email, FILTER_VALIDATE_EMAIL)) {
        http_response_code(400);
        return json_encode(array('status' => 'error', 'message' => 'Invalid user data'));
    }
    
    // Parse incoming request
    $method = $_SERVER['REQUEST_METHOD'] ?? 'POST';
    $data = ($method === 'POST') ? $_POST : $_GET;
    
    // Build order payload
    $payload = array(
        'wow_user_id' => $user_id,
        'wow_email' => $user_email,
        'wow_first_name' => $user['first_name'] ?? '',
        'wow_last_name' => $user['last_name'] ?? '',
        'transaction_kind' => strtoupper($data['transaction_kind'] ?? 'UNKNOWN'),
        'amount' => floatval($data['amount'] ?? 0),
        'currency' => strtoupper($data['currency'] ?? 'GHS'),
        'line_items' => array(),
    );
    
    // Parse line items
    if (!empty($data['product_id'])) {
        // Single product (legacy format)
        $payload['line_items'][] = array(
            'product_id' => intval($data['product_id']),
            'quantity' => intval($data['product_units'] ?? 1),
            'price' => floatval($data['product_price'] ?? $payload['amount']),
        );
    } elseif (!empty($data['line_items'])) {
        // Multiple items (JSON format)
        $items = is_array($data['line_items']) ? $data['line_items'] : json_decode($data['line_items'], true);
        if (is_array($items)) {
            $payload['line_items'] = $items;
        }
    }
    
    // Optional metadata
    if (!empty($data['wow_order_id'])) {
        $payload['wow_order_id'] = $data['wow_order_id'];
    }
    if (!empty($data['wow_post_id'])) {
        $payload['wow_post_id'] = $data['wow_post_id'];
    }
    if (!empty($data['address_id'])) {
        $payload['address_id'] = $data['address_id'];
    }
    
    // Send to WordPress PGB endpoint
    $result = pgb_send_to_wordpress($payload);
    
    if ($result['success']) {
        http_response_code(200);
        return json_encode(array(
            'status' => 'success',
            'order_id' => $result['order_id'],
            'payment_url' => $result['payment_url'],
        ));
    } else {
        http_response_code(500);
        return json_encode(array(
            'status' => 'error',
            'message' => $result['error'] ?? 'Unknown error',
        ));
    }
}

/**
 * Send payment request to WordPress PGB endpoint with HMAC signature
 * 
 * POST /wp-json/bzj-pgb/v1/order
 * Headers:
 * - X-BZJ-PGB-Timestamp: Current Unix timestamp
 * - X-BZJ-PGB-Request-ID: Unique request identifier
 * - X-BZJ-PGB-Signature: HMAC-SHA256 signature
 * 
 * @param array $payload Order data
 * @return array ['success' => bool, 'order_id' => int, 'payment_url' => string, 'error' => string]
 */
function pgb_send_to_wordpress($payload) {
    // Get configuration
    $db_config = pgb_get_database_config();
    if (!$db_config) {
        return array('success' => false, 'error' => 'Database configuration unavailable');
    }
    
    $wordpress_url = $db_config['wordpress_url'] ?? 'https://buzzjuice.net';
    $secret = $db_config['pgb_secret'] ?? getenv('BZJ_PGB_SECRET');
    
    if (!$secret) {
        return array('success' => false, 'error' => 'PGB secret not configured');
    }
    
    // Generate request metadata
    $timestamp = time();
    $request_id = 'pgb_' . uniqid() . '_' . $timestamp;
    $json_body = json_encode($payload);
    
    // Build HMAC signature
    $signed_msg = $timestamp . "\n" . $request_id . "\n" . $json_body;
    $signature = base64_encode(hash_hmac('sha256', $signed_msg, $secret, true));
    
    // Prepare headers
    $headers = array(
        'Content-Type: application/json',
        'X-BZJ-PGB-Timestamp: ' . $timestamp,
        'X-BZJ-PGB-Request-ID: ' . $request_id,
        'X-BZJ-PGB-Signature: ' . $signature,
    );
    
    // Send POST request
    $endpoint = $wordpress_url . '/wp-json/bzj-pgb/v1/order';
    
    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, $endpoint);
    curl_setopt($ch, CURLOPT_POST, 1);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $json_body);
    curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_TIMEOUT, 30);
    
    // SSL verification - use full CA verification in production
    if (strpos($wordpress_url, 'https') === 0) {
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, true);
        curl_setopt($ch, CURLOPT_SSL_VERIFYHOST, 2);
    }
    
    $response_body = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $curl_error = curl_error($ch);
    curl_close($ch);
    
    // Parse response
    if ($curl_error) {
        return array('success' => false, 'error' => 'cURL error: ' . $curl_error);
    }
    
    if ($http_code !== 200) {
        return array('success' => false, 'error' => 'HTTP ' . $http_code . ': ' . $response_body);
    }
    
    $response = json_decode($response_body, true);
    if (!$response || !is_array($response)) {
        return array('success' => false, 'error' => 'Invalid response format');
    }
    
    if ($response['success'] ?? false) {
        return array(
            'success' => true,
            'order_id' => $response['order_id'],
            'payment_url' => $response['payment_url'],
        );
    } else {
        return array('success' => false, 'error' => $response['message'] ?? 'Unknown error');
    }
}

/**
 * Get database configuration from Streams
 * 
 * Reads from: shared/db_helpers.php
 * 
 * @return array Configuration or null
 */
function pgb_get_database_config() {
    // From streams context, db_helpers is in shared/
    $db_helpers = dirname(__DIR__, 2) . '/db_helpers.php';
    
    if (!file_exists($db_helpers)) {
        return null;
    }
    
    require_once $db_helpers;
    
    // Extract configuration from db_helpers
    $config = array();
    
    if (defined('WORDPRESS_URL')) {
        $config['wordpress_url'] = WORDPRESS_URL;
    }
    
    if (defined('BZJ_PGB_SECRET')) {
        $config['pgb_secret'] = BZJ_PGB_SECRET;
    }
    
    return !empty($config) ? $config : null;
}
```
