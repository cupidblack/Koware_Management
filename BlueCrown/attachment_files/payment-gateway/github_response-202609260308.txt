# **Buzzjuice Streams (WoWonder) Payment Gateway Bridge - Comprehensive Upgrade Analysis & Proposal**

## **Executive Summary**

The current payment gateway bridge exhibits architectural fragmentation and reliability issues due to:
- **Scattered logic** across multiple files without clear separation of concerns
- **Missing idempotency** mechanisms leading to duplicate order processing
- **Inadequate error handling and recovery** workflows
- **Weak audit trails** for debugging dropped/failed orders
- **Poor integration cohesion** between WoWonder, WordPress, and WooCommerce systems

The proposed upgrade restructures the system into a **centralized, plugin-driven architecture** with comprehensive logging, transaction tracking, and addon/integration support.

**Critical Problems:**
1. **No server-side session validation** in wow-pgb_init.php
2. **Order data loss** if user abandons WooCommerce checkout
3. **No retry logic** if webhook fails
4. **Affiliate WP logic embedded** in wow-pgb_webhook.php (tight coupling)
5. **No idempotency guards** – duplicate webhook calls = duplicate credits/activations
6. **Logging is scattered** – difficult to trace order lifecycle
7. **Database operations across systems** without transaction boundaries
8. **Hard-coded configurations** vs. dynamic settings

### **3.1 `shared/payment-gateway/pgb-core.php`**

**Purpose:** Central PGB client handling order creation, validation, and state management.

```php
<?php
/**
 * Payment Gateway Bridge - Core Client
 * shared/payment-gateway/pgb-core.php
 */

if (!defined('PGB_CORE_LOADED')) {
    define('PGB_CORE_LOADED', true);

    /**
     * PGB_Client: Main interface for payment gateway operations
     */
    class PGB_Client {
        
        private $config = [];
        private $logger = null;
        private $validator = null;
        private $idempotency = null;
        
        // Database connections
        private $wp_db = null;
        private $streams_db = null;
        
        public function __construct($config = []) {
            // Load configuration
            $this->config = array_merge($this->get_default_config(), $config);
            
            // Initialize utilities
            require_once __DIR__ . '/pgb-logger.php';
            require_once __DIR__ . '/pgb-validator.php';
            require_once __DIR__ . '/pgb-idempotency.php';
            
            $this->logger = new PGB_Logger($this->config['logging']);
            $this->validator = new PGB_Validator($this->logger);
            $this->idempotency = new PGB_Idempotency($this->logger);
        }
        
        /**
         * Get database connections (lazy load)
         */
        public function get_wp_db() {
            if (!$this->wp_db) {
                require_once dirname(__DIR__) . '/db_helpers.php';
                $this->wp_db = get_wp_db_conn();
            }
            return $this->wp_db;
        }
        
        public function get_streams_db() {
            if (!$this->streams_db) {
                require_once dirname(__DIR__) . '/db_helpers.php';
                $this->streams_db = get_wowonder_db();
            }
            return $this->streams_db;
        }
        
        /**
         * Create order from Streams side
         * Called by: wow-pgb_init.php
         * 
         * @param array $order_data {
         *     'user_id'        => int,
         *     'amount'         => float,
         *     'product_name'   => string,
         *     'product_id'     => int,
         *     'product_price'  => float,
         *     'product_units'  => int,
         *     'product_owner_id' => int,
         *     'transaction_kind' => 'PRODUCT|PRO|WALLET|DONATE',
         *     'currency_code'  => string,
         *     'affiliate_ctx'  => array (optional)
         * }
         * @return array {status: int, order_id?: string, error?: string}
         */
        public function create_order(array $order_data) {
            $this->logger->log('info', 'PGB_Client::create_order() initiated', ['data_keys' => array_keys($order_data)]);
            
            // 1. Validate input
            $validation = $this->validator->validate_order_creation($order_data);
            if (!$validation['valid']) {
                $this->logger->log('error', 'Order validation failed', $validation['errors']);
                return [
                    'status' => 400,
                    'error' => implode('; ', $validation['errors'])
                ];
            }
            
            // 2. Sanitize
            $order_data = $this->validator->sanitize_order_data($order_data);
            
            // 3. Create temporary order record in Streams DB
            try {
                $db = $this->get_streams_db();
                $order_id = 'pgb_' . uniqid() . '_' . time();
                
                $sql = "INSERT INTO Wo_Payment_Transactions 
                        (userid, amount, order_id, kind, currency_code, notes, transaction_dt) 
                        VALUES (?, ?, ?, ?, ?, ?, NOW())";
                
                $stmt = $db->prepare($sql);
                if (!$stmt) {
                    throw new Exception("Prepare failed: " . $db->error);
                }
                
                $notes = "Order initiated (Streams) - Kind: {$order_data['transaction_kind']}";
                $stmt->bind_param(
                    'idsss',
                    $order_data['user_id'],
                    $order_data['amount'],
                    $order_id,
                    $order_data['transaction_kind'],
                    $order_data['currency_code'],
                    $notes
                );
                
                if (!$stmt->execute()) {
                    throw new Exception("Execute failed: " . $stmt->error);
                }
                $stmt->close();
                
                $this->logger->log('info', 'Order created in Streams DB', ['order_id' => $order_id]);
                
                return [
                    'status' => 200,
                    'order_id' => $order_id,
                    'redirect_url' => $this->build_woocommerce_redirect($order_id, $order_data)
                ];
                
            } catch (Exception $e) {
                $this->logger->log('error', 'Order creation failed', ['exception' => $e->getMessage()]);
                return [
                    'status' => 500,
                    'error' => 'Failed to create order'
                ];
            }
        }
        
        /**
         * Handle webhook from WooCommerce
         * Called by: buzzjuice-pgb-core.php (MU plugin)
         * 
         * @param array $woo_order_data Parsed WooCommerce webhook
         * @return array {status: int, message?: string}
         */
        public function handle_webhook(array $woo_order_data) {
            $this->logger->log('info', 'PGB_Client::handle_webhook() initiated', ['woo_order_id' => $woo_order_data['id'] ?? null]);
            
            // 1. Validate webhook signature (should be done by caller, but double-check)
            if (!$this->validator->validate_woocommerce_webhook($woo_order_data)) {
                $this->logger->log('error', 'Webhook validation failed');
                return ['status' => 403, 'message' => 'Invalid webhook'];
            }
            
            // 2. Check idempotency
            $woo_order_id = $woo_order_data['id'];
            if ($this->idempotency->is_processed('woo_order', $woo_order_id)) {
                $this->logger->log('info', 'Webhook already processed (idempotent)', ['woo_order_id' => $woo_order_id]);
                return ['status' => 200, 'message' => 'Already processed'];
            }
            
            // 3. Extract key data
            $order_status = $woo_order_data['status'] ?? null;
            $customer_email = $woo_order_data['billing']['email'] ?? '';
            $total_paid = floatval($woo_order_data['total'] ?? 0);
            
            // 4. Resolve WoWonder user
            $wow_user_id = $this->extract_wow_user_id($woo_order_data);
            if (!$wow_user_id) {
                $this->logger->log('error', 'Failed to resolve WoWonder user', ['email' => $customer_email]);
                return ['status' => 400, 'message' => 'User not found'];
            }
            
            // 5. Update Streams transaction record
            try {
                $this->update_transaction_from_webhook($woo_order_id, $wow_user_id, $woo_order_data);
            } catch (Exception $e) {
                $this->logger->log('error', 'Failed to update transaction', ['exception' => $e->getMessage()]);
                return ['status' => 500, 'message' => 'Transaction update failed'];
            }
            
            // 6. Process order completion (if status is 'completed')
            if (strtolower($order_status) === 'completed') {
                try {
                    $this->process_order_completion($wow_user_id, $woo_order_id, $woo_order_data);
                } catch (Exception $e) {
                    $this->logger->log('error', 'Order completion processing failed', ['exception' => $e->getMessage()]);
                    // Don't fail the webhook, log and continue
                }
            }
            
            // 7. Mark as processed (idempotency)
            $this->idempotency->mark_processed('woo_order', $woo_order_id);
            
            $this->logger->log('info', 'Webhook processing completed', ['woo_order_id' => $woo_order_id]);
            return ['status' => 200, 'message' => 'Webhook processed'];
        }
        
        /**
         * Extract WoWonder user ID from webhook data
         */
        private function extract_wow_user_id(array $woo_order_data) {
            // 1. Check for explicit metadata
            if (!empty($woo_order_data['meta_data'])) {
                foreach ($woo_order_data['meta_data'] as $meta) {
                    if (($meta['key'] ?? '') === 'bz_wow_user_id') {
                        return intval($meta['value']);
                    }
                }
            }
            
            // 2. Fallback: lookup by email
            $customer_email = $woo_order_data['billing']['email'] ?? '';
            if (!empty($customer_email)) {
                $db = $this->get_streams_db();
                $stmt = $db->prepare("SELECT user_id FROM Wo_Users WHERE email = ? LIMIT 1");
                if ($stmt) {
                    $stmt->bind_param('s', $customer_email);
                    $stmt->execute();
                    $stmt->bind_result($user_id);
                    if ($stmt->fetch()) {
                        return intval($user_id);
                    }
                    $stmt->close();
                }
            }
            
            return 0;
        }
        
        /**
         * Update Streams transaction record from WooCommerce webhook
         */
        private function update_transaction_from_webhook($woo_order_id, $wow_user_id, array $woo_order_data) {
            $db = $this->get_streams_db();
            
            $order_status = $woo_order_data['status'] ?? '';
            $payment_method = $woo_order_data['payment_method'] ?? '';
            $total_paid = floatval($woo_order_data['total'] ?? 0);
            
            // Find Streams order_id by woo_order_id (may be stored in notes or as reference)
            $sql = "UPDATE Wo_Payment_Transactions 
                    SET payment_status = ?, payment_method = ?, woo_order_id = ?, amount = ?, transaction_dt = NOW() 
                    WHERE userid = ? AND (order_id LIKE 'pgb_%' OR woo_order_id = ?)
                    ORDER BY transaction_dt DESC LIMIT 1";
            
            $stmt = $db->prepare($sql);
            if (!$stmt) {
                throw new Exception("Prepare failed: " . $db->error);
            }
            
            $woo_id_str = (string)$woo_order_id;
            $stmt->bind_param(
                'sssdis',
                $order_status,
                $payment_method,
                $woo_id_str,
                $total_paid,
                $wow_user_id,
                $woo_order_id
            );
            
            if (!$stmt->execute()) {
                throw new Exception("Execute failed: " . $stmt->error);
            }
            $stmt->close();
        }
        
        /**
         * Process order completion actions
         * Triggers hooks for addons/integrations
         */
        private function process_order_completion($wow_user_id, $woo_order_id, array $woo_order_data) {
            // Trigger action for addons
            do_action('pgb_order_completed', $wow_user_id, $woo_order_id, $woo_order_data);
        }
        
        /**
         * Build WooCommerce redirect URL
         */
        private function build_woocommerce_redirect($order_id, array $order_data) {
            $woo_url = $this->config['woocommerce_store_url'];
            
            $params = [
                'add-to-cart' => $this->config['woocommerce_bridge_product_id'],
                'quantity' => intval($order_data['product_units']),
                'pgb_order_id' => urlencode($order_id),
                'amount' => number_format($order_data['amount'], 2),
                'product_name' => urlencode($order_data['product_name']),
                'product_id' => intval($order_data['product_id']),
                'wow_user' => intval($order_data['user_id'])
            ];
            
            return $woo_url . '/checkout/?' . http_build_query($params);
        }
        
        /**
         * Get default configuration
         */
        private function get_default_config() {
            $this->load_config_file();
            
            return [
                'woocommerce_store_url' => getenv('WOO_STORE_URL') ?: 'https://shop.buzzjuice.net',
                'woocommerce_api_key' => getenv('WOO_API_KEY'),
                'woocommerce_api_secret' => getenv('WOO_API_SECRET'),
                'woocommerce_bridge_product_id' => getenv('WOO_BRIDGE_PRODUCT_ID') ?: 694,
                'webhook_secret' => getenv('WOO_WEBHOOK_SECRET'),
                'logging' => [
                    'enabled' => (getenv('PGB_LOGGING_ENABLED') ?: '1') === '1',
                    'level' => getenv('PGB_LOGGING_LEVEL') ?: 'info',
                    'directory' => dirname(__FILE__) . '/logs'
                ]
            ];
        }
        
        private function load_config_file() {
            $config_file = __DIR__ . '/pgb-config.php';
            if (file_exists($config_file)) {
                include_once $config_file;
            }
        }
        
        public function get_logger() {
            return $this->logger;
        }
    }
    
}
?>
```

### **3.2 `shared/payment-gateway/pgb-logger.php`**

**Purpose:** Centralized, structured logging with file rotation and filtering.

```php
<?php
/**
 * Payment Gateway Bridge - Logging Utility
 * shared/payment-gateway/pgb-logger.php
 */

if (!defined('PGB_LOGGER_LOADED')) {
    define('PGB_LOGGER_LOADED', true);

    class PGB_Logger {
        
        private $config = [];
        private $log_directory = '';
        private $current_log_file = '';
        
        public function __construct($config = []) {
            $this->config = array_merge([
                'enabled' => true,
                'level' => 'info',      // debug, info, warn, error
                'directory' => sys_get_temp_dir() . '/pgb-logs',
                'max_size' => 10485760, // 10MB
                'retention_days' => 30
            ], $config);
            
            $this->log_directory = $this->config['directory'];
            
            // Ensure directory exists
            if (!is_dir($this->log_directory)) {
                @mkdir($this->log_directory, 0755, true);
            }
            
            // Set current log file
            $this->current_log_file = $this->log_directory . '/pgb-' . date('Y-m-d') . '.log';
        }
        
        /**
         * Log a message
         * 
         * @param string $level debug|info|warn|error
         * @param string $message
         * @param array $context Additional data
         */
        public function log($level = 'info', $message = '', $context = []) {
            if (!$this->config['enabled']) {
                return;
            }
            
            $level_priority = [
                'debug' => 0,
                'info' => 1,
                'warn' => 2,
                'error' => 3
            ];
            
            $current_priority = $level_priority[$this->config['level']] ?? 1;
            $message_priority = $level_priority[$level] ?? 1;
            
            if ($message_priority < $current_priority) {
                return;
            }
            
            // Format log entry
            $timestamp = date('Y-m-d H:i:s.u');
            $context_json = !empty($context) ? ' | ' . json_encode($context, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE) : '';
            $log_entry = "[{$timestamp}] [{$level}] {$message}{$context_json}\n";
            
            // Write to file (with lock)
            $this->write_log($this->current_log_file, $log_entry);
            
            // Check file size and rotate if needed
            $this->check_rotation();
        }
        
        /**
         * Write log safely
         */
        private function write_log($file, $entry) {
            if (!is_writable(dirname($file))) {
                // Fallback to PHP error_log
                error_log($entry);
                return;
            }
            
            $fp = @fopen($file, 'a');
            if ($fp) {
                if (flock($fp, LOCK_EX)) {
                    fwrite($fp, $entry);
                    flock($fp, LOCK_UN);
                }
                fclose($fp);
            }
        }
        
        /**
         * Check and rotate log files
         */
        private function check_rotation() {
            if (!file_exists($this->current_log_file)) {
                return;
            }
            
            if (filesize($this->current_log_file) > $this->config['max_size']) {
                $backup_file = $this->current_log_file . '.bak.' . time();
                rename($this->current_log_file, $backup_file);
            }
            
            // Cleanup old logs
            $this->cleanup_old_logs();
        }
        
        /**
         * Remove logs older than retention period
         */
        private function cleanup_old_logs() {
            $now = time();
            $retention_seconds = $this->config['retention_days'] * 86400;
            
            foreach (glob($this->log_directory . '/pgb-*.log*') as $file) {
                if ($now - filemtime($file) > $retention_seconds) {
                    @unlink($file);
                }
            }
        }
        
        public function get_log_directory() {
            return $this->log_directory;
        }
    }
```

### **3.3 `shared/payment-gateway/pgb-idempotency.php`**

**Purpose:** Prevent duplicate webhook processing and duplicate operations.

```php
<?php
/**
 * Payment Gateway Bridge - Idempotency Guard
 * shared/payment-gateway/pgb-idempotency.php
 */

if (!defined('PGB_IDEMPOTENCY_LOADED')) {
    define('PGB_IDEMPOTENCY_LOADED', true);

    class PGB_Idempotency {
        
        private $logger = null;
        private $db = null;
        private $cache_duration = 3600; // 1 hour
        
        public function __construct($logger = null) {
            $this->logger = $logger;
        }
        
        /**
         * Get database connection
         */
        private function get_db() {
            if (!$this->db) {
                require_once dirname(__DIR__) . '/db_helpers.php';
                $this->db = get_wp_db_conn();
            }
            return $this->db;
        }
        
        /**
         * Check if operation was already processed
         * 
         * @param string $operation_type e.g., 'woo_order', 'affiliate_referral'
         * @param string|int $operation_id
         * @return bool
         */
        public function is_processed($operation_type, $operation_id) {
            $db = $this->get_db();
            $hash = $this->generate_idempotency_key($operation_type, $operation_id);
            
            $sql = "SELECT id FROM wp_options 
                    WHERE option_name = %s AND option_value = %s LIMIT 1";
            
            $option_name = 'pgb_idempotency_' . $operation_type;
            
            $stmt = $db->prepare($sql);
            if (!$stmt) {
                // Fallback to transient-style check
                return $this->check_transient('pgb_' . $hash);
            }
            
            $stmt->bind_param('ss', $option_name, $hash);
            $stmt->execute();
            $stmt->store_result();
            $found = $stmt->num_rows > 0;
            $stmt->close();
            
            return $found;
        }
        
        /**
         * Mark operation as processed
         * 
         * @param string $operation_type
         * @param string|int $operation_id
         */
        public function mark_processed($operation_type, $operation_id) {
            $db = $this->get_db();
            $hash = $this->generate_idempotency_key($operation_type, $operation_id);
            $timestamp = current_time('mysql');
            $expires = date('Y-m-d H:i:s', strtotime("+{$this->cache_duration} seconds"));
            
            $option_name = 'pgb_idempotency_' . $operation_type;
            
            // Try insert or update
            $sql = "INSERT INTO wp_options (option_name, option_value, autoload) 
                    VALUES (%s, %s, 'no')
                    ON DUPLICATE KEY UPDATE option_value = VALUES(option_value)";
            
            $stmt = $db->prepare($sql);
            if ($stmt) {
                $stmt->bind_param('ss', $option_name, $hash);
                $stmt->execute();
                $stmt->close();
                
                if ($this->logger) {
                    $this->logger->log('debug', 'Idempotency mark recorded', [
                        'type' => $operation_type,
                        'id' => $operation_id,
                        'hash' => $hash
                    ]);
                }
            }
        }
        
        /**
         * Generate idempotency key
         */
        private function generate_idempotency_key($operation_type, $operation_id) {
            return hash('sha256', "{$operation_type}::{$operation_id}");
        }
        
        /**
         * Transient fallback for checking processed operations
         */
        private function check_transient($key) {
            return get_transient($key) !== false;
        }
    }
    
}
```

```php
        // Check for rebate credit in order meta
        $rebate_cents = intval($wc_order->get_meta('bz_rebate_credit_cents') ?: 0);

        if ($rebate_cents <= 0) {
            $logger->log('debug', '[AffiliateWP] No rebate credit for order', ['woo_order' => $woo_order_id]);
            return;

        require_once dirname(dirname(__DIR__)) . '/db_helpers.php';

            $logger->log('error', '[AffiliateWP] Failed to get WoWonder DB connection');
            return;

        $stmt = $db->prepare("SELECT id FROM Wo_Payment_Transactions 
                             WHERE userid = ? AND woo_order_id = ? AND notes LIKE '%Rebate%' LIMIT 1");

            $stmt->bind_param('is', $wow_user_id, $woo_order_id);
            $stmt->execute();
            $stmt->store_result();

                $stmt->close();
                return;
            }
            $stmt->close();
        }

        // Convert cents to currency amount
        $rebate_amount = round($rebate_cents / 100.0, 2);

        try {
            $db->begin_transaction();

            // Update wallet
            $upd = $db->prepare("UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ? LIMIT 1");

            if (!$ins->execute()) throw new Exception("execute failed: " . $ins->error);

                'wow_user' => $wow_user_id,
                'amount' => $rebate_amount,
                'woo_order' => $woo_order_id
            ]);

        } catch (Exception $e) {
            $db->rollback();
            $logger->log('error', '[AffiliateWP] Failed to apply rebate', [
                'exception' => $e->getMessage(),
                'wow_user' => $wow_user_id
            ]);
        }
```
