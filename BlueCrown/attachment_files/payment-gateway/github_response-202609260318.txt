```php
<?php
/**
 * Centralized logging system for Payment Gateway Bridge
 * 
 * Features:
 * - Multi-channel logging (file, database, external services)
 * - Log level filtering (DEBUG, INFO, WARN, ERROR, CRITICAL)
 * - Toggle enable/disable per component
 * - Automatic log rotation
 * - Structured logging with context
 */

if (!class_exists('PGB_Logger')) {
    class PGB_Logger {
        private static $instance = null;
        private $log_dir;
        private $enabled_channels = [];
        private $log_levels = ['DEBUG', 'INFO', 'WARN', 'ERROR', 'CRITICAL'];
        private $current_level = 'INFO';  // Default level
        
        // Log level priority
        private $level_priority = [
            'DEBUG'    => 0,
            'INFO'     => 1,
            'WARN'     => 2,
            'ERROR'    => 3,
            'CRITICAL' => 4
        ];

        public static function get_instance() {
            if (null === self::$instance) {
                self::$instance = new self();
            }
            return self::$instance;
        }

        private function __construct() {
            $this->log_dir = dirname(__FILE__) . '/logs';
            $this->ensure_log_dir();
            $this->load_config();
        }

        /**
         * Main logging method
         */
        public function log($component, $level, $message, $context = []) {
            // Check if this level should be logged
            if (!$this->should_log($level)) {
                return;
            }

            $timestamp = gmdate('Y-m-d H:i:s');
            $trace = $this->get_trace();
            
            $entry = [
                'timestamp' => $timestamp,
                'component' => $component,
                'level'     => $level,
                'message'   => $message,
                'context'   => $context,
                'trace'     => $trace
            ];

            // Write to component-specific file
            $this->write_log($component, $entry);
            
            // Write to main log
            $this->write_log('pgb-core', $entry);

            // Also use WordPress error_log for critical issues
            if ($level === 'CRITICAL' || $level === 'ERROR') {
                error_log(sprintf(
                    '[%s] [%s] %s - %s',
                    $timestamp,
                    $component,
                    $level,
                    $message
                ));
            }
        }

        private function write_log($component, $entry) {
            $filename = $this->log_dir . '/pgb-' . $component . '.log';
            $formatted = $this->format_entry($entry);
            
            // Append to file
            file_put_contents($filename, $formatted . PHP_EOL, FILE_APPEND | LOCK_EX);
            
            // Rotate if exceeds size
            $this->rotate_if_needed($filename);
        }

        private function format_entry($entry) {
            return sprintf(
                "[%s] %s | %s | %s | Context: %s | %s",
                $entry['timestamp'],
                $entry['level'],
                $entry['component'],
                $entry['message'],
                json_encode($entry['context']),
                $entry['trace']
            );
        }

        private function get_trace() {
            $trace = debug_backtrace(DEBUG_BACKTRACE_IGNORE_ARGS, 5);
            if (count($trace) > 2) {
                $caller = $trace[2];
                return isset($caller['file']) ? basename($caller['file']) . ':' . $caller['line'] : 'unknown';
            }
            return 'unknown';
        }

        private function should_log($level) {
            return $this->level_priority[$level] >= $this->level_priority[$this->current_level];
        }

        private function ensure_log_dir() {
            if (!is_dir($this->log_dir)) {
                wp_mkdir_p($this->log_dir);
            }
        }

        private function load_config() {
            // Load from wp_options or environment
            $config = get_option('pgb_logger_config', [
                'enabled' => true,
                'level'   => 'INFO',
                'channels' => ['file', 'database']
            ]);

            $this->current_level = $config['level'] ?? 'INFO';
            $this->enabled_channels = $config['channels'] ?? ['file'];
        }

        private function rotate_if_needed($filename, $max_size = 5242880) { // 5MB
            if (file_exists($filename) && filesize($filename) > $max_size) {
                $backup = $filename . '.' . date('YmdHis');
                rename($filename, $backup);
                // Optionally compress or delete old backups
                $this->cleanup_old_logs(dirname($filename));
            }
        }

        private function cleanup_old_logs($dir, $days = 30) {
            $cutoff = time() - ($days * 86400);
            foreach (glob($dir . '/pgb-*.log.*') as $file) {
                if (filemtime($file) < $cutoff) {
                    unlink($file);
                }
            }
        }

        // Convenience methods
        public function debug($component, $message, $context = []) {
            $this->log($component, 'DEBUG', $message, $context);
        }

        public function info($component, $message, $context = []) {
            $this->log($component, 'INFO', $message, $context);
        }

        public function warn($component, $message, $context = []) {
            $this->log($component, 'WARN', $message, $context);
        }

        public function error($component, $message, $context = []) {
            $this->log($component, 'ERROR', $message, $context);
        }

        public function critical($component, $message, $context = []) {
            $this->log($component, 'CRITICAL', $message, $context);
        }
    }
}

// Global accessor
function pgb_log($component, $level, $message, $context = []) {
    PGB_Logger::get_instance()->log($component, $level, $message, $context);
}

function pgb_log_error($component, $message, $context = []) {
    PGB_Logger::get_instance()->error($component, $message, $context);
}
```

```php
<?php
/**
 * Transaction Queue System
 * 
 * Handles:
 * - Queue persistence for failed orders
 * - Automatic retry with exponential backoff
 * - Idempotency checks
 * - Transaction state tracking
 */

if (!class_exists('PGB_Transaction_Queue')) {
    class PGB_Transaction_Queue {
        private static $instance = null;
        private $db_table;
        private $max_retries = 5;
        private $base_retry_delay = 300; // 5 minutes

        public static function get_instance() {
            if (null === self::$instance) {
                self::$instance = new self();
            }
            return self::$instance;
        }

        private function __construct() {
            global $wpdb;
            $this->db_table = $wpdb->prefix . 'pgb_transaction_queue';
        }

        /**
         * Enqueue a transaction
         */
        public function enqueue($transaction_data) {
            global $wpdb;

            $enqueued_at = current_time('mysql', true);
            $next_retry = $this->calculate_retry_time(0);

            $result = $wpdb->insert(
                $this->db_table,
                [
                    'transaction_data'  => wp_json_encode($transaction_data),
                    'status'            => 'pending',
                    'retry_count'       => 0,
                    'enqueued_at'       => $enqueued_at,
                    'next_retry_at'     => $next_retry,
                    'idempotency_key'   => $this->generate_idempotency_key($transaction_data),
                    'created_at'        => $enqueued_at
                ],
                ['%s', '%s', '%d', '%s', '%s', '%s', '%s']
            );

            if ($result) {
                pgb_log('transaction-queue', 'INFO', 'Transaction enqueued', [
                    'id' => $wpdb->insert_id,
                    'idempotency_key' => $transaction_data['order_id'] ?? 'unknown'
                ]);
                return $wpdb->insert_id;
            }

            pgb_log('transaction-queue', 'ERROR', 'Failed to enqueue transaction', [
                'error' => $wpdb->last_error
            ]);
            return false;
        }

        /**
         * Check for duplicate based on idempotency key
         */
        public function is_duplicate($idempotency_key) {
            global $wpdb;

            $existing = $wpdb->get_var($wpdb->prepare(
                "SELECT id FROM $this->db_table WHERE idempotency_key = %s AND status IN ('completed', 'processing')",
                $idempotency_key
            ));

            return !empty($existing);
        }

        /**
         * Get pending transactions ready for retry
         */
        public function get_pending_retries() {
            global $wpdb;

            $now = current_time('mysql', true);
            $transactions = $wpdb->get_results($wpdb->prepare(
                "SELECT * FROM $this->db_table 
                 WHERE status = %s AND next_retry_at <= %s AND retry_count < %d
                 ORDER BY next_retry_at ASC 
                 LIMIT 10",
                'pending',
                $now,
                $this->max_retries
            ));

            return $transactions ?: [];
        }

        /**
         * Mark transaction as processing
         */
        public function mark_processing($queue_id) {
            global $wpdb;

            return $wpdb->update(
                $this->db_table,
                ['status' => 'processing'],
                ['id' => $queue_id],
                ['%s'],
                ['%d']
            );
        }

        /**
         * Mark transaction as completed
         */
        public function mark_completed($queue_id, $result_data = []) {
            global $wpdb;

            return $wpdb->update(
                $this->db_table,
                [
                    'status'        => 'completed',
                    'completed_at'  => current_time('mysql', true),
                    'result_data'   => wp_json_encode($result_data)
                ],
                ['id' => $queue_id],
                ['%s', '%s', '%s'],
                ['%d']
            );
        }

        /**
         * Mark transaction for retry
         */
        public function mark_retry($queue_id, $error_message = '') {
            global $wpdb;

            $current = $wpdb->get_row(
                $wpdb->prepare("SELECT retry_count FROM $this->db_table WHERE id = %d", $queue_id)
            );

            if (!$current) {
                return false;
            }

            $retry_count = intval($current->retry_count) + 1;
            $next_retry = $this->calculate_retry_time($retry_count);

            return $wpdb->update(
                $this->db_table,
                [
                    'retry_count'   => $retry_count,
                    'next_retry_at' => $next_retry,
                    'last_error'    => $error_message,
                    'last_attempt'  => current_time('mysql', true)
                ],
                ['id' => $queue_id],
                ['%d', '%s', '%s', '%s'],
                ['%d']
            );
        }

        /**
         * Mark transaction as failed (max retries exceeded)
         */
        public function mark_failed($queue_id, $error_message = '') {
            global $wpdb;

            $result = $wpdb->update(
                $this->db_table,
                [
                    'status'         => 'failed',
                    'failed_at'      => current_time('mysql', true),
                    'last_error'     => $error_message
                ],
                ['id' => $queue_id],
                ['%s', '%s', '%s'],
                ['%d']
            );

            // Alert administrators
            if ($result) {
                do_action('pgb_transaction_failed', $queue_id, $error_message);
                pgb_log('transaction-queue', 'CRITICAL', 'Transaction marked as failed', [
                    'queue_id' => $queue_id,
                    'error' => $error_message
                ]);
            }

            return $result;
        }

        /**
         * Calculate next retry time with exponential backoff
         */
        private function calculate_retry_time($retry_count) {
            $delay = $this->base_retry_delay * pow(2, $retry_count); // Exponential backoff
            $delay = min($delay, 86400); // Cap at 24 hours
            return date('Y-m-d H:i:s', time() + $delay);
        }

        /**
         * Generate idempotency key from transaction data
         */
        private function generate_idempotency_key($transaction_data) {
            $key_parts = [
                $transaction_data['wow_order_id'] ?? '',
                $transaction_data['woo_order_id'] ?? '',
                $transaction_data['user_id'] ?? '',
                $transaction_data['amount'] ?? ''
            ];

            return hash('sha256', implode('|', array_filter($key_parts)));
        }

        /**
         * Create database table (called during activation)
         */
        public static function create_table() {
            global $wpdb;
            $table = $wpdb->prefix . 'pgb_transaction_queue';
            $charset = $wpdb->get_charset_collate();

            $sql = "CREATE TABLE IF NOT EXISTS $table (
                id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                transaction_data LONGTEXT NOT NULL,
                status VARCHAR(50) NOT NULL DEFAULT 'pending',
                retry_count TINYINT UNSIGNED NOT NULL DEFAULT 0,
                idempotency_key VARCHAR(255) NOT NULL UNIQUE,
                enqueued_at DATETIME NOT NULL,
                next_retry_at DATETIME NOT NULL,
                last_attempt DATETIME,
                last_error TEXT,
                completed_at DATETIME,
                failed_at DATETIME,
                result_data LONGTEXT,
                created_at DATETIME NOT NULL,
                updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
                KEY status_retry (status, next_retry_at),
                KEY idempotency (idempotency_key)
            ) $charset;";

            require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
            dbDelta($sql);
        }
    }
}
```

#### **3. ORDER PROCESSOR (pgb-order-processor.php)**

```php
<?php
/**
 * Order Processing Engine
 * 
 * Orchestrates:
 * - Order data validation
 * - WooCommerce API calls
 * - State management
 * - Addon/integration hook firing
 * - Error handling with fallback
 */

if (!class_exists('PGB_Order_Processor')) {
    class PGB_Order_Processor {
        private static $instance = null;
        private $queue;
        private $logger;
        private $validator;
        private $woo_integration;

        public static function get_instance() {
            if (null === self::$instance) {
                self::$instance = new self();
            }
            return self::$instance;
        }

        private function __construct() {
            $this->queue = PGB_Transaction_Queue::get_instance();
            $this->logger = PGB_Logger::get_instance();
            $this->validator = PGB_Validator::get_instance();
            $this->woo_integration = PGB_Int_WooCommerce::get_instance();
        }

        /**
         * Main entry point from Streams
         */
        public function process_order($order_data) {
            pgb_log('order-processor', 'INFO', 'Processing order from Streams', [
                'order_id' => $order_data['wow_order_id'] ?? 'unknown'
            ]);

            // 1. Validate input
            $validation = $this->validator->validate_order_data($order_data);
            if (!$validation['valid']) {
                return $this->error_response(400, $validation['errors']);
            }

            // 2. Check for duplicates (idempotency)
            $idempotency_key = hash('sha256', implode('|', [
                $order_data['wow_order_id'],
                $order_data['user_id'],
                $order_data['amount']
            ]));

            if ($this->queue->is_duplicate($idempotency_key)) {
                pgb_log('order-processor', 'WARN', 'Duplicate order detected', [
                    'order_id' => $order_data['wow_order_id']
                ]);
                return $this->error_response(409, 'Duplicate order');
            }

            // 3. Enqueue for processing
            $queue_id = $this->queue->enqueue($order_data);
            if (!$queue_id) {
                return $this->error_response(500, 'Failed to queue order');
            }

            // 4. Attempt immediate processing
            return $this->process_queued_transaction($queue_id, $order_data);
        }

        /**
         * Process transaction from queue
         */
        public function process_queued_transaction($queue_id, $order_data) {
            $this->queue->mark_processing($queue_id);

            try {
                // Enrich order data
                $enriched = $this->enrich_order_data($order_data);

                // Fire pre-process hook
                do_action('pgb_before_order_process', $enriched);

                // Create WooCommerce order
                $woo_response = $this->woo_integration->create_order($enriched);

                if (!$woo_response['success']) {
                    throw new Exception($woo_response['error'] ?? 'WooCommerce order creation failed');
                }

                $result_data = [
                    'woo_order_id' => $woo_response['order_id'],
                    'payment_url'  => $woo_response['payment_url']
                ];

                // Fire post-process hook
                do_action('pgb_after_order_process', $enriched, $result_data);

                // Process addons/integrations
                $this->process_addons($enriched, $result_data);

                // Mark as completed
                $this->queue->mark_completed($queue_id, $result_data);

                return $this->success_response($result_data);

            } catch (Exception $e) {
                pgb_log('order-processor', 'ERROR', 'Order processing failed', [
                    'queue_id' => $queue_id,
                    'error' => $e->getMessage()
                ]);

                // Determine if retryable
                if ($this->is_retryable_error($e)) {
                    $this->queue->mark_retry($queue_id, $e->getMessage());
                    return $this->error_response(503, 'Order queued for retry');
                } else {
                    $this->queue->mark_failed($queue_id, $e->getMessage());
                    return $this->error_response(400, $e->getMessage());
                }
            }
        }

        /**
         * Enrich order data with computed/fetched values
         */
        private function enrich_order_data($order_data) {
            $enriched = $order_data;

            // Add currency conversion if needed
            if (isset($order_data['wow_currency_code'])) {
                $enriched['converted_amount'] = apply_filters(
                    'pgb_convert_currency',
                    $order_data['amount'],
                    $order_data['wow_currency_code'],
                    'USD'
                );
            }

            // Get WP user ID from WoWonder user
            $wow_user_id = intval($order_data['user_id']);
            if ($wow_user_id > 0) {
                require_once dirname(__FILE__) . '/../db_helpers.php';
                $wow_db = get_wowonder_db();
                if ($wow_db) {
                    $stmt = $wow_db->prepare("SELECT wp_user_id FROM Wo_Users WHERE user_id = ? LIMIT 1");
                    if ($stmt) {
                        $stmt->bind_param('i', $wow_user_id);
                        $stmt->execute();
                        $stmt->bind_result($wp_user_id);
                        if ($stmt->fetch()) {
                            $enriched['wp_user_id'] = intval($wp_user_id);
                        }
                        $stmt->close();
                    }
                }
            }

            // Capture AffiliateWP context at processing time (not during Streams initiation)
            $enriched['affiliate_context'] = $this->capture_affiliate_context($enriched);

            return $enriched;
        }

        /**
         * Capture AffiliateWP context
         */
        private function capture_affiliate_context($order_data) {
            $context = [
                'affiliate_id' => null,
                'referral_data' => []
            ];

            // Check if user is linked to affiliate
            if (function_exists('affwp_get_affiliate_by_user_id')) {
                $wp_user_id = intval($order_data['wp_user_id'] ?? 0);
                if ($wp_user_id > 0) {
                    $affiliate = affwp_get_affiliate_by_user_id($wp_user_id);
                    if ($affiliate && $affiliate->ID) {
                        $context['affiliate_id'] = intval($affiliate->ID);
                        $context['is_affiliate'] = true;
                    }
                }
            }

            return $context;
        }

        /**
         * Process addon integrations
         */
        private function process_addons($enriched_order, $result_data) {
            // Allow addons to hook into processing
            do_action('pgb_process_addons', $enriched_order, $result_data);
        }

        /**
         * Determine if error is retryable
         */
        private function is_retryable_error(Exception $e) {
            $code = $e->getCode();
            $message = $e->getMessage();

            // Network-like errors are retryable
            $retryable_codes = [503, 502, 504, 429];
            $retryable_msgs = ['timeout', 'connection', 'temporarily unavailable'];

            foreach ($retryable_msgs as $msg) {
                if (stripos($message, $msg) !== false) {
                    return true;
                }
            }

            return in_array($code, $retryable_codes);
        }

        private function success_response($data) {
            return [
                'status'  => 200,
                'success' => true,
                'data'    => $data
            ];
        }

        private function error_response($code, $message, $details = []) {
            return [
                'status'  => $code,
                'success' => false,
                'message' => $message,
                'details' => $details
            ];
        }
    }
}
```

```php
            $this->logger = PGB_Logger::get_instance();
        }

        /**
         * Main entry point called from Streams checkout
         */
        public function initiate_payment($streams_order_data) {
            $this->logger->info('streams-client', 'Payment initiation requested', [
                'order_id' => $streams_order_data['wow_order_id'] ?? 'unknown'
            ]);

            // Transform Streams format to PGB format
            $pgb_order_data = $this->transform_streams_order($streams_order_data);

            // Process through order processor
            $result = $this->processor->process_order($pgb_order_data);

            // Return appropriate response
            if ($result['success']) {
                return [
                    'status' => 200,
                    'url'    => $result['data']['payment_url'] ?? '',
                    'order'  => [
                        'wow_order_id' => $pgb_order_data['wow_order_id'],
                        'woo_order_id' => $result['data']['woo_order_id'] ?? null
                    ]
                ];
            } else {
                $this->logger->error('streams-client', 'Payment initiation failed', [
                    'message' => $result['message'],
                    'order_id' => $pgb_order_data['wow_order_id'] ?? 'unknown'
                ]);

                return [
                    'status'  => $result['status'],
                    'message' => $result['message'],
                    'error'   => true
                ];
            }
        }

        /**
         * Transform Streams order format to PGB format
         */
        private function transform_streams_order($streams_data) {
            return [
                // Core identifiers
                'wow_order_id'         => $streams_data['wow_order_id'] ?? uniqid('wow_'),
                'user_id'              => intval($streams_data['user_id'] ?? 0),
                'product_owner_id'     => intval($streams_data['product_owner_id'] ?? 0),

                // Product/service details
                'product_name'         => htmlspecialchars(trim($streams_data['product_name'] ?? '')),
                'product_price'        => floatval($streams_data['product_price'] ?? 0),
                'product_units'        => intval($streams_data['product_units'] ?? 1),
                'wow_post_id'          => intval($streams_data['wow_post_id'] ?? 0),
                'amount'               => floatval($streams_data['amount'] ?? 0),

                // Transaction type
                'transaction_kind'     => preg_replace('/[^a-zA-Z0-9_-]/', '', $streams_data['transaction_kind'] ?? 'PRODUCT'),
                'wow_currency_code'    => htmlspecialchars($streams_data['wow_currency_code'] ?? 'USD'),

                // Optional address
                'address_id'           => intval($streams_data['address_id'] ?? 0),

                // Metadata
                'customer_email'       => filter_var($streams_data['customer_email'] ?? '', FILTER_VALIDATE_EMAIL) ?: '',
                'customer_name'        => htmlspecialchars($streams_data['customer_name'] ?? ''),
                'metadata'             => $streams_data['metadata'] ?? []
            ];
        }
    }
```

#### **5. WEBHOOK HANDLER (pgb-webhook-handler.php)**

```php
<?php
/**
 * Unified Webhook Handler
 * 
 * Processes WooCommerce webhooks with:
 * - Signature verification
 * - Order status tracking
 * - Idempotency checks
 * - Addon notification system
 */

if (!class_exists('PGB_Webhook_Handler')) {
    class PGB_Webhook_Handler {
        private static $instance = null;
        private $logger;
        private $queue;
        private $secret;

        public static function get_instance() {
            if (null === self::$instance) {
                self::$instance = new self();
            }
            return self::$instance;
        }

        private function __construct() {
            $this->logger = PGB_Logger::get_instance();
            $this->queue = PGB_Transaction_Queue::get_instance();
            $this->secret = get_option('pgb_webhook_secret', '');
        }

        /**
         * Main webhook handler
         */
        public function handle_webhook() {
            // Get payload and headers
            $payload = file_get_contents("php://input");
            $headers = getallheaders() ?: [];

            // Verify signature
            if (!$this->verify_signature($payload, $headers)) {
                pgb_log('webhook', 'ERROR', 'Webhook signature verification failed');
                http_response_code(403);
                wp_die('Forbidden');
            }

            // Parse payload
            $data = json_decode($payload, true);
            if (json_last_error() !== JSON_ERROR_NONE) {
                pgb_log('webhook', 'ERROR', 'Invalid JSON payload');
                http_response_code(400);
                wp_die('Bad Request');
            }

            // Check idempotency
            $idempotency_key = $headers['X-WC-Webhook-ID'] ?? $data['id'] ?? '';
            if ($this->is_webhook_duplicate($idempotency_key)) {
                pgb_log('webhook', 'WARN', 'Duplicate webhook detected', ['id' => $idempotency_key]);
                http_response_code(200);
                wp_die('OK');
            }

            // Process webhook
            $this->process_webhook_data($data);

            // Return success
            http_response_code(200);
            wp_die('OK');
        }

        /**
         * Process webhook data
         */
        private function process_webhook_data($data) {
            $woo_order_id = $data['id'] ?? null;
            $order_status = $data['status'] ?? null;

            pgb_log('webhook', 'INFO', 'Processing webhook', [
                'woo_order_id' => $woo_order_id,
                'status' => $order_status
            ]);

            // Find corresponding PGB transaction
            $wow_order_id = $this->extract_wow_order_id($data);
            $wow_user_id = $this->extract_wow_user_id($data);

            if (!$wow_order_id || !$wow_user_id) {
                pgb_log('webhook', 'WARN', 'Could not extract WoWonder IDs from webhook');
                return;
            }

            // Fire before-webhook hook
            do_action('pgb_before_webhook_process', $data);

            // Handle by status
            switch (strtolower($order_status)) {
                case 'completed':
                    $this->handle_completed_order($data, $wow_order_id, $wow_user_id);
                    break;
                case 'failed':
                case 'cancelled':
                    $this->handle_failed_order($data, $wow_order_id, $wow_user_id);
                    break;
                case 'pending':
                case 'processing':
                    $this->handle_pending_order($data, $wow_order_id, $wow_user_id);
                    break;
            }

            // Fire after-webhook hook
            do_action('pgb_after_webhook_process', $data);
        }

        /**
         * Handle completed order
         */
        private function handle_completed_order($data, $wow_order_id, $wow_user_id) {
            pgb_log('webhook', 'INFO', 'Processing completed order', [
                'wow_order_id' => $wow_order_id,
                'woo_order_id' => $data['id']
            ]);

            // Update Streams payment transaction
            $this->update_streams_payment_transaction($data, $wow_order_id, 'completed');

            // Notify addons
            do_action('pgb_order_completed', $data, $wow_order_id, $wow_user_id);
        }

        /**
         * Handle failed order
         */
        private function handle_failed_order($data, $wow_order_id, $wow_user_id) {
            pgb_log('webhook', 'WARN', 'Order failed', [
                'wow_order_id' => $wow_order_id,
                'reason' => $data['status'] ?? 'unknown'
            ]);

            $this->update_streams_payment_transaction($data, $wow_order_id, 'failed');
            do_action('pgb_order_failed', $data, $wow_order_id, $wow_user_id);
        }

        /**
         * Handle pending order
         */
        private function handle_pending_order($data, $wow_order_id, $wow_user_id) {
            pgb_log('webhook', 'INFO', 'Order pending', ['wow_order_id' => $wow_order_id]);
            $this->update_streams_payment_transaction($data, $wow_order_id, 'pending');
            do_action('pgb_order_pending', $data, $wow_order_id, $wow_user_id);
        }

        /**
         * Update Streams payment transaction
         */
        private function update_streams_payment_transaction($woo_data, $wow_order_id, $status) {
            require_once dirname(__FILE__) . '/../db_helpers.php';
            $wow_db = get_wowonder_db();

            if (!$wow_db) {
                pgb_log('webhook', 'ERROR', 'Cannot connect to WoWonder database');
                return;
            }

            $update_sql = "UPDATE Wo_Payment_Transactions SET payment_status = ?, payment_method = ?, woo_order_id = ?, transaction_dt = NOW() WHERE order_id = ?";
            $stmt = $wow_db->prepare($update_sql);

            if (!$stmt) {
                pgb_log('webhook', 'ERROR', 'Prepare failed for payment transaction update', [
                    'error' => $wow_db->error
                ]);
                return;
            }

            $payment_method = $woo_data['payment_method'] ?? 'unknown';
            $woo_order_id = intval($woo_data['id']);

            $stmt->bind_param('ssis', $status, $payment_method, $woo_order_id, $wow_order_id);
            if (!$stmt->execute()) {
                pgb_log('webhook', 'ERROR', 'Execute failed for payment transaction update', [
                    'error' => $stmt->error
                ]);
            }
            $stmt->close();
        }

        /**
         * Extract WoWonder order ID from webhook meta
         */
        private function extract_wow_order_id($data) {
            // Check meta_data
            if (!empty($data['meta_data']) && is_array($data['meta_data'])) {
                foreach ($data['meta_data'] as $meta) {
                    if (($meta['key'] ?? '') === 'wow_order_id') {
                        return $meta['value'] ?? null;
                    }
                }
            }
            return null;
        }

        /**
         * Extract WoWonder user ID from webhook
         */
        private function extract_wow_user_id($data) {
            // Check meta_data for bz_wow_user_id (set by MU-plugin)
            if (!empty($data['meta_data']) && is_array($data['meta_data'])) {
                foreach ($data['meta_data'] as $meta) {
                    if (($meta['key'] ?? '') === 'bz_wow_user_id') {
                        return intval($meta['value'] ?? 0);
                    }
                }
            }

            // Fallback: lookup by email
            if (!empty($data['billing']['email'])) {
                require_once dirname(__FILE__) . '/../db_helpers.php';
                $wow_db = get_wowonder_db();
                if ($wow_db) {
                    $stmt = $wow_db->prepare("SELECT user_id FROM Wo_Users WHERE email = ? LIMIT 1");
                    if ($stmt) {
                        $stmt->bind_param('s', $data['billing']['email']);
                        $stmt->execute();
                        $stmt->bind_result($user_id);
                        if ($stmt->fetch()) {
                            return intval($user_id);
                        }
                        $stmt->close();
                    }
                }
            }

            return null;
        }

        /**
         * Verify webhook signature
         */
        private function verify_signature($payload, $headers) {
            if (empty($this->secret)) {
                pgb_log('webhook', 'WARN', 'No webhook secret configured');
                return false;
            }

            $signature = $headers['X-WC-Webhook-Signature'] ?? '';
            $expected = base64_encode(hash_hmac('sha256', $payload, $this->secret, true));

            return hash_equals($signature, $expected);
        }

        /**
         * Check webhook idempotency
         */
        private function is_webhook_duplicate($webhook_id) {
            global $wpdb;
            $table = $wpdb->prefix . 'pgb_webhook_log';

            // Create table if not exists
            if ($wpdb->get_var("SHOW TABLES LIKE '$table'") !== $table) {
                return false; // Table doesn't exist yet, so not a duplicate
            }

            $existing = $wpdb->get_var($wpdb->prepare(
                "SELECT id FROM $table WHERE webhook_id = %s LIMIT 1",
                $webhook_id
            ));

            if ($existing) {
                return true;
            }

            // Log this webhook
            $wpdb->insert($table, [
                'webhook_id' => $webhook_id,
                'processed_at' => current_time('mysql', true)
            ]);

            return false;
        }
    }
}
```

```php
<?php
/**
 * Payment Gateway Bridge Bootstrap

if (!class_exists('PGB_Bootstrap')) {
    class PGB_Bootstrap {
        private static $instance = null;

        public static function get_instance() {
            if (null === self::$instance) {

            return self::$instance;

            add_action('plugins_loaded', [$this, 'load_dependencies']);
            add_action('init', [$this, 'maybe_install']);
            add_action('init', [$this, 'register_hooks']);

        public function load_dependencies() {
            // Load shared payment gateway components
            $components = [
                'pgb-logger.php',
                'pgb-config.php',
                'pgb-validator.php',
                'pgb-transaction-queue.php',

                'pgb-webhook-handler.php',
                'pgb-client.php',
                'pgb-client-streams.php',
                'pgb-hooks.php'
            ];

            foreach ($components as $component) {
                $path = PGB_SHARED_DIR . $component;
                if (file_exists($path)) {
                    require_once $path;
                }
            }

            // Load integrations
            $integrations = glob(PGB_SHARED_DIR . 'integrations/pgb-int-*.php');
            foreach ($integrations ?: [] as $integration) {
                require_once $integration;
            }

            // Load addons
            $addons = glob(PGB_SHARED_DIR . 'addons/pgb-addon-*.php');
            foreach ($addons ?: [] as $addon) {
                require_once $addon;
            }

            // Load MU-plugin components
            $mu_components = glob(PGB_DIR . 'class-pgb-*.php');
            foreach ($mu_components ?: [] as $component) {
                if (basename($component) !== 'class-pgb-bootstrap.php') {
                    require_once $component;
                }
            }
        }

        public function maybe_install() {
            $version = get_option('pgb_version', '0');
            $current_version = '2.0.0';

            if (version_compare($version, $current_version, '<')) {
                $this->install();
                update_option('pgb_version', $current_version);
            }
        }

        private function install() {
            // Create database tables
            PGB_Transaction_Queue::create_table();
            $this->create_webhook_log_table();

            // Set default options
            $defaults = [
                'pgb_enabled' => true,
                'pgb_logger_config' => [
                    'enabled' => true,
                    'level'   => 'INFO',
                    'channels' => ['file', 'database']
                ]
            ];

            foreach ($defaults as $key => $value) {
                if (!get_option($key)) {
                    add_option($key, $value);
                }
            }

            do_action('pgb_install');
        }

        public function register_hooks() {
            // Register REST/webhook endpoint
            add_action('rest_api_init', [$this, 'register_webhook_endpoint']);

            // Register cron for retrying failed transactions
            if (!wp_next_scheduled('pgb_retry_failed_transactions')) {
                wp_schedule_event(time(), 'every_5_minutes', 'pgb_retry_failed_transactions');
            }
            add_action('pgb_retry_failed_transactions', [$this, 'retry_failed_transactions']);

            // Register settings page
            add_action('admin_menu', [$this, 'add_settings_page']);
        }

        public function register_webhook_endpoint() {
            register_rest_route('pgb/v1', '/webhook/woocommerce', [
                'methods' => 'POST',
                'callback' => [$this, 'handle_woocommerce_webhook'],
                'permission_callback' => '__return_true'
            ]);
        }

        public function handle_woocommerce_webhook() {
            $handler = PGB_Webhook_Handler::get_instance();
            $handler->handle_webhook();
        }

        public function retry_failed_transactions() {
            $queue = PGB_Transaction_Queue::get_instance();
            $processor = PGB_Order_Processor::get_instance();
            $pending = $queue->get_pending_retries();

            foreach ($pending as $transaction) {
                $data = json_decode($transaction->transaction_data, true);
                $processor->process_queued_transaction($transaction->id, $data);
            }
        }

        private function create_webhook_log_table() {

            $table = $wpdb->prefix . 'pgb_webhook_log';
            $charset = $wpdb->get_charset_collate();

            $sql = "CREATE TABLE IF NOT EXISTS $table (
                id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
                webhook_id VARCHAR(255) NOT NULL UNIQUE,
                processed_at DATETIME NOT NULL,
                created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                KEY webhook_id (webhook_id)
            ) $charset;";

            require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
            dbDelta($sql);
        }

        public function add_settings_page() {
            add_options_page(
                'Payment Gateway Bridge Settings',
                'PGB Settings',
                'manage_options',
                'pgb-settings',
                [$this, 'render_settings_page']

                    <?php submit_button(); ?>
                </form>
            </div>
            <?php
        }
```
