<?php
/**
 * Payment Gateway Bridge - Logger
 * Centralized logging with file rotation and severity levels
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_DEBUG;
    private $file_handles = [];

    public function __construct($config = []) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = $config['logging_enabled'] ?? true;
        $this->min_level = $config['log_level'] ?? self::LOG_DEBUG;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Log a message
     */
    public function log($message, $level = self::LOG_INFO, $context = [], $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        $source = $source_file ? basename($source_file) : 'unknown';

        // Format message with context
        $formatted_msg = $message;
        if (!empty($context)) {
            $formatted_msg .= ' | ' . json_encode($context);
        }

        $log_line = "[{$timestamp}] [{$level_upper}] [{$source}] {$formatted_msg}\n";

        // Write to file
        $this->write_to_file($source, $log_line);

        // Also write to WordPress error log if critical
        if (in_array($level, [self::LOG_ERROR, self::LOG_CRITICAL])) {
            error_log($log_line);
        }
    }

    /**
     * Check if message should be logged based on level
     */
    private function should_log($level) {
        $levels = [
            self::LOG_DEBUG => 0,
            self::LOG_INFO => 1,
            self::LOG_WARNING => 2,
            self::LOG_ERROR => 3,
            self::LOG_CRITICAL => 4,
        ];
        return ($levels[$level] ?? 0) >= ($levels[$this->min_level] ?? 0);
    }

    /**
     * Write message to file
     */
    private function write_to_file($source, $message) {
        $file_path = $this->log_dir . str_replace('.php', '', $source) . '.log';

        if (!isset($this->file_handles[$file_path])) {
            if (file_exists($file_path) && filesize($file_path) > 10485760) { // 10MB
                $archived = $file_path . '.' . date('YmdHis');
                @rename($file_path, $archived);
            }
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    /**
     * Close all file handles
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}
?>

     */
    const SUBSCRIPTION_ACTIVATED = 'pgb_subscription_activated';

    /**
     * Hook: Affiliate reward processing
     * 
     * do_action('pgb_affiliate_reward', $wow_user_id, $order_id, $amount)
     */
    const AFFILIATE_REWARD = 'pgb_affiliate_reward';

    /**
     * Hook: Log entry created
     * 
     * do_action('pgb_log_entry', $level, $message, $context)
     */
    const LOG_ENTRY = 'pgb_log_entry';

    /**
     * Register a custom hook listener
     */
    public static function register_hook($hook_name, $callback, $priority = 10, $accepted_args = 1) {
        if (function_exists('add_action') || function_exists('add_filter')) {
            add_action($hook_name, $callback, $priority, $accepted_args);
        }
    }

    /**
     * Execute action hook
     */
    public static function do_action($hook, ...$args) {
        if (function_exists('do_action')) {
            do_action($hook, ...$args);
        }
    }

    /**
     * Apply filter hook
     */
    public static function apply_filters($hook, $value, ...$args) {
        if (function_exists('apply_filters')) {
            return apply_filters($hook, $value, ...$args);
        }
        return $value;
    }
}

// buzzjuice.net/shared/payment-gateway/core/class-pgb-manager.php

<?php
/**
 * Payment Gateway Bridge - Transaction Manager
 * Handles order creation, state management, and idempotency
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Manager {
    private $logger = null;
    private $hooks = null;
    private $wow_db = null;
    private $wp_db = null;

    public function __construct($logger, $hooks) {
        $this->logger = $logger;
        $this->hooks = $hooks;
        $this->init_databases();
    }

    /**
     * Initialize database connections
     */
    private function init_databases() {
        // Load db_helpers if available
        if (!function_exists('get_wowonder_db')) {
            $helpers = dirname(__FILE__, 2) . '/../db_helpers.php';
            if (file_exists($helpers)) {
                require_once $helpers;
            }
        }

        if (function_exists('get_wowonder_db')) {
            $this->wow_db = get_wowonder_db();
        }

        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * Create order in WooCommerce
     * 
     * @param array $order_data Order data prepared by client
     * @param array $wow_user_data WoWonder user data
     * @return array ['success' => bool, 'order_id' => int, 'payment_url' => string, 'error' => string]
     */
    public function create_order($order_data, $wow_user_data) {
        $this->logger->log(
            'Creating WooCommerce order',
            PGB_Logger::LOG_INFO,
            ['order_type' => $order_data['meta_data']['transaction_kind'] ?? 'unknown'],
            __FILE__
        );

        // Validate order data
        $validation = $this->validate_order_data($order_data, $wow_user_data);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed: ' . $validation['error'],
                PGB_Logger::LOG_ERROR,
                ['errors' => $validation['errors']],
                __FILE__
            );
            return [
                'success' => false,
                'error' => $validation['error'],
                'details' => $validation['errors'],
            ];
        }

        // Apply filters
        $order_data = $this->hooks->apply_filters(
            PGB_Hooks::BEFORE_ORDER_CREATE,
            $order_data,
            $order_data['meta_data']['transaction_kind'] ?? 'unknown'
        );

        // Create order in WooCommerce
        try {
            $result = $this->send_to_woocommerce($order_data);

            if ($result['success']) {
                // Store transaction in WoWonder database
                $this->store_transaction_record(
                    $result['woo_order_id'],
                    $wow_user_data,
                    $order_data,
                    'pending'
                );

                $this->logger->log(
                    'Order created successfully',
                    PGB_Logger::LOG_INFO,
                    ['woo_order_id' => $result['woo_order_id']],
                    __FILE__
                );

                // Fire action hook
                $this->hooks->do_action(
                    PGB_Hooks::AFTER_ORDER_CREATE,
                    $result['woo_order_id'],
                    $order_data,
                    $wow_user_data['user_id']
                );

                return [
                    'success' => true,
                    'order_id' => $result['woo_order_id'],
                    'payment_url' => $result['payment_url'],
                ];
            } else {
                $this->logger->log(
                    'WooCommerce API error',
                    PGB_Logger::LOG_ERROR,
                    ['error' => $result['error']],
                    __FILE__
                );
                return [
                    'success' => false,
                    'error' => $result['error'],
                ];
            }
        } catch (Exception $e) {
            $this->logger->log(
                'Exception creating order: ' . $e->getMessage(),
                PGB_Logger::LOG_CRITICAL,
                ['trace' => $e->getTraceAsString()],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Internal server error',
            ];
        }
    }

    /**
     * Validate order data
     */
    private function validate_order_data($order_data, $wow_user_data) {
        $errors = [];

        // Required fields
        $required = ['customer_id', 'currency', 'line_items', 'meta_data'];
        foreach ($required as $field) {
            if (empty($order_data[$field])) {
                $errors[] = "Missing required field: {$field}";
            }
        }

        // Validate customer ID
        if (empty($order_data['customer_id']) || !is_numeric($order_data['customer_id'])) {
            $errors[] = "Invalid customer_id (must be numeric)";
        }

        // Validate line items
        if (!is_array($order_data['line_items']) || empty($order_data['line_items'])) {
            $errors[] = "No line items provided";
        } else {
            foreach ($order_data['line_items'] as $idx => $item) {
                if (empty($item['product_id'])) {
                    $errors[] = "Line item {$idx}: missing product_id";
                }
                if (!isset($item['quantity']) || $item['quantity'] <= 0) {
                    $errors[] = "Line item {$idx}: invalid quantity";
                }
                if (!isset($item['price']) || $item['price'] < 0) {
                    $errors[] = "Line item {$idx}: invalid price";
                }
            }
        }

        // Validate WoWonder user
        if (empty($wow_user_data['user_id']) || empty($wow_user_data['email'])) {
            $errors[] = "Invalid WoWonder user data";
        }

        return [
            'valid' => empty($errors),
            'errors' => $errors,
            'error' => empty($errors) ? null : implode('; ', $errors),
        ];
    }

    /**
     * Send order to WooCommerce API
     */
    private function send_to_woocommerce($order_data) {
        // This is a placeholder - actual implementation would call WC REST API
        // Implementation details in pgb-int-woocommerce-api.php

        if (!function_exists('pgb_wc_create_order')) {
            require_once dirname(__FILE__, 2) . '/integrations/pgb-int-woocommerce-api.php';
        }

        return pgb_wc_create_order($order_data);
    }

    /**
     * Store transaction record in WoWonder
     */
    private function store_transaction_record($woo_order_id, $wow_user_data, $order_data, $status = 'pending') {
        if (!$this->wow_db) {
            $this->logger->log(
                'WoWonder DB not available',
                PGB_Logger::LOG_WARNING,
                [],
                __FILE__
            );
            return false;
        }

        try {
            // Generate or retrieve wow order ID
            $wow_order_id = $order_data['meta_data']['wow_order_id'] ?? uniqid('wow_');
            $transaction_kind = $order_data['meta_data']['transaction_kind'] ?? 'UNKNOWN';
            $amount = 0;
            foreach ($order_data['line_items'] as $item) {
                $amount += floatval($item['total'] ?? 0);
            }

            $sql = "INSERT INTO Wo_Payment_Transactions 
                    (userid, amount, order_id, kind, currency_code, payment_status, woo_order_id, transaction_dt) 
                    VALUES (?, ?, ?, ?, ?, ?, ?, NOW())";

            $stmt = $this->wow_db->prepare($sql);
            if (!$stmt) {
                throw new Exception("Prepare failed: " . $this->wow_db->error);
            }

            $user_id = intval($wow_user_data['user_id']);
            $currency = $order_data['currency'] ?? 'GHS';

            $stmt->bind_param('idssssi', 
                $user_id,
                $amount,
                $wow_order_id,
                $transaction_kind,
                $currency,
                $status,
                $woo_order_id
            );

            if (!$stmt->execute()) {
                throw new Exception("Execute failed: " . $stmt->error);
            }

            $this->logger->log(
                'Transaction record stored',
                PGB_Logger::LOG_DEBUG,
                ['wow_order_id' => $wow_order_id, 'woo_order_id' => $woo_order_id],
                __FILE__
            );

            $stmt->close();
            return true;
        } catch (Exception $e) {
            $this->logger->log(
                'Failed to store transaction: ' . $e->getMessage(),
                PGB_Logger::LOG_ERROR,
                [],
                __FILE__
            );
            return false;
        }
    }

    /**
     * Process payment completion (called from webhook)
     */
    public function process_payment_completion($woo_order_id, $payment_data) {
        $this->logger->log(
            'Processing payment completion',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $woo_order_id],
            __FILE__
        );

        $this->hooks->do_action(
            PGB_Hooks::BEFORE_PAYMENT_PROCESS,
            $woo_order_id,
            $payment_data['transaction_kind'] ?? 'unknown'
        );

        // Execute order type handlers
        if (function_exists('do_action')) {
            do_action(
                'pgb_handle_' . strtolower($payment_data['transaction_kind'] ?? 'unknown'),
                $payment_data,
                $woo_order_id
            );
        }

        $this->hooks->do_action(
            PGB_Hooks::AFTER_PAYMENT_COMPLETE,
            $woo_order_id,
            $payment_data
        );
    }
}
?>

/**

 * Input validation and sanitization
 */

if (!defined('ABSPATH')) {

class PGB_Validator {
    /**
     * Validate email
     */
    public static function validate_email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }

    /**
     * Sanitize email
     */
    public static function sanitize_email($email) {
        $email = filter_var($email, FILTER_VALIDATE_EMAIL);
        return $email ?: '';
    }

    /**
     * Validate amount (positive float)
     */
    public static function validate_amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }

    /**
     * Validate user ID (positive integer)
     */
    public static function validate_user_id($user_id) {
        return intval($user_id) > 0;
    }

    /**
     * Sanitize string with HTML entities
     */
    public static function sanitize_string($value, $allow_html = false) {
        $value = trim($value);
        if (!$allow_html) {
            $value = htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
        }
        return $value;
    }

    /**
     * Validate transaction type
     */
    public static function validate_transaction_type($type) {
        $allowed = ['PRODUCT', 'PRO', 'WALLET', 'DONATE', 'PURCHASE', 'SALE'];
        return in_array(strtoupper($type), $allowed);
    }

    /**
     * Validate currency code
     */
    public static function validate_currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;
    }

    /**
     * Validate webhook signature
     */
    public static function validate_webhook_signature($payload, $signature, $secret) {
        $expected = base64_encode(hash_hmac('sha256', $payload, $secret, true));
        return hash_equals($signature, $expected);
    }
}

/**
 * Validate payment input from POST
 */
function pgb_validate_payment_input($post, $wo) {
    $errors = [];

    // Required parameters
    $required = ['amount', 'product_name', 'product_price', 'product_units', 
                 'product_owner_id', 'transaction_kind', 'wow_currency_code'];

    foreach ($required as $param) {
        if (empty($post[$param])) {
            $errors[$param] = "Missing required parameter";
        }
    }

    if (!empty($errors)) {
        return [
            'valid' => false,
            'errors' => $errors,
            'message' => 'Missing required fields',
        ];
    }

    // Validate types and values
    $amount = floatval($post['amount']);
    if (!PGB_Validator::validate_amount($amount)) {
        $errors['amount'] = 'Invalid amount';
    }

    $product_owner_id = intval($post['product_owner_id']);
    if (!PGB_Validator::validate_user_id($product_owner_id)) {
        $errors['product_owner_id'] = 'Invalid product owner ID';
    }

    $product_price = floatval($post['product_price']);
    if (!PGB_Validator::validate_amount($product_price)) {
        $errors['product_price'] = 'Invalid product price';
    }

    $product_units = intval($post['product_units']);
    if ($product_units <= 0) {
        $errors['product_units'] = 'Invalid product units';
    }

    $transaction_kind = strtoupper(trim($post['transaction_kind']));
    if (!PGB_Validator::validate_transaction_type($transaction_kind)) {
        $errors['transaction_kind'] = 'Invalid transaction kind';
    }

    if (!PGB_Validator::validate_currency($post['wow_currency_code'])) {
        $errors['wow_currency_code'] = 'Invalid currency code';
    }

    if (!empty($errors)) {
        return [
            'valid' => false,
            'errors' => $errors,
            'message' => 'Validation failed',
        ];
    }

    // Return sanitized data
    return [
        'valid' => true,
        'errors' => [],
        'message' => '',
        'data' => [
            'amount' => $amount,
            'product_name' => PGB_Validator::sanitize_string($post['product_name']),
            'product_price' => $product_price,
            'product_units' => $product_units,
            'product_owner_id' => $product_owner_id,
            'transaction_kind' => $transaction_kind,
            'wow_currency_code' => trim($post['wow_currency_code']),
            'wow_post_id' => intval($post['wow_post_id'] ?? 0),
            'address_id' => intval($post['address_id'] ?? 0),
        ],
    ];
}

function pgb_prepare_wc_order_data($input, $wow_user, $wow_order_id, $wo) {

        ['key' => '_buzzjuice_origin', 'value' => 'streams'],

    if ($input['transaction_kind'] === 'PRO') {
        $pro_id = intval($input['wow_post_id']);
        if ($pro_id > 0 && !empty($wo['pro_packages'][$pro_id])) {
            $package = $wo['pro_packages'][$pro_id];
            $meta_data[] = ['key' => '_subscription_period', 'value' => $package['time'] ?? 'month'];
            $meta_data[] = ['key' => '_subscription_interval', 'value' => intval($package['count'] ?? 1)];
            $meta_data[] = ['key' => '_subscription_length', 'value' => intval($package['time_count'] ?? 0)];

    if (!empty($_COOKIE['affwp_ref']) || !empty($_COOKIE['affwp_affiliate_id'])) {
        $affwp_snapshot = [
            'affwp_ref' => $_COOKIE['affwp_ref'] ?? '',
            'affwp_visit' => $_COOKIE['affwp_visit'] ?? '',
            'affwp_visit_id' => $_COOKIE['affwp_visit_id'] ?? '',
            'affwp_affiliate_id' => $_COOKIE['affwp_affiliate_id'] ?? '',
            'affwp_campaign' => $_COOKIE['affwp_campaign'] ?? '',
            'affwp_coupon' => $_COOKIE['affwp_coupon'] ?? '',
            'http_referer' => $_SERVER['HTTP_REFERER'] ?? '',
            'timestamp' => gmdate('c'),
            'snapshot_version' => '2',

            'key' => '_buzzjuice_affwp_context_snapshot',
            'value' => json_encode($affwp_snapshot),
        ];

        [
            'product_id' => intval($input['wow_post_id']) ?: 694, // fallback product ID
            'quantity' => intval($input['product_units']),
            'name' => $input['product_name'],

                ['key' => 'transaction_kind', 'value' => $input['transaction_kind']],
            ],
        ],
    ];

        'customer_id' => $wordpress_user_id,
        'currency' => $input['wow_currency_code'],

        'billing' => $billing,
        'shipping' => $billing, // Use billing for shipping
        'line_items' => $line_items,
        'meta_data' => $meta_data,
    ];
}

/**
 * Send HTTP request
 */
function pgb_wc_send_request($url, $method = 'GET', $data = null, $headers = []) {
    $ch = curl_init();

    $options = [
        CURLOPT_URL => $url,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_CUSTOMREQUEST => strtoupper($method),
        CURLOPT_HTTPHEADER => array_merge([
            'Content-Type: application/json',
        ], $headers),
        CURLOPT_SSL_VERIFYHOST => 0,
        CURLOPT_SSL_VERIFYPEER => 0,
    ];

    if ($data !== null) {
        $options[CURLOPT_POSTFIELDS] = json_encode($data);
    }

    curl_setopt_array($ch, $options);

    $body = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $error = curl_error($ch);

    curl_close($ch);

    return [
        'http_code' => $http_code,
        'body' => $body,
        'error' => $error,
    ];
}

- WooCommerce API URL
- API credentials
- Webhook secret

## **KEY IMPROVEMENTS**

| Aspect | Before | After |
|--------|--------|-------|
| **Error Handling** | Basic try-catch | Comprehensive logging with levels |
| **State Management** | Loose coupling | Atomic transactions with rollback |
| **Idempotency** | Missing | Built-in duplicate detection |
| **Logging** | Scattered error_log() | Centralized with file rotation |
| **Testing** | Difficult | Hook-based allows unit testing |
| **Extensibility** | Hardcoded logic | Addon/integration framework |
| **Transaction Tracking** | Single record | Multi-stage audit trail |
| **Webhook Security** | Manual verification | Centralized signature validation |
