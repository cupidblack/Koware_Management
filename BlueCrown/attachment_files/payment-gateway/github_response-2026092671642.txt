 * POST /wp-json/bzj-pgb/v1/order

 * - X-BZJ-PGB-Timestamp

function pgb_rest_create_order(WP_REST_Request $request) {
    if (!class_exists('PGB_Bootstrap')) {
        return new WP_Error('pgb_not_loaded', 'PGB not loaded', array('status' => 500));
    }
    
    $pgb = PGB_Bootstrap::instance();
    $logger = $pgb->get_logger();
    $manager = $pgb->get_manager();
    
    // Extract headers
    $timestamp = $request->get_header('X-BZJ-PGB-Timestamp');
    $request_id = $request->get_header('X-BZJ-PGB-Request-ID');
    $signature = $request->get_header('X-BZJ-PGB-Signature');
    
    $body = file_get_contents('php://input');
    $params = json_decode($body, true);
    
    $logger->log(
        'REST endpoint called',
        PGB_Logger::LOG_DEBUG,
        array('request_id' => $request_id),
        __FILE__
    );
    
    // Validate JSON
    if (!$params) {
        $logger->log(
            'Invalid JSON in request',
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error('invalid_json', 'Invalid JSON', array('status' => 400));
    }
    
    // Verify signature
    if (!pgb_verify_signature($timestamp, $request_id, $body, $signature)) {
        $logger->log(
            'Signature verification failed',
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error('invalid_signature', 'Signature mismatch', array('status' => 401));
    }
    
    // Verify timestamp freshness (5 minutes)
    $ts_int = intval($timestamp);
    if (abs(time() - $ts_int) > 300) {
        $logger->log(
            'Stale timestamp',
            PGB_Logger::LOG_WARNING,
            array('request_id' => $request_id, 'age_seconds' => abs(time() - $ts_int)),
            __FILE__
        );
        return new WP_Error('stale_timestamp', 'Request timestamp too old', array('status' => 401));
    }
    
    // Check idempotency (prevent duplicate processing)
    $stored = get_transient('pgb_req_' . $request_id);
    if ($stored) {
        $logger->log(
            'Duplicate request detected',
            PGB_Logger::LOG_WARNING,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error('duplicate_request', 'Request already processed', array('status' => 409));
    }
    
    // Mark as processed
    set_transient('pgb_req_' . $request_id, time(), 86400); // 24hr
    
    // Create order
    $result = $manager->create_order($params);
    
    if ($result['success']) {
        $logger->log(
            'Order created successfully',
            PGB_Logger::LOG_INFO,
            array('request_id' => $request_id, 'woo_order_id' => $result['order_id']),
            __FILE__
        );
        
        return array(
            'success' => true,
            'order_id' => $result['order_id'],
            'payment_url' => $result['payment_url'],
        );
    } else {
        $logger->log(
            'Order creation failed: ' . $result['error'],
            PGB_Logger::LOG_ERROR,
            array('request_id' => $request_id),
            __FILE__
        );
        return new WP_Error('order_failed', $result['error'], array('status' => 500));
    }
}

/**
 * Verify HMAC-SHA256 signature
 * 
 * Signature format: HMAC-SHA256(timestamp + "\n" + request_id + "\n" + body, secret)
 */
function pgb_verify_signature($timestamp, $request_id, $body, $signature) {
    // Get secret from server environment (MUST be set in .env or server config)
    $secret = getenv('BZJ_PGB_SECRET');
    
    if (empty($secret)) {
        error_log('[PGB] ERROR: BZJ_PGB_SECRET not set in environment');
        return false;
    }
    
    // Build signed message
    $signed_msg = $timestamp . "\n" . $request_id . "\n" . $body;
    
    // Calculate expected signature
    $expected = base64_encode(hash_hmac('sha256', $signed_msg, $secret, true));
    
    // Compare using constant-time function to prevent timing attacks
    return hash_equals($expected, $signature);
}

<?php
/**
 * CLASS: PGB_Logger
 * 
 * Comprehensive logging system with:
 * - Per-file log output (class-pgb-manager.log, etc.)
 * - Log level filtering (debug, info, warning, error, critical)
 * - File rotation at 10MB
 * - Sensitive data redaction
 * - Consistent JSON context formatting
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_INFO;
    private $file_handles = array();
    private $level_priority = array(
        'debug' => 0,
        'info' => 1,
        'warning' => 2,
        'error' => 3,
        'critical' => 4,
    );

    /**
     * Constructor
     */
    public function __construct($config = array()) {
        $this->log_dir = PGB_LOG_DIR;
        $this->enabled = isset($config['logging_enabled']) ? (bool)$config['logging_enabled'] : true;
        $this->min_level = isset($config['log_level']) ? $config['log_level'] : self::LOG_INFO;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Main logging method
     * 
     * @param string $message The log message
     * @param string $level Log level (debug, info, warning, error, critical)
     * @param array $context Additional context data
     * @param string $source_file The file where log originated (__FILE__)
     */
    public function log($message, $level = self::LOG_INFO, $context = array(), $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        
        // Extract filename from full path
        $source = 'unknown';
        if ($source_file) {
            $source = str_replace('.php', '', basename($source_file));
        }

        // Build log line
        $log_line = "[{$timestamp}] [{$level_upper}] [{$source}] {$message}";

        // Add context
        if (!empty($context) && is_array($context)) {
            $context = $this->redact_sensitive_data($context);
            $log_line .= " | " . json_encode($context, JSON_UNESCAPED_SLASHES);
        }

        $log_line .= "\n";

        // Write to file
        $this->write_to_file($source, $log_line);

        // Also log critical/error to WordPress error log
        if (in_array($level, array(self::LOG_ERROR, self::LOG_CRITICAL))) {
            error_log('[PGB-' . $level_upper . '] ' . $message);
        }
    }

    /**
     * Check if log level should be output
     */
    private function should_log($level) {
        $current = $this->level_priority[$level] ?? 0;
        $minimum = $this->level_priority[$this->min_level] ?? 0;
        return $current >= $minimum;
    }

    /**
     * Redact sensitive values from context arrays
     */
    private function redact_sensitive_data($data) {
        if (!is_array($data)) {
            return $data;
        }

        $sensitive_keys = array(
            'password', 'secret', 'api_key', 'api_secret', 'token', 'signature',
            'credit_card', 'cvv', 'ssn', 'secret_key', 'private_key', 'consumer_secret'
        );

        foreach ($data as $key => &$value) {
            if (is_array($value)) {
                $value = $this->redact_sensitive_data($value);
            } else {
                $lower_key = strtolower($key);
                foreach ($sensitive_keys as $sensitive) {
                    if (strpos($lower_key, $sensitive) !== false) {
                        $value = '[REDACTED]';
                        break;
                    }
                }
            }
        }

        return $data;
    }

    /**
     * Write log message to file with rotation
     */
    private function write_to_file($source, $message) {
        $file_path = $this->log_dir . $source . '.log';

        // Rotate log if it exceeds 10MB
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        // Get or open file handle
        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        // Write to file
        if (is_resource($this->file_handles[$file_path])) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    /**
     * Get log file path for component
     */
    public function get_log_file($component) {
        return $this->log_dir . str_replace('.php', '', $component) . '.log';
    }

    /**
     * Cleanup - close all file handles
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}

                array('%d')
            );
        } else {
            return $this->wpdb->insert(
                $table_name,
                $data,
                array_fill(0, count($data), '%s')
            );
        }

    /**
     * Create order from Streams payment request
     * 
     * Called by REST endpoint with Streams order data.
     * Streams sends:
     * - amount
     * - wow_user_id, wow_email
     * - transaction_kind (PRO, PRODUCT, WALLET, DONATE)
     * - product IDs / metadata
     * 
     * Returns:
     * - success (bool)
     * - order_id (WooCommerce order ID)
     * - payment_url (checkout URL for customer)
     * - error (string if failed)
     */
    public function create_order($request_data) {
        $this->logger->log(
            'create_order called',
            PGB_Logger::LOG_INFO,
            array('transaction_kind' => $request_data['transaction_kind'] ?? 'UNKNOWN'),
            __FILE__
        );

        // Validate input
        $validation = $this->validate_order_request($request_data);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                array('errors' => $validation['errors']),
                __FILE__
            );
            return array(
                'success' => false,
                'error' => implode('; ', $validation['errors']),
            );
        }

        try {
            // Extract data
            $wow_user_id = intval($request_data['wow_user_id']);
            $wow_email = trim($request_data['wow_email'] ?? '');
            $transaction_kind = $request_data['transaction_kind'];
            $amount = floatval($request_data['amount']);
            $currency = $request_data['currency'] ?? 'GHS';
            $request_id = $request_data['request_id'] ?? uniqid('pgb_');

            // Map WoWonder user to WordPress user
            $wp_user_id = $this->get_wp_user_id($wow_user_id, $wow_email);
            if (!$wp_user_id) {
                // If no WordPress user, create one
                $wp_user_id = $this->create_wp_user_from_wow($wow_user_id, $wow_email);
                if (!$wp_user_id) {
                    return array(
                        'success' => false,
                        'error' => 'Could not map or create WordPress user',
                    );
                }
            }

            // Prepare WooCommerce order data
            $wc_order_data = $this->prepare_wc_order_data(
                $request_data,
                $wp_user_id,
                $wow_user_id
            );
            if (!$wc_order_data) {
                return array(
                    'success' => false,
                    'error' => 'Could not prepare WooCommerce order data',
                );
            }

            // Create WooCommerce order
            $woo_order = wc_create_order();
            if (!$woo_order) {
                return array(
                    'success' => false,
                    'error' => 'Failed to create WooCommerce order',
                );
            }

            $woo_order_id = $woo_order->get_id();

            // Add items to order
            foreach ($wc_order_data['line_items'] as $item_data) {
                $product = wc_get_product($item_data['product_id']);
                if ($product) {
                    $woo_order->add_product(
                        $product,
                        $item_data['quantity'],
                        array(
                            'subtotal' => $item_data['price'],
                            'total' => $item_data['total'],
                        )
                    );
                }
            }

            // Set customer
            if ($wp_user_id) {
                $woo_order->set_customer_id($wp_user_id);
            } else {
                $woo_order->set_billing_email($wow_email);
            }

            // Set order meta
            $woo_order->set_currency($currency);
            $woo_order->add_meta_data('wow_user_id', $wow_user_id);
            $woo_order->add_meta_data('wow_order_id', $request_data['wow_order_id'] ?? uniqid('wow_'));
            $woo_order->add_meta_data('transaction_kind', $transaction_kind);
            $woo_order->add_meta_data('pgb_request_id', $request_id);

            // Calculate totals
            $woo_order->calculate_totals();
            $woo_order->save();

            // Record in PGB transaction ledger
            $ledger_data = array(
                'woo_order_id' => $woo_order_id,
                'wow_order_id' => $request_data['wow_order_id'] ?? uniqid('wow_'),
                'wow_user_id' => $wow_user_id,
                'wp_user_id' => $wp_user_id,
                'request_id' => $request_id,
                'transaction_kind' => $transaction_kind,
                'status' => 'order_created',
                'fulfillment_status' => 'pending',
                'amount' => $amount,
                'currency' => $currency,
                'payload' => json_encode($request_data),
            );

            $this->schema->upsert_transaction($ledger_data);

            // Get checkout URL
            $checkout_url = $woo_order->get_checkout_payment_url();

            $this->logger->log(
                'Order created successfully',
                PGB_Logger::LOG_INFO,
                array(
                    'woo_order_id' => $woo_order_id,
                    'wow_user_id' => $wow_user_id,
                    'amount' => $amount,
                ),
                __FILE__
            );

            return array(
                'success' => true,
                'order_id' => $woo_order_id,
                'payment_url' => $checkout_url,
            );

        } catch (Exception $e) {
            $this->logger->log(
                'Exception creating order: ' . $e->getMessage(),
                PGB_Logger::LOG_CRITICAL,
                array('trace' => $e->getTraceAsString()),
                __FILE__
            );
            return array(
                'success' => false,
                'error' => 'Internal server error',
            );
        }

    /**
     * Validate order request data
     */
    private function validate_order_request($data) {
        $errors = array();

        // Required fields
        $required = array('wow_user_id', 'wow_email', 'transaction_kind', 'amount');
        foreach ($required as $field) {
            if (empty($data[$field])) {
                $errors[] = "Missing: $field";
            }
        }

        // Validate types
        if (!empty($data['wow_user_id']) && !is_numeric($data['wow_user_id'])) {
            $errors[] = "wow_user_id must be numeric";
        }

        if (!empty($data['wow_email']) && !filter_var($data['wow_email'], FILTER_VALIDATE_EMAIL)) {
            $errors[] = "Invalid wow_email";
        }

        if (!empty($data['amount']) && floatval($data['amount']) <= 0) {
            $errors[] = "amount must be positive";
        }

        if (!empty($data['transaction_kind']) && !PGB_Validator::validate_transaction_type($data['transaction_kind'])) {
            $errors[] = "Invalid transaction_kind";
        }

        return array(
            'valid' => empty($errors),
            'errors' => $errors,
        );
    }

    /**

    private function get_wp_user_id($wow_user_id, $wow_email) {

        $users = get_users(array(

            'meta_value' => $wow_user_id,
            'number' => 1,
        ));

        if (!empty($users)) {
            return $users[0]->ID;
        }

        // Try by email
        $user = get_user_by('email', $wow_email);
        if ($user) {
            return $user->ID;
        }

        return null;
    }

                array('wow_user_id' => $wow_user_id),

            );
            return null;
        }

        // Store WoWonder user ID mapping
        update_user_meta($wp_user_id, 'wow_user_id', $wow_user_id);

            array('wp_user_id' => $wp_user_id, 'wow_user_id' => $wow_user_id),

        $line_items[] = array(

            'quantity' => 1,
            'price' => $amount,
            'total' => $amount,

class PGB_Idempotency {
    /**
     * Check if request ID has been processed
     */
    public static function is_duplicate_request($request_id) {
        if (!function_exists('get_transient')) {
            return false;
        }
        return get_transient('pgb_req_' . $request_id) !== false;
    }

    /**
     * Mark request as processed
     */
    public static function mark_request_processed($request_id) {
        if (!function_exists('set_transient')) {
            return false;
        }
        return set_transient('pgb_req_' . $request_id, time(), 86400); // 24hr
    }

    /**
     * Check for duplicate order within time window
     */
    public static function check_duplicate_order($wow_user_id, $amount, $transaction_kind, $window_minutes = 60) {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'bzj_pgb_transactions';
        $cutoff = time() - ($window_minutes * 60);
        
        $sql = $wpdb->prepare(
            "SELECT COUNT(*) as cnt FROM $table_name 
             WHERE wow_user_id = %d 
             AND amount = %f 
             AND transaction_kind = %s 
             AND UNIX_TIMESTAMP(created_at) > %d",
            $wow_user_id,
            $amount,
            $transaction_kind,
            $cutoff
        );
        
        $result = $wpdb->get_var($sql);
        return intval($result) > 0;
    }

    );

    // Add optional metadata
    if (!empty($post_data['product_name'])) {
        $request_data['product_name'] = sanitize_text_field($post_data['product_name']);
    }
    if (!empty($post_data['product_units'])) {
        $request_data['product_units'] = intval($post_data['product_units']);
    }
    if (!empty($post_data['wow_post_id'])) {
        $request_data['wow_post_id'] = intval($post_data['wow_post_id']);
    }

    // Call WordPress REST endpoint with HMAC signature
    $response = pgb_call_wordpress_endpoint($request_data);

    header('Content-Type: application/json');
    echo json_encode($response);
}

/**
 * Call WordPress PGB REST endpoint with HMAC-SHA256 signature
 */
function pgb_call_wordpress_endpoint($request_data) {
    // Get WordPress URL and secret from Streams settings
    $wp_url = get_streams_pgb_setting('wordpress_url');
    $shared_secret = get_streams_pgb_setting('pgb_shared_secret');

    if (empty($wp_url) || empty($shared_secret)) {
        return array(
            'status' => 'error',
            'message' => 'WordPress URL or shared secret not configured',
        );
    }

    // Prepare request
    $endpoint = rtrim($wp_url, '/') . '/wp-json/bzj-pgb/v1/order';
    $timestamp = time();
    $request_id = $request_data['request_id'];
    $body_json = json_encode($request_data, JSON_UNESCAPED_SLASHES);

    // Create HMAC-SHA256 signature
    $signed_message = $timestamp . "\n" . $request_id . "\n" . $body_json;
    $signature = base64_encode(hash_hmac('sha256', $signed_message, $shared_secret, true));

    // Make HTTP request
    $response = wp_remote_post($endpoint, array(
        'headers' => array(
            'Content-Type' => 'application/json',
            'X-BZJ-PGB-Timestamp' => $timestamp,
            'X-BZJ-PGB-Request-ID' => $request_id,
            'X-BZJ-PGB-Signature' => $signature,
        ),
        'body' => $body_json,
        'timeout' => 30,
        'sslverify' => true, // IMPORTANT: Enable SSL verification
    ));

    // Handle response
    if (is_wp_error($response)) {
        return array(
            'status' => 'error',
            'message' => 'Failed to communicate with payment gateway: ' . $response->get_error_message(),
        );
    }

    $response_body = json_decode(wp_remote_retrieve_body($response), true);
    $response_code = wp_remote_retrieve_response_code($response);

    if ($response_code === 200 && isset($response_body['success']) && $response_body['success']) {
        return array(
            'status' => 'success',
            'url' => $response_body['payment_url'],
            'order_id' => $response_body['order_id'],
        );
    } else {
        return array(
            'status' => 'error',
            'message' => $response_body['message'] ?? 'Payment gateway error',
        );
    }
}

/**
 * Get Streams PGB setting value
 */
function get_streams_pgb_setting($key) {
    global $sqlConnect;
    
    if (!$sqlConnect) {
        return null;
    }

    // Query Streams settings table (adjust table name if different)
    $sql = "SELECT setting_value FROM Wo_Settings WHERE setting_name = ?";
    $stmt = $sqlConnect->prepare($sql);
    if (!$stmt) {
        return null;
    }

    $stmt->bind_param('s', $key);
    $stmt->execute();
    $result = $stmt->get_result();
    $row = $result->fetch_assoc();
    $stmt->close();

    return isset($row['setting_value']) ? $row['setting_value'] : null;
}

                __FILE__
            );
            return;

        $existing_referral = affiliate_wp()->referrals->get_by(
            'reference',
            $woo_order_id,
            'woocommerce'

    }

    $pgb = PGB_Bootstrap::instance();
    $logger = $pgb->get_logger();

    if (!$order) {

        'Subscription processing started',
        PGB_Logger::LOG_INFO,
        array('order_id' => $woo_order_id),
        __FILE__
    );

        // Check if order contains subscription products
        $has_subscription = false;
        foreach ($order->get_items() as $item) {
            $product = $item->get_product();
            if (WC_Subscriptions_Product::is_subscription($product)) {
                $has_subscription = true;
                break;
            }
        }

        if (!$has_subscription) {
            $logger->log(
                'No subscription products in order',
                PGB_Logger::LOG_DEBUG,
                array(),
                __FILE__
            );
            return;

        $subscription = wcs_create_subscription(array(
            'customer_id' => $order->get_customer_id(),
            'customer_note' => 'Subscription from order #' . $woo_order_id,
        ));

        if (is_wp_error($subscription)) {
            $logger->log(
                'Failed to create subscription: ' . $subscription->get_error_message(),
                PGB_Logger::LOG_ERROR,
                array(),
                __FILE__
            );
            return;
        }

        // Copy items
        foreach ($order->get_items() as $item) {
            $product = $item->get_product();
            if (WC_Subscriptions_Product::is_subscription($product)) {
                $subscription->add_product(
                    $product,
                    $item->get_quantity()
                );
            }
        }

        // Copy billing address
        $subscription->set_billing_address_1($order->get_billing_address_1());
        $subscription->set_billing_address_2($order->get_billing_address_2());
        $subscription->set_billing_city($order->get_billing_city());

        $subscription->update_dates(array(
