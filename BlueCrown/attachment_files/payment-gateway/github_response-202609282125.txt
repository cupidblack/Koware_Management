I'll analyze the existing payment gateway bridge system and the proposal comprehensively, then provide a detailed step-by-step implementation guide for the upgraded architecture.

## Analysis Summary

**Current Issues:**
1. Direct cURL calls to the same domain fail in cPanel environment (DNS resolution issue)
2. API call limits on shared hosting
3. Fragmented affiliate/subscription handling across multiple files
4. Race conditions and duplicate processing risks
5. Lack of centralized logging and error handling

**Proposed Solution:** Database-first approach with WordPress as the authoritative system, complemented by a must-use plugin that orchestrates the entire flow.

```sql
  `wow_order_id` VARCHAR(64) UNIQUE NOT NULL,
  `wow_user_id` INT UNSIGNED NOT NULL,
  `wp_customer_id` INT UNSIGNED DEFAULT NULL,
  `transaction_type` ENUM('PRODUCT', 'PRO', 'WALLET', 'DONATE') NOT NULL,

  `subscription_id` INT UNSIGNED DEFAULT NULL,

  `status` ENUM('pending', 'processing', 'completed', 'failed', 'cancelled') DEFAULT 'pending',
  `woocommerce_order_id` INT UNSIGNED DEFAULT NULL,
  `metadata` LONGTEXT, -- JSON storage for additional data
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  KEY `wow_user_id` (`wow_user_id`),
  KEY `wp_customer_id` (`wp_customer_id`),
```

```php
     * Generate unique order ID
     * 
     * @param string $prefix
     * @return string

    function pgb_generate_order_id($prefix = 'bzj_') {
        try {
            $random = bin2hex(random_bytes(8));
        } catch (Throwable $e) {
            $random = substr(
                hash('sha256', uniqid('', true) . mt_rand()),
                0,
                16
            );
        }
        return $prefix . $random;
    }

if (!function_exists('pgb_array_get')) {
    /**
     * Safely get nested array value
     * 
     * @param array $array
     * @param string $key (supports dot notation: 'key.subkey')
     * @param mixed $default
     * @return mixed
     */
    function pgb_array_get($array, $key, $default = null) {
        if (!is_array($array)) {
            return $default;
        }
        
        if (strpos($key, '.') === false) {
            return isset($array[$key]) ? $array[$key] : $default;
        }
        
        $keys = explode('.', $key);
        $value = $array;
        
        foreach ($keys as $k) {
            if (is_array($value) && isset($value[$k])) {
                $value = $value[$k];
            } else {
                return $default;
            }
        }
        
        return $value;
    }
}
```

```php
<?php
/**
 * Buzzjuice Payment Gateway Configuration
 * 
 * @version 1.0.0

class PGB_Config {
    
    private static $config = null;
    
    /**
     * Get payment gateway configuration from WoWonder
     * 
     * @return array
     */
    public static function get_all() {
        if (self::$config !== null) {
            return self::$config;
        }
        
        self::$config = self::load_from_wow_db();
        return self::$config;
    }
    
    /**
     * Get specific configuration value
     * 
     * @param string $key (supports dot notation)
     * @param mixed $default
     * @return mixed
     */
    public static function get($key, $default = null) {
        $config = self::get_all();
        return pgb_array_get($config, $key, $default);
    }
    
    /**
     * Load configuration from WoWonder database
     * 
     * @return array
     */
    private static function load_from_wow_db() {
        $wow_db = get_wowonder_db();
        
        if (!$wow_db) {
            pgb_log('Failed to load config: WoWonder DB unavailable', 'error', 'pgb-config');
            return self::get_defaults();
        }
        
        $config = [];
        
        // Define keys to retrieve
        $config_keys = [
            'wow_payment',
            'wow_store_url',
            'wow_api_url',
            'wow_api_key',
            'wow_api_secret',
            'wow_webhook_secret',
            'wow_market_id',
            'wow_pro_package_id',
            'wow_pro_package_id_2',
            'wow_pro_package_id_3',
            'wow_pro_package_id_4',
            'wow_wallet_topup_id',
            'wow_crowdfund_id',
            'currency',
        ];
        
        foreach ($config_keys as $key) {
            $result = $wow_db->query(
                sprintf("SELECT value FROM Wo_Settings WHERE setting_name = '%s' LIMIT 1",
                    $wow_db->real_escape_string($key)
                )
            );
            
            if ($result && $result->num_rows > 0) {
                $row = $result->fetch_assoc();
                $config[$key] = $row['value'] ?? null;
            }
        }
        
        return array_merge(self::get_defaults(), $config);
    }
    
    /**
     * Get default configuration
     * 
     * @return array
     */
    private static function get_defaults() {
        return [
            'wow_payment' => 'no',
            'wow_store_url' => '',
            'wow_api_url' => '',
            'wow_api_key' => '',
            'wow_api_secret' => '',
            'wow_webhook_secret' => '',
            'wow_market_id' => 0,
            'wow_pro_package_id' => 0,
            'wow_pro_package_id_2' => 0,
            'wow_pro_package_id_3' => 0,
            'wow_pro_package_id_4' => 0,
            'wow_wallet_topup_id' => 0,
            'wow_crowdfund_id' => 0,
            'currency' => 'USD',
            'pgb_log_enabled' => true,
            'pgb_log_level' => 'info',
        ];
    }
    
    /**
     * Update configuration value
     * 
     * @param string $key
     * @param mixed $value
     * @return bool
     */
    public static function set($key, $value) {
        $wow_db = get_wowonder_db();
        
        if (!$wow_db) {
            pgb_log('Failed to set config: WoWonder DB unavailable', 'error', 'pgb-config');
            return false;
        }
        
        $escaped_key = $wow_db->real_escape_string($key);
        $escaped_value = $wow_db->real_escape_string((string) $value);
        
        $query = sprintf(
            "INSERT INTO Wo_Settings (setting_name, value) VALUES ('%s', '%s')
             ON DUPLICATE KEY UPDATE value = '%s'",
            $escaped_key,
            $escaped_value,
            $escaped_value
        );
        
        $result = $wow_db->query($query);
        
        if ($result) {
            self::$config = null; // Clear cache
            pgb_log("Config updated: $key", 'debug', 'pgb-config');
            return true;
        }
        
        pgb_log("Failed to update config: $key", 'error', 'pgb-config');
        return false;
    }
    
    /**
     * Validate critical configuration
     * 
     * @return array (errors array, empty if valid)
     */
    public static function validate() {
        $errors = [];
        $config = self::get_all();
        
        if (empty($config['wow_payment']) || $config['wow_payment'] !== 'yes') {
            $errors[] = 'Payment gateway is not enabled';
        }
        
        if (empty($config['wow_store_url'])) {
            $errors[] = 'WooCommerce store URL is not configured';
        }
        
        if (empty($config['wow_api_url']) || empty($config['wow_api_key']) || empty($config['wow_api_secret'])) {
            $errors[] = 'WooCommerce API credentials are incomplete';
        }
        
        return $errors;
    }
}

if (!function_exists('pgb_config')) {
    function pgb_config($key = null, $default = null) {
        if ($key === null) {
            return PGB_Config::get_all();
        }
        return PGB_Config::get($key, $default);
    }
}
```

```php
<?php
/**
 * Buzzjuice Payment Gateway Database Operations
 * 
 * @version 1.0.0
 */

class PGB_Database {
    
    private static $wow_db = null;
    private static $wp_db = null;
    
    /**
     * Get WoWonder database connection
     * 
     * @return mysqli|false
     */
    public static function get_wow_db() {
        if (self::$wow_db) {
            return self::$wow_db;
        }
        
        self::$wow_db = get_wowonder_db();
        return self::$wow_db;
    }
    
    /**
     * Get WordPress database connection
     * 
     * @return mysqli|false
     */
    public static function get_wp_db() {
        if (self::$wp_db) {
            return self::$wp_db;
        }
        
        self::$wp_db = get_wp_db_conn();
        return self::$wp_db;
    }
    
    /**
     * Create payment order in database
     * 
     * @param array $order_data
     * @return string|false (returns wow_order_id on success)
     */
    public static function create_payment_order($order_data) {
        $db = self::get_wow_db();
        
        if (!$db) {
            pgb_log('WoWonder DB connection failed', 'error', 'pgb-database');
            return false;
        }
        
        $wow_order_id = pgb_generate_order_id();
        
        $stmt = $db->prepare(
            "INSERT INTO Wo_Payment_Orders 
             (wow_order_id, wow_user_id, wp_customer_id, transaction_type, amount, currency_code, product_id, product_name, affiliate_id, status, metadata) 
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)"
        );
        
        if (!$stmt) {
            pgb_log('Prepare failed: ' . $db->error, 'error', 'pgb-database');
            return false;
        }
        
        $metadata = pgb_json_encode(pgb_array_get($order_data, 'metadata', []));
        
        $stmt->bind_param(
            'siisdsisiss',
            $wow_order_id,
            $order_data['wow_user_id'],
            $order_data['wp_customer_id'],
            $order_data['transaction_type'],
            $order_data['amount'],
            $order_data['currency_code'],
            $order_data['product_id'],
            $order_data['product_name'],
            $order_data['affiliate_id'],
            $order_data['status'],
            $metadata
        );
        
        if (!$stmt->execute()) {
            pgb_log('Execute failed: ' . $stmt->error, 'error', 'pgb-database');
            return false;
        }
        
        pgb_log("Payment order created: $wow_order_id", 'info', 'pgb-database');
        return $wow_order_id;
    }
    
    /**
     * Get payment order from database
     * 
     * @param string $wow_order_id
     * @return array|false
     */
    public static function get_payment_order($wow_order_id) {
        $db = self::get_wow_db();
        
        if (!$db) {
            pgb_log('WoWonder DB connection failed', 'error', 'pgb-database');
            return false;
        }
        
        $result = $db->query(
            sprintf(
                "SELECT * FROM Wo_Payment_Orders WHERE wow_order_id = '%s' LIMIT 1",
                $db->real_escape_string($wow_order_id)
            )
        );
        
        if ($result && $result->num_rows > 0) {
            $order = $result->fetch_assoc();
            if ($order['metadata']) {
                $order['metadata'] = pgb_json_decode($order['metadata']);
            }
            return $order;
        }
        
        return false;
    }
    
    /**
     * Update payment order status
     * 
     * @param string $wow_order_id
     * @param string $status
     * @param array $updates (optional additional fields)
     * @return bool
     */
    public static function update_payment_order($wow_order_id, $status, $updates = []) {
        $db = self::get_wow_db();
        
        if (!$db) {
            pgb_log('WoWonder DB connection failed', 'error', 'pgb-database');
            return false;
        }
        
        if (!pgb_is_valid_order_status($status)) {
            pgb_log("Invalid status: $status", 'error', 'pgb-database');
            return false;
        }
        
        $set_clauses = ["status = '" . $db->real_escape_string($status) . "'"];
        
        foreach ($updates as $field => $value) {
            if (in_array($field, ['woocommerce_order_id', 'subscription_id', 'affiliate_id', 'wp_customer_id'])) {
                $set_clauses[] = sprintf(
                    "%s = %s",
                    $field,
                    is_numeric($value) ? $value : "'" . $db->real_escape_string((string) $value) . "'"
                );
            }
        }
        
        $query = sprintf(
            "UPDATE Wo_Payment_Orders SET %s WHERE wow_order_id = '%s'",
            implode(', ', $set_clauses),
            $db->real_escape_string($wow_order_id)
        );
        
        $result = $db->query($query);
        
        if ($result) {
            pgb_log("Order updated: $wow_order_id -> $status", 'info', 'pgb-database');
            return true;
        }
        
        pgb_log('Update failed: ' . $db->error, 'error', 'pgb-database');
        return false;
    }
    
    /**
     * Get affiliate ID for WoWonder user
     * 
     * @param int $wow_user_id
     * @return int|false
     */
    public static function get_user_affiliate_id($wow_user_id) {
        $wp_user_id = pgb_get_wp_user_from_wow($wow_user_id);
        
        if (!$wp_user_id) {
            return false;
        }
        
        if (!function_exists('affwp_get_affiliate_id')) {
            pgb_log('AffiliateWP not available', 'warning', 'pgb-database');
            return false;
        }
        
        return affwp_get_affiliate_id($wp_user_id);
    }
}

if (!function_exists('pgb_db')) {
    function pgb_db($method = null) {
        if ($method === 'wow') {
            return PGB_Database::get_wow_db();
        } elseif ($method === 'wp') {
            return PGB_Database::get_wp_db();
        }
        return PGB_Database::class;
    }
}
```

```php
<?php
/**
 * Buzzjuice Payment Gateway Logger
 * 
 * @version 1.0.0
 */

class PGB_Logger {
    
    private static $log_dir = null;
    private static $log_levels = ['debug' => 0, 'info' => 1, 'warning' => 2, 'error' => 3];
    
    /**
     * Initialize logger
     * 
     * @return string (log directory path)
     */
    public static function init() {
        if (self::$log_dir) {
            return self::$log_dir;
        }
        
        $log_dir = BZJ_PGB_BASE_PATH . '/log';
        
        if (!is_dir($log_dir)) {
            @mkdir($log_dir, 0750, true);
        }
        
        self::$log_dir = $log_dir;
        return $log_dir;
    }
    
    /**
     * Log message
     * 
     * @param string $message
     * @param string $level
     * @param string $source
     * @return void
     */
    public static function log($message, $level = 'info', $source = 'unknown') {
        if (!BZJ_PGB_LOG_ENABLED) {
            return;
        }
        
        $level = strtolower($level);
        
        // Check log level threshold
        if (!isset(self::$log_levels[$level])) {
            $level = 'info';
        }
        
        $min_level = self::$log_levels[strtolower(BZJ_PGB_LOG_LEVEL)];
        $current_level = self::$log_levels[$level];
        
        if ($current_level < $min_level) {
            return;
        }
        
        $log_dir = self::init();
        
        // Create source-specific log file
        $log_file = $log_dir . '/' . $source . '.log';
        
        $entry = [
            'timestamp' => gmdate('Y-m-d H:i:s'),
            'level' => strtoupper($level),
            'source' => $source,
            'message' => $message,
            'request_id' => self::get_request_id(),
            'memory_usage' => function_exists('memory_get_usage') 
                ? round(memory_get_usage(true) / 1024 / 1024, 2) . ' MB'
                : 'unknown'
        ];
        
        $log_line = pgb_json_encode($entry) . PHP_EOL;
        
        @file_put_contents($log_file, $log_line, FILE_APPEND | LOCK_EX);
    }
    
    /**
     * Get or create request ID
     * 
     * @return string
     */
    private static function get_request_id() {
        static $request_id = null;
        
        if ($request_id) {
            return $request_id;
        }
        
        if (!empty($_SERVER['HTTP_X_REQUEST_ID'])) {
            $request_id = substr(
                preg_replace('/[^a-zA-Z0-9._:-]/', '', (string) $_SERVER['HTTP_X_REQUEST_ID']),
                0,
                128
            );
        } else {
            try {
                $request_id = bin2hex(random_bytes(8));
            } catch (Throwable $e) {
                $request_id = substr(hash('sha256', uniqid('', true) . mt_rand()), 0, 16);
            }
        }
        
        return $request_id;
    }
    
    /**
     * Get recent log entries
     * 
     * @param string $source
     * @param int $limit
     * @return array
     */
    public static function get_recent($source = 'unknown', $limit = 50) {
        $log_dir = self::init();
        $log_file = $log_dir . '/' . $source . '.log';
        
        if (!file_exists($log_file)) {
            return [];
        }
        
        $lines = file($log_file, FILE_SKIP_EMPTY_LINES);
        if (!$lines) {
            return [];
        }
        
        $entries = [];
        $lines = array_slice($lines, -$limit);
        
        foreach ($lines as $line) {
            $entry = pgb_json_decode(trim($line), true);
            if ($entry) {
                $entries[] = $entry;
            }
        }
        
        return $entries;
    }
    
    /**
     * Clear old log files
     * 
     * @param int $days_old (default 30)
     * @return int (number of files deleted)
     */
    public static function cleanup($days_old = 30) {
        $log_dir = self::init();
        $deleted = 0;
        $cutoff_time = time() - ($days_old * 86400);
        
        foreach (glob($log_dir . '/*.log') as $file) {
            if (filemtime($file) < $cutoff_time) {
                if (@unlink($file)) {
                    $deleted++;
                }
            }
        }
        
        return $deleted;
    }
}
```

```php
<?php
/**
 * Buzzjuice Payment Gateway Currency Converter
 * Integrates with WooCommerce Currency Switcher (WOOCS)
 * 
 * @version 1.0.0
 */

class PGB_Currency_Converter {
    
    private static $rates = null;
    
    /**
     * Convert amount from one currency to another
     * 
     * @param float $amount
     * @param string $from_currency
     * @param string $to_currency
     * @return float
     */
    public static function convert($amount, $from_currency, $to_currency) {
        // If same currency, no conversion needed
        if ($from_currency === $to_currency) {
            return pgb_format_amount($amount);
        }
        
        $rate = self::get_conversion_rate($from_currency, $to_currency);
        
        if ($rate === false) {
            pgb_log(
                "Conversion rate not found: $from_currency -> $to_currency",
                'warning',
                'class-pgb-currency-converter'
            );
            return pgb_format_amount($amount); // Return original if rate not found
        }
        
        $converted = $amount * $rate;
        return pgb_format_amount($converted);
    }
    
    /**
     * Get conversion rate between two currencies
     * Uses WOOCS if available, otherwise falls back to config
     * 
     * @param string $from_currency
     * @param string $to_currency
     * @return float|false
     */
    public static function get_conversion_rate($from_currency, $to_currency) {
        if ($from_currency === $to_currency) {
            return 1.0;
        }
        
        // Try WOOCS first (if WooCommerce Subscriptions is active)
        if (function_exists('woocs_get_currencies')) {
            $rate = self::get_woocs_rate($from_currency, $to_currency);
            if ($rate !== false) {
                return $rate;
            }
        }
        
        // Fall back to config rates
        return self::get_config_rate($from_currency, $to_currency);
    }
    
    /**
     * Get rate from WooCommerce Currency Switcher
     * 
     * @param string $from_currency
     * @param string $to_currency
     * @return float|false
     */
    private static function get_woocs_rate($from_currency, $to_currency) {
        if (!function_exists('woocs_get_currencies')) {
            return false;
        }
        
        try {
            $currencies = woocs_get_currencies();
            
            if (empty($currencies[$from_currency]) || empty($currencies[$to_currency])) {
                return false;
            }
            
            $from_rate = floatval($currencies[$from_currency]['rate'] ?? 1);
            $to_rate = floatval($currencies[$to_currency]['rate'] ?? 1);
            
            if ($from_rate === 0 || $to_rate === 0) {
                return false;
            }
            
            return $to_rate / $from_rate;
        } catch (Throwable $e) {
            pgb_log(
                "WOOCS rate retrieval failed: " . $e->getMessage(),
                'warning',
                'class-pgb-currency-converter'
            );
            return false;
        }
    }
    
    /**
     * Get rate from configuration
     * 
     * @param string $from_currency
     * @param string $to_currency
     * @return float|false
     */
    private static function get_config_rate($from_currency, $to_currency) {
        // This would need to be populated from WoWonder config
        // For now, return false to indicate rates not available
        pgb_log(
            "Config rate not available: $from_currency -> $to_currency",
            'debug',
            'class-pgb-currency-converter'
        );
        return false;
    }
    
    /**
     * Cache exchange rates
     * 
     * @param array $rates
     * @return void
     */
    public static function cache_rates($rates) {
        self::$rates = $rates;
    }
}
```

```php
/**

class PGB_Subscription_Handler {
    
    /**
     * Get subscription data for a product
     * 
     * @param string $transaction_type
     * @param int $product_id (wow_post_id for PRO packages)
     * @return array|false
     */
    public static function get_subscription_data($transaction_type, $product_id) {
        if ($transaction_type !== 'PRO') {
            return false;
        }
        
        // Get PRO package ID from config
        $config_key = 'wow_pro_package_id';
        
        if ($product_id === 2) {
            $config_key = 'wow_pro_package_id_2';
        } elseif ($product_id === 3) {
            $config_key = 'wow_pro_package_id_3';
        } elseif ($product_id === 4) {
            $config_key = 'wow_pro_package_id_4';
        }
        
        $product_kind = pgb_config($config_key, 0);
        
        if (!$product_kind) {
            pgb_log("No product ID configured for PRO package $product_id", 'warning', 'class-pgb-subscription-handler');
            return false;
        }
        
        // Get subscription metadata from WoWonder
        $wow_db = PGB_Database::get_wow_db();
        if (!$wow_db) {
            return false;
        }
        
        // Query pro packages table
        $result = $wow_db->query(
            sprintf(
                "SELECT * FROM Wo_ProPackages WHERE id = %d LIMIT 1",
                (int) $product_id
            )
        );
        
        if (!$result || $result->num_rows === 0) {
            pgb_log("Pro package not found: $product_id", 'warning', 'class-pgb-subscription-handler');
            return false;
        }
        
        $package = $result->fetch_assoc();
        
        return [
            'period' => strtolower($package['time'] ?? 'month'),
            'interval' => (int) ($package['count'] ?? 1),
            'length' => (int) ($package['time_count'] ?? 0),
            'product_id' => $product_kind
        ];
    }
    
    /**
     * Process subscription after order completion
     * 
     * @param int $wc_order_id
     * @param int $wow_user_id
     * @return bool
     */
    public static function process_subscription($wc_order_id, $wow_user_id) {
        if (!function_exists('wcs_order_contains_subscription')) {
            pgb_log('WCS not available', 'warning', 'class-pgb-subscription-handler');
            return false;
        }
        
        if (!wcs_order_contains_subscription($wc_order_id)) {
            return true; // No subscription in order
        }
        
        try {
            $subscriptions = wcs_get_subscriptions_for_order($wc_order_id, [
                'order_type' => 'parent'
            ]);
            
            foreach ($subscriptions as $subscription) {
                // Activate subscription
                if (method_exists($subscription, 'update_status')) {
                    $subscription->update_status('active');
                    
                    pgb_log(
                        "Subscription activated: " . $subscription->get_id() . " for user $wow_user_id",
                        'info',
                        'class-pgb-subscription-handler'
                    );
                }
            }
            
            return true;
        } catch (Throwable $e) {
            pgb_log(
                "Subscription processing failed: " . $e->getMessage(),
                'error',
                'class-pgb-subscription-handler'
            );
            return false;
        }
    }
    
    /**
     * Activate PRO package for WoWonder user
     * 
     * @param int $wow_user_id
     * @param int $pro_package_id
     * @param int $duration_days
     * @return bool
     */
    public static function activate_pro_package($wow_user_id, $pro_package_id, $duration_days = 30) {
        $wow_db = PGB_Database::get_wow_db();
        if (!$wow_db) {
            return false;
        }
        
        $expiry_time = time() + ($duration_days * 86400);
        
        // Update user pro status
        $query = sprintf(
            "UPDATE Wo_Users SET pro_type = %d, pro_time = %d WHERE user_id = %d",
            (int) $pro_package_id,
            $expiry_time,
            (int) $wow_user_id
        );
        
        $result = $wow_db->query($query);
        
        if ($result) {
            pgb_log(
                "PRO package activated for user $wow_user_id (package: $pro_package_id, days: $duration_days)",
                'info',
                'class-pgb-subscription-handler'
            );
            return true;
        }
        
        pgb_log(
            "Failed to activate PRO package: " . $wow_db->error,
            'error',
            'class-pgb-subscription-handler'
        );
        
        return false;
    }
```

```php
        if (self::$api_url && self::$consumer_key && self::$consumer_secret) {
            return true;
        }
        
        self::$api_url = rtrim(pgb_config('wow_api_url', ''), '/') . '/wc/v3';
        self::$consumer_key = pgb_config('wow_api_key', '');
        self::$consumer_secret = pgb_config('wow_api_secret', '');
        
        if (!self::$api_url || !self::$consumer_key || !self::$consumer_secret) {
            pgb_log('WooCommerce API credentials missing', 'error', 'pgb-int-woocommerce');
            return false;
        }
        
        return true;
    }
    
    /**
     * Create order in WooCommerce
     * 
     * @param array $order_data
     * @return array (status, order_id, checkout_url, message)
     */
    public static function create_order($order_data) {
        if (!self::init()) {
            return [
                'status' => 'error',
                'message' => 'WooCommerce API not configured'
            ];
        }
        
        $response = self::api_request(
            '/orders',
            'POST',
            $order_data
        );
        
        if ($response['http_code'] !== 201) {
            $error_msg = pgb_array_get($response, 'error', 'Unknown error');
            pgb_log(
                "Order creation failed: $error_msg (Code: {$response['http_code']})",
                'error',
                'pgb-int-woocommerce'
            );
            
            return [
                'status' => 'error',
                'message' => 'Failed to create WooCommerce order: ' . $error_msg,
                'http_code' => $response['http_code']
            ];
        }
        
        $order = pgb_json_decode($response['response'], true);
        
        if (empty($order['id']) || empty($order['checkout_payment_url'])) {
            pgb_log(
                "Invalid order response: " . pgb_json_encode($order),
                'error',
                'pgb-int-woocommerce'
            );
            
            return [
                'status' => 'error',
                'message' => 'Invalid response from WooCommerce'
            ];
        }
        
        pgb_log(
            "WC Order created: {$order['id']}",
            'info',
            'pgb-int-woocommerce'
        );
        
        return [
            'status' => 'success',
            'order_id' => $order['id'],
            'checkout_url' => $order['checkout_payment_url'],
            'message' => 'Order created successfully'
        ];
    }
    
    /**
     * Get order from WooCommerce
     * 
     * @param int $order_id
     * @return array|false
     */
    public static function get_order($order_id) {
        if (!self::init()) {
            return false;
        }
        
        $response = self::api_request("/orders/$order_id", 'GET');
        
        if ($response['http_code'] !== 200) {
            pgb_log("Failed to get order $order_id", 'warning', 'pgb-int-woocommerce');
            return false;
        }
        
        return pgb_json_decode($response['response'], true);
    }
    
    /**
     * Update order status
     * 
     * @param int $order_id
     * @param string $status
     * @return bool
     */
    public static function update_order_status($order_id, $status) {
        if (!self::init()) {
            return false;
        }
        
        $data = ['status' => $status];
        $response = self::api_request("/orders/$order_id", 'POST', $data);
        
        if ($response['http_code'] !== 200) {
            pgb_log(
                "Failed to update order $order_id status to $status",
                'error',
                'pgb-int-woocommerce'
            );
            return false;
        }
        
        return true;
    }
    
    /**
     * Send API request
     * 
     * @param string $endpoint
     * @param string $method
     * @param array $data
     * @return array (http_code, response, error)
     */
    private static function api_request($endpoint, $method = 'GET', $data = null) {
        $url = self::$api_url . $endpoint;
        
        // Add auth parameters to URL
        $separator = strpos($url, '?') === false ? '?' : '&';
        $url .= $separator . 'consumer_key=' . urlencode(self::$consumer_key) . 
                '&consumer_secret=' . urlencode(self::$consumer_secret);
        
        $curl = curl_init();
        
        $options = [
            CURLOPT_URL => $url,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HTTPHEADER => ['Content-Type: application/json'],
            CURLOPT_CUSTOMREQUEST => strtoupper($method),
            CURLOPT_SSL_VERIFYHOST => 0,
            CURLOPT_SSL_VERIFYPEER => 0,
            CURLOPT_TIMEOUT => 30,
        ];
        
        if (!empty($data) && in_array($method, ['POST', 'PUT', 'PATCH'])) {
            $options[CURLOPT_POSTFIELDS] = pgb_json_encode($data);
        }
        
        curl_setopt_array($curl, $options);
        
        $response = curl_exec($curl);
        $http_code = curl_getinfo($curl, CURLINFO_HTTP_CODE);
        $error = curl_error($curl);
        curl_close($curl);
        
        if ($error) {
            pgb_log(
                "cURL error ($method $endpoint): $error",
                'error',
                'pgb-int-woocommerce'
            );
        }
        
        return [
            'http_code' => $http_code,
            'response' => $response,
            'error' => $error
        ];
    }
}
```

```php
    public static function process_commission($payment_order, $wc_order_id) {

                'pgb-addons-affiliate-wp'
            );
            return false;

        // Get WooCommerce order

            pgb_log("WC order not found: $wc_order_id", 'error', 'pgb-addons-affiliate-wp');
            return false;
        }
        
        // Calculate commission
        $commission = self::calculate_commission($payment_order, $wc_order);
        
        if ($commission <= 0) {
            pgb_log(
                "No commission for affiliate $affiliate_id (amount: {$payment_order['amount']})",
                'info',
                'pgb-addons-affiliate-wp'
            );
            return true;
        }
        
        // Create referral
        $referral_id = affwp_add_referral([
            'affiliate_id' => $affiliate_id,
            'visit_id' => 0,
            'customer_id' => $wc_order->get_customer_id(),
            'currency' => $wc_order->get_currency(),
            'amount' => $commission,
            'reference' => $wc_order_id,
            'context' => 'woocommerce_order',
            'status' => 'unpaid',
            'custom_meta' => [
                'wow_order_id' => $payment_order['wow_order_id'],
                'wow_user_id' => $payment_order['wow_user_id'],
                'transaction_type' => $payment_order['transaction_type'],
            ]
        ]);
        
        if ($referral_id) {
            pgb_log(
                "Referral created: $referral_id for affiliate $affiliate_id (commission: $commission)",
                'info',
                'pgb-addons-affiliate-wp'
            );
            
            // Update payment order
            PGB_Database::update_payment_order(
                $payment_order['wow_order_id'],
                'completed',
                ['affiliate_id' => $affiliate_id]
            );
            
            return true;
        }
        
        pgb_log(
            "Failed to create referral for affiliate $affiliate_id",
            'error',
            'pgb-addons-affiliate-wp'
        );
        
        return false;
    }
    
    /**
     * Calculate commission for order
     * 
     * @param array $payment_order
     * @param object $wc_order
     * @return float
     */
    private static function calculate_commission($payment_order, $wc_order) {
        $amount = $payment_order['amount'];
        $rate = 0;
        $rate_type = 'percentage';
        
        // Get affiliate commission rate
        if (function_exists('affwp_get_affiliate_rate')) {
            $affiliate_id = $payment_order['affiliate_id'];
            $affiliate = affwp_get_affiliate($affiliate_id);
            
            if ($affiliate) {
                $rate = (float) $affiliate->rate;
                $rate_type = $affiliate->rate_type;
            }
        }
        
        if ($rate_type === 'percentage') {
            return pgb_format_amount($amount * ($rate / 100));
        }
        
        return pgb_format_amount($rate);
    }
}
```

```php
        if (!empty($config_errors)) {
            return [
                'status' => 'error',
                'message' => 'Payment gateway not configured: ' . implode(', ', $config_errors)
            ];
        }
        
        // Create order
        return PGB_Order_Manager::create_order($order_data);
    }
}

if (!function_exists('pgb_streams_process_checkout')) {
    /**
     * Process order to checkout
     * 
     * @param string $wow_order_id
     * @return array
     */
    function pgb_streams_process_checkout($wow_order_id) {
        // Load payment gateway if not already loaded
        $pgb_path = dirname(dirname(__DIR__)) . '/payment-gateway/pgb-bootstrap.php';
        
        if (!defined('BZJ_PGB_LOADED') && file_exists($pgb_path)) {
            require_once $pgb_path;
        }
        
        return PGB_Order_Manager::process_to_checkout($wow_order_id);
    }
}

if (!function_exists('pgb_streams_get_config')) {
    /**
     * Get payment gateway configuration
     * 
     * @param string $key
     * @param mixed $default
     * @return mixed
     */
    function pgb_streams_get_config($key = null, $default = null) {
        $pgb_path = dirname(dirname(__DIR__)) . '/payment-gateway/pgb-bootstrap.php';
        
        if (!defined('BZJ_PGB_LOADED') && file_exists($pgb_path)) {
            require_once $pgb_path;
        }
        
        if ($key === null) {
            return PGB_Config::get_all();
        }
        
        return PGB_Config::get($key, $default);
    }
}
```

### Phase 9: Database Setup Script

Create: `shared/payment-gateway/install.php`

```php
<?php
/**
 * Buzzjuice Payment Gateway Installation
 * Run this once to create required database tables
 */

if (!defined('ABSPATH')) {
    // Run from command line or directly
    require_once dirname(dirname(__DIR__)) . '/shared/db_helpers.php';
}

function bzj_pgb_install() {
    $wow_db = get_wowonder_db();
    
    if (!$wow_db) {
        die('Error: Could not connect to WoWonder database');
    }
    
    $charset = 'utf8mb4';
    
    // Create payment orders table
    $sql = <<<SQL
    CREATE TABLE IF NOT EXISTS `Wo_Payment_Orders` (
      `id` INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
      `wow_order_id` VARCHAR(64) UNIQUE NOT NULL,
      `wow_user_id` INT UNSIGNED NOT NULL,
      `wp_customer_id` INT UNSIGNED DEFAULT NULL,
      `transaction_type` ENUM('PRODUCT', 'PRO', 'WALLET', 'DONATE') NOT NULL,
      `amount` DECIMAL(10, 2) NOT NULL,
      `currency_code` VARCHAR(3) NOT NULL,
      `product_id` INT UNSIGNED DEFAULT NULL,
      `product_name` VARCHAR(255) DEFAULT NULL,
      `subscription_id` INT UNSIGNED DEFAULT NULL,
      `affiliate_id` INT UNSIGNED DEFAULT NULL,
      `status` ENUM('pending', 'processing', 'completed', 'failed', 'cancelled') DEFAULT 'pending',
      `woocommerce_order_id` INT UNSIGNED DEFAULT NULL,
      `metadata` LONGTEXT,
      `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
      `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
      KEY `wow_user_id` (`wow_user_id`),
      KEY `wp_customer_id` (`wp_customer_id`),
      KEY `status` (`status`),
      KEY `transaction_type` (`transaction_type`),
      KEY `created_at` (`created_at`)
    ) ENGINE=InnoDB DEFAULT CHARSET={$charset};
    SQL;
    
    if (!$wow_db->query($sql)) {
        die('Error creating table: ' . $wow_db->error);
    }
    
    echo "✅ Database tables created successfully\n";
    return true;
}

// Run installation
bzj_pgb_install();
```
