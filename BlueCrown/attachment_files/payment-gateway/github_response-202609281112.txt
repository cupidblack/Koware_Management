<?php
/**
 * Payment Gateway Bridge - Logger
 * Centralized logging with file rotation and severity levels
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_DEBUG;
    private $file_handles = [];
    private $level_hierarchy = [
        self::LOG_DEBUG => 0,
        self::LOG_INFO => 1,
        self::LOG_WARNING => 2,
        self::LOG_ERROR => 3,
        self::LOG_CRITICAL => 4,
    ];

    public function __construct($config = []) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = $config['logging_enabled'] ?? true;
        $this->min_level = $config['log_level'] ?? self::LOG_DEBUG;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }

        // Create .gitkeep
        if (!file_exists($this->log_dir . '.gitkeep')) {
            @file_put_contents($this->log_dir . '.gitkeep', '');
        }
    }

    /**
     * Log a message to file
     */
    public function log($message, $level = self::LOG_INFO, $context = [], $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        
        // Extract filename from full path
        if ($source_file) {
            $filename = basename(str_replace('\\', '/', $source_file));
            $source = str_replace('.php', '', $filename);
        } else {
            $source = 'unknown';
        }

        // Redact sensitive context
        $context = $this->redact_sensitive_data($context);

        // Format log line
        $context_str = !empty($context) ? ' | ' . json_encode($context) : '';
        $log_line = "[{$timestamp}] [{$level_upper}] {$message}{$context_str}\n";

        // Write to component-specific log file
        $this->write_to_file($source, $log_line);

        // Also write to WordPress error log if critical
        if (in_array($level, [self::LOG_ERROR, self::LOG_CRITICAL])) {
            if (function_exists('error_log')) {
                error_log("[PGB-{$level_upper}] {$source}: {$message}");
            }
        }
    }

    /**
     * Check if message should be logged based on level
     */
    private function should_log($level) {
        $current_level = $this->level_hierarchy[$level] ?? 0;
        $min_level = $this->level_hierarchy[$this->min_level] ?? 0;
        return $current_level >= $min_level;
    }

    /**
     * Redact sensitive data from context
     */
    private function redact_sensitive_data($context) {
        if (!is_array($context)) {
            return $context;
        }

        $sensitive_keys = [
            'password', 'secret', 'key', 'token', 'api_key', 'api_secret',
            'consumer_key', 'consumer_secret', 'auth', 'authorization',
            'credit_card', 'card_number', 'cvv', 'ssn', 'account_number',
            'private_key', 'webhook_secret', 'hmac_key'
        ];

        foreach ($context as $key => $value) {
            $key_lower = strtolower($key);
            foreach ($sensitive_keys as $sensitive) {
                if (strpos($key_lower, $sensitive) !== false) {
                    $context[$key] = '[REDACTED]';
                    break;
                }
            }
        }

        return $context;
    }

    /**
     * Write message to component-specific log file
     */
    private function write_to_file($component, $message) {
        $file_path = $this->log_dir . $component . '.log';

        // Handle log rotation
        if (file_exists($file_path) && filesize($file_path) > 10485760) { // 10MB
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        // Open file handle
        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        // Write log
        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    /**
     * Close all file handles
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }

    /**
     * Get log directory
     */
    public function get_log_dir() {
        return $this->log_dir;
    }

    /**
     * Clear old logs (older than X days)
     */
    public function cleanup_old_logs($days = 30) {
        $cutoff = time() - ($days * 86400);
        
        if (!is_dir($this->log_dir)) {
            return;
        }

        foreach (glob($this->log_dir . '*.log.*') as $archived_file) {
            if (filemtime($archived_file) < $cutoff) {
                @unlink($archived_file);
            }
        }
    }
}

<?php
/**
 * Payment Gateway Bridge - Validator
 * Input validation and sanitization
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Validator {
    private $logger = null;

    public function __construct($logger = null) {
        $this->logger = $logger;
    }

    /**
     * Validate email
     */
    public function validate_email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }

    /**
     * Sanitize email
     */
    public function sanitize_email($email) {
        $email = filter_var($email, FILTER_VALIDATE_EMAIL);
        return $email ?: '';
    }

    /**
     * Validate amount (positive float)
     */
    public function validate_amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }

    /**
     * Validate user ID (positive integer)
     */
    public function validate_user_id($user_id) {
        return intval($user_id) > 0;
    }

    /**
     * Sanitize string
     */
    public function sanitize_string($value, $allow_html = false, $max_length = null) {
        $value = trim($value);
        
        if (!$allow_html) {
            $value = htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
        }
        
        if ($max_length && strlen($value) > $max_length) {
            $value = substr($value, 0, $max_length);
        }
        
        return $value;
    }

    /**
     * Validate transaction type
     */
    public function validate_transaction_type($type) {
        $allowed = ['PRODUCT', 'PURCHASE', 'PRO', 'WALLET', 'DONATE', 'SALE'];
        return in_array(strtoupper($type), $allowed);
    }

    /**
     * Validate currency code (3-letter ISO code)
     */
    public function validate_currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;
    }

    /**
     * Validate URL
     */
    public function validate_url($url) {
        return filter_var($url, FILTER_VALIDATE_URL) !== false;
    }

    /**
     * Validate HMAC signature
     */
    public function validate_hmac_signature($payload, $signature, $secret, $algorithm = 'sha256') {
        if (empty($secret) || empty($payload) || empty($signature)) {
            return false;
        }

        $expected = base64_encode(hash_hmac($algorithm, $payload, $secret, true));
        return hash_equals($signature, $expected);
    }

    /**
     * Validate request timestamp (prevent replay attacks)
     */
    public function validate_timestamp($timestamp, $tolerance_seconds = 300) {
        $now = time();
        $time_diff = abs($now - intval($timestamp));
        
        return $time_diff <= $tolerance_seconds;
    }

    /**
     * Validate order data structure
     */
    public function validate_order_data($order_data) {
        $errors = [];

        // Required fields
        $required = ['customer_id', 'currency', 'line_items', 'meta_data'];
        foreach ($required as $field) {
            if (empty($order_data[$field])) {
                $errors[] = "Missing required field: {$field}";
            }
        }

        // Validate customer ID
        if (!empty($order_data['customer_id']) && !$this->validate_user_id($order_data['customer_id'])) {
            $errors[] = "Invalid customer_id (must be positive integer)";
        }

        // Validate currency
        if (!empty($order_data['currency']) && !$this->validate_currency($order_data['currency'])) {
            $errors[] = "Invalid currency code (must be 3-letter ISO code)";
        }

        // Validate line items
        if (!is_array($order_data['line_items']) || empty($order_data['line_items'])) {
            $errors[] = "No line items provided";
        } else {
            foreach ($order_data['line_items'] as $idx => $item) {
                if (empty($item['product_id'])) {
                    $errors[] = "Line item {$idx}: missing product_id";
                }
                if (!isset($item['quantity']) || intval($item['quantity']) <= 0) {
                    $errors[] = "Line item {$idx}: invalid quantity";
                }
                if (!isset($item['price']) || floatval($item['price']) < 0) {
                    $errors[] = "Line item {$idx}: invalid price";
                }
            }
        }

        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }

    /**
     * Validate user data
     */
    public function validate_user_data($user_data) {
        $errors = [];

        if (empty($user_data['user_id']) || !$this->validate_user_id($user_data['user_id'])) {
            $errors[] = "Invalid or missing user_id";
        }

        if (empty($user_data['email']) || !$this->validate_email($user_data['email'])) {
            $errors[] = "Invalid or missing email";
        }

        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }
}

// buzzjuice.net/shared/payment-gateway/core/class-pgb-auth.php

<?php
/**
 * Payment Gateway Bridge - Authentication
 * Handles authentication and request signing
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Auth {
    private $logger = null;
    const SIGNATURE_ALGORITHM = 'sha256';

    public function __construct($logger = null) {
        $this->logger = $logger;
    }

    /**
     * Get the PGB shared secret from environment
     * MUST be set as server environment variable, never in code/DB
     */
    public function get_secret() {
        // 1. Check environment variable (primary)
        $secret = getenv('BZJ_PGB_SECRET');
        if (!empty($secret)) {
            return $secret;
        }

        // 2. Check PHP constant (fallback for testing)
        if (defined('BZJ_PGB_SECRET')) {
            return BZJ_PGB_SECRET;
        }

        // Log error if missing
        if ($this->logger) {
            $this->logger->log(
                'PGB secret not configured',
                PGB_Logger::LOG_CRITICAL,
                [],
                __FILE__
            );
        }

        return null;
    }

    /**
     * Generate signature for request
     * Signature = HMAC-SHA256(timestamp + "\n" + request_id + "\n" + body, secret)
     */
    public function generate_signature($timestamp, $request_id, $body, $secret = null) {
        if (!$secret) {
            $secret = $this->get_secret();
        }

        if (!$secret) {
            if ($this->logger) {
                $this->logger->log(
                    'Cannot generate signature: secret not available',
                    PGB_Logger::LOG_ERROR,
                    [],
                    __FILE__
                );
            }
            return null;
        }

        $payload = "{$timestamp}\n{$request_id}\n{$body}";
        $signature = base64_encode(
            hash_hmac(self::SIGNATURE_ALGORITHM, $payload, $secret, true)
        );

        return $signature;
    }

    /**
     * Validate incoming request signature
     */
    public function validate_request_signature($timestamp, $request_id, $body, $provided_signature, $secret = null) {
        if (!$secret) {
            $secret = $this->get_secret();
        }

        if (!$secret) {
            if ($this->logger) {
                $this->logger->log(
                    'Cannot validate signature: secret not available',
                    PGB_Logger::LOG_ERROR,
                    [],
                    __FILE__
                );
            }
            return false;
        }

        $expected_signature = $this->generate_signature($timestamp, $request_id, $body, $secret);
        
        if (!$expected_signature) {
            return false;
        }

        // Use timing-safe comparison
        $is_valid = hash_equals($expected_signature, $provided_signature);

        if (!$is_valid && $this->logger) {
            $this->logger->log(
                'Request signature validation failed',
                PGB_Logger::LOG_WARNING,
                ['request_id' => $request_id],
                __FILE__
            );
        }

        return $is_valid;
    }

    /**
     * Get request signature from headers
     */
    public function get_signature_from_headers() {
        $headers = $this->get_request_headers();
        return $headers['X-BZJ-PGB-Signature'] ?? null;
    }

    /**
     * Get timestamp from headers
     */
    public function get_timestamp_from_headers() {
        $headers = $this->get_request_headers();
        return $headers['X-BZJ-PGB-Timestamp'] ?? null;
    }

    /**
     * Get request ID from headers
     */
    public function get_request_id_from_headers() {
        $headers = $this->get_request_headers();
        return $headers['X-BZJ-PGB-Request-ID'] ?? null;
    }

    /**
     * Get all request headers
     */
    private function get_request_headers() {
        $headers = [];

        if (function_exists('getallheaders')) {
            return getallheaders();
        }

        // Fallback for environments without getallheaders()
        foreach ($_SERVER as $key => $value) {
            if (strpos($key, 'HTTP_') === 0) {
                $header = str_replace('HTTP_', '', $key);
                $header = str_replace('_', '-', $header);
                $headers[$header] = $value;
            }
        }

        return $headers;
    }

    /**
     * Validate Streams user session
     * Called from Streams context (where $wo['user'] is available)
     */
    public function validate_streams_session(&$wo_global) {
        // Check if logged in
        if (empty($wo_global['loggedin']) || $wo_global['loggedin'] !== true) {

                );

        // Check user data
        if (empty($wo_global['user']['user_id']) || empty($wo_global['user']['email'])) {
            if ($this->logger) {
                $this->logger->log(
                    'Streams session validation failed: missing user data',
                    PGB_Logger::LOG_WARNING,

                    __FILE__

class PGB_Idempotency {
    private $logger = null;
    private $wp_db = null;

    public function __construct($logger = null) {
        $this->logger = $logger;
        $this->init_wp_db();
    }

    /**
     * Initialize WordPress database connection
     */
    private function init_wp_db() {
        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * Check if order already exists by request ID
     */
    public function check_duplicate_by_request_id($request_id, $wow_user_id) {
        if (!$this->wp_db) {
            return false;
        }

        $query = $this->wp_db->prepare(
            "SELECT woo_order_id FROM wp_pgb_transactions 
             WHERE request_id = %s AND wow_user_id = %d 
             AND created_at > DATE_SUB(NOW(), INTERVAL 1 HOUR)",
            $request_id,
            $wow_user_id
        );

        $result = $this->wp_db->get_row($query);

        if ($result) {
            if ($this->logger) {
                $this->logger->log(
                    'Duplicate request detected',
                    PGB_Logger::LOG_WARNING,
                    ['request_id' => $request_id, 'existing_order' => $result->woo_order_id],
                    __FILE__
                );
            }
            return $result->woo_order_id;
        }

        return false;
    }

    /**
     * Check if order exists by WoWonder order ID
     */
    public function check_duplicate_by_wow_order_id($wow_order_id) {
        if (!$this->wp_db) {
            return false;
        }

        $query = $this->wp_db->prepare(
            "SELECT woo_order_id FROM wp_pgb_transactions 
             WHERE wow_order_id = %s LIMIT 1",
            $wow_order_id
        );

        $result = $this->wp_db->get_row($query);

        if ($result) {
            if ($this->logger) {
                $this->logger->log(
                    'Duplicate wow_order_id detected',
                    PGB_Logger::LOG_WARNING,
                    ['wow_order_id' => $wow_order_id, 'existing_order' => $result->woo_order_id],
                    __FILE__
                );
            }
            return $result->woo_order_id;
        }

        return false;
    }

    /**
     * Record transaction request
     */
    public function record_transaction($request_id, $wow_order_id, $wow_user_id, $woo_order_id, $status = 'initiated') {
        if (!$this->wp_db) {
            return false;
        }

        $result = $this->wp_db->query(
            $this->wp_db->prepare(
                "INSERT INTO wp_pgb_transactions 
                 (request_id, wow_order_id, wow_user_id, woo_order_id, status, created_at) 
                 VALUES (%s, %s, %d, %d, %s, NOW())
                 ON DUPLICATE KEY UPDATE woo_order_id = VALUES(woo_order_id), status = VALUES(status)",
                $request_id,
                $wow_order_id,
                $wow_user_id,
                $woo_order_id,
                $status
            )
        );

        if (!$result && $this->logger) {
            $this->logger->log(
                'Failed to record transaction',
                PGB_Logger::LOG_ERROR,
                ['db_error' => $this->wp_db->last_error],
                __FILE__
            );
        }

        return $result !== false;
    }

    /**
     * Update transaction status
     */
    public function update_transaction_status($woo_order_id, $status, $fulfillment_status = null) {
        if (!$this->wp_db) {
            return false;
        }

        $query = "UPDATE wp_pgb_transactions SET status = %s";
        $params = [$status];

        if ($fulfillment_status !== null) {
            $query .= ", fulfillment_status = %s";
            $params[] = $fulfillment_status;
        }

        $query .= ", updated_at = NOW() WHERE woo_order_id = %d";
        $params[] = $woo_order_id;

        $result = $this->wp_db->query(
            $this->wp_db->prepare($query, ...$params)
        );

        return $result !== false;
    }
}

// buzzjuice.net/shared/payment-gateway/core/class-pgb-manager.php

<?php
/**
 * Payment Gateway Bridge - Manager
 * Handles order creation, state management, and fulfillment
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Manager {
    private $logger = null;
    private $validator = null;
    private $idempotency = null;
    private $wow_db = null;
    private $wp_db = null;

    public function __construct($logger, $validator, $idempotency) {
        $this->logger = $logger;
        $this->validator = $validator;
        $this->idempotency = $idempotency;
        $this->init_databases();
    }

    /**
     * Initialize database connections
     */
    private function init_databases() {
        if (function_exists('get_wowonder_db')) {
            $this->wow_db = get_wowonder_db();
        }

        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * Create order in WooCommerce
     */
    public function create_order_from_streams($order_data, $wow_user_data, $request_id = null) {
        if (!$request_id) {
            $request_id = uniqid('req_');
        }

        $this->logger->log(
            'Creating WooCommerce order from Streams',
            PGB_Logger::LOG_INFO,
            ['request_id' => $request_id, 'wow_user_id' => $wow_user_data['user_id']],
            __FILE__
        );

        // 1. Validate order and user data
        $order_validation = $this->validator->validate_order_data($order_data);
        $user_validation = $this->validator->validate_user_data($wow_user_data);

        if (!$order_validation['valid'] || !$user_validation['valid']) {
            $all_errors = array_merge($order_validation['errors'], $user_validation['errors']);
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                ['errors' => $all_errors],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Validation failed',
                'details' => $all_errors,
            ];
        }

        // 2. Generate WoWonder order ID
        $wow_order_id = $order_data['meta_data']['wow_order_id'] ?? 'wow_' . uniqid();

        // 3. Check for duplicates
        $duplicate_woo_id = $this->idempotency->check_duplicate_by_wow_order_id($wow_order_id);
        if ($duplicate_woo_id) {
            $this->logger->log(
                'Duplicate order detected',
                PGB_Logger::LOG_WARNING,
                ['wow_order_id' => $wow_order_id, 'existing_woo_order' => $duplicate_woo_id],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Duplicate order',
                'existing_order_id' => $duplicate_woo_id,
            ];
        }

        try {
            // 4. Prepare WooCommerce order
            $wc_order_data = $this->prepare_wc_order_data($order_data, $wow_user_data);

            // 5. Create order via WooCommerce REST API
            $wc_result = $this->create_wc_order($wc_order_data);

            if (!$wc_result['success']) {
                $this->logger->log(
                    'WooCommerce order creation failed',
                    PGB_Logger::LOG_ERROR,
                    ['error' => $wc_result['error'], 'wc_response' => $wc_result['response'] ?? null],
                    __FILE__
                );
                return $wc_result;
            }

            $woo_order_id = $wc_result['order_id'];
            $payment_url = $wc_result['payment_url'];

            // 6. Store transaction record
            $this->store_transaction_record(
                $request_id,
                $wow_order_id,
                $wow_user_data['user_id'],
                $woo_order_id,
                $order_data,
                'initiated'
            );

            // 7. Store WoWonder fulfillment records (now in initiated state)
            $this->store_streams_fulfillment_records($order_data, $wow_user_data, $wow_order_id, 'initiated');

            $this->logger->log(
                'Order created successfully',
                PGB_Logger::LOG_INFO,
                [
                    'request_id' => $request_id,
                    'wow_order_id' => $wow_order_id,
                    'woo_order_id' => $woo_order_id,
                ],
                __FILE__
            );

            // Fire action hook
            if (function_exists('do_action')) {
                do_action('pgb_order_created', $woo_order_id, $order_data, $wow_user_data);
            }

            return [
                'success' => true,
                'order_id' => $woo_order_id,
                'payment_url' => $payment_url,
                'wow_order_id' => $wow_order_id,
                'request_id' => $request_id,
            ];

        } catch (Exception $e) {
            $this->logger->log(
                'Exception creating order',
                PGB_Logger::LOG_CRITICAL,
                ['exception' => $e->getMessage(), 'trace' => $e->getTraceAsString()],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Internal server error',
            ];
        }
    }

    /**
     * Prepare WooCommerce order data
     */
    private function prepare_wc_order_data($order_data, $wow_user_data) {
        $line_items = [];

        foreach ($order_data['line_items'] as $item) {
            $line_items[] = [
                'product_id' => intval($item['product_id']),
                'quantity' => intval($item['quantity'] ?? 1),
                'price' => floatval($item['price'] ?? 0),
                'total' => floatval($item['total'] ?? 0),
            ];
        }

        $billing_data = $this->build_address_from_streams_user($wow_user_data);

        $wc_order_data = [
            'customer_id' => intval($order_data['customer_id']),
            'currency' => $order_data['currency'] ?? 'GHS',
            'set_paid' => false,
            'line_items' => $line_items,
            'billing' => $billing_data,
            'shipping' => $billing_data,
            'meta_data' => array_merge(
                $order_data['meta_data'] ?? [],
                [
                    ['key' => '_buzzjuice_origin', 'value' => 'streams'],
                    ['key' => 'wow_user_id', 'value' => $wow_user_data['user_id']],
                    ['key' => 'wow_email', 'value' => $wow_user_data['email']],
                ]
            ),
        ];

        return $wc_order_data;
    }

    /**
     * Build address from Streams user data
     */
    private function build_address_from_streams_user($user_data) {
        return [
            'first_name' => $this->validator->sanitize_string($user_data['first_name'] ?? $user_data['username'] ?? 'User'),
            'last_name' => $this->validator->sanitize_string($user_data['last_name'] ?? $user_data['username'] ?? 'User'),
            'company' => 'Buzzjuice Network',
            'address_1' => $this->validator->sanitize_string($user_data['address'] ?? '1234 Street'),
            'city' => $this->validator->sanitize_string($user_data['city'] ?? 'City'),
            'state' => $this->validator->sanitize_string($user_data['state'] ?? 'GH'),
            'postcode' => $this->validator->sanitize_string($user_data['zip'] ?? '00000'),
            'country' => strtoupper($user_data['country_id'] ?? 'GH'),
            'email' => $user_data['email'],
            'phone' => $this->validator->sanitize_string($user_data['phone_number'] ?? '+1-555-0000'),
        ];
    }

    /**
     * Create order via WooCommerce integration
     */
    private function create_wc_order($wc_order_data) {
        // This would call the integration file
        if (!function_exists('pgb_wc_create_order')) {
            require_once dirname(__FILE__, 2) . '/integrations/pgb-int-woocommerce.php';
        }

        return pgb_wc_create_order($wc_order_data, $this->logger);
    }

    /**
     * Store transaction record in WordPress
     */
    private function store_transaction_record($request_id, $wow_order_id, $wow_user_id, $woo_order_id, $order_data, $status) {
        if (!$this->wp_db) {
            $this->logger->log('WordPress DB not available', PGB_Logger::LOG_WARNING, [], __FILE__);
            return false;
        }

        $amount = 0;
        foreach ($order_data['line_items'] as $item) {
            $amount += floatval($item['total'] ?? 0);
        }

        $transaction_kind = $order_data['meta_data']['transaction_kind'] ?? 'UNKNOWN';

        return $this->idempotency->record_transaction(
            $request_id,
            $wow_order_id,
            $wow_user_id,
            $woo_order_id,
            $status
        );
    }

    /**
     * Store Streams fulfillment records (in initiated state, not completed)
     */
    private function store_streams_fulfillment_records($order_data, $wow_user_data, $wow_order_id, $status = 'initiated') {
        if (!$this->wow_db) {
            $this->logger->log('WoWonder DB not available', PGB_Logger::LOG_WARNING, [], __FILE__);
            return false;
        }

        $transaction_kind = $order_data['meta_data']['transaction_kind'] ?? 'UNKNOWN';
        $amount = 0;
        foreach ($order_data['line_items'] as $item) {
            $amount += floatval($item['total'] ?? 0);
        }

        $timestamp = time();

        // Insert into Wo_Payment_Transactions
        $sql = "INSERT INTO Wo_Payment_Transactions 
                (userid, amount, order_id, kind, currency_code, payment_status) 
                VALUES (?, ?, ?, ?, ?, ?)";
        $stmt = $this->wow_db->prepare($sql);
        if ($stmt) {
            $stmt->bind_param('idsss', 
                $wow_user_data['user_id'],
                $amount,
                $wow_order_id,
                $transaction_kind,
                $order_data['currency'] ?? 'GHS',
                $status
            );
            $stmt->execute();
            $stmt->close();

            $this->logger->log(
                'WoWonder transaction record created',
                PGB_Logger::LOG_DEBUG,
                ['wow_order_id' => $wow_order_id],
                __FILE__
            );
        }

        return true;
    }

    /**
     * Process payment completion
     */
    public function process_payment_completion($woo_order_id, $payment_data = []) {
        $this->logger->log(
            'Processing payment completion',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $woo_order_id],
            __FILE__
        );

        // Update transaction status
        $this->idempotency->update_transaction_status($woo_order_id, 'paid');

        // Fire action hooks for addons to process fulfillment
        if (function_exists('do_action')) {
            do_action('pgb_payment_complete', $woo_order_id, $payment_data);
        }

        $this->logger->log(
            'Payment completion processing finished',
            PGB_Logger::LOG_INFO,
            ['woo_order_id' => $woo_order_id],
            __FILE__
        );
    }

$amount = floatval($_POST['amount']);
$product_name = htmlspecialchars(trim($_POST['product_name']), ENT_QUOTES, 'UTF-8');

$product_owner_id = intval($_POST['product_owner_id']);
$transaction_kind = preg_replace('/[^a-zA-Z0-9_-]/', '', $_POST['transaction_kind']);
$wow_currency_code = htmlspecialchars($_POST['wow_currency_code'], ENT_QUOTES, 'UTF-8');
$wow_post_id = isset($_POST['wow_post_id']) ? intval($_POST['wow_post_id']) : 0;
$address_id = isset($_POST['address_id']) ? intval($_POST['address_id']) : 0;

    header('Content-Type: application/json');
    echo json_encode(['status' => 'error', 'message' => 'Invalid amount/price/units']);
    exit;
}

    http_response_code(500);

    echo json_encode(['status' => 'error', 'message' => 'Database connection unavailable']);
    exit;
}

$wow_order_id = 'wow_' . uniqid();
$timestamp = time();

$sql = "INSERT INTO Wo_Payment_Transactions 
        (userid, amount, order_id, kind, currency_code, payment_status) 
        VALUES (?, ?, ?, ?, ?, ?)";

    exit;

$status = 'initiated';
$stmt->bind_param('idsss', $wow_user_id, $amount, $wow_order_id, $transaction_kind, $wow_currency_code, $status);

if (!$stmt->execute()) {
    http_response_code(500);
    header('Content-Type: application/json');
    echo json_encode(['status' => 'error', 'message' => 'Failed to store transaction']);
    exit;
}
$stmt->close();

// Verify storage
$result = $sqlConnect->query("SELECT order_id FROM Wo_Payment_Transactions WHERE order_id = '$wow_order_id' LIMIT 1");
if (!$result || $result->num_rows === 0) {

    echo json_encode(['status' => 'error', 'message' => 'Transaction verification failed']);
    exit;

        $stmt = $sqlConnect->prepare($sql);

        // Create Wo_UserOrders record

                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)";

        if ($stmt) {

            $stmt->bind_param('siiiiiddssi',

                $wow_post_id,

            );
            $stmt->execute();
            $stmt->close();
        }

        // For PRODUCT: redirect to checkout via MU plugin
        // The MU plugin will fetch from DB and redirect to checkout
        header('Content-Type: application/json');
        echo json_encode([
            'status' => 200,
            'message' => 'Order stored, redirecting to checkout',

            'redirect' => $wo['config']['site_url'] . '/pgb-checkout?wow_order_id=' . urlencode($wow_order_id)
        ]);
        exit;

    case 'PRO':
    case 'WALLET':
    case 'DONATE':
        // For subscriptions/wallet/donations: redirect to checkout via MU plugin
        header('Content-Type: application/json');
        echo json_encode([
            'status' => 200,

    default:

        header('Content-Type: application/json');

function pgb_verify_request_signature($request) {
    $bootstrap = PGB_Bootstrap::instance();
    $auth = $bootstrap->get_auth();
    $logger = $bootstrap->get_logger();

    // Get headers
    $timestamp = $request->get_header('X-BZJ-PGB-Timestamp');
    $request_id = $request->get_header('X-BZJ-PGB-Request-ID');
    $signature = $request->get_header('X-BZJ-PGB-Signature');

    if (!$timestamp || !$request_id || !$signature) {
        $logger->log(
            'Missing signature headers',
            PGB_Logger::LOG_WARNING,
            [],
            __FILE__
        );
        return false;
    }

    // Validate timestamp
    if (!$auth->validate_timestamp($timestamp)) {
        $logger->log(
            'Request timestamp validation failed',
            PGB_Logger::LOG_WARNING,
            ['timestamp' => $timestamp],
            __FILE__
        );
        return false;
    }

    // Get body
    $body = $request->get_body();

    // Validate signature
    if (!$auth->validate_request_signature($timestamp, $request_id, $body, $signature)) {
        return false;
    }

    return true;
}

    if (function_exists('wc_create_order')) {
        return pgb_wc_create_order_internal($order_data, $logger);

/**
 * Create order using internal WooCommerce functions
 */
function pgb_wc_create_order_internal($order_data, $logger = null) {
    try {
        $order = wc_create_order();

        // Set customer
        if (!empty($order_data['customer_id'])) {
            $order->set_customer_id(intval($order_data['customer_id']));
        }

        // Set currency
        if (!empty($order_data['currency'])) {
            $order->set_currency($order_data['currency']);
        }

        // Add line items
        if (!empty($order_data['line_items'])) {
            foreach ($order_data['line_items'] as $item) {
                $order->add_product(
                    wc_get_product(intval($item['product_id'])),
                    intval($item['quantity'] ?? 1),
                    [
                        'subtotal' => $item['price'] ?? 0,
                        'total' => $item['total'] ?? 0,
                    ]
                );
            }
        }

        // Set addresses
        if (!empty($order_data['billing'])) {
            $order->set_address($order_data['billing'], 'billing');
        }
        if (!empty($order_data['shipping'])) {
            $order->set_address($order_data['shipping'], 'shipping');
        }

        // Add meta data
        if (!empty($order_data['meta_data'])) {
            foreach ($order_data['meta_data'] as $meta) {
                $order->add_meta_data($meta['key'], $meta['value']);
            }
        }

        // Save order
        $order->save();
        $order_id = $order->get_id();

        // Get payment URL
        $payment_url = $order->get_checkout_payment_url();

        if ($logger) {
            $logger->log(
                'Order created successfully',
                PGB_Logger::LOG_INFO,
                ['order_id' => $order_id],
                __FILE__
            );
        }

        return [
            'success' => true,
            'order_id' => $order_id,
            'payment_url' => $payment_url,
        ];

    } catch (Exception $e) {
        if ($logger) {
            $logger->log(
                'Order creation failed: ' . $e->getMessage(),
                PGB_Logger::LOG_ERROR,
                ['exception' => $e->getMessage()],
                __FILE__
            );
        }
        return [
            'success' => false,
            'error' => 'Failed to create order',
        ];
    }
}

/**
 * Create order via REST API (fallback)
 */
function pgb_wc_create_order_via_rest($wc_api_url, $wc_api_key, $wc_api_secret, $order_data, $logger = null) {
    $endpoint = rtrim($wc_api_url, '/') . '/wp-json/wc/v3/orders';

    $args = [
        'method' => 'POST',
        'headers' => [
            'Content-Type' => 'application/json',
            'Authorization' => 'Basic ' . base64_encode("{$wc_api_key}:{$wc_api_secret}"),
        ],
        'body' => json_encode($order_data),
        'sslverify' => false,
        'timeout' => 30,
    ];

    $response = wp_remote_post($endpoint, $args);

    if (is_wp_error($response)) {
        if ($logger) {
            $logger->log(
                'WooCommerce REST API error',
                PGB_Logger::LOG_ERROR,
                ['error' => $response->get_error_message()],
                __FILE__
            );
        }
        return [
            'success' => false,
            'error' => 'API request failed',
        ];
    }

    $http_code = wp_remote_retrieve_response_code($response);
    $body = json_decode(wp_remote_retrieve_body($response), true);

    if ($http_code === 201 && !empty($body['id'])) {
        $order_id = $body['id'];
        
        // Get payment URL from order
        $payment_url = $body['checkout_payment_url'] ?? 
                       home_url('/checkout/order-pay/' . $order_id . '/?pay_for_order=true');

        return [
            'success' => true,
            'order_id' => $order_id,
            'payment_url' => $payment_url,
        ];
    }

    if ($logger) {
        $logger->log(
            'WooCommerce order creation failed',
            PGB_Logger::LOG_ERROR,
            ['http_code' => $http_code, 'response' => $body],
            __FILE__
        );
    }

    return [
        'success' => false,
        'error' => 'Failed to create order in WooCommerce',
        'details' => $body['message'] ?? 'Unknown error',
    ];
}

    $logger = $bootstrap->get_logger();

    $logger->log(
        'Processing AffiliateWP rewards',
        PGB_Logger::LOG_INFO,
        ['order_id' => $order_id],
        __FILE__
    );

    // Get order
    $order = wc_get_order($order_id);
    if (!$order) {
        return;
    }

    // Get meta data
    $affwp_snapshot = $order->get_meta('_buzzjuice_affwp_context_snapshot');
    $wow_user_id = $order->get_meta('wow_user_id');

    if (!$affwp_snapshot) {
        $logger->log(
            'No affiliate snapshot found',
            PGB_Logger::LOG_DEBUG,
            ['order_id' => $order_id],
            __FILE__
        );
        return;
    }

    $snapshot = json_decode($affwp_snapshot, true);

    // Attempt affiliate resolution
    $affiliate_resolved = false;
    $affiliate_id = null;

    // 1. Check snapshot for affiliate
    if (!empty($snapshot['affwp_affiliate_id'])) {
        $affiliate_id = intval($snapshot['affwp_affiliate_id']);
        $affiliate_resolved = affwp_is_affiliate($affiliate_id);
    }

    // 2. Check customer email for affiliate relationship
    if (!$affiliate_resolved && !empty($snapshot['customer_email'])) {
        $customer_email = sanitize_email($snapshot['customer_email']);
        $affiliate = affiliate_wp()->affiliates->get_by('email', $customer_email);
        if ($affiliate) {
            $affiliate_id = $affiliate->ID;
            $affiliate_resolved = true;
        }
    }

    // 3. Check WordPress user for affiliate status
    if (!$affiliate_resolved && !empty($snapshot['wp_user_id'])) {
        $wp_user_id = intval($snapshot['wp_user_id']);
        $affiliate_id = affwp_get_affiliate_id($wp_user_id);
        if ($affiliate_id) {
            $affiliate_resolved = true;
        }
    }

    if (!$affiliate_resolved || !$affiliate_id) {
        $logger->log(
            'Could not resolve affiliate',
            PGB_Logger::LOG_WARNING,
            ['order_id' => $order_id, 'wow_user_id' => $wow_user_id],
            __FILE__
        );
        return;
    }

    // Calculate commission
    $order_total = floatval($order->get_total());
    $commission = affiliate_wp()->commissions->calculate_commission(
        $order_total,
        $affiliate_id,
        'purchase'
    );

    if (!$commission) {
        $logger->log(
            'Commission calculation failed',
            PGB_Logger::LOG_WARNING,
            ['affiliate_id' => $affiliate_id, 'order_id' => $order_id],
            __FILE__
        );
        return;
    }

    // Create referral
    $referral_args = [
        'affiliate_id' => $affiliate_id,
        'visit_id' => 0,
        'customer_email' => $order->get_billing_email(),
        'product_id' => 0,

        'currency' => $order->get_currency(),
        'order_id' => $order_id,

            'wow_user_id' => $wow_user_id,
            'origin' => 'streams',
        ],
        'status' => 'unpaid',
    ];

    $referral_id = affiliate_wp()->referrals->add($referral_args);

            __FILE__
        );

// buzzjuice.net/shared/payment-gateway/addons/pgb-addon-subscriptions.php

<?php
/**
 * Payment Gateway Bridge - WooCommerce Subscriptions Addon
 * Handles subscription creation and management

 * @package Buzzjuice\PaymentGateway

if (!defined('ABSPATH')) {

/**
 * Create subscription on payment completion
 */

    $bootstrap = PGB_Bootstrap::instance();
    $logger = $bootstrap->get_logger();

    $order = wc_get_order($order_id);
    if (!$order) {
        return;
    }

    // Check if this is a subscription product
    $is_subscription = false;
    foreach ($order->get_items() as $item) {
        if (WC_Subscriptions_Product::is_subscription($item->get_product_id())) {
            $is_subscription = true;
            break;
        }
    }

    if (!$is_subscription) {
        return;
    }

    $logger->log(
        'Processing subscription creation',
        PGB_Logger::LOG_INFO,
        ['order_id' => $order_id],
        __FILE__
    );

    $subscription_period = $order->get_meta('_subscription_period') ?? 'month';
    $subscription_interval = intval($order->get_meta('_subscription_interval') ?? 1);
    $subscription_length = intval($order->get_meta('_subscription_length') ?? 0);

    // Get WoWonder user info
    $wow_user_id = $order->get_meta('wow_user_id');
    $transaction_kind = $order->get_meta('_transaction_kind');

    try {
        // Create subscription using WC Subscriptions API
        $subscription = wcs_create_subscription([
            'order_id' => $order_id,
            'customer_id' => $order->get_customer_id(),
            'start_date' => current_time('mysql'),
            'billing_period' => $subscription_period,
            'billing_interval' => $subscription_interval,
        ]);

        if (is_wp_error($subscription)) {
            $logger->log(
                'Subscription creation failed',
                PGB_Logger::LOG_ERROR,
                ['error' => $subscription->get_error_message()],
                __FILE__
            );
            return;
        }

        $subscription_id = $subscription->get_id();

        // Copy order line items to subscription
        foreach ($order->get_items() as $item) {
            $subscription->add_product(
                wc_get_product($item->get_product_id()),
                $item->get_quantity()
            );
        }

        // Set subscription as active
        $subscription->update_status('active');
        $subscription->add_meta_data('_buzzjuice_wow_user_id', $wow_user_id);
        $subscription->add_meta_data('_buzzjuice_pro_type', $transaction_kind);
        $subscription->save();

        $logger->log(
            'Subscription created successfully',
            PGB_Logger::LOG_INFO,
            ['subscription_id' => $subscription_id, 'order_id' => $order_id],
            __FILE__
        );

        // Fire custom hook
        do_action('pgb_subscription_created', $subscription_id, $wow_user_id, $transaction_kind);

    } catch (Exception $e) {
        $logger->log(
            'Exception creating subscription',
            PGB_Logger::LOG_ERROR,
            ['exception' => $e->getMessage()],
            __FILE__

// buzzjuice.net/shared/payment-gateway/addons/pgb-addon-currency.php

<?php
/**
 * Payment Gateway Bridge - Currency Conversion Addon
 * Handles WOOCS currency conversion for orders
 * 
 * Extracted from: wow-pgb_sync.php (legacy currency logic)
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

// Only load if WooCommerce Currency Switcher is active
if (!class_exists('WOOCOMMERCE_CURRENCY_SWITCHER')) {
    return;
}

/**
 * Apply currency conversion to order prices
 */
add_filter('pgb_before_order_create', function ($order_data) {
    $bootstrap = PGB_Bootstrap::instance();
    $logger = $bootstrap->get_logger();

    // Get current WOOCS currency
    $woocs = WOOCOMMERCE_CURRENCY_SWITCHER::__get_instance();
    $current_currency = $woocs->current_currency;
    $store_currency = get_option('woocommerce_currency', 'GHS');

    if ($current_currency === $store_currency) {
        return $order_data; // No conversion needed
    }

    $logger->log(
        'Converting order currency',
        PGB_Logger::LOG_DEBUG,
        ['from' => $store_currency, 'to' => $current_currency],
        __FILE__
    );

    // Get conversion rate
    $rate = $woocs->get_rate($current_currency);
    if (!$rate) {
        $logger->log(
            'Currency rate not found',
            PGB_Logger::LOG_WARNING,
            ['currency' => $current_currency],
            __FILE__
        );
        return $order_data;
    }

    // Convert all prices
    if (!empty($order_data['line_items'])) {
        foreach ($order_data['line_items'] as &$item) {
            $item['price'] = floatval($item['price']) * $rate;
            $item['total'] = floatval($item['total']) * $rate;
        }
    }

    // Update currency in order
    $order_data['currency'] = $current_currency;

    $logger->log(
        'Currency conversion applied',
        PGB_Logger::LOG_DEBUG,
        ['rate' => $rate],
        __FILE__
    );

    return $order_data;
}, 10, 1);

// buzzjuice.net/shared/payment-gateway/integrations/pgb-int-jewel-affiliate.php

<?php
/**
 * Payment Gateway Bridge - Jewel Affiliate Integration
 * Processes Jewel rebates and wallet credits after payment
 * 
 * Extracted from: streams/jewel-affiliate-webhook.php
 * Refactored to integrate with PGB payment lifecycle
 * 
 * @package Buzzjuice\PaymentGateway
 * @since   5.0.0
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

/**
 * Process Jewel rewards on payment completion
 */
add_action('pgb_wc_payment_complete', function ($order_id) {
    $bootstrap = PGB_Bootstrap::instance();
    $logger = $bootstrap->get_logger();
    $wow_db = get_wowonder_db();

    if (!$wow_db) {
        $logger->log('WoWonder DB unavailable', PGB_Logger::LOG_WARNING, [], __FILE__);
        return;
    }

    $logger->log(
        'Processing Jewel affiliate rewards',
        PGB_Logger::LOG_INFO,
        ['order_id' => $order_id],
        __FILE__
    );

    $order = wc_get_order($order_id);
    if (!$order) {
        return;
    }

    // Get WoWonder user
    $wow_user_id = $order->get_meta('wow_user_id');
    if (!$wow_user_id) {
        $logger->log(
            'No WoWonder user linked to order',
            PGB_Logger::LOG_WARNING,
            ['order_id' => $order_id],
            __FILE__
        );
        return;
    }

    // Check if already processed (idempotency)
    if ($order->get_meta('_bzj_jewel_processed')) {
        $logger->log(
            'Jewel processing already completed',
            PGB_Logger::LOG_DEBUG,
            ['order_id' => $order_id],
            __FILE__
        );
        return;
    }

    // Calculate rebate amount (example: 10% rebate)
    $order_total = floatval($order->get_total());
    $rebate_amount = round($order_total * 0.10, 2);

    try {
        // Credit wallet in WoWonder
        $sql = "INSERT INTO Wo_WalletTransaction 
                (user_id, amount, transaction_type, order_id, notes) 
                VALUES (?, ?, ?, ?, ?)";
        
        $stmt = $wow_db->prepare($sql);
        if (!$stmt) {
            throw new Exception($wow_db->error);
        }

        $transaction_type = 'rebate';
        $notes = "Jewel affiliate rebate for order #{$order_id}";

        $stmt->bind_param('idsss',
            $wow_user_id,
            $rebate_amount,
            $transaction_type,
            $order_id,
            $notes
        );

        if (!$stmt->execute()) {
            throw new Exception($stmt->error);
        }

        $stmt->close();

        // Mark order as processed
        $order->add_meta_data('_bzj_jewel_processed', time());
        $order->add_meta_data('_bzj_jewel_rebate_amount', $rebate_amount);
        $order->save();

        $logger->log(
            'Jewel rebate credited',
            PGB_Logger::LOG_INFO,
            ['order_id' => $order_id, 'wow_user_id' => $wow_user_id, 'amount' => $rebate_amount],
            __FILE__
        );

        // Fire custom hook
        do_action('pgb_jewel_rebate_credited', $order_id, $wow_user_id, $rebate_amount);

    } catch (Exception $e) {
        $logger->log(
            'Jewel rebate processing failed',
            PGB_Logger::LOG_ERROR,
            ['exception' => $e->getMessage()],
            __FILE__
        );
    }
});

| **Subscriptions** | Manual REST API calls | WC_Subscriptions native API |
| **Affiliate Processing** | Coupled to redirect | Async via payment_complete hook |
| **Code Reusability** | Low (scattered) | High (hooks + components) |
