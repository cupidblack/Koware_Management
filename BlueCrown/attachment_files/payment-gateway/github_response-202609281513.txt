/**
 * Payment Gateway Bridge - Logger
 * Centralized logging with file rotation and severity levels
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    const LOG_OFF = 0;
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_DEBUG;
    private $file_handles = [];
    private $sensitive_keys = ['secret', 'key', 'token', 'password', 'api_key', 'api_secret'];

    public function __construct($config = []) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = $config['logging_enabled'] ?? true;
        $this->min_level = $config['log_level'] ?? self::LOG_DEBUG;

        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }

        // Create .gitkeep to ensure directory is committed
        if (!file_exists($this->log_dir . '.gitkeep')) {
            @file_put_contents($this->log_dir . '.gitkeep', '');
        }
    }

    public function log($message, $level = self::LOG_INFO, $context = [], $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        $source = $source_file ? basename($source_file, '.php') : 'unknown';

        // Sanitize context
        $safe_context = $this->redact_sensitive_data($context);

        // Format message with context
        $formatted_msg = $message;
        if (!empty($safe_context)) {
            $formatted_msg .= ' | ' . json_encode($safe_context);
        }

        $log_line = "[{$timestamp}] [{$level_upper}] {$formatted_msg}\n";

        // Write to file
        $this->write_to_file($source, $log_line);

        // Also write to WordPress error log if critical
        if (function_exists('error_log') && in_array($level, [self::LOG_ERROR, self::LOG_CRITICAL])) {
            error_log("[PGB-{$level_upper}] " . $formatted_msg);
        }
    }

    private function should_log($level) {
        $levels = [
            self::LOG_DEBUG => 0,
            self::LOG_INFO => 1,
            self::LOG_WARNING => 2,
            self::LOG_ERROR => 3,
            self::LOG_CRITICAL => 4,
        ];
        return ($levels[$level] ?? 0) >= ($levels[$this->min_level] ?? 0);
    }

    private function redact_sensitive_data($data) {
        if (!is_array($data)) {
            return $data;
        }

        $redacted = [];
        foreach ($data as $key => $value) {
            $key_lower = strtolower($key);
            $is_sensitive = false;
            foreach ($this->sensitive_keys as $sensitive) {
                if (strpos($key_lower, $sensitive) !== false) {
                    $is_sensitive = true;
                    break;
                }
            }
            $redacted[$key] = $is_sensitive ? '***REDACTED***' : $value;
        }
        return $redacted;
    }

    private function write_to_file($source, $message) {
        $file_path = $this->log_dir . $source . '.log';

        // Rotate log files if they exceed 10MB
        if (file_exists($file_path) && filesize($file_path) > 10485760) {
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}
?>

class PGB_Validator {
    public static function validate_email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }

    public static function sanitize_email($email) {
        $email = filter_var($email, FILTER_VALIDATE_EMAIL);
        return $email ?: '';
    }

    public static function validate_amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }

    public static function validate_user_id($user_id) {
        return intval($user_id) > 0;
    }

    public static function validate_order_id($order_id) {
        return is_string($order_id) && !empty(trim($order_id)) && strlen($order_id) <= 100;
    }

    public static function sanitize_string($value, $allow_html = false, $max_length = 1000) {
        $value = trim((string)$value);
        if (strlen($value) > $max_length) {
            $value = substr($value, 0, $max_length);
        }
        if (!$allow_html) {
            $value = htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
        }
        return $value;
    }

    public static function validate_transaction_type($type) {
        $allowed = ['PRODUCT', 'PRO', 'WALLET', 'DONATE', 'PURCHASE', 'SALE', 'SUBSCRIPTION', 'MARKETPLACE'];
        return in_array(strtoupper($type), $allowed);
    }

    public static function validate_currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;
    }

    public static function validate_webhook_signature($payload, $signature, $secret) {
        if (empty($secret) || empty($signature)) {
            return false;
        }
        $expected = base64_encode(hash_hmac('sha256', $payload, $secret, true));
        return hash_equals($signature, $expected);
    }

    public static function validate_hmac_signature($timestamp, $request_id, $body, $signature, $secret) {
        if (empty($secret) || empty($signature)) {
            return false;
        }
        $message = $timestamp . "\n" . $request_id . "\n" . $body;
        $expected = base64_encode(hash_hmac('sha256', $message, $secret, true));
        return hash_equals($signature, $expected);
    }

    public static function validate_post_data($data, $required_fields = []) {
        $errors = [];
        foreach ($required_fields as $field) {
            if (!isset($data[$field]) || empty($data[$field])) {
                $errors[] = "Missing required field: {$field}";
            }
        }
        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }
}

/**
 * Payment Gateway Bridge - Idempotency Handler
 * Prevents duplicate transaction processing
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Idempotency {
    private $wp_db = null;
    private $logger = null;

    public function __construct($logger = null) {
        $this->logger = $logger;
        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * Check if order already exists (idempotency)
     * Returns existing order data or null if new
     */
    public function check_duplicate_order($request_id, $wow_user_id, $amount, $window_minutes = 60) {
        if (!$this->wp_db) {
            return null;
        }

        $window = $window_minutes * 60;
        $now = time();
        $cutoff = $now - $window;

        try {
            $sql = "SELECT order_id, woo_order_id, status FROM " . wp_table('bzj_pgb_transactions') . "
                    WHERE request_id = %s OR (wow_user_id = %d AND amount = %f AND created_at > %d)
                    ORDER BY created_at DESC LIMIT 1";

            $sql = $this->wp_db->prepare($sql, [$request_id, $wow_user_id, $amount, $cutoff]);
            $result = $this->wp_db->query($sql);

            if ($result && $result->num_rows > 0) {
                $row = $result->fetch_assoc();
                if ($this->logger) {
                    $this->logger->log(
                        'Duplicate order detected',
                        PGB_Logger::LOG_WARNING,
                        ['order_id' => $row['order_id'], 'request_id' => $request_id],
                        __FILE__
                    );
                }
                return $row;
            }
        } catch (Exception $e) {
            if ($this->logger) {
                $this->logger->log(
                    'Idempotency check failed: ' . $e->getMessage(),
                    PGB_Logger::LOG_WARNING,
                    [],
                    __FILE__
                );
            }
        }

        return null;
    }

    /**
     * Record transaction for idempotency tracking
     */
    public function record_transaction($request_id, $wow_user_id, $wp_user_id, $woo_order_id, $transaction_kind, $amount, $currency, $status = 'initiated') {
        if (!$this->wp_db) {
            return false;
        }

        try {
            $sql = "INSERT INTO " . wp_table('bzj_pgb_transactions') . "
                    (request_id, wow_user_id, wp_user_id, woo_order_id, transaction_kind, amount, currency, status, created_at)
                    VALUES (%s, %d, %d, %d, %s, %f, %s, %s, NOW())";

            $sql = $this->wp_db->prepare($sql, [$request_id, $wow_user_id, $wp_user_id, $woo_order_id, $transaction_kind, $amount, $currency, $status]);
            return $this->wp_db->query($sql);
        } catch (Exception $e) {
            if ($this->logger) {
                $this->logger->log(
                    'Failed to record transaction: ' . $e->getMessage(),
                    PGB_Logger::LOG_ERROR,
                    [],
                    __FILE__
                );
            }
            return false;
        }
    }
}

// buzzjuice.net/shared/payment-gateway/core/class-pgb-manager.php
/**
 * Payment Gateway Bridge - Transaction Manager
 * Handles order creation, state management, and fulfillment
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Manager {
    private $logger = null;
    private $hooks = null;
    private $idempotency = null;
    private $woo_db = null;
    private $wp_db = null;

    public function __construct($logger, $hooks) {
        $this->logger = $logger;
        $this->hooks = $hooks;
        $this->idempotency = new PGB_Idempotency($logger);
        $this->init_databases();
    }

    private function init_databases() {
        if (function_exists('get_wowonder_db')) {
            $this->woo_db = get_wowonder_db();
        }
        if (function_exists('get_wp_db_conn')) {
            $this->wp_db = get_wp_db_conn();
        }
    }

    /**
     * Process order initiation from Streams client
     * Called via REST endpoint or direct call
     */
    public function process_order_initiation($order_data, $wow_user_data, $request_id = null) {
        $this->logger->log(
            'Order initiation requested',
            PGB_Logger::LOG_INFO,
            ['request_id' => $request_id, 'user_id' => $wow_user_data['user_id'] ?? null],
            __FILE__
        );

        // Validate inputs
        $validation = $this->validate_order_data($order_data, $wow_user_data);
        if (!$validation['valid']) {
            $this->logger->log(
                'Order validation failed',
                PGB_Logger::LOG_ERROR,
                $validation['errors'],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Validation failed',
                'errors' => $validation['errors'],
            ];
        }

        // Generate request ID if not provided
        if (!$request_id) {
            $request_id = 'req_' . uniqid() . '_' . bin2hex(random_bytes(4));
        }

        // Check for duplicate orders (idempotency)
        if (get_pgb_config_value('check_duplicate_orders', true)) {
            $amount = $this->calculate_order_total($order_data);
            $duplicate = $this->idempotency->check_duplicate_order(
                $request_id,
                $wow_user_data['user_id'],
                $amount,
                get_pgb_config_value('duplicate_window_minutes', 60)
            );

            if ($duplicate) {
                $this->logger->log(
                    'Duplicate order detected, returning existing order',
                    PGB_Logger::LOG_WARNING,
                    ['existing_order_id' => $duplicate['woo_order_id']],
                    __FILE__
                );
                return [
                    'success' => true,
                    'duplicate' => true,
                    'order_id' => $duplicate['woo_order_id'],
                    'status' => $duplicate['status'],
                ];
            }
        }

        // Apply filters
        $order_data = $this->hooks->apply_filters(
            PGB_Hooks::BEFORE_ORDER_CREATE,
            $order_data,
            $order_data['transaction_kind'] ?? 'UNKNOWN'
        );

        // Prepare payment data for database storage
        $payment_record = [
            'request_id' => $request_id,
            'wow_user_id' => $wow_user_data['user_id'],
            'wp_user_id' => $wow_user_data['wp_user_id'] ?? null,
            'transaction_kind' => $order_data['transaction_kind'],
            'amount' => $this->calculate_order_total($order_data),
            'currency' => $order_data['currency'] ?? 'GHS',
            'order_data' => json_encode($order_data),
            'wow_user_data' => json_encode($wow_user_data),
            'status' => 'initiated',
            'created_at' => date('Y-m-d H:i:s'),
        ];

        // Store payment record
        $stored = $this->store_payment_initiation($payment_record);
        if (!$stored) {
            $this->logger->log(
                'Failed to store payment initiation',
                PGB_Logger::LOG_ERROR,
                [],
                __FILE__
            );
            return [
                'success' => false,
                'error' => 'Failed to store payment data',
            ];
        }

        $this->logger->log(
            'Payment initiation stored successfully',
            PGB_Logger::LOG_INFO,
            ['request_id' => $request_id],
            __FILE__
        );

        // Fire hook for addons to process
        $this->hooks->do_action(
            PGB_Hooks::AFTER_ORDER_CREATE,
            $request_id,
            $order_data,
            $wow_user_data
        );

        return [
            'success' => true,
            'request_id' => $request_id,
            'redirect_url' => site_url('/wp-json/bzj-pgb/v1/checkout') . '?request_id=' . urlencode($request_id),
        ];
    }

    /**
     * Validate order data
     */
    private function validate_order_data($order_data, $wow_user_data) {
        $errors = [];

        // User validation
        if (empty($wow_user_data['user_id']) || !is_numeric($wow_user_data['user_id'])) {
            $errors[] = 'Invalid user ID';
        }
        if (empty($wow_user_data['email']) || !PGB_Validator::validate_email($wow_user_data['email'])) {
            $errors[] = 'Invalid user email';
        }

        // Order data validation
        if (empty($order_data['transaction_kind']) || !PGB_Validator::validate_transaction_type($order_data['transaction_kind'])) {
            $errors[] = 'Invalid transaction kind';
        }
        if (empty($order_data['currency']) || !PGB_Validator::validate_currency($order_data['currency'])) {
            $errors[] = 'Invalid currency';
        }
        if (empty($order_data['line_items']) || !is_array($order_data['line_items'])) {
            $errors[] = 'No line items provided';
        } else {
            foreach ($order_data['line_items'] as $idx => $item) {
                if (empty($item['product_id']) || !is_numeric($item['product_id'])) {
                    $errors[] = "Line item $idx: invalid product_id";
                }
                if (empty($item['quantity']) || !is_numeric($item['quantity']) || $item['quantity'] <= 0) {
                    $errors[] = "Line item $idx: invalid quantity";
                }
                if (!isset($item['price']) || !is_numeric($item['price']) || $item['price'] < 0) {
                    $errors[] = "Line item $idx: invalid price";
                }
            }
        }

        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }

    /**
     * Calculate total order amount
     */
    private function calculate_order_total($order_data) {
        $total = 0;
        if (isset($order_data['line_items']) && is_array($order_data['line_items'])) {
            foreach ($order_data['line_items'] as $item) {
                $total += floatval($item['total'] ?? ($item['price'] * $item['quantity']));
            }
        }
        return $total;
    }

    /**
     * Store payment initiation in WordPress database
     */
    private function store_payment_initiation($payment_record) {
        if (!$this->wp_db) {
            $this->logger->log('WordPress DB not available', PGB_Logger::LOG_ERROR, [], __FILE__);
            return false;
        }

                    VALUES (%s, %d, %d, %s, %f, %s, %s, %s, %s, %s)";

            $sql = $this->wp_db->prepare($sql, [
                $payment_record['request_id'],
                $payment_record['wow_user_id'],
                $payment_record['wp_user_id'] ?? 0,

                $payment_record['amount'],
                $payment_record['currency'],
                $payment_record['order_data'],
                $payment_record['wow_user_data'],
                $payment_record['status'],
                $payment_record['created_at'],

        } catch (Exception $e) {

                [],

    /**

    private function ensure_payment_table_exists() {
        if (!$this->wp_db) {

        $table_name = wp_table('bzj_pgb_initiations');

        $sql = "CREATE TABLE IF NOT EXISTS " . $table_name . " (

            request_id VARCHAR(255) UNIQUE NOT NULL,
            wow_user_id BIGINT(20) UNSIGNED NOT NULL,
            wp_user_id BIGINT(20) UNSIGNED DEFAULT 0,
            transaction_kind VARCHAR(50) NOT NULL,
            amount DECIMAL(12, 2) NOT NULL,

    /**
     * Process order fulfillment
     */
    private function process_order_fulfillment($order) {
        if (!$order) {
            return;
        }

        $wow_user_id = $order->get_meta('wow_user_id');
        $transaction_kind = $order->get_meta('transaction_kind');

        if (!$wow_user_id || !$transaction_kind) {
            $this->logger->log(
                'Missing metadata for fulfillment',
                PGB_Logger::LOG_WARNING,
                ['order_id' => $order->get_id()],
                __FILE__
            );
            return;
        }

        $this->logger->log(
            'Starting order fulfillment',
            PGB_Logger::LOG_INFO,
            ['order_id' => $order->get_id(), 'transaction_kind' => $transaction_kind],
            __FILE__
        );

        // Execute fulfillment for transaction type
        $this->hooks->handle_order_type($transaction_kind, [], $order->get_id());
    }

    /**
     * REST endpoint: Handle order creation
     */
    public function handle_rest_order_creation($request) {
        $params = $request->get_json_params();
        
        // Verify signature
        $headers = $request->get_headers();
        $signature = $headers['x_bzj_pgb_signature'][0] ?? '';
        $timestamp = $headers['x_bzj_pgb_timestamp'][0] ?? '';
        $request_id = $headers['x_bzj_pgb_request_id'][0] ?? '';

        if (!$this->verify_request_signature($request->get_body(), $signature, $timestamp, $request_id)) {
            $this->logger->log(
                'Invalid request signature',
                PGB_Logger::LOG_ERROR,
                ['request_id' => $request_id],
                __FILE__
            );
            return new WP_REST_Response([
                'success' => false,
                'error' => 'Invalid signature',
            ], 403);
        }

        // Process order
        $result = $this->process_order_initiation(
            $params['order_data'] ?? [],
            $params['user_data'] ?? [],
            $request_id
        );

        $status = $result['success'] ? 200 : 400;
        return new WP_REST_Response($result, $status);
    }

    /**
     * REST endpoint: Handle payment completion
     */
    public function handle_rest_payment_complete($request) {
        $params = $request->get_json_params();
        
        if (empty($params['woo_order_id'])) {
            return new WP_REST_Response([
                'success' => false,
                'error' => 'Missing woo_order_id',
            ], 400);
        }

        $this->on_payment_complete($params['woo_order_id']);

        return new WP_REST_Response([
            'success' => true,
            'message' => 'Payment processed',
        ], 200);
    }

    /**
     * Verify HMAC signature from Streams
     */
    private function verify_request_signature($body, $signature, $timestamp, $request_id) {
        $secret = get_pgb_secret();
        if (empty($secret)) {
            return false;
        }

        // Check timestamp within tolerance
        $now = time();
        $ts = intval($timestamp);
        $tolerance = get_pgb_config_value('timestamp_tolerance_seconds', 300);

        if (abs($now - $ts) > $tolerance) {
            $this->logger->log(
                'Request timestamp outside tolerance window',
                PGB_Logger::LOG_WARNING,
                ['timestamp' => $ts, 'now' => $now, 'tolerance' => $tolerance],
                __FILE__
            );
            return false;
        }

        // Validate signature
        return PGB_Validator::validate_hmac_signature($timestamp, $request_id, $body, $signature, $secret);
    }
}

class PGB_Addon_AffiliateWP {
    private static $instance = null;
    private $logger = null;

    public static function instance($logger = null) {
        if (null === self::$instance) {
            self::$instance = new self($logger);
        }
        return self::$instance;
    }

    private function __construct($logger = null) {
        $this->logger = $logger;
        $this->init_hooks();

    private function init_hooks() {
        if (!function_exists('add_action')) {
            return;
        }

        // Hook on payment complete to process affiliate rewards
        add_action('pgb_payment_completed', [$this, 'process_affiliate_reward'], 10, 2);
    }

    /**
     * Process affiliate reward when payment is completed
     */
    public function process_affiliate_reward($order_id, $payment_data) {
        if (!class_exists('Affiliate_WP')) {
            return;
        }

        if ($this->logger) {
            $this->logger->log(
                'Processing AffiliateWP reward',
                PGB_Logger::LOG_INFO,
                ['order_id' => $order_id],
                __FILE__
            );
        }

        if (!function_exists('wc_get_order')) {
            return;
        }

        $order = wc_get_order($order_id);
        if (!$order) {
            return;
        }

        $customer_email = $order->get_billing_email();
        $amount = $order->get_total();
        $customer_id = $order->get_customer_id();

        // Check if customer is linked to an affiliate
        if (function_exists('affwp_get_affiliate_id')) {
            $affiliate_id = affwp_get_affiliate_id($customer_id);

            if ($affiliate_id) {
                // Calculate commission
                $commission = $this->calculate_commission($amount, $affiliate_id);

                // Create referral
                if (function_exists('affwp_add_referral')) {
                    $referral_args = [
                        'affiliate_id' => $affiliate_id,
                        'reference' => $order_id,
                        'amount' => $commission,
                        'status' => 'approved',
                        'context' => 'woocommerce',
                        'date' => current_time('mysql'),
                    ];

                    $referral_id = affwp_add_referral($referral_args);

                    if ($referral_id) {
                        if ($this->logger) {
                            $this->logger->log(
                                'AffiliateWP referral created',
                                PGB_Logger::LOG_INFO,
                                ['referral_id' => $referral_id, 'affiliate_id' => $affiliate_id, 'commission' => $commission],
                                __FILE__
                            );
                        }
                    }
                }
            }
        }
    }

    /**
     * Calculate commission based on affiliate and amount
     */
    private function calculate_commission($amount, $affiliate_id) {
        // Default: 10% commission
        $rate = apply_filters('pgb_affiliate_commission_rate', 0.10, $affiliate_id);
        return $amount * $rate;
    }
}

/**
 * Payment Gateway Bridge - Jewel Affiliate Integration
 * Handles Jewel Affiliate processing and rebates

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');

class PGB_Int_JewelAffiliate {
    private static $instance = null;
    private $logger = null;
    private $wow_db = null;

    public static function instance($logger = null) {
        if (null === self::$instance) {
            self::$instance = new self($logger);
        }
        return self::$instance;
    }

    private function __construct($logger = null) {
        $this->logger = $logger;
        if (function_exists('get_wowonder_db')) {
            $this->wow_db = get_wowonder_db();
        }
        $this->init_hooks();
    }

    private function init_hooks() {
        if (!function_exists('add_action')) {
            return;
        }

        // Hook on payment complete
        add_action('pgb_payment_completed', [$this, 'process_jewel_rebate'], 10, 2);
    }

    /**
     * Process Jewel Affiliate rebate
     */
    public function process_jewel_rebate($order_id, $payment_data) {
        if (!$this->wow_db) {
            return;
        }

        if ($this->logger) {
            $this->logger->log(
                'Processing Jewel Affiliate rebate',
                PGB_Logger::LOG_INFO,
                ['order_id' => $order_id],
                __FILE__
            );
        }

        if (!function_exists('wc_get_order')) {
            return;
        }

        $order = wc_get_order($order_id);
        if (!$order) {
            return;
        }

        // Check if already processed
        if ($order->get_meta('_pgb_jewel_processed')) {
            if ($this->logger) {
                $this->logger->log(
                    'Jewel rebate already processed',
                    PGB_Logger::LOG_WARNING,
                    ['order_id' => $order_id],
                    __FILE__
                );
            }
            return;
        }

        $wow_user_id = $order->get_meta('wow_user_id');
        $amount = $order->get_total();

        if (!$wow_user_id) {
            return;
        }

        // Calculate rebate
        $rebate = $this->calculate_rebate($amount);

        // Credit wallet
        try {
            $sql = "UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ?";
            $stmt = $this->wow_db->prepare($sql);
            $stmt->bind_param('di', $rebate, $wow_user_id);
            
            if ($stmt->execute()) {
                $order->update_meta_data('_pgb_jewel_processed', true);
                $order->update_meta_data('_pgb_jewel_rebate', $rebate);
                $order->save();

                if ($this->logger) {
                    $this->logger->log(
                        'Jewel rebate applied',
                        PGB_Logger::LOG_INFO,
                        ['order_id' => $order_id, 'user_id' => $wow_user_id, 'rebate' => $rebate],
                        __FILE__
                    );
                }
            }
        } catch (Exception $e) {
            if ($this->logger) {
                $this->logger->log(
                    'Failed to apply Jewel rebate: ' . $e->getMessage(),
                    PGB_Logger::LOG_ERROR,
                    [],
                    __FILE__
                );
            }
        }
    }

    /**
     * Calculate rebate amount
     */
    private function calculate_rebate($amount) {
        // Default: 5% rebate
        return apply_filters('pgb_jewel_rebate_rate', $amount * 0.05);
    }
}

// Initialize integration
if (function_exists('add_action')) {
    add_action('pgb_initialized', function($pgb) {
        PGB_Int_JewelAffiliate::instance($pgb->get_logger());
    });
}

<?php
// buzzjuice.net/streams/paymentInitiate.php
/**
 * Payment Initiation Handler
 * Retrieves payment data and prepares WooCommerce checkout
 * Called from PGB client redirect
 */

require_once 'config.php';
require_once 'assets/init.php';
require_once '../shared/db_helpers.php';
require_once ABSPATH . 'wp-load.php';

// Get request ID from query
$request_id = isset($_GET['request_id']) ? sanitize_text_field($_GET['request_id']) : '';

if (empty($request_id)) {
    wp_die('Invalid request ID');
}

// Load Bootstrap to get manager
require_once dirname(__DIR__) . '/shared/payment-gateway/core/class-pgb-bootstrap.php';
$pgb = PGB_Bootstrap::instance();
$manager = $pgb->get_manager();

// Retrieve payment initiation
$payment = $manager->get_payment_initiation($request_id);

if (!$payment) {
    wp_die('Payment not found');
}

// Extract payment data
$order_data = $payment['order_data'];
$wow_user_data = $payment['wow_user_data'];

// Create WooCommerce order
if (!function_exists('wc_create_order')) {
    wp_die('WooCommerce not available');
}

$order = wc_create_order();

// Set customer
if ($payment['wp_user_id']) {
    $order->set_customer_id($payment['wp_user_id']);
}

$order->set_billing_email($wow_user_data['email']);
$order->set_billing_first_name($wow_user_data['name'] ?? 'Customer');

// Add line items
if (isset($order_data['line_items'])) {
    foreach ($order_data['line_items'] as $item) {
        $order->add_product(
            wc_get_product($item['product_id']),
            $item['quantity'],
            [
                'subtotal' => $item['price'] * $item['quantity'],
                'total' => $item['total'],
            ]
        );
    }
}

// Set currency
if (function_exists('wc_set_cart_session')) {
    $order->set_currency($order_data['currency']);
}

// Add metadata
$order->add_meta_data('wow_user_id', $wow_user_data['user_id']);
$order->add_meta_data('wow_order_id', $order_data['meta_data']['wow_order_id']);
$order->add_meta_data('transaction_kind', $payment['transaction_kind']);
$order->add_meta_data('request_id', $request_id);

// Calculate totals
$order->calculate_totals();

// Save order
$order->save();

// Redirect to checkout page
$checkout_url = wc_get_checkout_url();
$payment_url = $order->get_checkout_payment_url();

wp_redirect($payment_url);
exit();
