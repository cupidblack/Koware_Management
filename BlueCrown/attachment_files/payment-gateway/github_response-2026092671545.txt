/**
 * REST API endpoint for creating orders from Streams
 * POST /wp-json/bzj-pgb/v1/order
 * 
 * Headers required:
 * - X-BZJ-PGB-Timestamp
 * - X-BZJ-PGB-Request-ID
 * - X-BZJ-PGB-Signature
 * - Content-Type: application/json
 */
function pgb_rest_create_order($request) {
    $pgb = PGB_Bootstrap::instance();
    $logger = $pgb->get_logger();
    
    // Get headers
    $timestamp = $request->get_header('X-BZJ-PGB-Timestamp');
    $request_id = $request->get_header('X-BZJ-PGB-Request-ID');
    $signature = $request->get_header('X-BZJ-PGB-Signature');
    
    // Get raw body for signature verification
    $body = file_get_contents('php://input');
    $params = json_decode($body, true);
    
    if (!$params) {
        $logger->log(
            'Invalid JSON in request body',
            PGB_Logger::LOG_ERROR,
            [],
            __FILE__
        );
        return new WP_Error('invalid_json', 'Invalid JSON', array('status' => 400));
    }
    
    // Verify signature
    if (!pgb_verify_signature($timestamp, $request_id, $body, $signature)) {
        $logger->log(
            'Signature verification failed',
            PGB_Logger::LOG_ERROR,
            ['request_id' => $request_id],
            __FILE__
        );
        return new WP_Error('invalid_signature', 'Invalid signature', array('status' => 401));
    }
    
    // Verify timestamp is recent (within 5 minutes)
    $timestamp_int = intval($timestamp);
    if (abs(time() - $timestamp_int) > 300) {
        $logger->log(
            'Request timestamp too old',
            PGB_Logger::LOG_WARNING,
            ['request_id' => $request_id, 'age' => abs(time() - $timestamp_int)],
            __FILE__
        );
        return new WP_Error('stale_request', 'Request is too old', array('status' => 401));
    }
    
    // Check for duplicate request (idempotency)
    $stored_request_id = get_transient('pgb_request_' . $request_id);
    if ($stored_request_id) {
        $logger->log(
            'Duplicate request detected',
            PGB_Logger::LOG_WARNING,
            ['request_id' => $request_id],
            __FILE__
        );
        return new WP_Error('duplicate_request', 'Request already processed', array('status' => 409));
    }
    
    // Store request ID to prevent duplicates (5 minute expiry)
    set_transient('pgb_request_' . $request_id, time(), 300);
    
    // Create the order
    $manager = $pgb->get_manager();
    $result = $manager->create_order($params, $request_id);
    
    if ($result['success']) {
        return array(
            'success' => true,
            'order_id' => $result['order_id'],
            'payment_url' => $result['payment_url'],
        );
    } else {
        return new WP_Error('order_creation_failed', $result['error'], array('status' => 500));
    }
}

/**
 * Verify webhook/request signature
 * 
 * Signature = HMAC-SHA256(timestamp + "\n" + request_id + "\n" + body, secret)
 */
function pgb_verify_signature($timestamp, $request_id, $body, $signature) {
    // Get the shared secret from environment
    $secret = getenv('BZJ_PGB_SECRET');
    
    if (empty($secret)) {
        error_log('[PGB] BZJ_PGB_SECRET not set in environment');
        return false;
    }
    
    // Build the signed message
    $signed_message = $timestamp . "\n" . $request_id . "\n" . $body;
    
    // Calculate expected signature
    $expected = base64_encode(hash_hmac('sha256', $signed_message, $secret, true));
    
    // Compare (constant time to prevent timing attacks)
    return hash_equals($expected, $signature);
}

<?php
/**
 * Payment Gateway Bridge - Logger
 * 
 * Comprehensive logging with file rotation, severity levels, and component tracking
 */

if (!defined('ABSPATH')) {
    exit('Direct access not permitted.');
}

class PGB_Logger {
    const LOG_DEBUG = 'debug';
    const LOG_INFO = 'info';
    const LOG_WARNING = 'warning';
    const LOG_ERROR = 'error';
    const LOG_CRITICAL = 'critical';

    private $log_dir = '';
    private $enabled = true;
    private $min_level = self::LOG_INFO;
    private $file_handles = array();
    private $level_priority = array(
        'debug' => 0,
        'info' => 1,
        'warning' => 2,
        'error' => 3,
        'critical' => 4,
    );

    /**
     * Constructor
     */
    public function __construct($config = array()) {
        $this->log_dir = dirname(__FILE__, 2) . '/log/';
        $this->enabled = isset($config['logging_enabled']) ? $config['logging_enabled'] : true;
        $this->min_level = isset($config['log_level']) ? $config['log_level'] : self::LOG_INFO;

        // Ensure log directory exists
        if (!is_dir($this->log_dir)) {
            @mkdir($this->log_dir, 0755, true);
        }
    }

    /**
     * Log a message
     * 
     * @param string $message The message to log
     * @param string $level The log level
     * @param array $context Additional context data
     * @param string $source_file The file where the log originated
     */
    public function log($message, $level = self::LOG_INFO, $context = array(), $source_file = null) {
        if (!$this->enabled || !$this->should_log($level)) {
            return;
        }

        $timestamp = date('Y-m-d H:i:s');
        $level_upper = strtoupper($level);
        $source = $source_file ? basename($source_file, '.php') : 'unknown';

        // Build log line
        $log_line = "[{$timestamp}] [{$level_upper}] [{$source}]";

        // Add message
        $log_line .= ' ' . $message;

        // Add context if provided
        if (!empty($context)) {
            // Redact sensitive data
            $context = $this->redact_sensitive_data($context);
            $log_line .= ' | ' . json_encode($context);
        }

        $log_line .= "\n";

        // Write to file
        $this->write_to_file($source, $log_line);

        // Also write critical/error to WordPress error log
        if (in_array($level, array(self::LOG_ERROR, self::LOG_CRITICAL))) {
            error_log('[PGB-' . $level_upper . '] ' . $message);
        }
    }

    /**
     * Check if message should be logged
     */
    private function should_log($level) {
        $current_priority = $this->level_priority[$level] ?? 0;
        $min_priority = $this->level_priority[$this->min_level] ?? 0;
        return $current_priority >= $min_priority;
    }

    /**
     * Redact sensitive data from context
     */
    private function redact_sensitive_data($data) {
        if (!is_array($data)) {
            return $data;
        }

        $sensitive_keys = array(
            'password', 'secret', 'api_key', 'api_secret', 'token', 'signature',
            'credit_card', 'cvv', 'ssn', 'secret_key', 'private_key'
        );

        foreach ($data as $key => &$value) {
            if (is_array($value)) {
                $value = $this->redact_sensitive_data($value);
            } else {
                // Check if key contains sensitive keywords
                $lower_key = strtolower($key);
                foreach ($sensitive_keys as $sensitive) {
                    if (strpos($lower_key, $sensitive) !== false) {
                        $value = '[REDACTED]';
                        break;
                    }
                }
            }
        }

        return $data;
    }

    /**
     * Write message to file
     */
    private function write_to_file($source, $message) {
        $file_path = $this->log_dir . $source . '.log';

        // Check file size and rotate if needed
        if (file_exists($file_path) && filesize($file_path) > 10485760) { // 10MB
            $archived = $file_path . '.' . date('YmdHis');
            @rename($file_path, $archived);
        }

        // Open file handle if not already open
        if (!isset($this->file_handles[$file_path])) {
            $this->file_handles[$file_path] = @fopen($file_path, 'a');
        }

        // Write message
        if ($this->file_handles[$file_path]) {
            fwrite($this->file_handles[$file_path], $message);
            fflush($this->file_handles[$file_path]);
        }
    }

    /**
     * Get log file path
     */
    public function get_log_file($component) {
        return $this->log_dir . $component . '.log';
    }

    /**
     * Close all file handles
     */
    public function __destruct() {
        foreach ($this->file_handles as $handle) {
            if (is_resource($handle)) {
                fclose($handle);
            }
        }
    }
}
?>

/**
 * Payment Gateway Bridge - Validator
 * 
 * Input validation and sanitization utilities
 */

if (!defined('ABSPATH')) {

class PGB_Validator {
    /**
     * Validate email
     */
    public static function validate_email($email) {
        return filter_var($email, FILTER_VALIDATE_EMAIL) !== false;
    }

    /**
     * Sanitize email
     */
    public static function sanitize_email($email) {
        $email = filter_var($email, FILTER_VALIDATE_EMAIL);
        return $email ?: '';
    }

    /**
     * Validate amount
     */
    public static function validate_amount($amount) {
        $amount = floatval($amount);
        return $amount > 0;
    }

    /**
     * Validate user ID
     */
    public static function validate_user_id($user_id) {
        return intval($user_id) > 0;
    }

    /**
     * Validate order ID
     */
    public static function validate_order_id($order_id) {
        return !empty($order_id) && is_string($order_id) && strlen($order_id) > 0;
    }

    /**
     * Sanitize string
     */
    public static function sanitize_string($value, $allow_html = false) {
        $value = trim($value);
        if (!$allow_html) {
            $value = htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
        }
        return $value;
    }

    /**
     * Validate transaction type
     */
    public static function validate_transaction_type($type) {
        $allowed = array('PRODUCT', 'PRO', 'WALLET', 'DONATE', 'PURCHASE', 'SALE', 'SUBSCRIPTION');
        return in_array(strtoupper($type), $allowed);
    }

    /**
     * Validate currency code
     */
    public static function validate_currency($code) {
        return preg_match('/^[A-Z]{3}$/', $code) === 1;

class PGB_Idempotency {
    /**
     * Check if request has been processed
     */
    public static function is_duplicate($request_id, $logger = null) {
        if (function_exists('get_transient')) {
            $stored = get_transient('pgb_request_' . $request_id);
            return $stored !== false;
        }
        return false;
    }

    /**
     * Mark request as processed
     */
    public static function mark_processed($request_id, $logger = null) {
        if (function_exists('set_transient')) {
            set_transient('pgb_request_' . $request_id, time(), 86400); // 24 hour expiry
            
            if ($logger) {
                $logger->log(
                    'Request marked as processed',
                    PGB_Logger::LOG_DEBUG,
                    array('request_id' => $request_id),
                    __FILE__
                );
            }
        }
    }

    /**
     * Check for duplicate order within window
     */
    public static function check_duplicate_order($wow_user_id, $amount, $transaction_kind, $db, $logger = null, $window_minutes = 60) {
        if (!$db) {
            return false;
        }

        try {
            $cutoff_time = time() - ($window_minutes * 60);
            
            $sql = "SELECT COUNT(*) as cnt FROM Wo_Payment_Transactions 
                    WHERE userid = ? 
                    AND amount = ? 
                    AND kind = ? 
                    AND UNIX_TIMESTAMP(transaction_dt) > ?
                    LIMIT 1";
            
            $stmt = $db->prepare($sql);
            if (!$stmt) {
                if ($logger) {
                    $logger->log(
                        'Prepare failed for duplicate check',
                        PGB_Logger::LOG_WARNING,
                        array('error' => $db->error),
                        __FILE__
                    );
                }
                return false;
            }
            
            $stmt->bind_param('idsi', $wow_user_id, $amount, $transaction_kind, $cutoff_time);
            $stmt->execute();
            $result = $stmt->get_result();
            $row = $result->fetch_assoc();
            $stmt->close();
            
            $is_duplicate = isset($row['cnt']) && $row['cnt'] > 0;
            
            if ($is_duplicate && $logger) {
                $logger->log(
                    'Duplicate order detected within window',
                    PGB_Logger::LOG_WARNING,
                    array(
                        'wow_user_id' => $wow_user_id,
                        'amount' => $amount,
                        'transaction_kind' => $transaction_kind,
                        'window_minutes' => $window_minutes,
                    ),
                    __FILE__
                );
            }
            
            return $is_duplicate;
        } catch (Exception $e) {
            if ($logger) {
                $logger->log(
                    'Exception checking for duplicate order: ' . $e->getMessage(),
                    PGB_Logger::LOG_ERROR,
                    array(),
                    __FILE__
                );
            }
            return false;
        }
    }

                    2 => 'wow_pro_package_id_2',
                    3 => 'wow_pro_package_id_3',
                    4 => 'wow_pro_package_id_4',
                );
                $wow_product_kind = $wo['config'][$config_keys[$wow_post_id]] ?? 0;
            }
            break;
        case 'WALLET':
            $wow_product_kind = $wo['config']['wow_wallet_topup_id'] ?? 0;
            break;
        case 'DONATE':
            $wow_product_kind = $wo['config']['wow_crowdfund_id'] ?? 0;
            break;
    }

    if ($wow_product_kind <= 0) {
        http_response_code(400);
        return json_encode(array(
            'status' => 'error',
            'message' => 'Invalid product configuration'
        ));
    }

    // 9. Get WordPress user ID
    $wordpress_user_id = 0;
    if ($sqlConnect) {
        try {
            $wp_query = $sqlConnect->query(
                "SELECT wp_user_id FROM Wo_Users WHERE user_id = " . intval($wow_user_id)
            );
            if ($wp_query && $wp_query->num_rows > 0) {
                $wp_row = $wp_query->fetch_assoc();
                $wordpress_user_id = intval($wp_row['wp_user_id'] ?? 0);
            }
        } catch (Exception $e) {
            error_log('[PGB] WordPress user lookup error: ' . $e->getMessage());
        }
    }

    if ($wordpress_user_id <= 0) {
        http_response_code(500);
        return json_encode(array(
            'status' => 'error',
            'message' => 'Could not find WordPress user mapping'
        ));
    }

    // 10. Capture AffiliateWP context BEFORE leaving Streams
    $affwp_snapshot = pgb_capture_affwp_context($wordpress_user_id, $wow_user_email);

    // 11. Prepare WooCommerce order data
    $order_data = pgb_prepare_wc_order(array(
        'customer_id' => $wordpress_user_id,
        'product_id' => $wow_product_kind,
        'product_name' => $product_name,
        'product_price' => $product_price,
        'product_units' => $product_units,
        'total_amount' => $amount,
        'currency' => $wow_currency_code,
        'wow_order_id' => $wow_order_id,
        'wow_post_id' => $wow_post_id,
        'transaction_kind' => $transaction_kind,
        'affwp_snapshot' => $affwp_snapshot,
        'wow_user_id' => $wow_user_id,
        'email' => $wow_user_email,
    ));

    // 12. Send to WordPress via REST API
    $result = pgb_send_to_wordpress_api($order_data);

    if ($result['success']) {
        http_response_code(200);
        return json_encode(array(
            'status' => 200,
            'order_id' => $wow_order_id,
            'url' => $result['payment_url'],
            'message' => 'Order created, redirecting to payment'
        ));
    } else {
        http_response_code(500);
        return json_encode(array(
            'status' => 'error',
            'message' => $result['error'] ?? 'Failed to create WooCommerce order'
        ));
    }
}

/**

 * This is the key improvement - context captured here, not after redirect

function pgb_capture_affwp_context($wp_user_id, $customer_email) {
    $snapshot = array(
        'timestamp' => gmdate('c'),
        'wp_user_id' => $wp_user_id,
        'customer_email' => $customer_email,
        'snapshot_version' => '2',
    );

    // Capture AffiliateWP cookies if they exist
    $affwp_keys = array(
        'affwp_ref',
        'affwp_visit',
        'affwp_visit_id',
        'affwp_affiliate_id',
        'affwp_campaign',
        'affwp_coupon',

    );

    $meta_data = array(

        array('key' => '_buzzjuice_affwp_context_snapshot', 'value' => json_encode($params['affwp_snapshot'])),
    );

    // Add subscription meta if PRO transaction
    if ($params['transaction_kind'] === 'PRO') {
        // Get subscription details from Streams config
        $subscription_period = 'month';
        $subscription_interval = 1;
        $subscription_length = 0;

            'email' => $params['email'],
        ),

/**
 * Send order to WordPress via REST API
 */
function pgb_send_to_wordpress_api($order_data) {
    global $wo;

    // Get WordPress URL
    $wp_url = getenv('WP_BASE_SITE_URL') ?: ($wo['config']['wp_base_site_url'] ?? '');
    if (empty($wp_url)) {
        return array(
            'success' => false,
            'error' => 'WordPress URL not configured',
        );
    }

    // Get the secret
    $secret = getenv('BZJ_PGB_SECRET');
    if (empty($secret)) {
        return array(
            'success' => false,
            'error' => 'Payment gateway secret not configured',
        );
    }

    // Create signature
    $timestamp = time();
    $request_id = uniqid('req_');
    $body = json_encode($order_data);
    $signed_message = $timestamp . "\n" . $request_id . "\n" . $body;
    $signature = base64_encode(hash_hmac('sha256', $signed_message, $secret, true));

    // Prepare headers
    $headers = array(
        'Content-Type: application/json',
        'X-BZJ-PGB-Timestamp: ' . $timestamp,
        'X-BZJ-PGB-Request-ID: ' . $request_id,
        'X-BZJ-PGB-Signature: ' . $signature,
    );

    // Send request
    $curl = curl_init();
    curl_setopt_array($curl, array(
        CURLOPT_URL => rtrim($wp_url, '/') . '/wp-json/bzj-pgb/v1/order',
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => $headers,
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => $body,
        CURLOPT_TIMEOUT => 30,
        CURLOPT_SSL_VERIFYPEER => true,
        CURLOPT_SSL_VERIFYHOST => 2,
    ));

    $response = curl_exec($curl);
    $http_code = curl_getinfo($curl, CURLINFO_HTTP_CODE);
    $error = curl_error($curl);
    curl_close($curl);

    if ($http_code === 200 || $http_code === 201) {
        $result = json_decode($response, true);
        return array(
            'success' => true,
            'payment_url' => $result['payment_url'] ?? '',
            'order_id' => $result['order_id'] ?? '',
        );
    } else {
        error_log('[PGB] WordPress API error: HTTP ' . $http_code . ' - ' . $error);
        return array(
            'success' => false,
            'error' => 'WordPress API error: ' . ($error ?: 'HTTP ' . $http_code),
        );
    }
}

    $snapshot = maybe_unserialize($order->get_meta('_buzzjuice_affwp_context_snapshot'));
    if (!is_array($snapshot)) {

    }

    if (!empty($snapshot['affwp_affiliate_id'])) {

    // Try to resolve from customer email

            // Look up affiliate by username/ref

            if ($affiliate) {
                $affiliate_id = $affiliate->ID;
            }
        } elseif (!empty($customer_email)) {
            // Look up customer in affiliate database
            $customer = affiliate_wp()->customers->get_customer_by('email', $customer_email);
            if ($customer && $customer->affiliate_id) {
                $affiliate_id = $customer->affiliate_id;
            }
        }
    }

    if ($affiliate_id <= 0) {
        return; // No affiliate to reward
    }

    // 3. Calculate commission
    $commission = pgb_affwp_calculate_commission($order_total, $affiliate_id);

    if ($commission <= 0) {
        return;
    }

    // 4. Check for duplicate referral (idempotency)
    $existing = affiliate_wp()->referrals->get_referrals(array(
        'order_id' => $woo_order_id,
        'affiliate_id' => $affiliate_id,
    ));

    if (!empty($existing)) {
        return; // Already processed
    }

    // 5. Create referral
    $referral_id = affiliate_wp()->referrals->add(array(
        'affiliate_id' => $affiliate_id,
        'order_id' => $woo_order_id,
        'customer_id' => $customer_id,
        'amount' => $commission,
        'currency' => $currency,
        'description' => 'Buzzjuice - ' . $order->get_order_number(),
        'status' => 'approved',
    ));

    if ($referral_id) {
        do_action('pgb_affiliate_reward_processed', $referral_id, $affiliate_id, $commission);
    }
}

/**
 * Calculate commission for affiliate
 */
function pgb_affwp_calculate_commission($amount, $affiliate_id) {
    if (!class_exists('Affiliate_WP')) {
        return 0;
    }

    // Get affiliate
    $affiliate = affwp_get_affiliate($affiliate_id);
    if (!$affiliate || $affiliate->status !== 'active') {
        return 0;
    }

    // Get commission rate
    $rate = $affiliate->rate ?? affiliate_wp()->settings->get('default_referral_rate', 20);
    $rate = floatval($rate);

    // Calculate
    $commission = ($amount / 100) * $rate;

    return round($commission, 2);
}

        if (!empty($meta_value)) {
            $maybe = @unserialize($meta_value);
            if ($maybe && is_array($maybe)) {
                $caps = $maybe;
            }
        }

        // Add roles
        $caps['jewel-affiliate'] = true;
        if (!empty($map_row['wp_role'])) {
            $caps[$map_row['wp_role']] = true;
        }

        // Remove other mapped roles
        foreach ($full_mapping as $r) {
            $role = trim($r['wp_role'] ?? '');
            if (!empty($role) && $role !== $map_row['wp_role']) {
                unset($caps[$role]);
            }
        }

        // Save
        $new_meta_value = serialize($caps);
        $stmt = $wpdb->prepare(
            "UPDATE `{$meta_table}` SET meta_value = ? WHERE user_id = ? AND meta_key = ?"
        );

        if ($stmt) {
            $stmt->bind_param('sis', $new_meta_value, $wp_user_id, $cap_key);
            $stmt->execute();
            $stmt->close();
        }
    } catch (Exception $e) {
        error_log('[PGB-Jewel] Error syncing WP roles: ' . $e->getMessage());
    }
}

/**
 * Apply rebate credit to wallet
 */
function pgb_jewel_apply_rebate_credit($wow_user_id, $credit_cents, $woo_order_id) {
    if (!function_exists('get_wowonder_db')) {
        $helpers = dirname(__FILE__, 3) . '/db_helpers.php';
        if (file_exists($helpers)) {
            require_once $helpers;
        }
    }

    if (!function_exists('get_wowonder_db')) {
        return;
    }

    $db = get_wowonder_db();
    if (!$db) {
        return;
    }

    try {
        $amount = round($credit_cents / 100.0, 2);

        // Check for duplicate
        $check_sql = "SELECT id FROM Wo_Payment_Transactions 
                      WHERE userid = ? AND notes LIKE '%Rebate%' AND woo_order_id = ? LIMIT 1";
        $chk = $db->prepare($check_sql);
        if ($chk) {
            $chk->bind_param('is', $wow_user_id, $woo_order_id);
            $chk->execute();
            $chk->store_result();
            if ($chk->num_rows > 0) {
                $chk->close();
                return; // Already applied
            }
            $chk->close();
        }

        // Update wallet
        $upd = $db->prepare("UPDATE Wo_Users SET wallet = wallet + ? WHERE user_id = ?");
        if ($upd) {
            $upd->bind_param('di', $amount, $wow_user_id);
            $upd->execute();
            $upd->close();
        }

        // Record transaction
        $note = "Rebate - Subscription Rebate credit";
        $payment_status = 'completed';
        $payment_method = 'rebate';
        $ins = $db->prepare(
            "INSERT INTO Wo_Payment_Transactions (userid, amount, notes, order_id, payment_status, payment_method, woo_order_id, kind, transaction_dt) 
             VALUES (?, ?, ?, ?, ?, ?, ?, 'REBATE', NOW())"
        );

        if ($ins) {
            $order_ref = strval($woo_order_id);
            $ins->bind_param('idsssss', $wow_user_id, $amount, $note, $order_ref, $payment_status, $payment_method, $order_ref);
            $ins->execute();
            $ins->close();
        }

        error_log("[PGB-Jewel] Rebate of {$amount} applied to user $wow_user_id");
    } catch (Exception $e) {
        error_log('[PGB-Jewel] Error applying rebate: ' . $e->getMessage());
    }
}

   ```
   BZJ_PGB_SECRET=your_64char_random_secret_here
   WP_BASE_SITE_URL=https://wordpress.domain.com
   WC_API_KEY=ck_your_wc_api_key
   WC_API_SECRET=cs_your_wc_api_secret
   ```

   - Copy `bzj-payment-gateway.php` to `wp-content/mu-plugins/`
   - WordPress automatically loads it (no activation needed)

   - Update `streams/requests.php` to add PGB endpoint
   - Update `streams/themes/sunshine/layout/extra_js/content.phtml` to call new endpoint
   - Test payment flow

6. **Configure WooCommerce Webhooks:**
   - Go to WooCommerce > Settings > Webhooks
   - Create webhook for order status changes
   - Set delivery URL to your WordPress site
   - Use environment secret for signature

7. **Enable logging:**
   - Go to WordPress admin
   - Set `WP_DEBUG = true` in `wp-config.php`
   - Logs will appear in `buzzjuice.net/shared/payment-gateway/log/`
