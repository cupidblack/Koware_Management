## COMPREHENSIVE ANALYSIS

### Current Issues Identified

From the error logs, the primary problems are:

1. **File Path Issues**: Incorrect `require_once` paths (e.g., `/streams/init.php` vs `/streams/assets/init.php`)
2. **WordPress Function Dependencies**: Using WordPress functions (`absint()`) in Streams context where `wp-load.php` is not loaded
3. **Table Name Issues**: Incorrect table references (e.g., `Wo_Users` vs actual table names)
4. **Database Column Mismatches**: Non-existent columns like `signature_hash`
5. **Architecture Problems**: Over-reliance on direct API calls that fail on shared hosting

### Root Cause

The previous implementations tried to directly call WordPress functions and use incorrect paths without accounting for Streams' isolated environment. The proposed solution correctly identifies this and offers a database-first approach with a must-use plugin engine.

// File: shared/payment-gateway/core/class-pgb-logger.php

class PGB_Logger {
    private static $instance = null;
    private $logs_dir;
    private $debug_enabled;
    private $component;

    public static function getInstance($component = 'pgb-core') {
        if (self::$instance === null) {
            self::$instance = new self($component);
        }
        return self::$instance;
    }

    public function __construct($component = 'pgb-core') {
        $this->component = $component;
        $this->logs_dir = dirname(__DIR__) . '/log';
        
        if (!is_dir($this->logs_dir)) {
            @mkdir($this->logs_dir, 0755, true);
        }

        $this->debug_enabled = defined('PGB_DEBUG') && PGB_DEBUG === true;
    }

    /**
     * Log an entry to the component-specific log file
     *
     * @param string $level    One of: 'info', 'warning', 'error', 'debug'
     * @param string $message  The message to log
     * @param array  $context  Additional contextual data
     * @return bool
     */
    public function log($level = 'info', $message = '', $context = array()) {
        if (!$this->debug_enabled && $level === 'debug') {
            return false;
        }

        $log_file = $this->logs_dir . '/' . $this->component . '.log';
        
        $timestamp = gmdate('Y-m-d\TH:i:s\Z');
        $level_upper = strtoupper($level);
        
        $entry = array(
            'timestamp' => $timestamp,
            'level' => $level_upper,
            'message' => (string) $message,
            'context' => is_array($context) ? $context : array('value' => $context),
            'component' => $this->component,
        );

        $log_line = json_encode($entry, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE) . PHP_EOL;

        return (bool) @file_put_contents(
            $log_file,
            $log_line,
            FILE_APPEND | LOCK_EX
        );
    }

    public function info($message, $context = array()) {
        return $this->log('info', $message, $context);
    }

    public function warning($message, $context = array()) {
        return $this->log('warning', $message, $context);
    }

    public function error($message, $context = array()) {
        return $this->log('error', $message, $context);
    }

    public function debug($message, $context = array()) {
        return $this->log('debug', $message, $context);
    }

    /**
     * Create a fingerprint of sensitive data for logging
     *
     * @param mixed $value
     * @return string
     */
    public static function fingerprint($value) {
        if ($value === null || $value === '') {
            return '';
        }
        return substr(hash('sha256', (string) $value), 0, 16);
    }

    /**
     * Sanitize data for logging (remove sensitive fields)
     *
     * @param array $data
     * @return array
     */
    public static function sanitize_for_log($data) {
        if (!is_array($data)) {
            return $data;
        }

        $sensitive_keys = array(
            'password', 'api_key', 'api_secret', 'auth_token', 'session_id',
            'cookie', 'jwt', 'secret', 'token', 'private_key'
        );

        $sanitized = $data;

        foreach ($sanitized as $key => &$value) {
            $key_lower = strtolower($key);
            
            foreach ($sensitive_keys as $sensitive) {
                if (strpos($key_lower, $sensitive) !== false) {
                    $value = self::fingerprint($value);
                    break;
                }
            }
        }

        return $sanitized;
    }
}

class PGB_Config {
    private static $instance = null;
    private $config = array();
    private $logger;

    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('class-pgb-config');
        $this->load_config();
    }

    /**
     * Load payment gateway configuration
     * Sources: DB (primary), ENV (fallback), defaults (final fallback)
     */
    private function load_config() {
        try {
            // 1. Load from database (Streams database)
            $this->load_from_streams_db();

            // 2. Fallback to environment variables
            if (empty($this->config)) {
                $this->load_from_env();
            }

            // 3. Set defaults
            $this->set_defaults();

            $this->logger->info('Configuration loaded successfully', array(
                'has_woocommerce_url' => !empty($this->config['wow_store_url']),
                'has_api_credentials' => !empty($this->config['wow_api_key']),
            ));

        } catch (Exception $e) {
            $this->logger->error('Configuration loading failed', array(
                'message' => $e->getMessage(),
                'file' => $e->getFile(),
                'line' => $e->getLine(),
            ));
        }
    }

    /**
     * Load configuration from Streams database
     */
    private function load_from_streams_db() {
        try {
            // Streams config is stored in Wo_Settings or similar
            // This must be called from Streams context where $wo is available
            global $wo;

            if (!isset($wo['config'])) {
                $this->logger->warning('Streams config not available in global $wo');
                return;
            }

            // Map Streams config keys to PGB config
            $streams_keys = array(
                'wow_store_url' => 'wow_store_url',
                'wow_api_url' => 'wow_api_url',
                'wow_api_key' => 'wow_api_key',
                'wow_api_secret' => 'wow_api_secret',
                'wow_pro_package_id' => 'wow_pro_package_id',
                'wow_pro_package_id_2' => 'wow_pro_package_id_2',
                'wow_pro_package_id_3' => 'wow_pro_package_id_3',
                'wow_pro_package_id_4' => 'wow_pro_package_id_4',
                'wow_wallet_topup_id' => 'wow_wallet_topup_id',
                'wow_crowdfund_id' => 'wow_crowdfund_id',
                'currency' => 'currency',
            );

            foreach ($streams_keys as $streams_key => $config_key) {
                if (isset($wo['config'][$streams_key])) {
                    $this->config[$config_key] = $wo['config'][$streams_key];
                }
            }

        } catch (Exception $e) {
            $this->logger->error('Failed to load from Streams DB', array(
                'message' => $e->getMessage(),
            ));
        }
    }

    /**
     * Load configuration from environment variables
     */
    private function load_from_env() {
        $env_keys = array(
            'PGB_STORE_URL' => 'wow_store_url',
            'PGB_API_URL' => 'wow_api_url',
            'PGB_API_KEY' => 'wow_api_key',
            'PGB_API_SECRET' => 'wow_api_secret',
            'PGB_CURRENCY' => 'currency',
        );

        foreach ($env_keys as $env_key => $config_key) {
            $value = getenv($env_key);
            if ($value !== false) {
                $this->config[$config_key] = $value;
            }
        }
    }

    /**
     * Set default configuration values
     */
    private function set_defaults() {
        $defaults = array(
            'wow_store_url' => '',
            'wow_api_url' => '',
            'wow_api_key' => '',
            'wow_api_secret' => '',
            'currency' => 'USD',
            'wow_pro_package_id' => 0,
            'wow_pro_package_id_2' => 0,
            'wow_pro_package_id_3' => 0,
            'wow_pro_package_id_4' => 0,
            'wow_wallet_topup_id' => 0,
            'wow_crowdfund_id' => 0,
            'enable_affiliate_wp' => true,
            'enable_subscriptions' => true,
            'max_retries' => 3,
            'retry_delay' => 5,
        );

        foreach ($defaults as $key => $default_value) {
            if (!isset($this->config[$key])) {
                $this->config[$key] = $default_value;
            }
        }
    }

    /**
     * Get a configuration value
     */
    public function get($key, $default = null) {
        return isset($this->config[$key]) ? $this->config[$key] : $default;
    }

    /**
     * Set a configuration value
     */
    public function set($key, $value) {
        $this->config[$key] = $value;
        return true;
    }

    /**
     * Get all configuration
     */
    public function get_all() {
        return $this->config;
    }

    /**
     * Check if configuration is valid
     */
    public function is_valid() {
        $required = array('wow_store_url', 'wow_api_key', 'wow_api_secret');
        
        foreach ($required as $key) {
            if (empty($this->config[$key])) {
                return false;
            }
        }

        return true;
    }

    /**
     * Get validation errors
     */
    public function get_validation_errors() {
        $errors = array();
        $required = array(
            'wow_store_url' => 'WooCommerce Store URL',
            'wow_api_key' => 'WooCommerce API Key',
            'wow_api_secret' => 'WooCommerce API Secret',
        );

        foreach ($required as $key => $label) {
            if (empty($this->config[$key])) {
                $errors[$key] = $label . ' is required';
            }
        }

        return $errors;
    }

// File: shared/payment-gateway/core/class-pgb-currency-handler.php

class PGB_Currency_Handler {
    private static $instance = null;
    private $logger;
    private $exchange_rates = array();

    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('class-pgb-currency-handler');
    }

    /**
     * Convert amount from one currency to another using WOOCS
     * 
     * @param float  $amount
     * @param string $from_currency
     * @param string $to_currency
     * @return float
     */
    public function convert($amount, $from_currency = '', $to_currency = '') {
        try {
            if (empty($from_currency) || empty($to_currency) || $from_currency === $to_currency) {
                return (float) $amount;
            }

            // Check if WOOCS is available
            if (!$this->is_woocs_available()) {
                $this->logger->warning('WOOCS not available for currency conversion', array(
                    'from' => $from_currency,
                    'to' => $to_currency,
                ));
                return (float) $amount;
            }

            // Get exchange rate using WOOCS functions
            $rate = $this->get_exchange_rate($from_currency, $to_currency);

            if ($rate === false) {
                $this->logger->warning('Exchange rate not found', array(
                    'from' => $from_currency,
                    'to' => $to_currency,
                ));
                return (float) $amount;
            }

            $converted = (float) $amount * $rate;

            $this->logger->debug('Currency conversion completed', array(
                'original_amount' => $amount,
                'from' => $from_currency,
                'to' => $to_currency,
                'rate' => $rate,
                'converted_amount' => $converted,
            ));

            return round($converted, 2);

        } catch (Exception $e) {
            $this->logger->error('Currency conversion failed', array(
                'message' => $e->getMessage(),
                'from' => $from_currency,
                'to' => $to_currency,
            ));
            return (float) $amount;
        }
    }

    /**
     * Get exchange rate between two currencies
     * Using WOOCS currency switcher plugin
     * 
     * @param string $from
     * @param string $to
     * @return float|false
     */
    private function get_exchange_rate($from, $to) {
        try {
            // Check cache first
            $cache_key = 'pgb_rate_' . $from . '_' . $to;
            if (isset($this->exchange_rates[$cache_key])) {
                return $this->exchange_rates[$cache_key];
            }

            // If WOOCS has the function available
            if (function_exists('woocs_get_currencies')) {
                $currencies = woocs_get_currencies();
                
                if (isset($currencies[$from]) && isset($currencies[$to])) {
                    $from_rate = (float) $currencies[$from]['rate'];
                    $to_rate = (float) $currencies[$to]['rate'];

                    if ($from_rate > 0) {
                        $rate = $to_rate / $from_rate;
                        $this->exchange_rates[$cache_key] = $rate;
                        return $rate;
                    }
                }
            }

            // Fallback: try to get from WordPress options if WOOCS data is stored
            if (function_exists('get_option')) {
                $woocs_currencies = get_option('woocs', array());
                
                if (is_array($woocs_currencies) && isset($woocs_currencies[$from]) && isset($woocs_currencies[$to])) {
                    $from_data = $woocs_currencies[$from];
                    $to_data = $woocs_currencies[$to];

                    $from_rate = isset($from_data['rate']) ? (float) $from_data['rate'] : 1;
                    $to_rate = isset($to_data['rate']) ? (float) $to_data['rate'] : 1;

                    if ($from_rate > 0) {
                        $rate = $to_rate / $from_rate;
                        $this->exchange_rates[$cache_key] = $rate;
                        return $rate;
                    }
                }
            }

            return false;

        } catch (Exception $e) {
            $this->logger->error('Exchange rate lookup failed', array(
                'message' => $e->getMessage(),
                'from' => $from,
                'to' => $to,
            ));
            return false;
        }
    }

    /**
     * Check if WOOCS is available
     * 
     * @return bool
     */
    private function is_woocs_available() {
        if (function_exists('woocs_get_currencies')) {
            return true;
        }

        if (class_exists('WOOCS') && defined('WOOCS_PLUGIN_DIR')) {
            return true;
        }

        return false;
    }

    /**
     * Format currency for display
     * 
     * @param float  $amount
     * @param string $currency
     * @return string
     */
    public function format_for_display($amount, $currency = '') {
        $amount = (float) $amount;
        
        // Use WOOCS formatting if available
        if (function_exists('woocs_format_currency')) {
            return woocs_format_currency($amount, $currency);
        }

        // Fallback to basic formatting
        return $currency . ' ' . number_format($amount, 2, '.', ',');
    }

    /**
     * Get current store currency
     * 
     * @return string
     */
    public function get_store_currency() {
        if (function_exists('woocs_get_current_currency')) {
            return woocs_get_current_currency();
        }

        if (function_exists('get_woocommerce_currency')) {
            return get_woocommerce_currency();
        }

        return 'USD';
    }
}

<?php
// File: shared/payment-gateway/clients/class-pgb-client.php

class PGB_Client {
    protected $logger;
    protected $config;
    protected $curl_timeout = 30;
    protected $max_retries = 3;
    protected $retry_delay = 5;

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('class-pgb-client');
        $this->config = PGB_Config::getInstance();
    }

    /**
     * Make a curl request with retry logic
     * 
     * @param string $url
     * @param string $method
     * @param array  $data
     * @param array  $headers
     * @param array  $options
     * @return array
     */
    protected function request($url, $method = 'GET', $data = null, $headers = array(), $options = array()) {
        try {
            // Validate URL
            if (filter_var($url, FILTER_VALIDATE_URL) === false) {
                $this->logger->error('Invalid URL provided to request', array(
                    'url' => $url,
                ));
                return $this->error_response('Invalid URL', 400);
            }

            // Prepare headers
            $default_headers = array(
                'Content-Type: application/json',
            );

            $final_headers = array_merge($default_headers, (array) $headers);

            // Prepare curl options
            $curl_options = array(
                CURLOPT_URL => $url,
                CURLOPT_RETURNTRANSFER => true,
                CURLOPT_HTTPHEADER => $final_headers,
                CURLOPT_CUSTOMREQUEST => strtoupper($method),
                CURLOPT_TIMEOUT => $this->curl_timeout,
                // SSL verification disabled for testing - enable in production
                CURLOPT_SSL_VERIFYHOST => 0,
                CURLOPT_SSL_VERIFYPEER => 0,
            );

            // Merge with additional options
            if (is_array($options)) {
                $curl_options = array_merge($curl_options, $options);
            }

            // Add data for POST/PUT
            if (!empty($data) && in_array($method, array('POST', 'PUT', 'PATCH'))) {
                $curl_options[CURLOPT_POSTFIELDS] = is_string($data) 
                    ? $data 
                    : json_encode($data);
            }

            // Execute with retry logic
            $attempt = 0;
            $last_error = null;

            while ($attempt < $this->max_retries) {
                try {
                    $response = $this->execute_curl($curl_options);

                    if ($response['success']) {
                        $this->logger->debug('Request successful', array(
                            'url' => $url,
                            'method' => $method,
                            'http_code' => $response['http_code'],
                            'attempt' => $attempt + 1,
                        ));
                        return $response;
                    }

                    // Determine if we should retry
                    if ($this->should_retry($response['http_code'])) {
                        $attempt++;
                        if ($attempt < $this->max_retries) {
                            $delay = $this->retry_delay * ($attempt + 1);
                            $this->logger->debug('Retrying request', array(
                                'url' => $url,
                                'attempt' => $attempt + 1,
                                'delay' => $delay,
                            ));
                            sleep($delay);
                            continue;
                        }
                    }

                    $last_error = $response;
                    break;

                } catch (Exception $e) {
                    $attempt++;
                    $last_error = array(
                        'success' => false,
                        'error' => $e->getMessage(),
                        'http_code' => 0,
                    );

                    if ($attempt < $this->max_retries) {
                        sleep($this->retry_delay);
                        continue;
                    }
                    break;
                }
            }

            $this->logger->error('Request failed after retries', array(
                'url' => $url,
                'method' => $method,
                'attempts' => $attempt,
                'last_error' => $last_error,
            ));

            return $last_error ?? $this->error_response('Request failed', 0);

        } catch (Exception $e) {
            $this->logger->error('Request exception', array(
                'message' => $e->getMessage(),
                'file' => $e->getFile(),
                'line' => $e->getLine(),
            ));

            return $this->error_response($e->getMessage(), 0);
        }
    }

    /**
     * Execute curl request
     * 
     * @param array $options
     * @return array
     */
    private function execute_curl($options) {
        $curl = curl_init();
        curl_setopt_array($curl, $options);

        $response = curl_exec($curl);
        $http_code = (int) curl_getinfo($curl, CURLINFO_HTTP_CODE);
        $error = curl_error($curl);
        curl_close($curl);

        if ($error) {
            return array(
                'success' => false,
                'error' => $error,
                'http_code' => $http_code,
                'response' => null,
            );
        }

        return array(
            'success' => $http_code >= 200 && $http_code < 300,
            'http_code' => $http_code,
            'error' => null,
            'response' => $response,
        );
    }

    /**
     * Determine if request should be retried
     * 
     * @param int $http_code
     * @return bool
     */
    protected function should_retry($http_code) {
        // Retry on server errors and timeouts
        $retryable = array(
            408, // Request Timeout
            429, // Too Many Requests
            500, // Internal Server Error
            502, // Bad Gateway
            503, // Service Unavailable
            504, // Gateway Timeout
        );

        return in_array($http_code, $retryable);
    }

    /**
     * Create error response array
     * 
     * @param string $message
     * @param int    $code
     * @return array
     */
    protected function error_response($message, $code = 0) {
        return array(
            'success' => false,
            'error' => $message,
            'http_code' => $code,
            'response' => null,
        );
    }

    /**
     * Parse JSON response
     * 
     * @param string $response
     * @return array|false
     */
    protected function parse_json_response($response) {
        try {
            return json_decode($response, true);
        } catch (Exception $e) {
            $this->logger->error('JSON parsing failed', array(
                'message' => $e->getMessage(),
                'response_sample' => substr($response, 0, 200),
            ));
            return false;
        }
    }
}

// File: shared/payment-gateway/clients/class-pgb-client-streams.php

class PGB_Client_Streams extends PGB_Client {
    private $wo_connection;
    private $streams_db;

    public function __construct() {
        parent::__construct();
        $this->logger = PGB_Logger::getInstance('class-pgb-client-streams');
    }

    /**
     * Initialize connection to Streams database
     * Called when operating in Streams context
     * 
     * @param mysqli $connection
     * @return bool
     */
    public function set_database_connection($connection) {
        if (!($connection instanceof mysqli)) {
            $this->logger->error('Invalid database connection provided');

        }

        $this->streams_db = $connection;
        return true;
    }

    /**
     * Store payment order in Streams database
     * This is called from Streams context before redirect to WooCommerce
     * 
     * @param array $order_data
     * @return array
     */
    public function store_order_in_database($order_data) {
        try {
            if (!$this->streams_db) {
                throw new Exception('Streams database connection not initialized');
            }

            // Validate required fields
            $required = array('user_id', 'amount', 'transaction_kind', 'currency_code');
            foreach ($required as $field) {
                if (empty($order_data[$field])) {
                    throw new Exception("Missing required field: $field");
                }
            }

            // Generate order ID
            $order_id = 'wow_' . uniqid('', true);
            $timestamp = time();

            // Prepare data
            $user_id = (int) $order_data['user_id'];
            $amount = (float) $order_data['amount'];
            $transaction_kind = $order_data['transaction_kind'];
            $currency_code = $order_data['currency_code'];
            $product_name = isset($order_data['product_name']) ? $order_data['product_name'] : 'Product';
            $product_id = isset($order_data['product_id']) ? (int) $order_data['product_id'] : 0;
            $product_owner_id = isset($order_data['product_owner_id']) ? (int) $order_data['product_owner_id'] : 0;
            $product_units = isset($order_data['product_units']) ? (int) $order_data['product_units'] : 1;
            $address_id = isset($order_data['address_id']) ? (int) $order_data['address_id'] : 0;
            $notes = isset($order_data['notes']) ? $order_data['notes'] : '';

            // Store in Wo_Payment_Transactions
            $sql = "INSERT INTO Wo_Payment_Transactions 
                    (userid, amount, order_id, kind, currency_code, notes, timestamp) 
                    VALUES (?, ?, ?, ?, ?, ?, ?)";

                throw new Exception('Prepare failed: ' . $this->streams_db->error);

                'idssssi',
                $user_id,
                $amount,
                $order_id,
                $transaction_kind,
                $currency_code,
                $notes,
                $timestamp
            );

            if (!$stmt->execute()) {
                throw new Exception('Execute failed: ' . $stmt->error);
            }

            $stmt->close();

            // Store order metadata for later retrieval
            $metadata = array(
                'order_id' => $order_id,
                'user_id' => $user_id,
                'product_name' => $product_name,
                'product_id' => $product_id,
                'product_owner_id' => $product_owner_id,
                'product_units' => $product_units,
                'address_id' => $address_id,
                'transaction_kind' => $transaction_kind,
                'currency_code' => $currency_code,
                'amount' => $amount,
                'timestamp' => $timestamp,
                'status' => 'pending',
                'woocommerce_order_id' => null,
                'woocommerce_transaction_id' => null,
            );

            // Store metadata in Streams session or cache
            $this->store_order_metadata($order_id, $metadata);

            $this->logger->info('Order stored successfully', array(
                'order_id' => $order_id,
                'user_id' => $user_id,
                'amount' => $amount,
                'kind' => $transaction_kind,
            ));

            return array(

                'message' => 'Order stored successfully',
            );

<?php
// File: shared/payment-gateway/core/class-pgb-bootstrap.php

class PGB_Bootstrap {
    private static $instance = null;
    private $logger;
    private $config;
    private $addons = array();
    private $integrations = array();

    public static function getInstance() {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('class-pgb-bootstrap');
        $this->config = PGB_Config::getInstance();
        
        $this->logger->info('Payment Gateway Bootstrap initialized');
    }

    /**
     * Initialize the payment gateway
     * Called from must-use plugin
     */
    public function init() {
        try {
            $this->logger->info('Initializing Payment Gateway Bridge');

            // Load core classes
            $this->load_core_classes();

            // Load addons
            $this->load_addons();

            // Load integrations
            $this->load_integrations();

            // Setup WordPress hooks
            $this->setup_hooks();

            // Validate configuration
            if (!$this->config->is_valid()) {
                $errors = $this->config->get_validation_errors();
                $this->logger->warning('Configuration validation failed', $errors);
            }

            $this->logger->info('Payment Gateway Bridge initialization complete');

            return true;

        } catch (Exception $e) {
            $this->logger->error('Bootstrap initialization failed', array(
                'message' => $e->getMessage(),
                'file' => $e->getFile(),
                'line' => $e->getLine(),
            ));
            return false;
        }
    }

    /**
     * Load core classes
     */
    private function load_core_classes() {
        $core_classes = array(
            'class-pgb-logger',
            'class-pgb-config',
            'class-pgb-currency-handler',
        );

        foreach ($core_classes as $class_file) {
            $file_path = dirname(__DIR__) . '/core/' . $class_file . '.php';
            
            if (file_exists($file_path)) {
                require_once $file_path;
                $this->logger->debug('Core class loaded', array('class' => $class_file));
            } else {
                $this->logger->warning('Core class not found', array('file' => $file_path));
            }
        }
    }

    /**
     * Load client classes
     */
    public function load_client_classes() {
        $client_classes = array(
            'class-pgb-client',
            'class-pgb-client-streams',
            'class-pgb-client-wordpress',
        );

        foreach ($client_classes as $class_file) {
            $file_path = dirname(__DIR__) . '/clients/' . $class_file . '.php';
            
            if (file_exists($file_path)) {
                require_once $file_path;
                $this->logger->debug('Client class loaded', array('class' => $class_file));
            } else {
                $this->logger->warning('Client class not found', array('file' => $file_path));
            }
        }
    }

    /**
     * Load addons from addons directory
     */
    private function load_addons() {
        $addons_dir = dirname(__DIR__) . '/addons';

        if (!is_dir($addons_dir)) {
            $this->logger->debug('Addons directory does not exist', array('path' => $addons_dir));
            return;
        }

        $addon_files = glob($addons_dir . '/pgb-addon-*.php');

        foreach ($addon_files as $addon_file) {
            try {
                require_once $addon_file;
                
                $addon_name = basename($addon_file, '.php');
                $this->addons[$addon_name] = $addon_file;
                
                $this->logger->debug('Addon loaded', array('addon' => $addon_name));

            } catch (Exception $e) {
                $this->logger->error('Failed to load addon', array(
                    'file' => basename($addon_file),
                    'message' => $e->getMessage(),
                ));
            }
        }

        $this->logger->info('Addons loaded', array('count' => count($this->addons)));
    }

    /**
     * Setup WordPress hooks
     */
    private function setup_hooks() {
        // Hook for order completion processing
        add_action('woocommerce_order_status_completed', array($this, 'on_order_completed'), 999, 1);
        
        // Hook for order status change
        add_action('woocommerce_order_status_changed', array($this, 'on_order_status_changed'), 10, 4);
        
        // Admin page
        add_action('admin_menu', array($this, 'register_admin_pages'));

        $this->logger->debug('WordPress hooks registered');
    }

    /**
     * Handle order completion
     * 
     * @param int $order_id
     */
    public function on_order_completed($order_id) {
        try {
            $this->logger->info('Order completion hook triggered', array('order_id' => $order_id));

            // Get order from database
            $order = wc_get_order($order_id);

            if (!$order) {
                throw new Exception('Order not found');
            }

            // Check if this is a PGB order
            $is_pgb_order = $order->get_meta('_buzzjuice_pgb_order');

            if (!$is_pgb_order) {
                $this->logger->debug('Not a PGB order, skipping', array('order_id' => $order_id));
                return;
            }

            // Process order completion
            $this->process_order_completion($order);

        } catch (Exception $e) {
            $this->logger->error('Order completion processing failed', array(
                'message' => $e->getMessage(),
                'order_id' => $order_id,
            ));
        }
    }

    /**
     * Handle order status changes
     * 
     * @param int    $order_id
     * @param string $old_status
     * @param string $new_status
     * @param object $order
     */
    public function on_order_status_changed($order_id, $old_status, $new_status, $order) {
        try {
            $this->logger->debug('Order status changed', array(
                'order_id' => $order_id,
                'old_status' => $old_status,
                'new_status' => $new_status,
            ));

            // Handle specific status transitions
            if ($new_status === 'completed') {
                $this->on_order_completed($order_id);
            } elseif ($new_status === 'failed' || $new_status === 'cancelled') {
                $this->process_order_failure($order);
            }

        } catch (Exception $e) {
            $this->logger->error('Order status change handler failed', array(
                'message' => $e->getMessage(),
                'order_id' => $order_id,
            ));
        }
    }

    /**
     * Process order completion
     * 
     * @param WC_Order $order
     */
    private function process_order_completion($order) {
        try {

class PGB_Helpers_Order {
    private static $logger;

    /**
     * Build WooCommerce order data from Streams order
     * 
     * @param array $streams_order
     * @param int   $wordpress_customer_id
     * @return array
     */
    public static function build_woocommerce_order_data($streams_order, $wordpress_customer_id) {
        try {
            $currency_handler = PGB_Currency_Handler::getInstance();
            $config = PGB_Config::getInstance();

            // Convert currency if necessary
            $order_amount = $streams_order['amount'];
            $order_currency = $streams_order['currency_code'] ?? 'USD';
            $woocommerce_currency = $currency_handler->get_store_currency();

            if ($order_currency !== $woocommerce_currency) {
                $order_amount = $currency_handler->convert(
                    $order_amount,
                    $order_currency,
                    $woocommerce_currency
                );
            }

            // Get product mapping
            $product_id = self::get_woocommerce_product_id(
                $streams_order['transaction_kind'],
                $streams_order['wow_post_id'] ?? 0
            );

            // Build line item
            $line_items = array(
                array(
                    'product_id' => $product_id,
                    'quantity' => $streams_order['product_units'] ?? 1,
                    'price' => number_format((float) $order_amount, 2, '.', ''),
                    'total' => number_format((float) $order_amount, 2, '.', ''),
                    'subtotal' => number_format((float) $order_amount, 2, '.', ''),
                    'name' => $streams_order['product_name'] ?? 'Product',
                    'meta_data' => array(
                        array(
                            'key' => 'streams_order_id',
                            'value' => $streams_order['order_id'] ?? '',
                        ),
                        array(
                            'key' => 'streams_user_id',
                            'value' => $streams_order['user_id'] ?? 0,
                        ),
                        array(
                            'key' => 'transaction_kind',
                            'value' => $streams_order['transaction_kind'] ?? '',
                        ),
                    ),
                ),
            );

            // Build order data
            $order_data = array(
                'customer_id' => (int) $wordpress_customer_id,
                'currency' => $woocommerce_currency,
                'set_paid' => false,
                'line_items' => $line_items,
                'meta_data' => array(
                    array(
                        'key' => '_buzzjuice_pgb_order',
                        'value' => '1',
                    ),
                    array(
                        'key' => '_buzzjuice_streams_order_id',
                        'value' => $streams_order['order_id'] ?? '',
                    ),
                    array(
                        'key' => '_buzzjuice_streams_user_id',
                        'value' => (int) ($streams_order['user_id'] ?? 0),
                    ),
                    array(
                        'key' => '_buzzjuice_transaction_kind',
                        'value' => $streams_order['transaction_kind'] ?? '',
                    ),
                    array(
                        'key' => '_buzzjuice_product_owner_id',
                        'value' => (int) ($streams_order['product_owner_id'] ?? 0),
                    ),
                ),
            );

            return $order_data;

        } catch (Exception $e) {
            self::$logger->error('Failed to build WooCommerce order data', array(
                'message' => $e->getMessage(),
            ));
            return array();
        }
    }

    /**
     * Get WooCommerce product ID from Streams transaction type
     * 
     * @param string $transaction_kind
     * @param int    $wow_post_id
     * @return int
     */
    private static function get_woocommerce_product_id($transaction_kind, $wow_post_id = 0) {
        $config = PGB_Config::getInstance();

        switch ($transaction_kind) {
            case 'PRO':
                // Map to pro package product ID
                return (int) $config->get('wow_pro_package_id');

            case 'WALLET':
                return (int) $config->get('wow_wallet_topup_id');

            case 'DONATE':
                return (int) $config->get('wow_crowdfund_id');

            case 'PRODUCT':
            case 'PURCHASE':
                return (int) $wow_post_id;

            default:
                return 0;
        }
    }

    public static function get_checkout_payment_url($woocommerce_order_id) {

                return false;

        } catch (Exception $e) {
            self::$logger->error('Failed to get checkout payment URL', array(
                'message' => $e->getMessage(),
                'order_id' => $woocommerce_order_id,
            ));
            return false;
        }
    }
}

<?php
// File: shared/payment-gateway/addons/pgb-addon-affiliate-wp.php

class PGB_Addon_Affiliate_WP {
    private $logger;
    private $config;

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('pgb-addon-affiliate-wp');
        $this->config = PGB_Config::getInstance();

        if (!$this->is_affiliate_wp_active()) {
            $this->logger->warning('AffiliateWP plugin is not active');
            return;
        }

        // Hook into order completion
        add_action('woocommerce_order_status_completed', array($this, 'process_order_for_affiliates'), 10, 1);
    }

    /**
     * Check if AffiliateWP is active
     */
    private function is_affiliate_wp_active() {
        return class_exists('Affiliate_WP') && function_exists('affiliate_wp');
    }

    /**
     * Process order for affiliate commissions
     * 
     * @param int $order_id
     */
    public function process_order_for_affiliates($order_id) {
        try {
            if (!$this->is_affiliate_wp_active()) {
                return;
            }

            $order = wc_get_order($order_id);

            if (!$order) {
                return;
            }

            // Check if this is a PGB order
            if (!$order->get_meta('_buzzjuice_pgb_order')) {
                return;
            }

            $this->logger->info('Processing order for affiliates', array(
                'order_id' => $order_id,
            ));

            // Get customer
            $customer_id = $order->get_customer_id();

            if (!$customer_id) {
                $this->logger->warning('No customer found for order', array(
                    'order_id' => $order_id,
                ));
                return;
            }

            // Get WooCommerce customer email for AffiliateWP lookup
            $customer = new WC_Customer($customer_id);
            $customer_email = $customer->get_email();

            // Check for affiliate referrer
            $affiliate = affwp_get_affiliate_by('email', $customer_email);

            if (!$affiliate || !$affiliate->ID) {
                $this->logger->debug('No affiliate found for customer', array(
                    'order_id' => $order_id,
                    'email' => $customer_email,
                ));
                return;
            }

            $this->logger->info('Affiliate found for customer', array(
                'order_id' => $order_id,
                'affiliate_id' => $affiliate->ID,
            ));

            // Calculate commission
            $commission = $this->calculate_commission($order, $affiliate);

            if (!$commission) {
                $this->logger->warning('No commission calculated', array(
                    'order_id' => $order_id,
                    'affiliate_id' => $affiliate->ID,
                ));
                return;
            }

            // Create referral
            $referral_data = array(
                'customer_id' => affwp_get_customer_id_by_user_id($affiliate->user_id),
                'affiliate_id' => $affiliate->ID,
                'amount' => $commission,
                'reference' => $order_id,
                'description' => 'WooCommerce Order #' . $order_id,
                'status' => 'approved',
                'order_total' => $order->get_total(),
                'order_id' => $order_id,
            );

            $referral_id = affwp_add_referral($referral_data);

            if ($referral_id) {
                $this->logger->info('Referral created', array(
                    'referral_id' => $referral_id,
                    'affiliate_id' => $affiliate->ID,
                    'amount' => $commission,
                ));
            } else {
                $this->logger->error('Failed to create referral', array(
                    'affiliate_id' => $affiliate->ID,
                ));
            }

        } catch (Exception $e) {
            $this->logger->error('Affiliate processing failed', array(
                'message' => $e->getMessage(),
                'order_id' => $order_id,
            ));
        }
    }

    /**
     * Calculate commission for order
     * 
     * @param WC_Order $order
     * @param object   $affiliate
     * @return float|false
     */
    private function calculate_commission($order, $affiliate) {
        try {
            $commission_rate = affwp_get_affiliate_meta($affiliate->ID, 'commission_rate', true);

            if (!$commission_rate) {
                // Use default rate if available
                $commission_rate = affiliate_wp()->settings->get('referral_rate', 0);
            }

            if (!$commission_rate) {
                return false;
            }

            $order_total = $order->get_total();
            $commission = ($order_total * $commission_rate) / 100;

            $this->logger->debug('Commission calculated', array(
                'affiliate_id' => $affiliate->ID,
                'order_total' => $order_total,
                'rate' => $commission_rate,
                'commission' => $commission,
            ));

            return $commission;

        } catch (Exception $e) {
            $this->logger->error('Commission calculation failed', array(
                'message' => $e->getMessage(),
            ));
            return false;
        }
    }
}

### **PHASE 8: STREAMS CLIENT INTEGRATION**

```php
<?php
// File: shared/payment-gateway/clients/pgb-streams-handler.php
// This file is called from Streams context

// Include Streams init to access $wo global
if (file_exists(__DIR__ . '/../../streams/assets/init.php')) {
    require_once __DIR__ . '/../../streams/assets/init.php';
}

// This provides the bridge between Streams and the Payment Gateway

class PGB_Streams_Handler {
    private $logger;
    private $streams_client;

    public function __construct() {
        $this->logger = PGB_Logger::getInstance('pgb-streams-handler');
        $this->streams_client = new PGB_Client_Streams();
    }

    /**
     * Initialize payment from Streams checkout button
     * Called via AJAX from Streams frontend
     * 
     * @param array $order_data
     * @return array
     */
    public function initiate_payment($order_data) {
        try {
            global $wo;

            if (!isset($wo['user']) || !$wo['user']) {
                throw new Exception('User not authenticated');
            }

            // Validate order data
            $required = array(
                'amount', 'product_name', 'product_price', 'product_units',
                'transaction_kind', 'currency_code'
            );

            foreach ($required as $field) {
                if (empty($order_data[$field])) {
                    throw new Exception("Missing: $field");
                }
            }

            // Set database connection
            global $sqlConnect;
            $this->streams_client->set_database_connection($sqlConnect);

            // Enrich order data with user info
            $order_data['user_id'] = $wo['user']['user_id'] ?? $wo['user']['id'];

            // Store order in Streams database
            $store_result = $this->streams_client->store_order_in_database($order_data);

            if (!$store_result['success']) {
                throw new Exception($store_result['message']);
            }

            $this->logger->info('Payment initiated from Streams', array(
                'order_id' => $store_result['order_id'],
                'user_id' => $order_data['user_id'],
                'amount' => $order_data['amount'],
            ));

            return array(
                'success' => true,
                'order_id' => $store_result['order_id'],
                'message' => 'Order created successfully',
                'proceed_to_woocommerce' => true,
            );

        } catch (Exception $e) {
            $this->logger->error('Payment initiation failed', array(
                'message' => $e->getMessage(),
            ));

            return array(
                'success' => false,
                'message' => $e->getMessage(),
            );
        }
    }

    /**
     * Get WordPress user ID for Streams user
     * 
     * @param int $streams_user_id
     * @return int|false
     */
    public function get_wordpress_user_id($streams_user_id) {
        try {
            global $sqlConnect;

            if (!$sqlConnect) {
                throw new Exception('Database connection not available');
            }

            // Query Streams users table for wp_user_id mapping
            $sql = "SELECT wp_user_id FROM Wo_Users WHERE user_id = ? LIMIT 1";
            $stmt = $sqlConnect->prepare($sql);

            if (!$stmt) {
                throw new Exception('Prepare failed: ' . $sqlConnect->error);
            }

            $stmt->bind_param('i', $streams_user_id);
            $stmt->execute();
            $result = $stmt->get_result();

            $wp_user_id = false;

            if ($result && $result->num_rows > 0) {
                $row = $result->fetch_assoc();
                $wp_user_id = (int) $row['wp_user_id'];
            }

            $stmt->close();

            return $wp_user_id;

        } catch (Exception $e) {
            $this->logger->error('Failed to get WordPress user ID', array(
                'message' => $e->getMessage(),
                'streams_user_id' => $streams_user_id,
            ));
            return false;
        }
    }
}
?>
```
