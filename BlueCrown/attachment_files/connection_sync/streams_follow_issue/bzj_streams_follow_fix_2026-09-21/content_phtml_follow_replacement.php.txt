/*
 * Replace the existing Wo_RegisterFollow() function in:
 * streams/themes/sunshine/layout/extra_js/content.phtml
 *
 * This is intentionally self-contained and does not change unrelated
 * JavaScript in the file.
 */
(function () {
    'use strict';

    window.BuzzjuiceFollowState = window.BuzzjuiceFollowState || {};

    function followLog(eventName, details) {
        if (window.console && console.debug) {
            console.debug('[BuzzjuiceFollow]', eventName, details || {});
        }
    }

    function escapeHtml(value) {
        return $('<div>').text(value == null ? '' : String(value)).html();
    }

    function renderFollowButton(id, state) {
        var container = $('[id="Follow-' + id + '"]');

        if (!container.length) {
            followLog('container_missing', { id: id });
            return;
        }

        var followingText = <?php echo json_encode($wo['lang']['following_btn'] ?? 'Following'); ?>;
        var followText = <?php echo json_encode($wo['lang']['follow'] ?? 'Follow'); ?>;

        var isFollowing = state === 'following';
        var label = isFollowing ? followingText : followText;

        var buttonClass = isFollowing
            ? 'btn-active btn btn-default btn-sm wo_following_btn'
            : 'btn btn-default btn-sm wo_follow_btn';

        var icon = isFollowing
            ? '<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M16 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2"></path><circle cx="8.5" cy="7" r="4"></circle><polyline points="17 11 19 13 23 9"></polyline></svg>'
            : '<svg width="16" height="16" viewBox="0 0 16 16" fill="none" xmlns="http://www.w3.org/2000/svg"><path d="M14.75 8.44434H11.75M13.25 9.94434L13.25 6.94434" class="icon_main" stroke-width="1.5" stroke-linecap="round"/><path d="M1.25 13.7779C1.25 11.9681 2.52299 10.4266 4.25303 10.1413L4.40886 10.1156C5.79419 9.88724 7.20581 9.88724 8.59113 10.1156L8.74697 10.1413C10.477 10.4266 11.75 11.9681 11.75 13.7779C11.75 14.5602 11.1361 15.1943 10.3789 15.1943H2.6211C1.86386 15.1943 1.25 14.5602 1.25 13.7779Z" stroke="currentColor" stroke-width="1.5"/><path d="M9.5625 4.64746C9.5625 6.27843 8.19137 7.60059 6.5 7.60059C4.80863 7.60059 3.4375 6.27843 3.4375 4.64746C3.4375 3.0165 4.80863 1.69434 6.5 1.69434C8.19137 1.69434 9.5625 4.64746 9.5625 4.64746Z" class="icon_main" stroke-width="1.5"/></svg>';

        var html =
            '<button type="button" ' +
            'class="' + buttonClass + '" ' +
            'id="wo_useract_btn" ' +
            'data-follow-state="' + state + '" ' +
            'aria-pressed="' + (isFollowing ? 'true' : 'false') + '" ' +
            'onclick="return Wo_RegisterFollow(' + parseInt(id, 10) + ')">' +
            icon +
            '<span class="button-text">' + escapeHtml(label) + '</span>' +
            '</button>';

        container.html(html);
    }

    window.Wo_RegisterFollow = function (id) {
        id = parseInt(id, 10);

        if (!id || id < 1) {
            followLog('invalid_id', { id: id });
            return false;
        }

        if (window.BuzzjuiceFollowState[id] === 'pending') {
            followLog('duplicate_click_ignored', { id: id });
            return false;
        }

        var container = $('[id="Follow-' + id + '"]');
        var button = container.find('button').first();

        if (!container.length || !button.length) {
            followLog('button_missing', { id: id });
            return false;
        }

        var previousState =
            button.attr('data-follow-state') ||
            (button.hasClass('btn-active') ? 'following' : 'follow');

        window.BuzzjuiceFollowState[id] = 'pending';

        button.prop('disabled', true);
        button.attr('aria-busy', 'true');

        followLog('request_start', {
            id: id,
            previousState: previousState
        });

        $.ajax({
            url: Wo_Ajax_Requests_File(),
            type: 'GET',
            dataType: 'json',
            cache: false,
            data: {
                f: 'follow_user',
                following_id: id,
                _: Date.now()
            }
        })
        .done(function (data) {
            followLog('request_complete', {
                id: id,
                response: data
            });

            if (!data || parseInt(data.status, 10) !== 200) {
                throw new Error(
                    data && data.error
                        ? data.error
                        : 'The follow operation failed'
                );
            }

            var finalState = data.state === 'following'
                ? 'following'
                : 'follow';

            window.BuzzjuiceFollowState[id] = finalState;

            renderFollowButton(id, finalState);

            if (
                finalState === 'following' &&
                typeof node_socket_flow !== 'undefined' &&
                String(node_socket_flow) === '1' &&
                typeof socket !== 'undefined'
            ) {
                socket.emit('user_notification', {
                    to_id: id,
                    user_id: _getCookie('user_id'),
                    type: 'added'
                });
            } else if (
                finalState === 'follow' &&
                typeof node_socket_flow !== 'undefined' &&
                String(node_socket_flow) === '1' &&
                typeof socket !== 'undefined'
            ) {
                socket.emit('user_notification', {
                    to_id: id,
                    user_id: _getCookie('user_id'),
                    type: 'removed'
                });
            }

            if (data.can_send == 1 && typeof Wo_SendMessages === 'function') {
                Wo_SendMessages();
            }
        })
        .fail(function (xhr, status, error) {
            followLog('request_failed', {
                id: id,
                httpStatus: xhr.status,
                status: status,
                error: error,
                response: xhr.responseText
            });

            window.BuzzjuiceFollowState[id] = previousState;
            renderFollowButton(id, previousState);

            if (typeof $('#modal-alert').modal === 'function') {
                $('#modal-alert').modal('show');

                if (typeof Wo_Delay === 'function') {
                    Wo_Delay(function () {
                        $('#modal-alert').modal('hide');
                    }, 3000);
                }
            }
        });

        return false;
    };
}());
