```php
            Wo_SessionDiagnosticLog(
                'create_main_session_failed',
                array(
                    'session_key' =>
                        'main_hash_id',

                    'reason' =>
                        'php_session_not_active'
                )
            );

            return '';
        }

        $existing =
            isset($_SESSION['main_hash_id'])
                ? (string) $_SESSION['main_hash_id']
                : '';

        if (
            preg_match(
                '/^[a-f0-9]{20}$/i',
                $existing
            )
        ) {

            Wo_SessionDiagnosticLog(
                'create_main_session_reused',
                array(
                    'session_key' =>
                        'main_hash_id',

                    'value' =>
                        Wo_SessionValueDiagnostics(
                            $existing
                        )
                )
            );

            return $existing;
        }

        try {

            $hash =
                bin2hex(
                    random_bytes(10)
                );

        } catch (Throwable $exception) {

            Wo_SessionDiagnosticLog(
                'create_main_session_random_bytes_failed',
                array(
                    'session_key' =>
                        'main_hash_id',

                    'exception_class' =>
                        get_class($exception),

                    'exception_message' =>
                        $exception->getMessage()
                )
            );

            return '';
        }

        $_SESSION['main_hash_id'] =
            $hash;

        Wo_SessionDiagnosticLog(
            'create_main_session_created',
            array(
                'session_key' =>
                    'main_hash_id',

                'old_value' =>
                    Wo_SessionValueDiagnostics(
                        $existing
                    ),

                'new_value' =>
                    Wo_SessionValueDiagnostics(
                        $hash
                    ),

                'session_write_pending' =>
                    true
            )
        );

        return $hash;
    }
}

if (!function_exists('Wo_CheckMainSession')) {

    function Wo_CheckMainSession($hash = '')
    {
        $provided =
            is_scalar($hash)
                ? (string) $hash
                : '';

        if (!Wo_EnsurePhpSession()) {

            Wo_SessionDiagnosticLog(
                'check_main_session_failed',
                array(
                    'session_key' =>
                        'main_hash_id',

                    'reason' =>
                        'php_session_not_active',

                    'provided' =>
                        Wo_SessionValueDiagnostics(
                            $provided
                        )
                )
            );

            return false;
        }

        $stored =
            isset($_SESSION['main_hash_id'])
                ? (string) $_SESSION['main_hash_id']
                : '';

        $provided_valid =
            (bool) preg_match(
                '/^[a-f0-9]{20}$/i',
                $provided
            );

        $stored_valid =
            (bool) preg_match(
                '/^[a-f0-9]{20}$/i',
                $stored
            );

        $matched =
            $provided_valid &&
            $stored_valid &&
            hash_equals(
                $stored,
                $provided
            );

        Wo_SessionDiagnosticLog(
            $matched
                ? 'check_main_session_passed'
                : 'check_main_session_failed',

            array(
                'session_key' =>
                    'main_hash_id',

                'provided' =>
                    Wo_SessionValueDiagnostics(
                        $provided
                    ),

                'stored' =>
                    Wo_SessionValueDiagnostics(
                        $stored
                    ),

                'provided_format_valid' =>
                    $provided_valid,

                'stored_format_valid' =>
                    $stored_valid,

                'match' =>
                    $matched,

                'reason' =>
                    $matched
                        ? 'match'
                        : (
                            $stored === ''
                                ? 'empty_stored_hash'
                                : (
                                    $provided === ''
                                        ? 'empty_provided_hash'
                                        : 'fingerprint_or_format_mismatch'
                                )
                        )
            )
        );

        return $matched;
    }
}
```

This retains the intended distinction:

```text
hash_id       = 40 hex characters
main_hash_id  = 20 hex characters
```

and removes the weak:

```php
sha1(rand(...))
```

generation. GitHub correctly identified that the old generation mechanism had an extremely small random space.

# 16. Step 12 — Add client-side diagnostic logging

Add this once to the global Streams JavaScript:

```javascript
function Wo_SessionDiagnosticLogClient(
    event,
    context
) {

    context = context || {};

    var mainSession =
        $('.main_session').first().val() || '';

    var cookieNames = [];

    if (document.cookie) {

        cookieNames =
            document.cookie
                .split(';')
                .map(function(item) {

                    return item
                        .trim()
                        .split('=')[0];

                })
                .filter(function(name) {

                    return name.length > 0;

                });
    }

    var payload = {

        event: event,

        context: {

            page:
                window.location.href,

            session_cookie_names:
                cookieNames,

            streams_session_cookie_visible:
                cookieNames.indexOf(
                    'BUZZSTREAMSESSID'
                ) !== -1,

            social_session_cookie_visible:
                cookieNames.indexOf(
                    'BUZZSOCIALSESSID'
                ) !== -1,

            legacy_php_session_cookie_visible:
                cookieNames.indexOf(
                    'PHPSESSID'
                ) !== -1,

            main_session_present:
                mainSession.length > 0,

            main_session_length:
                mainSession.length,

            main_session_sha256:
                '',

            extra:
                context
        }
    };

    /*
     * main_session is deliberately not sent in raw form.
     */

    if (
        window.crypto &&
        window.crypto.subtle &&
        window.TextEncoder
    ) {

        window.crypto.subtle.digest(
            'SHA-256',
            new TextEncoder().encode(
                mainSession
            )
        )
        .then(function(buffer) {

            var bytes =
                new Uint8Array(buffer);

            var hex = '';

            bytes.forEach(
                function(byte) {

                    hex +=
                        byte
                            .toString(16)
                            .padStart(2, '0');
                }
            );

            payload.context.main_session_sha256 =
                hex;

            if (
                navigator.sendBeacon
            ) {

                navigator.sendBeacon(
                    Wo_Ajax_Requests_File() +
                    '?f=session_status_client_log',

                    JSON.stringify(
                        payload
                    )
                );
            }

        })
        .catch(function() {});
    }
}
```

The client-side logger is diagnostic only; it must never become part of the security decision. This agrees with GitHub's recommendation.

# 18. Step 14 — Do not disable `Wo_CheckMainSession()`

This:

```php
}
```

must be removed.

The fact that this makes Follow work is actually valuable diagnostic evidence: it proves the failure is occurring at the session gate. It does **not** mean the session gate should be permanently bypassed.

The proper architecture is:

```text
correct session
+
correct token
+

```

not:


# 19. Step 15 — Disable page caching for Streams

This is particularly important because the page contains a session-specific token.

Imagine:

```text
User A
main_hash_id = AAA
```

The page gets cached.

Then:

```text
User B
receives cached page
main_session = AAA
```

but User B's actual PHP session contains:

```text
main_hash_id = BBB
```

The Follow request sends:

```text
AAA
```

PHP checks:

```text
AAA == BBB
```

and correctly rejects it.

GitHub specifically identified this possibility in its analysis. 

The `Cache-Control` headers in the new `init.php` address application-level caching, but you should also exclude:

```text
/streams/
/streams/requests.php
/streams/ajax_loading.php
/streams/ww-sso-bridge.php
```

from any server/CDN/full-page cache.

# 25. Final architecture after this change

The most important improvement is that Buzzjuice will no longer have three applications implicitly competing over one PHP session namespace.

### WordPress

```text
WordPress authentication
    ↓
WP cookies
    ↓
SSO JWT
```

### Streams

```text
/streams
    ↓
BUZZSTREAMSESSID
    ↓
$_SESSION
    ├── hash_id
    ├── main_hash_id
    ├── wo_user_id
    └── WP identity metadata
```

### Socials

```text
/social
    ↓
BUZZSOCIALSESSID
    ↓
$_SESSION
    ├── qd_user_id
    ├── user_id
    ├── wp_sso_login
    └── QuickDate state
```

This is much cleaner for the larger Buzzjuice architecture as well: **WordPress remains the identity/source-of-truth authority, while Streams and Socials maintain isolated application sessions.**
