The `invalid_main_session` log entry identifies the immediate failure:

> The request is being rejected before `Wo_RegisterFollow()` or `Wo_Followers` is reached.

Therefore, account attributes, `Wo_RegisterUser()`, and SSO registration cannot explain that particular failed follow attempt. The browser request must include the same `$hash_id` that `streams/requests.php` expects.

There are also two design problems in the previous fix:

1. The client-side code still allowed the old `Requested` logic to run in some page variants.
2. The endpoint did not consistently return a verified final state.
3. The proposed replacement can still fail when the client does not submit `hash_id`.
4. `Wo_Followers.active = 0` must not be treated as a valid pending state if Buzzjuice is intended to use immediate follows.

The following is the corrected implementation. It is intentionally non-mutating with respect to user accounts and does not change SSO registration.

if (!function_exists('Buzzjuice_FollowLog')) {
    function Buzzjuice_FollowLog($event, array $context = array())
    {
        $base_dir = dirname(__DIR__, 3) . '/data.logs/follow';

function Wo_IsFollowing($following_id, $user_id = 0)
{
    global $sqlConnect, $wo;

    $following_id = (int) $following_id;

    if (
        $user_id === 0 ||
        $user_id === null ||
        $user_id === '' ||
        !is_numeric($user_id)
    ) {
        $user_id = !empty($wo['user']['user_id'])
            ? (int) $wo['user']['user_id']
            : 0;
    } else {
        $user_id = (int) $user_id;
    }

    if ($following_id < 1 || $user_id < 1) {
        return false;
    }

    $followers_table = defined('T_FOLLOWERS')
        ? T_FOLLOWERS
        : 'Wo_Followers';

    $query = mysqli_query(
        $sqlConnect,
        "SELECT 1
         FROM {$followers_table}
         WHERE `following_id` = {$following_id}
           AND `follower_id` = {$user_id}
           AND `active` = 1
         LIMIT 1"
    );

    if (!$query) {
        Buzzjuice_FollowLog('is_following_query_failed', array(
            'following_id' => $following_id,
            'follower_id'  => $user_id,
            'mysql_error'  => mysqli_error($sqlConnect)
        ));

        return false;
    }

    return mysqli_num_rows($query) > 0;
}

function Wo_IsFollowRequested($following_id = 0, $follower_id = 0)
{
    global $sqlConnect, $wo;

    /*
     * Immediate-follow mode does not expose pending requests. This function
     * remains available for compatibility with older WoWonder code.
     */
    $following_id = (int) $following_id;

    if (
        $follower_id === 0 ||
        $follower_id === null ||
        $follower_id === '' ||
        !is_numeric($follower_id)
    ) {
        $follower_id = !empty($wo['user']['user_id'])
            ? (int) $wo['user']['user_id']
            : 0;
    } else {
        $follower_id = (int) $follower_id;
    }

    if ($following_id < 1 || $follower_id < 1) {
        return false;
    }

    return false;
}

function Wo_RegisterFollow($following_id = 0, $followers_id = 0)
{
    global $wo, $sqlConnect;

    $following_id = (int) $following_id;

    if (!is_array($followers_id)) {
        $followers_id = array($followers_id);
    }

    if ($following_id < 1) {
        Buzzjuice_FollowLog('register_rejected', array(
            'reason'       => 'invalid_following_id',
            'following_id' => $following_id
        ));

        return false;
    }

    $followers_table = defined('T_FOLLOWERS')
        ? T_FOLLOWERS
        : 'Wo_Followers';

    $users_table = defined('T_USERS')
        ? T_USERS
        : 'Wo_Users';

    $result = false;

    foreach ($followers_id as $raw_follower_id) {
        $follower_id = (int) $raw_follower_id;

        if ($follower_id < 1 || $follower_id === $following_id) {
            Buzzjuice_FollowLog('register_rejected', array(
                'reason'        => 'invalid_follower_id',
                'following_id'  => $following_id,
                'follower_id'   => $follower_id
            ));

            continue;
        }

        $user_query = mysqli_query(
            $sqlConnect,
            "SELECT `user_id`, `active`
             FROM {$users_table}
             WHERE `user_id` IN ({$following_id}, {$follower_id})"
        );

        if (!$user_query) {
            Buzzjuice_FollowLog('register_failed', array(
                'reason'        => 'user_validation_query_failed',
                'following_id'  => $following_id,
                'follower_id'   => $follower_id,
                'mysql_error'   => mysqli_error($sqlConnect)
            ));

            continue;
        }

        $valid_users = array();

        while ($user = mysqli_fetch_assoc($user_query)) {
            $valid_users[(int) $user['user_id']] = $user;
        }

        if (
            empty($valid_users[$following_id]) ||
            empty($valid_users[$follower_id]) ||
            (int) $valid_users[$following_id]['active'] !== 1 ||
            (int) $valid_users[$follower_id]['active'] !== 1
        ) {
            Buzzjuice_FollowLog('register_rejected', array(
                'reason'        => 'missing_or_inactive_user',
                'following_id'  => $following_id,
                'follower_id'   => $follower_id,
                'users_found'   => array_keys($valid_users)
            ));

            continue;
        }

        if (function_exists('Wo_IsBlocked') && Wo_IsBlocked($following_id)) {
            Buzzjuice_FollowLog('register_rejected', array(
                'reason'        => 'blocked',
                'following_id'  => $following_id,
                'follower_id'   => $follower_id
            ));

            continue;
        }

        $existing_query = mysqli_query(
            $sqlConnect,
            "SELECT `id`, `active`
             FROM {$followers_table}
             WHERE `following_id` = {$following_id}
               AND `follower_id` = {$follower_id}
             ORDER BY `id` DESC"
        );

        if (!$existing_query) {
            Buzzjuice_FollowLog('register_failed', array(
                'reason'        => 'relationship_read_failed',
                'following_id'  => $following_id,
                'follower_id'   => $follower_id,
                'mysql_error'   => mysqli_error($sqlConnect)
            ));

            continue;
        }

        $rows = array();

        while ($row = mysqli_fetch_assoc($existing_query)) {
            $rows[] = array(
                'id'     => (int) $row['id'],
                'active' => (int) $row['active']
            );
        }

        if (!empty($rows)) {
            $relationship_id = $rows[0]['id'];

            $update_query = mysqli_query(
                $sqlConnect,
                "UPDATE {$followers_table}
                 SET `active` = 1
                 WHERE `id` = {$relationship_id}"
            );

            if (!$update_query) {
                Buzzjuice_FollowLog('register_failed', array(
                    'reason'          => 'relationship_activation_failed',
                    'following_id'    => $following_id,
                    'follower_id'     => $follower_id,
                    'relationship_id' => $relationship_id,
                    'mysql_error'     => mysqli_error($sqlConnect)
                ));

                continue;
            }

            if (count($rows) > 1) {
                $duplicate_ids = array();

                foreach (array_slice($rows, 1) as $duplicate) {
                    $duplicate_ids[] = (int) $duplicate['id'];
                }

                if (!empty($duplicate_ids)) {
                    $delete_query = mysqli_query(
                        $sqlConnect,
                        "DELETE FROM {$followers_table}
                         WHERE `id` IN (" . implode(',', $duplicate_ids) . ")"
                    );

                    if (!$delete_query) {
                        Buzzjuice_FollowLog('duplicate_cleanup_failed', array(
                            'following_id'  => $following_id,
                            'follower_id'   => $follower_id,
                            'retained_id'   => $relationship_id,
                            'duplicate_ids' => $duplicate_ids,
                            'mysql_error'   => mysqli_error($sqlConnect)
                        ));
                    }
                }
            }

            Buzzjuice_FollowLog('existing_relationship_activated', array(
                'following_id'    => $following_id,
                'follower_id'     => $follower_id,
                'relationship_id' => $relationship_id,
                'previous_active' => $rows[0]['active']
            ));
        } else {
            $insert_query = mysqli_query(
                $sqlConnect,
                "INSERT INTO {$followers_table}
                    (`following_id`, `follower_id`, `active`)
                 VALUES
                    ({$following_id}, {$follower_id}, 1)"
            );

            if (!$insert_query) {
                Buzzjuice_FollowLog('register_failed', array(
                    'reason'        => 'relationship_insert_failed',
                    'following_id'  => $following_id,
                    'follower_id'   => $follower_id,
                    'mysql_error'   => mysqli_error($sqlConnect)
                ));

                continue;
            }

            $relationship_id = (int) mysqli_insert_id($sqlConnect);

            Buzzjuice_FollowLog('new_relationship_inserted', array(
                'following_id'    => $following_id,
                'follower_id'     => $follower_id,
                'relationship_id' => $relationship_id,
                'active'          => 1
            ));
        }

        $verify_query = mysqli_query(
            $sqlConnect,
            "SELECT `id`
             FROM {$followers_table}
             WHERE `following_id` = {$following_id}
               AND `follower_id` = {$follower_id}
               AND `active` = 1
             ORDER BY `id` DESC
             LIMIT 1"
        );

        if (!$verify_query || mysqli_num_rows($verify_query) < 1) {
            Buzzjuice_FollowLog('register_verification_failed', array(
                'following_id'    => $following_id,
                'follower_id'     => $follower_id,
                'relationship_id' => $relationship_id,
                'mysql_error'     => mysqli_error($sqlConnect)
            ));

            continue;
        }

        $verified = mysqli_fetch_assoc($verify_query);

        if (function_exists('cache')) {
            @cache($following_id, 'users', 'delete');
            @cache($follower_id, 'users', 'delete');
        }

        $result = true;

        Buzzjuice_FollowLog('register_complete', array(
            'following_id'    => $following_id,
            'follower_id'     => $follower_id,
            'relationship_id' => (int) $verified['id'],
            'state'           => 'following',
            'active'          => 1
        ));
    }

    return $result;
}

function Wo_DeleteFollow($following_id = 0, $follower_id = 0)
{
    global $wo, $sqlConnect;

    $following_id = (int) $following_id;
    $follower_id  = (int) $follower_id;

    if (
        $following_id < 1 ||
        $follower_id < 1 ||
        $following_id === $follower_id
    ) {
        Buzzjuice_FollowLog('delete_rejected', array(
            'reason'        => 'invalid_relationship',
            'following_id'  => $following_id,
            'follower_id'   => $follower_id
        ));

        return false;
    }

    $followers_table = defined('T_FOLLOWERS')
        ? T_FOLLOWERS
        : 'Wo_Followers';

    $query = mysqli_query(
        $sqlConnect,
        "DELETE FROM {$followers_table}
         WHERE `following_id` = {$following_id}
           AND `follower_id` = {$follower_id}"
    );

    if (!$query) {
        Buzzjuice_FollowLog('delete_failed', array(
            'following_id' => $following_id,
            'follower_id'  => $follower_id,
            'mysql_error'  => mysqli_error($sqlConnect)
        ));

        return false;
    }

    $deleted_rows = mysqli_affected_rows($sqlConnect);

    if (
        !empty($wo['config']['connectivitySystem']) &&
        (int) $wo['config']['connectivitySystem'] === 1
    ) {
        mysqli_query(
            $sqlConnect,
            "DELETE FROM {$followers_table}
             WHERE `following_id` = {$follower_id}
               AND `follower_id` = {$following_id}"
        );
    }

    if (function_exists('Wo_DeleteSelectedActivity')) {
        if (
            !empty($wo['config']['connectivitySystem']) &&
            (int) $wo['config']['connectivitySystem'] === 1
        ) {
            Wo_DeleteSelectedActivity($follower_id, 'friend', $following_id);
            Wo_DeleteSelectedActivity($following_id, 'friend', $follower_id);
        } else {
            Wo_DeleteSelectedActivity($follower_id, 'following', $following_id);
        }
    }

    if (function_exists('bzj_connection_client')) {
        try {
            bzj_connection_client(
                'streams',
                'unfollow',
                $follower_id,
                $following_id
            );
        } catch (Throwable $exception) {
            Buzzjuice_FollowLog('sync_exception', array(
                'operation'    => 'unfollow',
                'following_id' => $following_id,
                'follower_id'  => $follower_id,
                'error'        => $exception->getMessage()
            ));
        }
    }

    if (function_exists('cache')) {
        @cache($following_id, 'users', 'delete');
        @cache($follower_id, 'users', 'delete');
    }

    Buzzjuice_FollowLog('delete_complete', array(
        'following_id' => $following_id,
        'follower_id'  => $follower_id,
        'deleted_rows' => $deleted_rows,
        'state'        => 'follow'
    ));

    /*
     * DELETE is idempotent for the requested final state. If there was no
     * existing row, the requested state is still "follow".
     */
    return true;
}

Find pending rows:

```sql name=streams_follow_pending_rows.sql
SELECT
    f.id,
    f.following_id,
    f.follower_id,
    f.active,
    f.is_typing,
    f.notify,
    f.time,
    follower.username AS follower_username,
    following.username AS following_username
FROM Wo_Followers AS f
LEFT JOIN Wo_Users AS follower
    ON follower.user_id = f.follower_id
LEFT JOIN Wo_Users AS following
    ON following.user_id = f.following_id
WHERE f.active = 0
ORDER BY f.id DESC;
```

Find duplicate relationships:

```sql name=streams_follow_duplicates.sql
SELECT
    following_id,
    follower_id,
    COUNT(*) AS relationship_count,
    GROUP_CONCAT(
        CONCAT(id, ':active=', active)
        ORDER BY id DESC
        SEPARATOR ', '
    ) AS rows_found
FROM Wo_Followers
GROUP BY following_id, follower_id
HAVING COUNT(*) > 1
ORDER BY relationship_count DESC;
```
