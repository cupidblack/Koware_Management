function Wo_RegisterFollow(id) {
  var _follow_con = $('[id=Follow-' + id + ']');

  if (!_follow_con.length) {
    console.warn('[BZJ Follow] Follow container not found', id);
    return false;
  }

  var button = _follow_con.find('button').first();

  if (!button.length) {
    console.warn('[BZJ Follow] Follow button not found', id);
    return false;
  }

  if (button.data('bzj-follow-busy') === true) {
    return false;
  }

  button.data('bzj-follow-busy', true);
  button.prop('disabled', true);

  var old_html = _follow_con.html();

  console.log('[BZJ Follow] request start', {
    following_id: id,
    old_class: button.attr('class')
  });

  $.ajax({
    url: Wo_Ajax_Requests_File(),
    type: 'GET',
    dataType: 'json',
    cache: false,
    data: {
      f: 'follow_user',
      following_id: id
    }
  })
  .done(function(data) {
    console.log('[BZJ Follow] server response', data);

    if (!data || typeof data !== 'object') {
      console.error('[BZJ Follow] invalid server response', data);
      _follow_con.html(old_html);
      return;
    }

    if (data.status != 200) {
      console.error('[BZJ Follow] server rejected follow operation', data);
      _follow_con.html(old_html);

      if (data.error) {
        $('#modal-alert').modal('show');
        $('#modal-alert .modal-body').text(data.error);
        Wo_Delay(function() {
          $('#modal-alert').modal('hide');
        }, 3000);
      }
      return;
    }

    var state = data.state;

    if (state === 'following') {
      var following_html =
        '<button type="button" ' +
        'onclick="Wo_RegisterFollow(' + parseInt(id, 10) + ')" ' +
        'class="btn-active btn btn-default btn-sm wo_following_btn" ' +
        'id="wo_useract_btn">' +
        '<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" ' +
        'viewBox="0 0 24 24" fill="none" stroke="currentColor" ' +
        'stroke-width="2" stroke-linecap="round" stroke-linejoin="round">' +
        '<path d="M16 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2"></path>' +
        '<circle cx="8.5" cy="7" r="4"></circle>' +
        '<polyline points="17 11 19 13 23 9"></polyline>' +
        '</svg>' +
        '<span class="button-text"> <?php echo $wo["lang"]["following_btn"];?></span>' +
        '</button>';

      _follow_con.html(following_html);

      if (typeof node_socket_flow !== 'undefined' && node_socket_flow == "1" &&
          typeof socket !== 'undefined') {
        socket.emit("user_notification", {
          to_id: id,
          user_id: _getCookie("user_id"),
          type: "added"
        });
      }

      if (data.can_send == 1 && typeof Wo_SendMessages === 'function') {
        Wo_SendMessages();
      }

      return;
    }

    if (state === 'follow') {
      var follow_html =
        '<button type="button" ' +
        'onclick="Wo_RegisterFollow(' + parseInt(id, 10) + ')" ' +
        'class="btn btn-default btn-sm wo_follow_btn" ' +
        'id="wo_useract_btn">' +
        '<svg width="16" height="16" viewBox="0 0 16 16" fill="none" ' +
        'xmlns="http://www.w3.org/2000/svg">' +
        '<path d="M14.75 8.44434H11.75M13.25 9.94434L13.25 6.94434" ' +
        'class="icon_main" stroke-width="1.5" stroke-linecap="round"/>' +
        '<path d="M1.25 13.7779C1.25 11.9681 2.52299 10.4266 4.25303 10.1413L4.40886 10.1156C5.79419 9.88724 7.20581 9.88724 8.59113 10.1156L8.74697 10.1413C10.477 10.4266 11.75 11.9681 11.75 13.7779C11.75 14.5602 11.1361 15.1943 10.3789 15.1943H2.6211C1.86386 15.1943 1.25 14.5602 1.25 13.7779Z" stroke="currentColor" stroke-width="1.5"/>' +
        '<path d="M9.5625 4.64746C9.5625 6.27843 8.19137 7.60059 6.5 7.60059C4.80863 7.60059 3.4375 6.27843 3.4375 4.64746C3.4375 3.0165 4.80863 1.69434 6.5 1.69434C8.19137 1.69434 9.5625 3.0165 9.5625 4.64746Z" class="icon_main" stroke-width="1.5"/>' +
        '</svg>' +
        '<span class="button-text"> <?php echo $wo["lang"]["follow"];?></span>' +
        '</button>';

      _follow_con.html(follow_html);

      if (typeof node_socket_flow !== 'undefined' && node_socket_flow == "1" &&
          typeof socket !== 'undefined') {
        socket.emit("user_notification", {
          to_id: id,
          user_id: _getCookie("user_id"),
          type: "removed"
        });
      }

      return;
    }

    console.error('[BZJ Follow] unexpected state', data);
    _follow_con.html(old_html);
  })
  .fail(function(xhr, status, error) {
    console.error('[BZJ Follow] AJAX failure', {
      status: status,
      error: error,
      http_status: xhr.status,
      response: xhr.responseText
    });

    _follow_con.html(old_html);
  })
  .always(function() {
    var current_button = _follow_con.find('button').first();
    current_button.data('bzj-follow-busy', false);
    current_button.prop('disabled', false);
  });

  return false;
}
