Below is a refactored web-based version for:

`streams/bzj-orphan-user-cleaner.php`

It is:

- Dry-run by default.
- Protected by an environment-configured access key.
- CSRF-protected.
- Never modifies `Wo_Users`.
- Limited to explicitly approved WoWonder user-reference tables.
- Uses a fresh temporary snapshot of existing users.
- Supports optional user-ID and table filters.
- Shows dry-run results before cleanup.
- Uses one transaction per table.
- Performs post-cleanup verification.
- Does not delete media files, S3 objects, WordPress data, QuickDate data, or Palmier data.
- Does not call `Wo_DeleteUser()`, `Wo_DeletePost()`, `Wo_DeletePage()`, or `Wo_DeleteGroup()` because those functions may perform unrelated filesystem or application-side actions.

Configure an access key in `.env`:

```dotenv
BZJ_ORPHAN_CLEANER_KEY=replace-this-with-a-long-random-secret
```

Then access:

```text
https://buzzjuice.net/streams/bzj-orphan-user-cleaner.php
```

function h(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

function fail(string $message, int $status = 500): never
{
    http_response_code($status);
    exit($message);
}

function quoteIdentifier(string $identifier): string
{
    if (!preg_match('/^[A-Za-z0-9_]+$/', $identifier)) {
        throw new RuntimeException('Unsafe SQL identifier.');
    }

    return '`' . $identifier . '`';
}

function queryOrFail(mysqli $connection, string $sql): mysqli_result|bool
{
    $result = $connection->query($sql);

    if ($result === false) {
        throw new RuntimeException($connection->error);
    }

    return $result;
}

function tableExists(mysqli $connection, string $table): bool
{
    $escaped = $connection->real_escape_string($table);
    $result = queryOrFail(
        $connection,
        "SHOW TABLES LIKE '{$escaped}'"
    );

    return $result instanceof mysqli_result && $result->num_rows > 0;
}

function getColumns(mysqli $connection, string $table): array
{
    $result = queryOrFail(
        $connection,
        'SHOW COLUMNS FROM ' . quoteIdentifier($table)
    );

    $columns = [];

    while ($row = $result->fetch_assoc()) {
        if (!isset($row['Field'])) {
            continue;
        }

        $actualName = (string)$row['Field'];
        $columns[strtolower($actualName)] = $actualName;
    }

    return $columns;
}

function resolveTable(mysqli $connection, string $configuredTable): ?string
{
    $candidates = BZJ_TABLE_ALIASES[$configuredTable] ?? [$configuredTable];

    foreach ($candidates as $candidate) {
        if (tableExists($connection, $candidate)) {
            return $candidate;
        }
    }

    return null;
}

function createExistingUserSnapshot(mysqli $connection): void
{
    queryOrFail(
        $connection,
        'DROP TEMPORARY TABLE IF EXISTS `bzj_existing_user_ids`'
    );

    queryOrFail(
        $connection,
        '
        CREATE TEMPORARY TABLE `bzj_existing_user_ids` (
            `user_id` BIGINT UNSIGNED NOT NULL,
            PRIMARY KEY (`user_id`)
        ) ENGINE=InnoDB
        '
    );

    queryOrFail(
        $connection,
        '
        INSERT INTO `bzj_existing_user_ids` (`user_id`)
        SELECT DISTINCT CAST(`user_id` AS UNSIGNED)
        FROM `Wo_Users`
        WHERE `user_id` IS NOT NULL
          AND CAST(`user_id` AS UNSIGNED) > 0
        '
    );
}

function getExistingUserCount(mysqli $connection): int
{
    $result = queryOrFail(
        $connection,
        'SELECT COUNT(*) AS total FROM `bzj_existing_user_ids`'
    );

    $row = $result->fetch_assoc();

    return (int)($row['total'] ?? 0);
}

function buildOrphanCondition(
    string $column,
    ?int $userId = null
): string {
    $quotedColumn = quoteIdentifier($column);

    $condition = "
        {$quotedColumn} IS NOT NULL
        AND {$quotedColumn} <> ''
        AND CAST({$quotedColumn} AS UNSIGNED) > 0
        AND NOT EXISTS (
            SELECT 1
            FROM `bzj_existing_user_ids` AS existing_users
            WHERE existing_users.`user_id` =
                  CAST({$quotedColumn} AS UNSIGNED)
        )
    ";

    if ($userId !== null) {
        $condition .=
            ' AND CAST(' . $quotedColumn . ' AS UNSIGNED) = ' .
            (int)$userId;
    }

    return $condition;
}

function countRows(
    mysqli $connection,
    string $table,
    string $condition
): int {
    $result = queryOrFail(
        $connection,
        'SELECT COUNT(*) AS total FROM ' .
        quoteIdentifier($table) .
        ' WHERE ' .
        $condition
    );

    $row = $result->fetch_assoc();

    return (int)($row['total'] ?? 0);
}

function sampleOrphanIds(
    mysqli $connection,
    string $table,
    string $column,
    string $condition,
    int $limit
): array {
    $limit = max(
        1,
        min($limit, BZJ_MAX_SAMPLE_LIMIT)
    );

    $result = queryOrFail(
        $connection,
        'SELECT DISTINCT ' .
        quoteIdentifier($column) .
        ' AS orphan_user_id FROM ' .
        quoteIdentifier($table) .
        ' WHERE ' .
        $condition .
        ' ORDER BY ' .
        quoteIdentifier($column) .
        ' LIMIT ' .
        $limit
    );

    $ids = [];

    while ($row = $result->fetch_assoc()) {
        $ids[] = (string)$row['orphan_user_id'];
    }

    return $ids;
}

function buildConfirmedFinding(
    mysqli $connection,
    string $table,
    array $configuredColumns,
    ?int $userId,
    int $sampleLimit
): ?array {
    if (
        $table === BZJ_USERS_TABLE ||
        !tableExists($connection, $table)
    ) {
        return null;
    }

    $actualColumns = getColumns($connection, $table);
    $conditions = [];
    $columnFindings = [];

    foreach ($configuredColumns as $configuredColumn) {
        $key = strtolower($configuredColumn);

        if (!isset($actualColumns[$key])) {
            continue;
        }

        $actualColumn = $actualColumns[$key];
        $condition = buildOrphanCondition($actualColumn, $userId);
        $count = countRows($connection, $table, $condition);

        if ($count < 1) {
            continue;
        }

        $columnFindings[] = [
            'column' => $actualColumn,
            'count' => $count,
            'sample_user_ids' => sampleOrphanIds(
                $connection,
                $table,
                $actualColumn,
                $condition,
                $sampleLimit
            ),
        ];

        $conditions[] = '(' . $condition . ')';
    }

    if (!$conditions) {
        return null;
    }

    $combinedCondition = implode(' OR ', $conditions);

    return [
        'table' => $table,
        'columns' => $columnFindings,
        'condition' => $combinedCondition,
        'orphan_row_count' => countRows(
            $connection,
            $table,
            $combinedCondition
        ),
    ];
}

function discoverAdvisoryReferences(
    mysqli $connection,
    array $confirmedTables,
    ?int $userId,
    int $sampleLimit
): array {
    $results = [];
    $tableResult = queryOrFail($connection, 'SHOW TABLES');

    while ($tableRow = $tableResult->fetch_row()) {
        $table = (string)($tableRow[0] ?? '');

        if (
            $table === '' ||
            $table === BZJ_USERS_TABLE ||
            isset($confirmedTables[$table])
        ) {
            continue;
        }

        $columns = getColumns($connection, $table);

        foreach (BZJ_DISCOVERY_COLUMNS as $candidateColumn) {
            $key = strtolower($candidateColumn);

            if (!isset($columns[$key])) {
                continue;
            }

            $actualColumn = $columns[$key];
            $condition = buildOrphanCondition($actualColumn, $userId);
            $count = countRows($connection, $table, $condition);

            if ($count < 1) {
                continue;
            }

            $results[] = [
                'table' => $table,
                'column' => $actualColumn,
                'count' => $count,
                'sample_user_ids' => sampleOrphanIds(
                    $connection,
                    $table,
                    $actualColumn,
                    $condition,
                    $sampleLimit
                ),
                'automatic_cleanup' => false,
                'reason' => 'Schema-discovered reference; manual review required.',
            ];
        }
    }

    return $results;
}

function generateCsrfToken(): string
{
    if (
        empty($_SESSION[BZJ_CSRF_SESSION_KEY]) ||
        !is_string($_SESSION[BZJ_CSRF_SESSION_KEY])
    ) {
        $_SESSION[BZJ_CSRF_SESSION_KEY] = bin2hex(random_bytes(32));
    }

    return $_SESSION[BZJ_CSRF_SESSION_KEY];
}

function checkCsrfToken(string $token): bool
{
    return isset($_SESSION[BZJ_CSRF_SESSION_KEY]) &&
        is_string($_SESSION[BZJ_CSRF_SESSION_KEY]) &&
        hash_equals($_SESSION[BZJ_CSRF_SESSION_KEY], $token);
}

function cleanerAccessKey(): string
{
    $key = getenv('BZJ_ORPHAN_CLEANER_KEY');

    return is_string($key) ? trim($key) : '';
}

function isAuthenticated(): bool
{
    if (
        !isset($_SESSION['bzj_orphan_cleaner_authenticated']) ||
        $_SESSION['bzj_orphan_cleaner_authenticated'] !== true
    ) {
        return false;
    }

    return true;
}

function requireAuthenticated(): void
{
    if (!isAuthenticated()) {
        fail('Authentication required.', 403);
    }
}

function authenticate(string $providedKey): bool
{
    $configuredKey = cleanerAccessKey();

    if ($configuredKey === '' || $providedKey === '') {
        return false;
    }

    if (!hash_equals($configuredKey, $providedKey)) {
        return false;
    }

    $_SESSION['bzj_orphan_cleaner_authenticated'] = true;

    return true;
}

function parsePositiveInteger(
    string $value,
    int $default,
    int $maximum
): int {
    if ($value === '' || !ctype_digit($value)) {
        return $default;
    }

    return max(1, min((int)$value, $maximum));
}

function scanDatabase(
    mysqli $connection,
    ?int $userId,
    ?string $tableFilter,
    int $sampleLimit,
    bool $includeDiscovery
): array {
    if (!tableExists($connection, BZJ_USERS_TABLE)) {
        throw new RuntimeException(
            'The authoritative Wo_Users table does not exist.'
        );
    }

    createExistingUserSnapshot($connection);

    $resolvedReferences = [];

    foreach (BZJ_CONFIRMED_REFERENCES as $configuredTable => $columns) {
        $resolvedTable = resolveTable($connection, $configuredTable);

        if ($resolvedTable === null) {
            continue;
        }

        if (
            $tableFilter !== null &&
            $tableFilter !== $configuredTable &&
            $tableFilter !== $resolvedTable
        ) {
            continue;
        }

        $resolvedReferences[$resolvedTable] = $columns;
    }

    $findings = [];

    foreach ($resolvedReferences as $table => $columns) {
        $finding = buildConfirmedFinding(
            $connection,
            $table,
            $columns,
            $userId,
            $sampleLimit
        );

        if ($finding !== null) {
            $findings[] = $finding;
        }
    }

    $advisory = [];

    if ($includeDiscovery && $tableFilter === null) {
        $advisory = discoverAdvisoryReferences(
            $connection,
            array_fill_keys(array_keys($resolvedReferences), true),
            $userId,
            $sampleLimit
        );
    }

    return [
        'existing_user_count' => getExistingUserCount($connection),
        'confirmed_findings' => $findings,
        'advisory_findings' => $advisory,
    ];
}

function executeCleanup(
    mysqli $connection,
    array $findings,
    ?int $userId
): array {
    $deleted = [];

    /*
     * Refresh the authoritative user snapshot immediately before deletion.
     * A user may have been created after the dry run.
     */
    createExistingUserSnapshot($connection);

    foreach ($findings as $finding) {
        $table = (string)$finding['table'];

        if (
            $table === BZJ_USERS_TABLE ||
            !tableExists($connection, $table)
        ) {
            continue;
        }

        $actualColumns = getColumns($connection, $table);
        $conditions = [];

        foreach ($finding['columns'] as $columnFinding) {
            $configuredColumn = (string)$columnFinding['column'];
            $key = strtolower($configuredColumn);

            if (!isset($actualColumns[$key])) {
                continue;
            }

            $conditions[] = '(' .
                buildOrphanCondition(
                    $actualColumns[$key],
                    $userId
                ) .
                ')';
        }

        if (!$conditions) {
            continue;
        }

        $condition = implode(' OR ', $conditions);

        if (!$connection->begin_transaction()) {
            throw new RuntimeException(
                "Unable to begin transaction for {$table}."
            );
        }

        try {
            $sql = 'DELETE FROM ' .
                quoteIdentifier($table) .
                ' WHERE ' .
                $condition;

            queryOrFail($connection, $sql);

            $affectedRows = $connection->affected_rows;

            if (!$connection->commit()) {
                throw new RuntimeException(
                    "Unable to commit transaction for {$table}."
                );
            }

            $deleted[] = [
                'table' => $table,
                'deleted_rows' => $affectedRows,
                'status' => 'committed',
            ];
        } catch (Throwable $exception) {
            $connection->rollback();

            $deleted[] = [
                'table' => $table,
                'deleted_rows' => 0,
                'status' => 'rolled_back',
                'error' => $exception->getMessage(),
            ];

            throw $exception;
        }
    }

    return $deleted;
}

function verifyCleanup(
    mysqli $connection,
    array $findings,
    ?int $userId
): array {
    createExistingUserSnapshot($connection);

    $remaining = [];

    foreach ($findings as $finding) {
        $table = (string)$finding['table'];

        if (!tableExists($connection, $table)) {
            continue;
        }

        $actualColumns = getColumns($connection, $table);
        $conditions = [];

        foreach ($finding['columns'] as $columnFinding) {
            $column = (string)$columnFinding['column'];
            $key = strtolower($column);

            if (!isset($actualColumns[$key])) {
                continue;
            }

            $conditions[] = '(' .
                buildOrphanCondition(
                    $actualColumns[$key],
                    $userId
                ) .
                ')';
        }

        if (!$conditions) {
            continue;
        }

        $count = countRows(
            $connection,
            $table,
            implode(' OR ', $conditions)
        );

        if ($count > 0) {
            $remaining[] = [
                'table' => $table,
                'remaining_rows' => $count,
            ];
        }
    }

    return $remaining;
}

function renderFindingRows(array $findings): string
{
    if (!$findings) {
        return '<p class="success">No confirmed orphan rows found.</p>';
    }

    $html = '<table>';
    $html .= '<thead><tr>';
    $html .= '<th>Table</th>';
    $html .= '<th>Orphan rows</th>';
    $html .= '<th>Columns and sample IDs</th>';
    $html .= '</tr></thead><tbody>';

    foreach ($findings as $finding) {
        $html .= '<tr>';
        $html .= '<td><code>' . h((string)$finding['table']) . '</code></td>';
        $html .= '<td>' . h((string)$finding['orphan_row_count']) . '</td>';
        $html .= '<td>';

        foreach ($finding['columns'] as $column) {
            $samples = $column['sample_user_ids'] ?? [];

            $html .= '<div class="column-finding">';
            $html .= '<strong>' . h((string)$column['column']) . '</strong>: ';
            $html .= h((string)$column['count']) . ' row(s)';

            if ($samples) {
                $html .= '<br><small>Sample IDs: ' .
                    h(implode(', ', $samples)) .
                    '</small>';
            }

            $html .= '</div>';
        }

        $html .= '</td>';
        $html .= '</tr>';
    }

    $html .= '</tbody></table>';

    return $html;
}

    $html .= '<th>Column</th>';
    $html .= '<th>Rows</th>';
    $html .= '<th>Sample IDs</th>';
    $html .= '</tr></thead><tbody>';

    foreach ($findings as $finding) {
        $html .= '<tr>';

        $html .= '<td>' . h((string)$finding['count']) . '</td>';

            throw new RuntimeException('Invalid security token.');
        }

        if (!authenticate((string)($_POST['access_key'] ?? ''))) {
            throw new RuntimeException('Invalid access key.');
        }

        $message = 'Authentication successful.';
    }

    if (
        $_SERVER['REQUEST_METHOD'] === 'POST' &&
        isset($_POST['action']) &&
        $_POST['action'] === 'logout'
    ) {
        if (!checkCsrfToken((string)($_POST['csrf_token'] ?? ''))) {
            throw new RuntimeException('Invalid security token.');
        }

        unset($_SESSION['bzj_orphan_cleaner_authenticated']);
        $message = 'Logged out.';
    }

    if (
        $_SERVER['REQUEST_METHOD'] === 'POST' &&
        isset($_POST['action']) &&
        in_array(
            $_POST['action'],
            ['dry_run', 'clean'],
            true
        )
    ) {
        requireAuthenticated();

        if (!checkCsrfToken((string)($_POST['csrf_token'] ?? ''))) {
            throw new RuntimeException('Invalid security token.');
        }

        $userId = null;
        $userIdInput = trim((string)($_POST['user_id'] ?? ''));

        if ($userIdInput !== '') {
            if (
                !ctype_digit($userIdInput) ||
                (int)$userIdInput < 1
            ) {
                throw new RuntimeException(
                    'User ID must be a positive integer.'
                );
            }

            $userId = (int)$userIdInput;
        }

        $tableFilter = trim((string)($_POST['table'] ?? ''));

        if (
            $tableFilter !== '' &&
            !preg_match('/^[A-Za-z0-9_]+$/', $tableFilter)
        ) {
            throw new RuntimeException('Invalid table name.');
        }

        $tableFilter = $tableFilter === ''
            ? null
            : $tableFilter;

        $sampleLimit = parsePositiveInteger(
            trim((string)($_POST['sample_limit'] ?? '')),
            BZJ_DEFAULT_SAMPLE_LIMIT,
            BZJ_MAX_SAMPLE_LIMIT
        );

        $connection = get_wowonder_db();

        if (
            !$connection instanceof mysqli ||
            $connection->connect_errno
        ) {
            throw new RuntimeException(
                'Unable to connect to the WoWonder database.'
            );
        }

        $connection->set_charset('utf8mb4');

        $scan = scanDatabase(
            $connection,
            $userId,
            $tableFilter,
            $sampleLimit,
            true
        );

        if ($_POST['action'] === 'clean') {
            $confirmed = (string)($_POST['confirm_clean'] ?? '');

            if ($confirmed !== 'CLEAN') {
                throw new RuntimeException(
                    'Type CLEAN in the confirmation field before proceeding.'
                );
            }

            if (!$scan['confirmed_findings']) {
                $message = 'No confirmed orphan rows were found.';
            } else {
                $cleanupResults = executeCleanup(
                    $connection,
                    $scan['confirmed_findings'],
                    $userId
                );

                $remaining = verifyCleanup(
                    $connection,
                    $scan['confirmed_findings'],
                    $userId
                );

                if ($remaining) {
                    $message =
                        'Cleanup completed, but some confirmed orphan rows remain.';
                } else {
                    $message =
                        'Cleanup completed successfully. No confirmed orphan rows remain.';
                }

                $scan['cleanup_results'] = $cleanupResults;
                $scan['verification'] = $remaining;
            }
        } else {
            $message = 'Dry run completed. No database rows were changed.';
        }
    }
} catch (Throwable $exception) {
    $error = $exception->getMessage();

Recommended operating procedure:

1. Add `BZJ_ORPHAN_CLEANER_KEY` to the server environment.
2. Create a full database backup.
3. Open the cleaner page.
4. Authenticate.
5. Run a dry run without filters.
6. Review the confirmed and advisory results.
7. Initially clean one table at a time, for example `Wo_Followers`.
8. Re-run the dry run after each cleanup.
9. Do not automatically delete advisory schema-discovered results without confirming their meaning in the application schema.
10. Remove or restrict access to this file after the cleanup is complete.
