Use the following standalone CLI utility. It is **dry-run by default**, never deletes from `Wo_Users`, does not remove files/S3 objects, and only automatically purges tables explicitly mapped from `Wo_DeleteUser()`.

Save it outside the web-accessible directory if possible, for example:

```text
streams/tools/bzj-orphan-user-cleaner.php
```

Run a dry run first:

```bash
php streams/tools/bzj-orphan-user-cleaner.php
```

Write a JSON report:

```bash
php streams/tools/bzj-orphan-user-cleaner.php \
  --report=/secure/path/orphan-users-report.json
```

Execute only after reviewing the report:

```bash
php streams/tools/bzj-orphan-user-cleaner.php --execute --yes
```

Limit the operation to one user ID or table:

```bash
php streams/tools/bzj-orphan-user-cleaner.php --user-id=123
php streams/tools/bzj-orphan-user-cleaner.php --table=Wo_Followers --execute --yes
```

function quoteIdentifier(string $identifier): string
{
    if (!preg_match('/^[A-Za-z0-9_]+$/', $identifier)) {
        throw new RuntimeException("Unsafe SQL identifier: {$identifier}");
    }

    return '`' . $identifier . '`';
}

function queryOrFail(mysqli $conn, string $sql): mysqli_result|bool
{
    $result = $conn->query($sql);

    if ($result === false) {
        throw new RuntimeException($conn->error . "\nSQL: " . $sql);
    }

    return $result;
}

function tableExists(mysqli $conn, string $table): bool
{
    $escaped = $conn->real_escape_string($table);
    $result = queryOrFail($conn, "SHOW TABLES LIKE '{$escaped}'");

    return $result instanceof mysqli_result && $result->num_rows > 0;
}

function getTables(mysqli $conn): array
{
    $tables = [];
    $result = queryOrFail($conn, "SHOW TABLES");

    while ($row = $result->fetch_row()) {
        if (isset($row[0])) {
            $tables[] = (string)$row[0];
        }
    }

    return $tables;
}

function getColumns(mysqli $conn, string $table): array
{
    $result = queryOrFail(
        $conn,
        'SHOW COLUMNS FROM ' . quoteIdentifier($table)
    );

    $columns = [];

    while ($row = $result->fetch_assoc()) {
        if (!isset($row['Field'])) {
            continue;
        }

        $name = (string)$row['Field'];
        $columns[strtolower($name)] = $name;
    }

    return $columns;
}

function createExistingUserSet(mysqli $conn): void
{
    queryOrFail(
        $conn,
        'DROP TEMPORARY TABLE IF EXISTS `bzj_existing_user_ids`'
    );

    queryOrFail(
        $conn,
        '
        CREATE TEMPORARY TABLE `bzj_existing_user_ids` (
            `user_id` BIGINT UNSIGNED NOT NULL,
            PRIMARY KEY (`user_id`)
        ) ENGINE=InnoDB
        '
    );

    queryOrFail(
        $conn,
        '
        INSERT INTO `bzj_existing_user_ids` (`user_id`)
        SELECT DISTINCT CAST(`user_id` AS UNSIGNED)
        FROM `Wo_Users`
        WHERE `user_id` IS NOT NULL
          AND CAST(`user_id` AS UNSIGNED) > 0
        '
    );
}

function getExistingUserCount(mysqli $conn): int
{
    $result = queryOrFail(
        $conn,
        'SELECT COUNT(*) AS total FROM `bzj_existing_user_ids`'
    );

    $row = $result->fetch_assoc();

    return (int)($row['total'] ?? 0);
}

function orphanCondition(
    string $column,
    ?int $filterUserId = null
): string {
    $quotedColumn = quoteIdentifier($column);

    $condition = "
        {$quotedColumn} IS NOT NULL
        AND {$quotedColumn} <> ''
        AND CAST({$quotedColumn} AS UNSIGNED) > 0
        AND NOT EXISTS (
            SELECT 1
            FROM `bzj_existing_user_ids` AS existing_users
            WHERE existing_users.`user_id`
                = CAST({$quotedColumn} AS UNSIGNED)
        )
    ";

    if ($filterUserId !== null) {
        $condition .=
            ' AND CAST(' . $quotedColumn . ' AS UNSIGNED) = '
            . (int)$filterUserId;
    }

    return $condition;
}

function getColumnCount(
    mysqli $conn,
    string $table,
    string $condition
): int {
    $result = queryOrFail(
        $conn,
        'SELECT COUNT(*) AS total FROM '
        . quoteIdentifier($table)
        . ' WHERE '
        . $condition
    );

    $row = $result->fetch_assoc();

    return (int)($row['total'] ?? 0);
}

function getDistinctSampleIds(
    mysqli $conn,
    string $table,
    string $column,
    string $condition,
    int $limit
): array {
    $result = queryOrFail(
        $conn,
        'SELECT DISTINCT '
        . quoteIdentifier($column)
        . ' AS orphan_user_id FROM '
        . quoteIdentifier($table)
        . ' WHERE '
        . $condition
        . ' ORDER BY '
        . quoteIdentifier($column)
        . ' LIMIT '
        . $limit
    );

    $ids = [];

    while ($row = $result->fetch_assoc()) {
        $ids[] = (string)$row['orphan_user_id'];
    }

    return $ids;
}

/**
 * Build one condition per table.
 *
 * Combining all reference columns into one DELETE condition ensures that a
 * row referenced by multiple orphan columns is deleted only once.
 */
function buildTableFindings(
    mysqli $conn,
    string $table,
    array $configuredColumns,
    ?int $filterUserId,
    int $sampleLimit
): ?array {
    if ($table === BZJ_USERS_TABLE || !tableExists($conn, $table)) {
        return null;
    }

    $actualColumns = getColumns($conn, $table);
    $columns = [];

    foreach ($configuredColumns as $configuredColumn) {
        $key = strtolower($configuredColumn);

        if (isset($actualColumns[$key])) {
            $columns[] = $actualColumns[$key];
        }
    }

    if (!$columns) {
        return null;
    }

    $conditions = [];
    $columnFindings = [];

    foreach ($columns as $column) {
        $condition = orphanCondition($column, $filterUserId);
        $count = getColumnCount($conn, $table, $condition);

        if ($count === 0) {
            continue;
        }

        $columnFindings[] = [
            'column' => $column,
            'count' => $count,
            'sample_user_ids' => getDistinctSampleIds(
                $conn,
                $table,
                $column,
                $condition,
                $sampleLimit
            ),
        ];

        $conditions[] = '(' . $condition . ')';
    }

    if (!$conditions) {
        return null;
    }

    $combinedCondition = implode(' OR ', $conditions);
    $rowCount = getColumnCount($conn, $table, $combinedCondition);

    return [
        'table' => $table,
        'columns' => $columnFindings,
        'condition' => $combinedCondition,
        'orphan_row_count' => $rowCount,
    ];
}

function discoverPossibleReferences(
    mysqli $conn,
    array $knownTables,
    array $candidateColumns,
    ?int $filterUserId,
    int $sampleLimit
): array {
    $results = [];

    foreach ($knownTables as $table) {
        if (
            $table === BZJ_USERS_TABLE ||
            isset($knownTables[$table]) ||
            !tableExists($conn, $table)
        ) {
            continue;
        }

        $actualColumns = getColumns($conn, $table);

        foreach ($candidateColumns as $candidateColumn) {
            $key = strtolower($candidateColumn);

            if (!isset($actualColumns[$key])) {
                continue;
            }

            $column = $actualColumns[$key];
            $condition = orphanCondition($column, $filterUserId);
            $count = getColumnCount($conn, $table, $condition);

            if ($count === 0) {
                continue;
            }

            $results[] = [
                'table' => $table,
                'column' => $column,
                'count' => $count,
                'sample_user_ids' => getDistinctSampleIds(
                    $conn,
                    $table,
                    $column,
                    $condition,
                    $sampleLimit
                ),
                'automatic_cleanup' => false,
                'reason' => 'Schema-discovered reference; not in confirmed map',
            ];
        }
    }

    return $results;
}

function writeReport(string $path, array $report): void
{
    $directory = dirname($path);

    if (!is_dir($directory)) {
        if (!mkdir($directory, 0750, true) && !is_dir($directory)) {
            throw new RuntimeException(
                "Unable to create report directory: {$directory}"
            );
        }
    }

    $json = json_encode(
        $report,
        JSON_PRETTY_PRINT |
        JSON_UNESCAPED_SLASHES |
        JSON_UNESCAPED_UNICODE
    );

    if ($json === false || file_put_contents($path, $json . PHP_EOL) === false) {
        throw new RuntimeException("Unable to write report: {$path}");
    }
}

if (!tableExists($conn, BZJ_USERS_TABLE)) {
    fwrite(
        STDERR,
        'Required table `' . BZJ_USERS_TABLE . '` does not exist.' . PHP_EOL
    );
    exit(1);
}

if (isset($options['report'])) {
    $reportPath = (string)$options['report'];

    if ($reportPath === '' || $reportPath[0] !== '/') {
        fwrite(STDERR, "--report must be an absolute path.\n");
        exit(1);
    }
} else {
    $reportPath =
        dirname(__DIR__, 2)
        . '/data/bzj-orphan-user-report-'
        . date('Ymd-His')
        . '.json';
}

try {
    createExistingUserSet($conn);

    $existingUserCount = getExistingUserCount($conn);

    $tables = getTables($conn);
    $tableLookup = array_fill_keys($tables, true);

    $findings = [];

    foreach ($confirmedReferences as $table => $columns) {
        if ($filterTable !== null && $filterTable !== $table) {
            continue;
        }

        $finding = buildTableFindings(
            $conn,
            $table,
            $columns,
            $filterUserId,
            $limit
        );

        if ($finding !== null) {
            $findings[] = $finding;
        }
    }

    $discovered = [];

    if (!$noDiscovery && !$filterTable) {
        $discovered = discoverPossibleReferences(
            $conn,
            $tableLookup,
            $discoveryColumns,
            $filterUserId,
            $limit
        );
    }

    $report = [
        'tool' => 'Buzzjuice WoWonder Orphan User Cleaner',
        'generated_at_utc' => gmdate('c'),
        'mode' => $execute ? 'execute' : 'dry-run',
        'database' => defined('WOWONDER_DB_NAME')
            ? WOWONDER_DB_NAME
            : null,
        'users_table' => BZJ_USERS_TABLE,
        'existing_user_count' => $existingUserCount,
        'filter_user_id' => $filterUserId,
        'filter_table' => $filterTable,
        'confirmed_findings' => $findings,
        'discovered_findings' => $discovered,
        'deleted_rows' => [],
        'verification' => null,
    ];

    if ($discoverOnly) {
        $report['mode'] = 'discover-only';
    }

    writeReport($reportPath, $report);

    echo $execute
        ? "EXECUTION MODE: confirmed orphan rows will be deleted.\n"
        : "DRY-RUN MODE: no database rows will be modified.\n";

    echo "Existing users: {$existingUserCount}\n";
    echo "Report: {$reportPath}\n\n";

    if (!$findings && !$discovered) {
        echo "No orphan user references were found.\n";
        exit(0);
    }

    if ($findings) {
        echo "Confirmed orphan references:\n";

        foreach ($findings as $finding) {
            echo sprintf(
                "  %s: %d orphan row(s)\n",
                $finding['table'],
                $finding['orphan_row_count']
            );

            foreach ($finding['columns'] as $columnFinding) {
                echo sprintf(
                    "    - %s: %d reference(s); sample IDs: %s\n",
                    $columnFinding['column'],
                    $columnFinding['count'],
                    $columnFinding['sample_user_ids']
                        ? implode(', ', $columnFinding['sample_user_ids'])
                        : '(none)'
                );
            }
        }
    }

    if ($discovered) {
        echo "\nPossible schema-discovered references:\n";

        foreach ($discovered as $finding) {
            echo sprintf(
                "  %s.%s: %d row(s); sample IDs: %s\n",
                $finding['table'],
                $finding['column'],
                $finding['count'],
                $finding['sample_user_ids']
                    ? implode(', ', $finding['sample_user_ids'])
                    : '(none)'
            );
        }

        echo "\nThese discovered references are advisory only and will not be deleted.\n";
    }

    if (!$execute || $discoverOnly) {
        echo "\nDry run complete. No rows were changed.\n";
        echo "Review the JSON report before using --execute --yes.\n";
        exit(0);
    }

    if (!$findings) {
        echo "\nThere are no confirmed findings eligible for deletion.\n";
        exit(0);
    }

    $deletedRows = [];

    foreach ($findings as $finding) {
        $table = $finding['table'];
        $condition = $finding['condition'];

        echo "\nPurging {$table}...\n";

        if (!$conn->begin_transaction()) {
            throw new RuntimeException(
                "Unable to start transaction for {$table}: {$conn->error}"
            );
        }

        try {
            $sql =
                'DELETE FROM '
                . quoteIdentifier($table)
                . ' WHERE '
                . $condition;

            queryOrFail($conn, $sql);

            $affectedRows = $conn->affected_rows;

            if (!$conn->commit()) {
                throw new RuntimeException(
                    "Unable to commit transaction for {$table}: {$conn->error}"
                );
            }

            $deletedRows[] = [
                'table' => $table,
                'deleted_rows' => $affectedRows,
                'status' => 'committed',
            ];

            echo "Deleted {$affectedRows} row(s).\n";
        } catch (Throwable $exception) {
            $conn->rollback();

            $deletedRows[] = [
                'table' => $table,
                'deleted_rows' => 0,
                'status' => 'rolled_back',
                'error' => $exception->getMessage(),
            ];

            throw $exception;
        }
    }

    /*
     * Re-audit after deletion. This confirms whether confirmed orphan rows
     * remain. The temporary existing-user table remains available because the
     * connection has not been closed.
     */
    $remaining = [];

    foreach ($findings as $finding) {
        $table = $finding['table'];

        $remainingCount = getColumnCount(
            $conn,
            $table,
            $finding['condition']
        );

        if ($remainingCount > 0) {
            $remaining[] = [
                'table' => $table,
                'remaining_rows' => $remainingCount,
            ];
        }
    }

    $report['deleted_rows'] = $deletedRows;
    $report['verification'] = [
        'completed_at_utc' => gmdate('c'),
        'remaining_confirmed_orphan_rows' => $remaining,
        'status' => $remaining ? 'remaining_rows_found' : 'clean',
    ];

    writeReport($reportPath, $report);

    echo "\nCleanup completed.\n";
    echo "Updated report: {$reportPath}\n";

    if ($remaining) {
        echo "Warning: some confirmed orphan rows remain:\n";

        foreach ($remaining as $item) {
            echo sprintf(
                "  %s: %d row(s)\n",
                $item['table'],
                $item['remaining_rows']
            );
        }

        exit(2);
    }

    echo "Post-cleanup verification: no confirmed orphan rows remain.\n";
    exit(0);
} catch (Throwable $exception) {
    fwrite(
        STDERR,
        "\nCleanup failed:\n"
        . $exception->getMessage()
        . PHP_EOL
    );

    exit(1);
}
