Below is a complete standalone implementation for:

`buzzjuice.net/shared/bzj-wwqd-orphan-user-cleaner.php`

It is intentionally **not pushed** to the repository.

Key corrections included:

- Uses the configured `BZJ_ORPHAN_CLEANER_KEY`.
- Uses WordPress nonces instead of a separate PHP-session token, avoiding the previous “Invalid security token” problem.
- Requires a logged-in WordPress administrator.
- Uses a fresh scan immediately before every destructive operation.
- Never deletes from `Wo_Users` or QuickDate `users`.
- Does not delete media or cloud-storage objects.
- Only deletes rows from explicitly approved reference tables.
- Treats schema-discovered tables as advisory only.
- Repairs `wo_user_id` and `qd_user_id` only when username/email matching is unambiguous.
- Reuses the existing `bzj_log()` function when available without redeclaring it.
- Supports table-name aliases and configurable WoWonder prefixes.
- Shows dry-run results, cleanup results, mapping anomalies, and post-operation verification.

````php name=shared/bzj-wwqd-orphan-user-cleaner.php
<?php
/**
 * Buzzjuice WoWonder + QuickDate orphan-user cleaner.
 *
 * Install:
 *   buzzjuice.net/shared/bzj-wwqd-orphan-user-cleaner.php
 *
 * Required:
 *   - WordPress installation one directory above this file.
 *   - shared/db_helpers.php beside this file.
 *   - BZJ_ORPHAN_CLEANER_KEY in .env.
 *   - mysqli.
 *
 * Safety:
 *   - WordPress administrator authentication is required.
 *   - WordPress nonce and BZJ_ORPHAN_CLEANER_KEY are required for POST requests.
 *   - Dry-run is read-only.
 *   - Authoritative user tables are never deleted from.
 *   - Media and cloud objects are never deleted.
 *   - Only explicitly approved reference tables are automatically cleaned.
 *   - Mapping repairs require unique username/email matches.
 */

declare(strict_types=1);

if (PHP_SAPI === 'cli') {
    http_response_code(403);
    exit("This utility is web-only.\n");
}

error_reporting(E_ALL);
ini_set('display_errors', '0');
ini_set('log_errors', '1');

const BZJ_ORPHAN_CLEANER_VERSION = '2026-09-18.3';

const BZJ_SAMPLE_LIMIT = 25;

/*
 * Load WordPress first so authentication, nonce functions, and the actual
 * WordPress database prefix are available before loading database helpers.
 */
$wp_load = dirname(__DIR__) . '/wp-load.php';

if (!is_file($wp_load)) {
    http_response_code(500);
    exit('WordPress bootstrap could not be located.');
}

require_once $wp_load;
require_once __DIR__ . '/db_helpers.php';

if (
    !function_exists('current_user_can') ||
    !function_exists('wp_create_nonce') ||
    !function_exists('wp_verify_nonce')
) {
    http_response_code(500);
    exit('Required WordPress functions are unavailable.');
}

if (!is_user_logged_in() || !current_user_can('manage_options')) {
    http_response_code(403);
    exit('Administrator access is required.');
}

/* -------------------------------------------------------------------------
 * General helpers

function bzj_ou_h($value): string
{
    return htmlspecialchars(
        (string) $value,
        ENT_QUOTES | ENT_SUBSTITUTE,
        'UTF-8'
    );
}

function bzj_ou_fail(string $message): void
{
    throw new RuntimeException($message);
}

function bzj_ou_identifier(string $identifier): string
{
    /*
     * A single optional trailing space is supported because the supplied
     * WoWonder deletion function contains a historical "from_id " column.
     */
    if (!preg_match('/^[A-Za-z0-9_]+ ?$/', $identifier)) {
        bzj_ou_fail('Unsafe SQL identifier: ' . $identifier);
    }

    return '`' . $identifier . '`';
}

function bzj_ou_query(mysqli $db, string $sql)
{
    $result = $db->query($sql);

    if ($result === false) {
        throw new RuntimeException(
            $db->error . ' | SQL: ' . $sql
        );
    }

    return $result;
}

function bzj_ou_table_exists(mysqli $db, string $table): bool
{
    $escaped = $db->real_escape_string($table);
    $result = bzj_ou_query($db, "SHOW TABLES LIKE '{$escaped}'");

    return $result instanceof mysqli_result && $result->num_rows > 0;
}

function bzj_ou_columns(mysqli $db, string $table): array
{
    $result = bzj_ou_query(
        $db,
        'SHOW COLUMNS FROM ' . bzj_ou_identifier($table)
    );

    $columns = [];

    while ($row = $result->fetch_assoc()) {
        if (isset($row['Field'])) {
            $columns[strtolower((string) $row['Field'])] =
                (string) $row['Field'];
        }
    }

    return $columns;
}

function bzj_ou_engine(mysqli $db, string $table): array
{
    $escaped = $db->real_escape_string($table);
    $result = bzj_ou_query($db, "SHOW TABLE STATUS LIKE '{$escaped}'");
    $row = $result instanceof mysqli_result ? $result->fetch_assoc() : [];

    $engine = (string) ($row['Engine'] ?? '');

    return [
        'engine' => $engine,
        'transactional' => strcasecmp($engine, 'InnoDB') === 0,
    ];
}

function bzj_ou_normalize($value): string
{
    $value = trim((string) $value);

    return function_exists('mb_strtolower')
        ? mb_strtolower($value, 'UTF-8')
        : strtolower($value);
}

function bzj_ou_json($value): string
{
    $json = json_encode(
        $value,
        JSON_UNESCAPED_SLASHES |
        JSON_UNESCAPED_UNICODE |
        JSON_INVALID_UTF8_SUBSTITUTE
    );

    return $json === false ? '{}' : $json;
}

function bzj_ou_audit(string $event, array $data = []): void
{
    /*
     * The registration kernel already defines bzj_log($type, $data).
     * Do not redeclare it.
     */
    if (function_exists('bzj_log')) {
        try {
            bzj_log($event, $data);
            return;
        } catch (Throwable $e) {
            error_log('bzj_log failed: ' . $e->getMessage());
        }
    }

    $directory = dirname(__DIR__) . '/data/logs';

    if (!is_dir($directory)) {
        @mkdir($directory, 0750, true);
    }

    $record = [
        'timestamp_utc' => gmdate('c'),
        'event' => $event,
        'admin_user_id' => function_exists('get_current_user_id')
            ? (int) get_current_user_id()
            : 0,
        'data' => $data,
    ];

    @file_put_contents(
        $directory . '/bzj-wwqd-orphan-user-cleaner.log',
        bzj_ou_json($record) . PHP_EOL,
        FILE_APPEND | LOCK_EX
    );
}

function bzj_ou_report(array $data): ?string
{
    $directory = dirname(__DIR__) . '/data/logs';

    if (
        !is_dir($directory) &&
        !@mkdir($directory, 0750, true) &&
        !is_dir($directory)
    ) {
        return null;
    }

    $filename =
        'bzj-wwqd-orphan-report-' .
        gmdate('Ymd-His') .
        '-' .
        bin2hex(random_bytes(3)) .
        '.json';

    $path = $directory . '/' . $filename;

    $json = json_encode(
        $data,
        JSON_PRETTY_PRINT |
        JSON_UNESCAPED_SLASHES |
        JSON_UNESCAPED_UNICODE |
        JSON_INVALID_UTF8_SUBSTITUTE
    );

    if ($json === false || @file_put_contents($path, $json . PHP_EOL, LOCK_EX) === false) {
        return null;
    }

    return $path;
}

/* -------------------------------------------------------------------------
 * Security
 * ---------------------------------------------------------------------- */

function bzj_ou_configured_key(): string
{
    $key = getenv('BZJ_ORPHAN_CLEANER_KEY');

    if (!is_string($key) || trim($key) === '') {
        bzj_ou_fail(
            'BZJ_ORPHAN_CLEANER_KEY is missing from the environment.'
        );
    }

    return trim($key);
}

function bzj_ou_check_post_security(): void
{
    $nonce = isset($_POST['bzj_nonce'])
        ? (string) $_POST['bzj_nonce']
        : '';

    if ($nonce === '' || !wp_verify_nonce($nonce, 'bzj_ou_action')) {
        bzj_ou_fail('Invalid security token. Please reload the page.');
    }

    $postedKey = isset($_POST['bzj_access_key'])
        ? trim((string) $_POST['bzj_access_key'])
        : '';

    if (
        $postedKey === '' ||
        !hash_equals(bzj_ou_configured_key(), $postedKey)
    ) {
        bzj_ou_fail('Invalid cleaner access key.');
    }
}

function bzj_ou_connections(): array
{
    $wp = get_wp_db_conn();
    $wo = get_wowonder_db();
    $qd = get_qd_db_conn();

    foreach ([$wp, $wo, $qd] as $connection) {
        if (!$connection instanceof mysqli || $connection->connect_errno) {
            bzj_ou_fail('One or more database connections could not be opened.');
        }

        $connection->set_charset('utf8mb4');
    }

    return [
        'wp' => $wp,
        'wo' => $wo,
        'qd' => $qd,
    ];
}

function bzj_ou_wp_table(string $name): string
{
    global $table_prefix;

    $prefix = isset($table_prefix) && is_string($table_prefix)
        ? $table_prefix
        : 'wp_';

    if (!preg_match('/^[A-Za-z0-9_]+$/', $prefix)) {
        bzj_ou_fail('Unsafe WordPress table prefix.');
    }

    return $prefix . $name;
}

function bzj_ou_resolve_reference(
    mysqli $db,
    string $configuredTable
): ?string {
    $aliases = [
        'Wo_AppSessions' => ['Wo_AppSessions', 'Wo_AppsSessions'],
        'Wo_AppsSessions' => ['Wo_AppsSessions', 'Wo_AppSessions'],
        'Wo_EventsInterested' => ['Wo_EventsInterested', 'Wo_EventsInt'],
        'Wo_EventsInt' => ['Wo_EventsInt', 'Wo_EventsInterested'],
        'Wo_PagesInvites' => ['Wo_PagesInvites', 'Wo_PagesInvaites'],
        'Wo_PagesInvaites' => ['Wo_PagesInvaites', 'Wo_PagesInvites'],
        'Wo_LiveSubscriptions' => ['Wo_LiveSubscriptions', 'Wo_LiveSub'],
        'Wo_LiveSub' => ['Wo_LiveSub', 'Wo_LiveSubscriptions'],
        'Wo_Email' => ['Wo_Email', 'Wo_Emails'],
        'Wo_Emails' => ['Wo_Emails', 'Wo_Email'],
        'Wo_UserMonetization' => [
            'Wo_UserMonetization',
            'Wo_UserMonetizations',
        ],
        'Wo_UserMonetizations' => [
            'Wo_UserMonetizations',
            'Wo_UserMonetization',
        ],
        'Wo_MonetizationSubscription' => [
            'Wo_MonetizationSubscription',
            'Wo_MonetizationSubscribtion',
            'Wo_MonetizationSubscriptions',
        ],
        'Wo_MonetizationSubscribtion' => [
            'Wo_MonetizationSubscribtion',
            'Wo_MonetizationSubscription',
            'Wo_MonetizationSubscriptions',
        ],
    ];

    $candidates = $aliases[$configuredTable] ?? [$configuredTable];

    foreach ($candidates as $candidate) {
        if (bzj_ou_table_exists($db, $candidate)) {
            return $candidate;
        }
    }

    return null;
}

function bzj_ou_authoritative_config(
    mysqli $db,
    array $platform
): array {
    if (!bzj_ou_table_exists($db, $platform['users_table'])) {
        bzj_ou_fail(
            'Authoritative table does not exist: ' .
            $platform['users_table']
        );
    }

    $columns = bzj_ou_columns($db, $platform['users_table']);

    foreach ([
        $platform['id_column'],
        $platform['username_column'],
        $platform['email_column'],
    ] as $required) {
        if (!isset($columns[strtolower($required)])) {
            bzj_ou_fail(
                'Required column is missing from ' .
                $platform['users_table'] . ': ' . $required
            );
        }
    }

    return [
        'table' => $platform['users_table'],
        'id' => $columns[strtolower($platform['id_column'])],
        'username' => $columns[strtolower($platform['username_column'])],
        'email' => $columns[strtolower($platform['email_column'])],
    ];
}

/* -------------------------------------------------------------------------
 * Orphan scanning
 * ---------------------------------------------------------------------- */

function bzj_ou_orphan_condition(
    string $referenceColumn,
    string $authoritativeTable,
    string $authoritativeId
): string {
    $ref = bzj_ou_identifier($referenceColumn);
    $authTable = bzj_ou_identifier($authoritativeTable);
    $authId = bzj_ou_identifier($authoritativeId);

    return "
        {$ref} IS NOT NULL
        AND TRIM(CAST({$ref} AS CHAR)) REGEXP '^[0-9]+$'
        AND CAST({$ref} AS UNSIGNED) > 0
        AND NOT EXISTS (
            SELECT 1
            FROM {$authTable} AS authoritative_users
            WHERE CAST(authoritative_users.{$authId} AS UNSIGNED)
                = CAST({$ref} AS UNSIGNED)
        )
    ";
}

function bzj_ou_scan_platform(
    mysqli $db,
    string $platformKey,
    int $sampleLimit = BZJ_SAMPLE_LIMIT
): array {
    $platform = bzj_ou_platforms()[$platformKey];
    $authoritative = bzj_ou_authoritative_config($db, $platform);
    $sampleLimit = max(1, min($sampleLimit, 100));

    $findings = [];

    foreach ($platform['references'] as $reference) {
        $configuredTable = $reference[0];
        $configuredColumns = $reference[1];

        $table = bzj_ou_resolve_reference($db, $configuredTable);

        if ($table === null) {
            continue;
        }

        if (strcasecmp($table, $authoritative['table']) === 0) {
            continue;
        }

        $actualColumns = bzj_ou_columns($db, $table);
        $columnFindings = [];
        $deleteConditions = [];

        foreach ($configuredColumns as $configuredColumn) {
            $columnKey = strtolower($configuredColumn);

            if (!isset($actualColumns[$columnKey])) {
                continue;
            }

            $column = $actualColumns[$columnKey];

            $condition = bzj_ou_orphan_condition(
                $column,
                $authoritative['table'],
                $authoritative['id']
            );

            $countResult = bzj_ou_query(
                $db,
                'SELECT COUNT(*) AS total FROM ' .
                bzj_ou_identifier($table) .
                ' WHERE ' . $condition
            );

            $countRow = $countResult->fetch_assoc();
            $count = (int) ($countRow['total'] ?? 0);

            if ($count < 1) {
                continue;
            }

            $sampleResult = bzj_ou_query(
                $db,
                'SELECT DISTINCT ' .
                bzj_ou_identifier($column) .
                ' AS orphan_id FROM ' .
                bzj_ou_identifier($table) .
                ' WHERE ' . $condition .
                ' ORDER BY CAST(' .
                bzj_ou_identifier($column) .
                ' AS UNSIGNED) LIMIT ' .
                $sampleLimit
            );

            $sampleIds = [];

            while ($sample = $sampleResult->fetch_assoc()) {
                $sampleIds[] = (string) ($sample['orphan_id'] ?? '');
            }

            $columnFindings[] = [
                'column' => $column,
                'orphan_references' => $count,
                'sample_ids' => $sampleIds,
            ];

            $deleteConditions[] = '(' . $condition . ')';
        }

        if (!$deleteConditions) {
            continue;
        }

        $rowCountResult = bzj_ou_query(
            $db,
            'SELECT COUNT(*) AS total FROM ' .
            bzj_ou_identifier($table) .
            ' WHERE ' .
            implode(' OR ', $deleteConditions)
        );

        $rowCount = $rowCountResult->fetch_assoc();
        $engine = bzj_ou_engine($db, $table);

        $findings[] = [
            'table' => $table,
            'columns' => $columnFindings,
            'delete_conditions' => $deleteConditions,
            'orphan_rows' => (int) ($rowCount['total'] ?? 0),
            'engine' => $engine['engine'],
            'transactional' => $engine['transactional'],
        ];
    }

    return [
        'platform' => $platformKey,
        'label' => $platform['label'],
        'authoritative_table' => $authoritative['table'],
        'authoritative_id_column' => $authoritative['id'],
        'existing_user_count' => bzj_ou_count_users(
            $db,
            $authoritative['table'],
            $authoritative['id']
        ),
        'findings' => $findings,
    ];
}

function bzj_ou_count_users(
    mysqli $db,
    string $table,
    string $idColumn
): int {
    $result = bzj_ou_query(
        $db,
        'SELECT COUNT(*) AS total FROM ' .
        bzj_ou_identifier($table) .
        ' WHERE ' .
        bzj_ou_identifier($idColumn) .
        ' IS NOT NULL'
    );

    $row = $result->fetch_assoc();

    return (int) ($row['total'] ?? 0);
}

function bzj_ou_load_platform_users(
    mysqli $db,
    string $platformKey
): array {
    $platform = bzj_ou_platforms()[$platformKey];
    $authoritative = bzj_ou_authoritative_config($db, $platform);

    $result = bzj_ou_query(
        $db,
        'SELECT ' .
        bzj_ou_identifier($authoritative['id']) .
        ' AS platform_id, ' .
        bzj_ou_identifier($authoritative['username']) .
        ' AS username, ' .
        bzj_ou_identifier($authoritative['email']) .
        ' AS email FROM ' .
        bzj_ou_identifier($authoritative['table'])
    );

    $users = [];

    while ($row = $result->fetch_assoc()) {
        $id = trim((string) ($row['platform_id'] ?? ''));

        if (!ctype_digit($id) || (int) $id < 1) {
            continue;
        }

        $users[(string) ((int) $id)] = [
            'id' => (int) $id,
            'username' => (string) ($row['username'] ?? ''),
            'email' => (string) ($row['email'] ?? ''),
        ];
    }

    return $users;
}

function bzj_ou_index_platform_users(array $users): array
{
    $index = [
        'username' => [],
        'email' => [],
    ];

    foreach ($users as $user) {
        $username = bzj_ou_normalize($user['username']);
        $email = bzj_ou_normalize($user['email']);

        if ($username !== '') {
            $index['username'][$username][] = $user;
        }

        if ($email !== '') {
            $index['email'][$email][] = $user;
        }
    }

    return $index;
}

function bzj_ou_get_meta(
    mysqli $wp,
    string $metaTable,
    int $wpUserId,
    string $metaKey
): string {
    $escapedKey = $wp->real_escape_string($metaKey);

    $result = bzj_ou_query(
        $wp,
        'SELECT meta_value FROM ' .
        bzj_ou_identifier($metaTable) .
        ' WHERE user_id = ' .
        $wpUserId .
        " AND meta_key = '{$escapedKey}'" .
        ' ORDER BY umeta_id ASC LIMIT 1'
    );

    $row = $result->fetch_assoc();

    return trim((string) ($row['meta_value'] ?? ''));
}

function bzj_ou_mapping_scan(
    mysqli $wp,
    mysqli $wo,
    mysqli $qd
): array {
    $usersTable = bzj_ou_wp_table('users');
    $metaTable = bzj_ou_wp_table('usermeta');

    if (
        !bzj_ou_table_exists($wp, $usersTable) ||
        !bzj_ou_table_exists($wp, $metaTable)
    ) {
        bzj_ou_fail('WordPress users/usermeta tables were not found.');
    }

    $platformUsers = [
        'wowonder' => bzj_ou_load_platform_users($wo, 'wowonder'),
        'quickdate' => bzj_ou_load_platform_users($qd, 'quickdate'),
    ];

    $indexes = [
        'wowonder' => bzj_ou_index_platform_users($platformUsers['wowonder']),
        'quickdate' => bzj_ou_index_platform_users($platformUsers['quickdate']),
    ];

    $wpUsers = bzj_ou_query(
        $wp,
        'SELECT ID, user_login, user_email FROM ' .
        bzj_ou_identifier($usersTable) .
        ' ORDER BY ID ASC'
    );

    $findings = [];
    $summary = [
        'wordpress_users' => 0,
        'wowonder_users' => count($platformUsers['wowonder']),
        'quickdate_users' => count($platformUsers['quickdate']),
        'ok' => 0,
        'repairable' => 0,
        'ambiguous' => 0,
        'unmatched' => 0,
    ];

    while ($wpUser = $wpUsers->fetch_assoc()) {
        $summary['wordpress_users']++;

        $wpId = (int) ($wpUser['ID'] ?? 0);
        $wpUsername = bzj_ou_normalize($wpUser['user_login'] ?? '');
        $wpEmail = bzj_ou_normalize($wpUser['user_email'] ?? '');

        foreach (['wowonder', 'quickdate'] as $platformKey) {
            $platform = bzj_ou_platforms()[$platformKey];
            $index = $indexes[$platformKey];
            $users = $platformUsers[$platformKey];

            $emailMatches = $wpEmail !== '' &&
                isset($index['email'][$wpEmail])
                ? $index['email'][$wpEmail]
                : [];

            $usernameMatches = $wpUsername !== '' &&
                isset($index['username'][$wpUsername])
                ? $index['username'][$wpUsername]
                : [];

            $candidateIds = [];

            foreach (array_merge($emailMatches, $usernameMatches) as $match) {
                $candidateIds[(string) $match['id']] = true;
            }

            $candidateIds = array_keys($candidateIds);

            $storedId = bzj_ou_get_meta(
                $wp,
                $metaTable,
                $wpId,
                $platform['meta_key']
            );

            $storedValid =
                ctype_digit($storedId) &&
                (int) $storedId > 0;

            $storedExists =
                $storedValid &&
                isset($users[(string) ((int) $storedId)]);

            $status = 'ok';
            $repairable = false;
            $candidateId = null;

            if (count($candidateIds) === 0) {
                $status = $storedId === ''
                    ? 'no_platform_match'
                    : (
                        $storedExists
                            ? 'stored_id_has_no_identity_match'
                            : 'stored_id_missing_from_platform'
                    );
            } elseif (
                count($candidateIds) > 1 ||
                count($emailMatches) > 1 ||
                count($usernameMatches) > 1
            ) {
                $status = 'ambiguous_identity_match';
            } else {
                $candidateId = (int) $candidateIds[0];

                $emailAgrees =
                    $wpEmail === '' ||
                    (
                        count($emailMatches) === 1 &&
                        (int) $emailMatches[0]['id'] === $candidateId
                    );

                $usernameAgrees =
                    $wpUsername === '' ||
                    (
                        count($usernameMatches) === 1 &&
                        (int) $usernameMatches[0]['id'] === $candidateId
                    );

                if ($emailAgrees && $usernameAgrees) {
                    $repairable =
                        $storedId !== (string) $candidateId;

                    $status = $repairable
                        ? 'repairable_mismatch'
                        : 'ok';
                } else {
                    $status = 'conflicting_identity_match';
                }
            }

            if ($status === 'ok') {
                $summary['ok']++;
                continue;
            }

            if ($repairable) {
                $summary['repairable']++;
            } elseif (
                in_array(
                    $status,
                    [
                        'ambiguous_identity_match',
                        'conflicting_identity_match',
                    ],
                    true
                )
            ) {
                $summary['ambiguous']++;
            } else {
                $summary['unmatched']++;
            }

            $findings[] = [
                'wp_user_id' => $wpId,
                'wp_username' => (string) ($wpUser['user_login'] ?? ''),
                'wp_email' => (string) ($wpUser['user_email'] ?? ''),
                'platform' => $platformKey,
                'meta_key' => $platform['meta_key'],
                'stored_id' => $storedId,
                'stored_id_exists' => $storedExists,
                'candidate_id' => $candidateId,
                'repairable' => $repairable,
                'status' => $status,
                'email_match_count' => count($emailMatches),
                'username_match_count' => count($usernameMatches),
            ];
        }
    }

    return [
        'summary' => $summary,
        'findings' => $findings,
    ];
}

    return [

function bzj_ou_clean_platform(
    mysqli $db,
    string $platformKey,
    array $freshScan
): array {
    $results = [];

    foreach ($freshScan['findings'] as $finding) {
        $table = (string) $finding['table'];

        if (!bzj_ou_table_exists($db, $table)) {
            continue;
        }

        $engine = bzj_ou_engine($db, $table);

        $sql =
            'DELETE FROM ' .
            bzj_ou_identifier($table) .
            ' WHERE ' .
            implode(' OR ', $finding['delete_conditions']);

        $transactionStarted = false;

        try {
            if ($engine['transactional']) {
                if (!$db->begin_transaction()) {
                    throw new RuntimeException(
                        'Could not begin transaction for ' . $table
                    );
                }

                $transactionStarted = true;
            }

            bzj_ou_query($db, $sql);
            $deleted = (int) $db->affected_rows;

            if ($transactionStarted && !$db->commit()) {
                throw new RuntimeException(
                    'Could not commit transaction for ' . $table
                );
            }

            $results[] = [
                'platform' => $platformKey,
                'table' => $table,
                'deleted_rows' => $deleted,
                'status' => 'committed',
                'engine' => $engine['engine'],
                'transactional' => $engine['transactional'],
            ];
        } catch (Throwable $e) {
            if ($transactionStarted) {
                $db->rollback();
            }

            $results[] = [
                'platform' => $platformKey,
                'table' => $table,
                'deleted_rows' => 0,
                'status' => 'failed',
                'engine' => $engine['engine'],
                'transactional' => $engine['transactional'],
                'error' => $e->getMessage(),
            ];

            throw $e;
        }
    }

    return $results;
}

function bzj_ou_repair_mappings(
    mysqli $wp,
    array $mappingScan
): array {
    $metaTable = bzj_ou_wp_table('usermeta');
    $results = [];

    foreach ($mappingScan['findings'] as $finding) {
        if (
            empty($finding['repairable']) ||
            $finding['candidate_id'] === null
        ) {
            continue;
        }

        $wpUserId = (int) $finding['wp_user_id'];
        $metaKey = (string) $finding['meta_key'];
        $newValue = (string) $finding['candidate_id'];

        $escapedKey = $wp->real_escape_string($metaKey);
        $escapedValue = $wp->real_escape_string($newValue);

        $existing = bzj_ou_query(
            $wp,
            'SELECT umeta_id FROM ' .
            bzj_ou_identifier($metaTable) .
            ' WHERE user_id = ' .
            $wpUserId .
            " AND meta_key = '{$escapedKey}'" .
            ' ORDER BY umeta_id ASC LIMIT 1'
        );

        $existingRow = $existing->fetch_assoc();

        try {
            if (!$wp->begin_transaction()) {
                throw new RuntimeException(
                    'Could not begin WordPress metadata transaction.'
                );
            }

            if ($existingRow) {
                bzj_ou_query(
                    $wp,
                    'UPDATE ' .
                    bzj_ou_identifier($metaTable) .
                    " SET meta_value = '{$escapedValue}'" .
                    ' WHERE umeta_id = ' .
                    (int) $existingRow['umeta_id']
                );

                $action = 'updated';
            } else {
                bzj_ou_query(
                    $wp,
                    'INSERT INTO ' .
                    bzj_ou_identifier($metaTable) .
                    ' (user_id, meta_key, meta_value) VALUES (' .
                    $wpUserId .
                    ", '{$escapedKey}', '{$escapedValue}')"
                );

                $action = 'inserted';
            }

            if (!$wp->commit()) {
                throw new RuntimeException(
                    'Could not commit WordPress metadata transaction.'
                );
            }

            $results[] = [
                'platform' => $finding['platform'],
                'wp_user_id' => $wpUserId,
                'meta_key' => $metaKey,
                'old_value' => $finding['stored_id'],
                'new_value' => $newValue,
                'action' => $action,
                'status' => 'committed',
            ];
        } catch (Throwable $e) {
            $wp->rollback();

            $results[] = [
                'platform' => $finding['platform'],
                'wp_user_id' => $wpUserId,
                'meta_key' => $metaKey,
                'old_value' => $finding['stored_id'],
                'new_value' => $newValue,
                'status' => 'failed',
                'error' => $e->getMessage(),
            ];

            throw $e;
        }
    }

    return $results;
}

function bzj_ou_render_orphans(array $scan): string
{
    $html = '';

    foreach (['wowonder', 'quickdate'] as $platformKey) {
        $platform = $scan[$platformKey];

        $html .= '<h3>' .
            bzj_ou_h($platform['label']) .
            '</h3>';

        $html .= '<p>Authoritative users in <code>' .
            bzj_ou_h($platform['authoritative_table']) .
            '</code>: <strong>' .
            (int) $platform['existing_user_count'] .
            '</strong></p>';

        if (!$platform['findings']) {
            $html .= '<p class="success">No confirmed orphan rows found.</p>';
            continue;
        }

        $html .= '<table><thead><tr>' .
            '<th>Table</th>' .
            '<th>Reference columns</th>' .
            '<th>Rows to delete</th>' .
            '<th>Engine</th>' .
            '<th>Sample IDs</th>' .

        foreach ($platform['findings'] as $finding) {
            $html .= '<tr><td><code>' .

            foreach ($finding['columns'] as $column) {
                $html .= '<div><code>' .
                    bzj_ou_h($column['column']) .
                    '</code>: ' .
                    (int) $column['orphan_references'] .
                    '</div>';
            }

            $engine = $finding['engine'] !== ''
                ? $finding['engine']
                : 'unknown';

            $engine .= $finding['transactional']
                ? ' / transactional'
                : ' / non-transactional';

            $html .= '</td><td><strong>' .
                (int) $finding['orphan_rows'] .
                '</strong></td><td>' .
                bzj_ou_h($engine) .
                '</td><td>';

            foreach ($finding['columns'] as $column) {
                $html .= '<div><code>' .
                    bzj_ou_h($column['column']) .
                    '</code>: ' .
                    bzj_ou_h(implode(', ', $column['sample_ids'])) .
                    '</div>';
            }

            $html .= '</td></tr>';
        }

        $html .= '</tbody></table>';
    }

    return $html;
}

function bzj_ou_render_results(array $results): string
{
    if (!$results) {
        return '<p class="muted">No rows required processing.</p>';
    }

    $html = '<table><thead><tr>' .
        '<th>Platform</th>' .
        '<th>Target</th>' .
        '<th>Action</th>' .
        '<th>Status</th>' .
        '<th>Details</th>' .
        '</tr></thead><tbody>';

    foreach ($results as $result) {
        $target = $result['table'] ??
            ('WP user ' . ($result['wp_user_id'] ?? ''));

        if (isset($result['deleted_rows'])) {
            $action = 'Deleted ' .
                (int) $result['deleted_rows'] .
                ' row(s)';
        } else {
            $action = ucfirst($result['action'] ?? 'metadata update');
        }

        $details = $result['error'] ?? '';

        if ($details === '' && isset($result['old_value'])) {
            $details =
                'Old: ' . $result['old_value'] .
                ' → New: ' . $result['new_value'];
        }

        $html .= '<tr>' .
            '<td>' . bzj_ou_h($result['platform'] ?? '') . '</td>' .
            '<td>' . bzj_ou_h($target) . '</td>' .
            '<td>' . bzj_ou_h($action) . '</td>' .
            '<td>' . bzj_ou_h($result['status'] ?? '') . '</td>' .
            '<td>' . bzj_ou_h($details) . '</td>' .
            '</tr>';
    }

    return $html . '</tbody></table>';
}

$action = isset($_POST['bzj_action'])
    ? trim((string) $_POST['bzj_action'])
    : '';

$scan = null;
$operationResults = [];
$message = '';
$error = '';
$reportPath = null;

try {
    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        bzj_ou_check_post_security();

        $connections = bzj_ou_connections();

        if ($action === 'scan') {
            $scan = bzj_ou_full_scan($connections);

            $reportPath = bzj_ou_report([
                'type' => 'dry_run',
                'scan' => $scan,
            ]);

            bzj_ou_audit('orphan_cleaner_dry_run', [
                'scan' => $scan,
                'report' => $reportPath,
            ]);

            $message = 'Dry run completed. No database rows were changed.';
        } elseif ($action === 'clean') {
            $phrase = trim(
                (string) ($_POST['confirmation_phrase'] ?? '')
            );

            if (!hash_equals(BZJ_CLEAN_CONFIRMATION, $phrase)) {
                bzj_ou_fail('The cleanup confirmation phrase is incorrect.');
            }

            /*
             * Never trust the previously displayed scan.
             */
            $fresh = bzj_ou_full_scan($connections);
            $operationResults = [];

            $operationResults = array_merge(
                $operationResults,
                bzj_ou_clean_platform(
                    $connections['wo'],
                    'wowonder',
                    $fresh['wowonder']
                )
            );

            $operationResults = array_merge(
                $operationResults,
                bzj_ou_clean_platform(
                    $connections['qd'],
                    'quickdate',
                    $fresh['quickdate']
                )
            );

            $scan = bzj_ou_full_scan($connections);

            $reportPath = bzj_ou_report([
                'type' => 'orphan_cleanup',
                'before' => $fresh,
                'results' => $operationResults,
                'after' => $scan,
            ]);

            bzj_ou_audit('orphan_cleaner_cleanup', [
                'before' => $fresh,
                'results' => $operationResults,
                'after' => $scan,
                'report' => $reportPath,
            ]);

            $message = 'Orphan cleanup completed and verified.';
        } elseif ($action === 'repair') {
            $phrase = trim(
                (string) ($_POST['confirmation_phrase'] ?? '')
            );

            if (!hash_equals(BZJ_REPAIR_CONFIRMATION, $phrase)) {
                bzj_ou_fail('The mapping-repair confirmation phrase is incorrect.');
            }

            $freshMapping = bzj_ou_mapping_scan(
                $connections['wp'],
                $connections['wo'],
                $connections['qd']
            );

            $operationResults = bzj_ou_repair_mappings(
                $connections['wp'],
                $freshMapping
            );

            $scan = bzj_ou_full_scan($connections);

            $reportPath = bzj_ou_report([
                'type' => 'mapping_repair',
                'before' => $freshMapping,
                'results' => $operationResults,
                'after' => $scan['mapping'],
            ]);

            bzj_ou_audit('orphan_cleaner_mapping_repair', [
                'before' => $freshMapping,
                'results' => $operationResults,
                'after' => $scan['mapping'],
                'report' => $reportPath,
            ]);

            $message = 'WordPress platform-ID repair completed and verified.';
        }
    }
} catch (Throwable $e) {
    $error = $e->getMessage();

    bzj_ou_audit('orphan_cleaner_error', [
        'action' => $action,
        'error' => $error,
    ]);
}
````

### Deployment notes

1. Place the file at:

   `buzzjuice.net/shared/bzj-wwqd-orphan-user-cleaner.php`

2. Confirm `.env` contains:

   ```dotenv
   BZJ_ORPHAN_CLEANER_KEY=your-long-random-secret
   ```

3. Log in to Buzzjuice as a WordPress administrator.

4. Open the cleaner URL.

5. Run the dry run first.

6. Review the orphan rows and mapping anomalies.

7. Enter the exact confirmation phrase before cleaning or repairing.

8. After use, restrict or remove access to the utility, since it is a powerful database-maintenance endpoint.
