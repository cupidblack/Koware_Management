Below is a complete, standalone replacement for:

`shared/bzj-wwqd-orphan-user-cleaner.php`

It is intentionally **not pushed** and does not modify the repository.

Key safeguards included:

- WordPress administrator authentication.
- CSRF protection.
- Dry-run by default.
- Separate orphan cleanup and metadata repair actions.
- Fresh scan immediately before destructive operations.
- Per-table transactions.
- No deletion from `Wo_Users` or QuickDate `users`.
- No media/S3 deletion.
- No automatic repair when username/email matches are ambiguous.
- Database table and column existence checks.
- Audit logging.
- Post-clean verification.
- WordPress metadata is repaired only when the authoritative platform account is uniquely identified.

````php name=shared/bzj-wwqd-orphan-user-cleaner.php
<?php
/**
 * Buzzjuice WoWonder + QuickDate orphan-user cleaner and ID-integrity auditor.
 *
 * Location:
 *   /shared/bzj-wwqd-orphan-user-cleaner.php
 *
 * Required:
 *   - /shared/db_helpers.php
 *   - WordPress installed at the project root
 *   - mysqli
 *
 * Important:
 *   - This script never deletes rows from Wo_Users or QuickDate users.
 *   - This script never deletes media files or S3 objects.
 *   - This script only removes rows from configured reference tables when
 *     their user references no longer exist in the authoritative user table.
 *   - WordPress metadata repairs are restricted to uniquely identifiable users.
 */

declare(strict_types=1);

error_reporting(E_ALL);
ini_set('display_errors', '0');
ini_set('log_errors', '1');

if (PHP_SAPI === 'cli') {
    http_response_code(403);
    exit("This utility is web-admin only.\n");
}

const BZJ_CLEANER_MAX_SAMPLE_LIMIT = 100;

require_once __DIR__ . '/db_helpers.php';

/*
 * Load WordPress so that the utility can require an authenticated
 * administrator session before displaying or processing anything.
 */
$wpLoadCandidates = array(
    dirname(__DIR__) . '/wp-load.php',
    dirname(__DIR__, 2) . '/wp-load.php',
);

$wpLoaded = false;

foreach ($wpLoadCandidates as $wpLoad) {
    if (is_file($wpLoad)) {
        require_once $wpLoad;
        $wpLoaded = true;

    }

}

}

if (session_status() !== PHP_SESSION_ACTIVE) {
    session_start();
}

if (empty($_SESSION[BZJ_CLEANER_CSRF])) {
    $_SESSION[BZJ_CLEANER_CSRF] = bin2hex(random_bytes(32));
}

$csrfToken = (string) $_SESSION[BZJ_CLEANER_CSRF];

function bzj_cleaner_h($value): string
{
    return htmlspecialchars(
        (string) $value,
        ENT_QUOTES | ENT_SUBSTITUTE,
        'UTF-8'
    );
}

function bzj_cleaner_fail(string $message): void
{
    throw new RuntimeException($message);
}

function bzj_cleaner_identifier(string $identifier): string
{
    if (!preg_match('/^[A-Za-z0-9_]+$/', $identifier)) {
        bzj_cleaner_fail('Unsafe SQL identifier: ' . $identifier);
    }

    return '`' . $identifier . '`';
}

function bzj_cleaner_query(mysqli $db, string $sql)
{
    $result = $db->query($sql);

    if ($result === false) {
        throw new RuntimeException(
            $db->error . ' | SQL: ' . $sql
        );
    }

    return $result;
}

function bzj_cleaner_table_exists(mysqli $db, string $table): bool
{
    $table = $db->real_escape_string($table);
    $result = bzj_cleaner_query(
        $db,
        "SHOW TABLES LIKE '{$table}'"
    );

    return $result instanceof mysqli_result && $result->num_rows > 0;
}

function bzj_cleaner_columns(mysqli $db, string $table): array
{
    $result = bzj_cleaner_query(
        $db,
        'SHOW COLUMNS FROM ' . bzj_cleaner_identifier($table)
    );

    $columns = array();

    while ($row = $result->fetch_assoc()) {
        if (isset($row['Field'])) {
            $columns[strtolower((string) $row['Field'])] =
                (string) $row['Field'];
        }
    }

    return $columns;
}

function bzj_cleaner_tables(mysqli $db): array
{
    $result = bzj_cleaner_query($db, 'SHOW TABLES');
    $tables = array();

    while ($row = $result->fetch_row()) {
        if (isset($row[0])) {
            $tables[] = (string) $row[0];
        }
    }

    return $tables;
}

function bzj_cleaner_normalize($value): string
{
    return function_exists('mb_strtolower')
        ? mb_strtolower(trim((string) $value), 'UTF-8')
        : strtolower(trim((string) $value));
}

function bzj_cleaner_log(array $event): void
{
    $directory = dirname(__DIR__) . '/data/logs';

    if (!is_dir($directory)) {
        @mkdir($directory, 0750, true);
    }

    $event['timestamp_utc'] = gmdate('c');
    $event['admin_user_id'] = get_current_user_id();

    $line = json_encode(
        $event,
        JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE
    );

    if ($line !== false) {
        @file_put_contents(
            $directory . '/bzj-wwqd-orphan-user-cleaner.log',
            $line . PHP_EOL,
            FILE_APPEND | LOCK_EX
        );
    }
}

function bzj_cleaner_wp_table(string $table): string
{
    $prefix = defined('WP_TABLE_PREFIX')
        ? WP_TABLE_PREFIX
        : 'wp_';

    return $prefix . $table;
}

function bzj_cleaner_existing_ids(
    mysqli $db,
    string $usersTable,
    string $idColumn
): array {
    if (!bzj_cleaner_table_exists($db, $usersTable)) {
        bzj_cleaner_fail(
            'Required table does not exist: ' . $usersTable
        );
    }

    $columns = bzj_cleaner_columns($db, $usersTable);

    if (!isset($columns[strtolower($idColumn)])) {
        bzj_cleaner_fail(
            'Required ID column does not exist: ' .
            $usersTable . '.' . $idColumn
        );
    }

    $actualIdColumn = $columns[strtolower($idColumn)];

    $result = bzj_cleaner_query(
        $db,
        'SELECT DISTINCT CAST(' .
        bzj_cleaner_identifier($actualIdColumn) .
        ' AS UNSIGNED) AS user_id
         FROM ' . bzj_cleaner_identifier($usersTable) . '
         WHERE ' . bzj_cleaner_identifier($actualIdColumn) . ' IS NOT NULL
           AND TRIM(CAST(' .
        bzj_cleaner_identifier($actualIdColumn) .
        ' AS CHAR)) REGEXP \'^[0-9]+$\'
           AND CAST(' .
        bzj_cleaner_identifier($actualIdColumn) .
        ' AS UNSIGNED) > 0'
    );

    $ids = array();

    while ($row = $result->fetch_assoc()) {
        $ids[(string) $row['user_id']] = true;
    }

    return $ids;
}

function bzj_cleaner_orphan_condition(
    string $column,
    array $existingIds
): string {
    if (!$existingIds) {
        return '1 = 0';
    }

    $quotedIds = array();

    foreach (array_keys($existingIds) as $id) {
        $quotedIds[] = (string) ((int) $id);
    }

    return bzj_cleaner_identifier($column) .
        ' IS NOT NULL
         AND TRIM(CAST(' . bzj_cleaner_identifier($column) .
        ' AS CHAR)) REGEXP \'^[0-9]+$\'
         AND CAST(' . bzj_cleaner_identifier($column) .
        ' AS UNSIGNED) > 0
         AND CAST(' . bzj_cleaner_identifier($column) .
        ' AS UNSIGNED) NOT IN (' . implode(',', $quotedIds) . ')';
}

function bzj_cleaner_scan_platform(
    string $platform,
    mysqli $db,
    int $sampleLimit = BZJ_CLEANER_SAMPLE_LIMIT
): array {
    $authoritative = bzj_cleaner_authoritative_map()[$platform];
    $referenceMap = bzj_cleaner_reference_maps()[$platform];

    $existingIds = bzj_cleaner_existing_ids(
        $db,
        $authoritative['table'],
        $authoritative['id']
    );

    $findings = array();
    $allTables = bzj_cleaner_tables($db);

    foreach ($referenceMap as $configuredTable => $configuredColumns) {
        $table = null;

        /*
         * Prefer exact table names. The map contains common WoWonder
         * spelling/version variants as separate entries.
         */
        foreach ($allTables as $availableTable) {
            if (strcasecmp($availableTable, $configuredTable) === 0) {
                $table = $availableTable;
                break;
            }
        }

        if ($table === null) {
            continue;
        }

        if (
            strcasecmp($table, $authoritative['table']) === 0
        ) {
            continue;
        }

        $actualColumns = bzj_cleaner_columns($db, $table);
        $conditions = array();
        $columnFindings = array();

        foreach ($configuredColumns as $configuredColumn) {
            $columnKey = strtolower($configuredColumn);

            if (!isset($actualColumns[$columnKey])) {
                continue;
            }

            $column = $actualColumns[$columnKey];
            $condition = bzj_cleaner_orphan_condition(
                $column,
                $existingIds
            );

            $countResult = bzj_cleaner_query(
                $db,
                'SELECT COUNT(*) AS total
                 FROM ' . bzj_cleaner_identifier($table) . '
                 WHERE ' . $condition
            );

            $countRow = $countResult->fetch_assoc();
            $count = (int) ($countRow['total'] ?? 0);

            if ($count < 1) {
                continue;
            }

            $sampleResult = bzj_cleaner_query(
                $db,
                'SELECT DISTINCT ' .
                bzj_cleaner_identifier($column) .
                ' AS orphan_id
                 FROM ' . bzj_cleaner_identifier($table) . '
                 WHERE ' . $condition . '
                 ORDER BY CAST(' .
                bzj_cleaner_identifier($column) .
                ' AS UNSIGNED)
                 LIMIT ' . (int) $sampleLimit
            );

            $sampleIds = array();

            while ($sample = $sampleResult->fetch_assoc()) {
                $sampleIds[] = (string) $sample['orphan_id'];
            }

            $conditions[] = '(' . $condition . ')';

            $columnFindings[] = array(
                'column' => $column,
                'orphan_references' => $count,
                'sample_user_ids' => $sampleIds,
            );
        }

        if (!$conditions) {
            continue;
        }

        $combinedCondition = implode(' OR ', $conditions);

        $rowCountResult = bzj_cleaner_query(
            $db,
            'SELECT COUNT(*) AS total
             FROM ' . bzj_cleaner_identifier($table) . '
             WHERE ' . $combinedCondition
        );

        $rowCount = $rowCountResult->fetch_assoc();

        $findings[] = array(
            'table' => $table,
            'columns' => $columnFindings,
            'orphan_rows' => (int) ($rowCount['total'] ?? 0),
        );
    }

    return array(
        'platform' => $platform,
        'authoritative_table' => $authoritative['table'],
        'existing_user_count' => count($existingIds),
        'findings' => $findings,
        'existing_ids' => $existingIds,
    );
}

function bzj_cleaner_load_platform_users(
    mysqli $db,
    string $platform
): array {
    $config = bzj_cleaner_authoritative_map()[$platform];
    $columns = bzj_cleaner_columns($db, $config['table']);

    $required = array(
        strtolower($config['id']),
        strtolower($config['username']),
        strtolower($config['email']),
    );

    foreach ($required as $requiredColumn) {
        if (!isset($columns[$requiredColumn])) {
            return array();
        }
    }

    $idColumn = $columns[strtolower($config['id'])];
    $usernameColumn = $columns[strtolower($config['username'])];
    $emailColumn = $columns[strtolower($config['email'])];

    $result = bzj_cleaner_query(
        $db,
        'SELECT ' .
        bzj_cleaner_identifier($idColumn) . ' AS platform_id, ' .
        bzj_cleaner_identifier($usernameColumn) . ' AS username, ' .
        bzj_cleaner_identifier($emailColumn) . ' AS email
         FROM ' . bzj_cleaner_identifier($config['table'])
    );

    $users = array();

    while ($row = $result->fetch_assoc()) {
        $id = (string) $row['platform_id'];

        if (!ctype_digit($id) || (int) $id < 1) {
            continue;
        }

        $users[$id] = array(
            'id' => (int) $id,
            'username' => (string) ($row['username'] ?? ''),
            'email' => (string) ($row['email'] ?? ''),
        );
    }

    return $users;
}

function bzj_cleaner_index_platform_users(array $users): array
{
    $index = array(
        'email' => array(),
        'username' => array(),
    );

    foreach ($users as $user) {
        $email = bzj_cleaner_normalize($user['email']);
        $username = bzj_cleaner_normalize($user['username']);

        if ($email !== '') {
            if (!isset($index['email'][$email])) {
                $index['email'][$email] = array();
            }

            $index['email'][$email][] = $user;
        }

        if ($username !== '') {
            if (!isset($index['username'][$username])) {
                $index['username'][$username] = array();
            }

            $index['username'][$username][] = $user;
        }
    }

    return $index;
}

function bzj_cleaner_meta_value(
    mysqli $db,
    string $metaTable,
    int $userId,
    string $metaKey
): string {
    $escapedKey = $db->real_escape_string($metaKey);

    $result = bzj_cleaner_query(
        $db,
        'SELECT meta_value
         FROM ' . bzj_cleaner_identifier($metaTable) . '
         WHERE user_id = ' . (int) $userId . '
           AND meta_key = \'' . $escapedKey . '\'
         ORDER BY umeta_id ASC
         LIMIT 1'
    );

    $row = $result->fetch_assoc();

    return (string) ($row['meta_value'] ?? '');
}

function bzj_cleaner_wp_mapping_scan(
    mysqli $wp,
    mysqli $wo,
    mysqli $qd
): array {
    $usersTable = bzj_cleaner_wp_table('users');
    $metaTable = bzj_cleaner_wp_table('usermeta');

    if (
        !bzj_cleaner_table_exists($wp, $usersTable) ||
        !bzj_cleaner_table_exists($wp, $metaTable)
    ) {
        bzj_cleaner_fail('WordPress users/usermeta tables were not found.');
    }

    $woUsers = bzj_cleaner_load_platform_users($wo, 'wowonder');
    $qdUsers = bzj_cleaner_load_platform_users($qd, 'quickdate');

    $woIndex = bzj_cleaner_index_platform_users($woUsers);
    $qdIndex = bzj_cleaner_index_platform_users($qdUsers);

    $wpResult = bzj_cleaner_query(
        $wp,
        'SELECT ID, user_login, user_email
         FROM ' . bzj_cleaner_identifier($usersTable) . '
         ORDER BY ID ASC'
    );

    $findings = array();

    while ($wpUser = $wpResult->fetch_assoc()) {
        $wpId = (int) $wpUser['ID'];
        $wpUsername = bzj_cleaner_normalize($wpUser['user_login']);
        $wpEmail = bzj_cleaner_normalize($wpUser['user_email']);

        $woMeta = bzj_cleaner_meta_value(
            $wp,
            $metaTable,
            $wpId,
            'wo_user_id'
        );

        $qdMeta = bzj_cleaner_meta_value(
            $wp,
            $metaTable,
            $wpId,
            'qd_user_id'
        );

        $platforms = array(
            'wowonder' => array(
                'meta_key' => 'wo_user_id',
                'meta_value' => $woMeta,
                'users' => $woUsers,
                'index' => $woIndex,
            ),
            'quickdate' => array(
                'meta_key' => 'qd_user_id',
                'meta_value' => $qdMeta,
                'users' => $qdUsers,
                'index' => $qdIndex,
            ),
        );

        foreach ($platforms as $platform => $data) {
            $emailMatches = array();
            $usernameMatches = array();

            if (
                $wpEmail !== '' &&
                isset($data['index']['email'][$wpEmail])
            ) {
                $emailMatches = $data['index']['email'][$wpEmail];
            }

            if (
                $wpUsername !== '' &&
                isset($data['index']['username'][$wpUsername])
            ) {
                $usernameMatches = $data['index']['username'][$wpUsername];
            }

            $candidateIds = array();

            foreach (array_merge($emailMatches, $usernameMatches) as $match) {
                $candidateIds[(string) $match['id']] = true;
            }

            $candidateIds = array_keys($candidateIds);
            $storedId = trim($data['meta_value']);
            $storedValid = ctype_digit($storedId) && (int) $storedId > 0;
            $storedExists = $storedValid &&
                isset($data['users'][(string) ((int) $storedId)]);

            $emailIds = array();
            foreach ($emailMatches as $match) {
                $emailIds[(string) $match['id']] = true;
            }

            $usernameIds = array();
            foreach ($usernameMatches as $match) {
                $usernameIds[(string) $match['id']] = true;
            }

            $sameEvidence = array_intersect(
                array_keys($emailIds),
                array_keys($usernameIds)
            );

            $repairable = false;
            $repairId = null;
            $status = 'ok';

            if (count($candidateIds) === 0) {
                if ($storedId !== '') {
                    $status = $storedExists
                        ? 'stored_id_has_no_username_or_email_match'
                        : 'stored_id_missing_from_platform';
                } else {
                    $status = 'no_platform_match';
                }
            } elseif (count($candidateIds) > 1) {
                $status = 'ambiguous_username_or_email_match';
            } elseif (
                count($emailMatches) > 1 ||
                count($usernameMatches) > 1
            ) {
                $status = 'ambiguous_duplicate_platform_identity';
            } else {
                $repairId = (int) $candidateIds[0];

                /*
                 * If both email and username are available, they must point
                 * to the same platform account. A single-field match remains
                 * repairable when the other WordPress field is empty.
                 */
                $hasEmail = $wpEmail !== '';
                $hasUsername = $wpUsername !== '';

                $emailSupports = !$hasEmail || count($emailMatches) === 1;
                $usernameSupports =
                    !$hasUsername || count($usernameMatches) === 1;

                $bothAgree = !$hasEmail ||
                    !$hasUsername ||
                    count($sameEvidence) === 1;

                if ($emailSupports && $usernameSupports && $bothAgree) {
                    $repairable = true;
                    $status = $storedId === (string) $repairId
                        ? 'ok'
                        : 'repairable_mismatch';
                } else {
                    $status = 'conflicting_identity_match';
                }
            }

            if ($status !== 'ok') {
                $findings[] = array(
                    'wp_user_id' => $wpId,
                    'wp_username' => (string) $wpUser['user_login'],
                    'wp_email' => (string) $wpUser['user_email'],
                    'platform' => $platform,
                    'meta_key' => $data['meta_key'],
                    'stored_id' => $storedId,
                    'stored_id_exists' => $storedExists,
                    'candidate_id' => $repairId,
                    'repairable' => $repairable,
                    'status' => $status,
                    'email_match_count' => count($emailMatches),
                    'username_match_count' => count($usernameMatches),
                );
            }
        }
    }

    return array(
        'findings' => $findings,
        'wowonder_user_count' => count($woUsers),
        'quickdate_user_count' => count($qdUsers),
    );
}

function bzj_cleaner_render_orphan_findings(array $scan): string
{
    $html = '';
    $totalRows = 0;

    foreach ($scan['findings'] as $finding) {
        $totalRows += (int) $finding['orphan_rows'];

        $html .= '<h4>' .
            bzj_cleaner_h($finding['table']) .
            ' — ' .
            (int) $finding['orphan_rows'] .
            ' row(s)</h4>';

        $html .= '<ul>';

        foreach ($finding['columns'] as $column) {
            $html .= '<li><strong>' .
                bzj_cleaner_h($column['column']) .
                '</strong>: ' .
                (int) $column['orphan_references'] .
                ' reference(s); sample IDs: ' .

                    $column['sample_user_ids']
                        ? implode(', ', $column['sample_user_ids'])

    }

        $totalRows . '</p>' . $html;
}

function bzj_cleaner_clean_orphans(
    mysqli $db,
    string $platform,
    array $scan
): array {
    $authoritative = bzj_cleaner_authoritative_map()[$platform];
    $results = array();

    /*
     * Fresh snapshot immediately before deletion.
     */
    $existingIds = bzj_cleaner_existing_ids(
        $db,
        $authoritative['table'],
        $authoritative['id']
    );

    foreach ($scan['findings'] as $finding) {
        $table = $finding['table'];

        if (!bzj_cleaner_table_exists($db, $table)) {
            continue;
        }

        $actualColumns = bzj_cleaner_columns($db, $table);
        $conditions = array();

        foreach ($finding['columns'] as $columnFinding) {
            $key = strtolower($columnFinding['column']);

            if (!isset($actualColumns[$key])) {
                continue;
            }

            $column = $actualColumns[$key];

            $conditions[] = '(' .
                bzj_cleaner_orphan_condition(
                    $column,
                    $existingIds
                ) .
                ')';
        }

        if (!$conditions) {
            continue;
        }

        $sql = 'DELETE FROM ' .
            bzj_cleaner_identifier($table) .
            ' WHERE ' .
            implode(' OR ', $conditions);

        if (!$db->begin_transaction()) {
            throw new RuntimeException(
                'Unable to begin transaction for ' . $table
            );
        }

        try {
            bzj_cleaner_query($db, $sql);
            $deleted = (int) $db->affected_rows;

            if (!$db->commit()) {
                throw new RuntimeException(
                    'Unable to commit transaction for ' . $table
                );
            }

            $results[] = array(
                'platform' => $platform,
                'table' => $table,
                'deleted_rows' => $deleted,
                'status' => 'committed',
            );
        } catch (Throwable $exception) {
            $db->rollback();

            $results[] = array(
                'platform' => $platform,
                'table' => $table,
                'deleted_rows' => 0,
                'status' => 'rolled_back',
                'error' => $exception->getMessage(),
            );

            throw $exception;
        }
    }

    return $results;
}

function bzj_cleaner_repair_mappings(
    mysqli $wp,
    array $mappingScan
): array {
    $metaTable = bzj_cleaner_wp_table('usermeta');
    $results = array();

    if (!bzj_cleaner_table_exists($wp, $metaTable)) {
        bzj_cleaner_fail('WordPress usermeta table was not found.');
    }

    foreach ($mappingScan['findings'] as $finding) {
        if (!$finding['repairable']) {
            continue;
        }

        $wpUserId = (int) $finding['wp_user_id'];
        $metaKey = (string) $finding['meta_key'];
        $newValue = (string) $finding['candidate_id'];

        $escapedKey = $wp->real_escape_string($metaKey);
        $escapedValue = $wp->real_escape_string($newValue);

        if (!$wp->begin_transaction()) {
            throw new RuntimeException(
                'Unable to begin WordPress metadata transaction.'
            );
        }

        try {
            $update = bzj_cleaner_query(
                $wp,
                'UPDATE ' . bzj_cleaner_identifier($metaTable) . '
                 SET meta_value = \'' . $escapedValue . '\'
                 WHERE user_id = ' . $wpUserId . '
                   AND meta_key = \'' . $escapedKey . '\''
            );

            if ($update instanceof mysqli_result) {
                $affected = 0;
            } else {
                $affected = (int) $wp->affected_rows;
            }

            if ($affected === 0) {
                bzj_cleaner_query(
                    $wp,
                    'INSERT INTO ' .
                    bzj_cleaner_identifier($metaTable) .
                    ' (user_id, meta_key, meta_value)
                     VALUES (' .
                    $wpUserId . ', \'' .
                    $escapedKey . '\', \'' .
                    $escapedValue . '\')'
                );
            }

            if (!$wp->commit()) {
                throw new RuntimeException(
                    'Unable to commit WordPress metadata transaction.'
                );
            }

            $results[] = array(
                'wp_user_id' => $wpUserId,
                'platform' => $finding['platform'],
                'meta_key' => $metaKey,
                'new_value' => $newValue,
                'status' => 'committed',
            );
        } catch (Throwable $exception) {
            $wp->rollback();

            $results[] = array(
                'wp_user_id' => $wpUserId,
                'platform' => $finding['platform'],
                'meta_key' => $metaKey,
                'new_value' => $newValue,
                'status' => 'rolled_back',
                'error' => $exception->getMessage(),
            );

            throw $exception;
        }
    }

    return $results;
}

function bzj_cleaner_connections(): array
{
    $wp = get_wp_db_conn();
    $wo = get_wowonder_db();
    $qd = get_qd_db_conn();

    if (!$wp instanceof mysqli || $wp->connect_errno) {
        throw new RuntimeException('WordPress database connection failed.');
    }

    if (!$wo instanceof mysqli || $wo->connect_errno) {
        throw new RuntimeException('WoWonder database connection failed.');
    }

    if (!$qd instanceof mysqli || $qd->connect_errno) {
        throw new RuntimeException('QuickDate database connection failed.');
    }

    $wp->set_charset('utf8mb4');
    $wo->set_charset('utf8mb4');
    $qd->set_charset('utf8mb4');

    return array(
        'wp' => $wp,
        'wo' => $wo,
        'qd' => $qd,
    );
}

function bzj_cleaner_full_scan(array $connections): array
{
    $woScan = bzj_cleaner_scan_platform(
        'wowonder',
        $connections['wo']
    );

    $qdScan = bzj_cleaner_scan_platform(
        'quickdate',
        $connections['qd']
    );

    $mappingScan = bzj_cleaner_wp_mapping_scan(
        $connections['wp'],
        $connections['wo'],
        $connections['qd']
    );

    unset($woScan['existing_ids']);

    unset($qdScan['existing_ids']);

    return array(
        'wowonder' => $woScan,
        'quickdate' => $qdScan,
        'mapping' => $mappingScan,
        'generated_at_utc' => gmdate('c'),
    );
}

        'event' => 'error',

    ));
````

### Important deployment notes

1. Save the file as:

   `buzzjuice.net/shared/bzj-wwqd-orphan-user-cleaner.php`

2. The script assumes WordPress is located at one of:

   - `buzzjuice.net/wp-load.php`
   - the parent project root relative to `shared/`

3. The script uses these WordPress metadata keys:

   - `wo_user_id`
   - `qd_user_id`

4. The WoWonder and QuickDate tables are treated as authoritative:

   - WoWonder: `Wo_Users.user_id`
   - QuickDate: `users.id`

5. The script intentionally does **not** call `Wo_DeleteUser()` or QuickDate’s `delete_user()` because those functions are designed for deleting a complete existing account and can delete files, pages, groups, posts, and other owned content. An orphan-reference cleanup utility cannot safely infer ownership of every media object.

   - the WoWonder database;
   - the QuickDate database.
