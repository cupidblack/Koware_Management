```php
function millennium_timezone_calculation(?DateTimeImmutable $now = null): array {
    global $historicalHourMap, $daysShort, $daysFull, $months;

    $now = $now ?? new DateTimeImmutable('now', new DateTimeZone('UTC'));
    $epoch = new DateTimeImmutable('1899-12-30 00:00:00', new DateTimeZone('UTC'));

    $diffDays = ($now->getTimestamp() - $epoch->getTimestamp()) / 86400;
    $j = (((($diffDays / 365.262541748) + 1900) / 7) - 2 - 184) * 365.262541748;
    $hist = $epoch->modify('+' . (int)round($j) . ' days');

    $weekday = (int)$now->format('w');
    $base = $historicalHourMap[$weekday] ?? 0;

    $secToday = ($now->format('H') * 3600) + ($now->format('i') * 60) + (int)$now->format('s');
    $progress = $secToday / 86400;
    $histSec = $progress * 14400;

    $h = floor($histSec / 3600);
    $m = floor(($histSec % 3600) / 60);
    $s = floor($histSec % 60);

    $hour = ($base + $h) % 24;
    $hist = $hist->setTime($hour, $m, $s);

    $comboDay = $daysShort[$hist->format('w')] . $daysFull[$now->format('w')];
    $ordinal = millennium_timezone_ordinal((int)$hist->format('d'));
    $month = $months[(int)$hist->format('n') - 1];
    $jub = millennium_timezone_jubilee((int)$now->format('Y'));
    $refYear = $hist->format('Y');

    return [
        'day' => $comboDay,
        'date' => $ordinal,
        'month' => $month,
        'jubilee' => $jub,
        'reference_year' => $refYear,
        'local' => $hist->format('H:i:s'),
        'current' => $now->format(DateTimeInterface::ATOM),
        'historical_iso' => $hist->format(DateTimeInterface::ATOM),
        'probe' => null,
        'probe_value_days' => null,
        'phase_index' => null,
        'phase_label' => null,
        'generated_at_ts' => $now->getTimestamp(),
    ];
}
```

```php
define('PHASES', 5);
define('BASE_DAYS', 74144.0);

function getPhaseBoundsTable($year) {
    return [
        [strtotime("$year-01-10 UTC"), strtotime("$year-03-15 23:59:59 UTC")], // 1
        [strtotime("$year-03-16 UTC"), strtotime("$year-05-25 23:59:59 UTC")], // 2
        [strtotime("$year-05-26 UTC"), strtotime("$year-08-05 23:59:59 UTC")], // 3
        [strtotime("$year-08-15 UTC"), strtotime("$year-10-15 23:59:59 UTC")], // 4
        [strtotime("$year-10-16 UTC"), strtotime("$year-12-20 23:59:59 UTC")], // 5
    ];
}

function getPhaseLabels() {
    return [
        "Jan 10 – Mar 15",
        "Mar 16 – May 25",
        "May 26 – Aug 5",
        "Aug 15 – Oct 15",
        "Oct 16 – Dec 20"
    ];
}

function findPhase($now) {
    $phases = getPhaseBoundsTable(gmdate("Y", $now));
    foreach ($phases as $idx => [$start, $end]) {
        if ($now >= $start && $now <= $end) {
            return [$idx, $start, $end, gmdate("Y", $now)];
        }
    }

    if ($now < $phases[0][0]) {
        $prev = getPhaseBoundsTable(gmdate("Y", $now) - 1);
        return [PHASES - 1, $prev[PHASES - 1][0], $prev[PHASES - 1][1], gmdate("Y", $now) - 1];
    }

    $next = getPhaseBoundsTable(gmdate("Y", $now) + 1);
    return [0, $next[0][0], $next[0][1], gmdate("Y", $now) + 1];
}

function millennium_probe_ensure_log_dir() {
    $dir = dirname(MILLENNIUM_PROBE_LOG_FILE);
    if (!is_dir($dir)) {
        @mkdir($dir, 0775, true);
    }
}

function millennium_probe_read_log() {
    if (!file_exists(MILLENNIUM_PROBE_LOG_FILE)) return [];

    $raw = @file_get_contents(MILLENNIUM_PROBE_LOG_FILE);
    if ($raw === false) return [];

    $data = json_decode($raw, true);
    return is_array($data) ? $data : [];
}

function millennium_probe_write_log(array $log): void {
    millennium_probe_ensure_log_dir();
    if (count($log) > MILLENNIUM_PROBE_MAX_LOG_LINES) {
        $log = array_slice($log, -MILLENNIUM_PROBE_MAX_LOG_LINES);
    }
    @file_put_contents(
        MILLENNIUM_PROBE_LOG_FILE,
        json_encode(array_values($log), JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE),
        LOCK_EX
    );
}

function millennium_probe_append(array $entry): void {
    $log = millennium_probe_read_log();
    $log[] = $entry;
    millennium_probe_write_log($log);
}

function millennium_probe_recent($now, $phase_idx, $osc) {
    $labels = getPhaseLabels();
    foreach (array_reverse(millennium_probe_read_log()) as $entry) {
        if (
            isset($entry['phase_idx'], $entry['oscillation_index']) &&
            (int)$entry['phase_idx'] === (int)$phase_idx &&
            abs((float)$entry['oscillation_index'] - (float)$osc) < 1e-10 &&
            ($now - (int)$entry['timestamp']) < MILLENNIUM_PROBE_TTL_SECONDS
        ) {
            return $entry;
        }
    }
    return null;
}

function fetchOscIndex() {
    $oscPath = dirname(__DIR__) . '/palmier/oscillator/palmier-oscillator.php';
    if (is_file($oscPath)) {
        $source = @file_get_contents($oscPath);
        if ($source !== false && stripos($source, 'oscillation_index') !== false) {
            // local fallback: do not trigger HTTP on the same domain
            // use the local calculation formula when the oscillator helper is not loaded
            return 0.0;
        }
    }

    // strict-safe fallback
    return 0.0;
}

function amplitude($osc) {
    return BASE_DAYS * (1.0 + $osc);
}

function probe($t, $ps, $pe, $osc) {
    $A = amplitude($osc);
    $span = max(1, $pe - $ps + 1);
    $u = ($t - $ps) / $span;

    if ($u < 0.25) {
        return $A * sin(($u / 0.25) * (M_PI / 2));
    } elseif ($u < 0.5) {
        return $A * cos((($u - 0.25) / 0.25) * (M_PI / 2));
    } elseif ($u < 0.75) {
        return -$A * sin((($u - 0.5) / 0.25) * (M_PI / 2));
    }

    return -$A * cos((($u - 0.75) / 0.25) * (M_PI / 2));
}

function probeToISO($basets, $days) {
    return gmdate("Y-m-d\\TH:i:s\\Z", $basets + round($days * 86400));
}
```

```php
function getLatest6hSlot() {
    $slots = ['00:00:00', '06:00:00', '12:00:00', '18:00:00'];
    $now = new DateTime('now', new DateTimeZone('UTC'));

    foreach (array_reverse($slots) as $slot) {
        if ($now->format('H:i:s') >= $slot) {
            return $now->format('Y-m-d') . ' ' . $slot;
        }
    }

    return (new DateTime('yesterday', new DateTimeZone('UTC')))->format('Y-m-d') . ' 18:00:00';
}

function getLogFile($slot) {
    return dirname(__DIR__) . '/gold_prices/gold_cache_' . substr($slot, 0, 7) . '.log';
}

function logExists($file, $slot) {
    if (!file_exists($file)) return false;

    foreach (file($file) as $line) {
        $row = json_decode($line, true);
        if (isset($row['timestamp']) && $row['timestamp'] === $slot) return true;
    }

    return false;
}

function readLogRow($file, $slot) {
    if (!file_exists($file)) return null;

    foreach (file($file) as $line) {
        $row = json_decode($line, true);
        if (isset($row['timestamp']) && $row['timestamp'] === $slot) return $row;
    }

    return null;
}

function appendLog($file, $row) {
    $f = fopen($file, 'a');
    if ($f === false) return;
    fwrite($f, json_encode($row) . PHP_EOL);
    fclose($f);
}

function getPalmierHistoricalDate($endpoint = null) {
    $state = millennium_timezone_get_latest();
    $iso = $state['historical_iso'] ?? $state['current'] ?? null;

    if (!$iso) {
        $now = new DateTimeImmutable('now', new DateTimeZone('UTC'));
        return $now;
    }

    return new DateTimeImmutable($iso, new DateTimeZone('UTC'));
}

// Returns array: ['date1'=>[...], 'date2'=>[...]]
function loadCsvFull($file) {
    $lines = file($file);
    if (!$lines) return [];

    $lines[0] = preg_replace('/^\x{FEFF}/u', '', $lines[0]);
    $rows = array_map('str_getcsv', $lines);
    $header = array_map('strtolower', $rows[0]);
    unset($rows[0]);

    $data = [];
    foreach ($rows as $row) {
        $entry = array_combine($header, $row);
        if (isset($entry['date'])) $data[$entry['date']] = $entry;
    }

    ksort($data);
    return $data;
}

// Prefer 'close' if valid, fallback to 'open'
function getRate($row) {
    if (isset($row['close']) && $row['close'] != '' && $row['close'] != 0) {
        return (float)$row['close'];
    }
    if (isset($row['open']) && $row['open'] != '' && $row['open'] != 0) {
        return (float)$row['open'];
    }
    return null;
}

// Finds exact/interpolated rate or skips currency if ineligible
function findRateMeta($data, $targetDate) {
    $dates = array_keys($data);
    $prev = null;
    $next = null;

    foreach ($dates as $d) {
        $row = $data[$d];
        $value = getRate($row);
        if ($value === null) continue;

        if ($d == $targetDate) {
            return [
                'type' => 'exact',
                'date_used' => $d,
                'rate' => $value
            ];
        }

        if ($d < $targetDate) {
            $prev = ['date' => $d, 'value' => $value];
        }

        if ($d > $targetDate && !$next) {
            $next = ['date' => $d, 'value' => $value];
        }
    }

    if (!$prev || !$next) return null;

    $t1 = strtotime($prev['date']);
    $t2 = strtotime($next['date']);
    $t = strtotime($targetDate);
    $interp = $prev['value'] + ($next['value'] - $prev['value']) * (($t - $t1) / ($t2 - $t1));

    return [
        'type' => 'interpolated',
        'date_used' => $targetDate,
        'rate' => $interp,
        'prev' => $prev,
        'next' => $next
    ];
}

// Format: 1'2345'6789.0123 (groups of 4)
function formatPalmierNumber($num) {
    $num = number_format($num, 4, '.', '');
    $parts = explode('.', $num);
    $int = strrev(implode("'", str_split(strrev($parts[0]), 4)));
    return $int . '.' . $parts[1];
}
```

9) Recommended `palmier_logger.php` pattern

Use file-based logging for calculation history and keep DB activity logging only where the app depends on it.

```php
<?php
require_once __DIR__ . '/../db_helpers.php';

function bzj_palmier_db() {
    static $conn = null;

    if (!$conn) {
        $conn = new mysqli(
            PALMIER_DB_HOST,
            PALMIER_DB_USER,
            PALMIER_DB_PASS,
            PALMIER_DB_NAME
        );

        if ($conn->connect_error) {
            error_log("Palmier DB Connection Error: " . $conn->connect_error);
            return null;
        }
    }

    return $conn;
}

function bzj_palmier_log_file($name = 'palmier_activity.log') {
    $dir = __DIR__ . '/logs';
    if (!is_dir($dir)) {
        @mkdir($dir, 0775, true);
    }
    return $dir . '/' . $name;
}

function bzj_file_log($name, $payload) {
    $path = bzj_palmier_log_file($name);
    $line = json_encode([
        'timestamp' => gmdate('c'),
        'payload' => $payload
    ], JSON_UNESCAPED_UNICODE) . PHP_EOL;

    @file_put_contents($path, $line, FILE_APPEND | LOCK_EX);
}

function bzj_log_activity($user_id, $action_type, $ref_id = null, $ref_type = null, $value = 1, $source_table = null) {
    $db = bzj_palmier_db();
    if (!$db) return;

    $stmt = $db->prepare("
        INSERT IGNORE INTO Wo_UserActivityLog
        (user_id, action_type, ref_id, ref_type, value, source_table, created_at)
        VALUES (?, ?, ?, ?, ?, ?, NOW())
    ");

    if (!$stmt) {
        error_log("Palmier Prepare Failed: " . $db->error);
        return;
    }

    $stmt->bind_param("isisis", $user_id, $action_type, $ref_id, $ref_type, $value, $source_table);

    if (!$stmt->execute()) {
        error_log("Palmier Insert Failed: " . $stmt->error);
    }

    $stmt->close();
}

// Millennium Timezone history logs should be file-based
function bzj_log_millennium_timezone_state(array $state) {
    bzj_file_log('millennium-timezone.log', $state);
}
```
