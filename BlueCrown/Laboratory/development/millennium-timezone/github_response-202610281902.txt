function millennium_timezone_db_insert($table, $payload = []) {
    $conn = get_iapd_db_conn();

    $safeTable = preg_replace('/[^A-Za-z0-9_]/', '', $table);

    $placeholders = [];
    $values = [];

    foreach ($payload as $key => $value) {
        $fields[] = '`' . preg_replace('/[^A-Za-z0-9_]/', '', $key) . '`';
        $placeholders[] = '?';
        $values[] = $value;

    if (!$fields) return false;

    $sql = 'INSERT INTO `' . MILLENNIUM_TIMEZONE_DB_NAME . '`.`' . $fullTable . '` ('
        . implode(', ', $fields) . ') VALUES (' . implode(', ', $placeholders) . ')';

function ordinal($n) {

    $now = ($dt instanceof DateTimeImmutable)
        ? $dt

    $epoch = new DateTimeImmutable('1899-12-30 00:00:00', new DateTimeZone('UTC'));

function millennium_probe_read_log() {

function millennium_probe_value($t, $phaseStart, $phaseEnd, $osc) {
    $A = millennium_probe_amplitude($osc);
    $span = max(1, (int)$phaseEnd - (int)$phaseStart + 1);
    $u = ($t - $phaseStart) / $span;

    if ($u < 0.25) {
        return $A * sin(($u / 0.25) * (M_PI / 2));
    }
    if ($u < 0.5) {
        return $A * cos((($u - 0.25) / 0.25) * (M_PI / 2));
    }
    if ($u < 0.75) {
        return -$A * sin((($u - 0.5) / 0.25) * (M_PI / 2));
    }
    return -$A * cos((($u - 0.75) / 0.25) * (M_PI / 2));
}

function loadCsvFull($file) {

// Finds exact/interpolated rate or skips currency if ineligible
function findRateMeta($data, $targetDate) {
    $dates = array_keys($data);
    $prev = null;
    $next = null;

    foreach ($dates as $d) {
        $row = $data[$d];
        $value = getRate($row);
        if ($value === null) continue;

        if ($d == $targetDate) {
            return [
                'type' => 'exact',
                'date_used' => $d,
                'rate' => $value
            ];
        }
        if ($d < $targetDate) $prev = ['date' => $d, 'value' => $value];
        if ($d > $targetDate && !$next) $next = ['date' => $d, 'value' => $value];
    }

    if (!$prev || !$next) return null;

    $t1 = strtotime($prev['date']);
    $t2 = strtotime($next['date']);
    $t = strtotime($targetDate);

    $interp = $prev['value'] + ($next['value'] - $prev['value']) * (($t - $t1) / ($t2 - $t1));

    return [
        'type' => 'interpolated',
        'date_used' => $targetDate,
        'rate' => $interp,
        'prev' => $prev,
        'next' => $next
    ];
}
