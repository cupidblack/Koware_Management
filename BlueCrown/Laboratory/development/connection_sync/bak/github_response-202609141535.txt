2. **Corrected BuddyBoss API calls** — `friends_accept_friendship()` now uses actual friendship ID
3. **Proper HMAC-SHA256 authentication** — Timestamp + body signature for secure external reporting

```php
<?php
/**
 * Plugin Name: Buzzjuice Connections Sync
 * Description: Canonical synchronization of BuddyBoss connections and moderation 
 *              across WordPress, WoWonder Streams, and QuickDate Socials.
 * Version: 4.1.0
 * Author: Buzzjuice
 * License: GPL-2.0-or-later
 *
 * PRODUCTION ARCHITECTURE:
 * - WordPress/BuddyBoss is the canonical authority
 * - Relationships and blocks stored separately (non-overlapping)
 * - Directional blocks: block_a_to_b and block_b_to_a are independent
 * - Durable event ledger serves as audit trail and retry queue
 * - External platforms sync via HMAC-signed REST API after canonical change
 * - Idempotent event UUIDs prevent duplicate processing
 * - No internal recursion flag during external processing
 * - Log file for diagnostics; two tables for durable operational state
 */

function bzj_connections_sync_log( $message, $context = [] ) {
    $timestamp = gmdate( 'Y-m-d H:i:s' );
    $line = "[{$timestamp}] {$message}";

    if ( ! empty( $context ) ) {
        $line .= ' ' . wp_json_encode(
            $context,
            JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE
        );
    }

    $line .= PHP_EOL;

    @error_log( $line, 3, BZJ_CONNECTIONS_SYNC_LOG );
}

// ==================== INSTALLATION ====================

function bzj_connections_sync_install() {
    global $wpdb;

    $current_version = get_option( 'bzj_connections_sync_schema_version', '' );

    if ( $current_version === BZJ_CONNECTIONS_SYNC_VERSION ) {
        return;
    }

    require_once ABSPATH . 'wp-admin/includes/upgrade.php';

    $charset = $wpdb->get_charset_collate();
    $rel_table = bzj_connections_sync_relationships_table();
    $evt_table = bzj_connections_sync_events_table();

    $sql = "

    CREATE TABLE IF NOT EXISTS {$rel_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

        user_a BIGINT UNSIGNED NOT NULL,
        user_b BIGINT UNSIGNED NOT NULL,

        relationship_state VARCHAR(20) NOT NULL DEFAULT 'none',

        block_a_to_b TINYINT(1) NOT NULL DEFAULT 0,
        block_b_to_a TINYINT(1) NOT NULL DEFAULT 0,

        requested_by BIGINT UNSIGNED NULL,

        created_at DATETIME NOT NULL,
        updated_at DATETIME NOT NULL,

        last_event_uuid CHAR(36) NULL,

        PRIMARY KEY (id),
        UNIQUE KEY relationship_pair (user_a, user_b),
        KEY relationship_state (relationship_state),
        KEY block_a_to_b (block_a_to_b),
        KEY block_b_to_a (block_b_to_a),
        KEY updated_at (updated_at)
    ) {$charset};

    CREATE TABLE IF NOT EXISTS {$evt_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

        event_uuid CHAR(36) NOT NULL,

        origin VARCHAR(20) NOT NULL,

        event_type VARCHAR(60) NOT NULL,

        operation VARCHAR(30) NOT NULL,

        actor_wp_id BIGINT UNSIGNED NOT NULL,

        target_wp_id BIGINT UNSIGNED NOT NULL,

        relationship_state VARCHAR(20) NOT NULL,

        payload LONGTEXT NULL,

        status VARCHAR(20) NOT NULL DEFAULT 'pending',

        attempts INT UNSIGNED NOT NULL DEFAULT 0,

        next_attempt_at DATETIME NOT NULL,

        last_error TEXT NULL,

        created_at DATETIME NOT NULL,

        processed_at DATETIME NULL,

        PRIMARY KEY (id),
        UNIQUE KEY event_uuid (event_uuid),
        KEY retry_lookup (status, next_attempt_at),
        KEY pair_lookup (actor_wp_id, target_wp_id),
        KEY created_at (created_at)
    ) {$charset};

    ";

    dbDelta( $sql );

    update_option(
        'bzj_connections_sync_schema_version',
        BZJ_CONNECTIONS_SYNC_VERSION,
        false
    );

    bzj_connections_sync_log(
        'Schema installed/updated',
        [ 'version' => BZJ_CONNECTIONS_SYNC_VERSION ]
    );
}

// ==================== UTILITIES ====================

function bzj_connections_sync_uuid() {
    return wp_generate_uuid4();
}

/**
 * Normalize user pair so user_a < user_b.
 * Prevents (A,B) and (B,A) from being stored as separate records.
 */
function bzj_connections_sync_pair( $a, $b ) {
    $a = absint( $a );
    $b = absint( $b );

    if ( ! $a || ! $b || $a === $b ) {
        return false;
    }

    if ( $a > $b ) {
        [ $a, $b ] = [ $b, $a ];
    }

    return [
        'user_a' => $a,
        'user_b' => $b,
    ];
}

/**
 * MySQL named lock for atomic operations.
 */
function bzj_connections_sync_lock( $a, $b, $timeout = 5 ) {
    global $wpdb;

    $pair = bzj_connections_sync_pair( $a, $b );

    if ( ! $pair ) {
        return false;
    }

    $lock_name = sprintf( 'bzj_rel_%d_%d', $pair['user_a'], $pair['user_b'] );

    $result = (int) $wpdb->get_var(
        $wpdb->prepare( 'SELECT GET_LOCK( %s, %d )', $lock_name, $timeout )
    );

    return 1 === $result ? $lock_name : false;
}

function bzj_connections_sync_unlock( $lock_name ) {
    global $wpdb;

    if ( ! $lock_name ) {
        return;
    }

    $wpdb->get_var(
        $wpdb->prepare( 'SELECT RELEASE_LOCK( %s )', $lock_name )
    );
}

// ==================== CANONICAL STATE READER ====================

function bzj_connections_sync_get_relationship( $a, $b ) {
    global $wpdb;

    $pair = bzj_connections_sync_pair( $a, $b );

    if ( ! $pair ) {
        return false;
    }

    return $wpdb->get_row(
        $wpdb->prepare(
            'SELECT * FROM ' . bzj_connections_sync_relationships_table() . '
            WHERE user_a = %d AND user_b = %d LIMIT 1',
            $pair['user_a'],
            $pair['user_b']
        )
    );
}

/**
 * Check if any block exists between users.
 */
function bzj_connections_sync_is_blocked( $a, $b ) {
    $row = bzj_connections_sync_get_relationship( $a, $b );

    if ( ! $row ) {
        return false;
    }

    return (int) $row->block_a_to_b === 1 || (int) $row->block_b_to_a === 1;
}

/**
 * Check if one user blocks another (directional).
 */
function bzj_connections_sync_is_directionally_blocked( $blocker, $blocked ) {
    $pair = bzj_connections_sync_pair( $blocker, $blocked );

    if ( ! $pair ) {
        return false;
    }

    $row = bzj_connections_sync_get_relationship( $blocker, $blocked );

    if ( ! $row ) {
        return false;
    }

    // Determine which side the blocker is on
    if ( $pair['user_a'] === absint( $blocker ) ) {
        return (int) $row->block_a_to_b === 1;
    }

    return (int) $row->block_b_to_a === 1;
}

// ==================== EVENT CREATION ====================

function bzj_connections_sync_create_event(
    $uuid,
    $origin,
    $event_type,
    $operation,
    $actor,
    $target,
    $state,
    $payload
) {
    global $wpdb;

    $table = bzj_connections_sync_events_table();

    // Idempotency check: return existing event ID if UUID already processed
    $existing = $wpdb->get_var(
        $wpdb->prepare(
            "SELECT id FROM {$table} WHERE event_uuid = %s LIMIT 1",
            $uuid
        )
    );

    if ( $existing ) {
        return absint( $existing );
    }

    $now = current_time( 'mysql', true );

    $ok = $wpdb->insert(
        $table,
        [
            'event_uuid'         => $uuid,
            'origin'             => sanitize_key( $origin ),
            'event_type'         => sanitize_key( $event_type ),
            'operation'          => sanitize_key( $operation ),
            'actor_wp_id'        => absint( $actor ),
            'target_wp_id'       => absint( $target ),
            'relationship_state' => sanitize_key( $state ),
            'payload'            => wp_json_encode( $payload ),
            'status'             => BZJ_EVENT_STATUS_PENDING,
            'attempts'           => 0,
            'next_attempt_at'    => $now,
            'created_at'         => $now,
        ],
        [ '%s', '%s', '%s', '%s', '%d', '%d', '%s', '%s', '%s', '%d', '%s', '%s' ]
    );

    if ( ! $ok ) {
        bzj_connections_sync_log(
            'Event creation failed',
            [
                'error' => $wpdb->last_error,
                'uuid'  => $uuid,
            ]
        );
        return false;
    }

    return $wpdb->insert_id;
}

// ==================== CANONICAL STATE WRITER ====================

function bzj_connections_sync_set_relationship(
    $actor,
    $target,
    $relationship_state,
    $options = []
) {
    global $wpdb;

    $actor  = absint( $actor );
    $target = absint( $target );

    $pair = bzj_connections_sync_pair( $actor, $target );

    if ( ! $pair ) {
        return false;
    }

    $relationship_state = sanitize_key( $relationship_state );

    if ( ! in_array(
        $relationship_state,
        [ BZJ_REL_NONE, BZJ_REL_REQUESTED, BZJ_REL_CONNECTED ],
        true
    ) ) {
        return false;
    }

    $lock = bzj_connections_sync_lock( $actor, $target );

    if ( ! $lock ) {
        bzj_connections_sync_log(
            'Cannot acquire lock',
            [ 'actor' => $actor, 'target' => $target ]
        );
        return false;
    }

    try {

        $existing = bzj_connections_sync_get_relationship( $actor, $target );

        $now = current_time( 'mysql', true );

        $block_a_to_b = isset( $options['block_a_to_b'] )
            ? (int) $options['block_a_to_b']
            : ( $existing ? (int) $existing->block_a_to_b : 0 );

        $block_b_to_a = isset( $options['block_b_to_a'] )
            ? (int) $options['block_b_to_a']
            : ( $existing ? (int) $existing->block_b_to_a : 0 );

        $requested_by = isset( $options['requested_by'] )
            ? absint( $options['requested_by'] )
            : ( $existing ? absint( $existing->requested_by ) : 0 );

        $event_uuid = ! empty( $options['event_uuid'] )
            ? sanitize_text_field( $options['event_uuid'] )
            : bzj_connections_sync_uuid();

        $data = [
            'relationship_state' => $relationship_state,
            'block_a_to_b'       => $block_a_to_b,
            'block_b_to_a'       => $block_b_to_a,
            'requested_by'       => $requested_by ?: null,
            'updated_at'         => $now,
            'last_event_uuid'    => $event_uuid,
        ];

        if ( $existing ) {

            $wpdb->update(
                bzj_connections_sync_relationships_table(),
                $data,
                [
                    'user_a' => $pair['user_a'],
                    'user_b' => $pair['user_b'],
                ],
                [ '%s', '%s', '%s', '%d', '%s', '%s' ],
                [ '%d', '%d' ]
            );

        } else {

            $data['user_a']    = $pair['user_a'];
            $data['user_b']    = $pair['user_b'];
            $data['created_at'] = $now;

            $wpdb->insert(
                bzj_connections_sync_relationships_table(),
                $data,
                [ '%d', '%d', '%s', '%d', '%d', '%d', '%s', '%s' ]
            );
        }

        bzj_connections_sync_log(
            'Canonical state updated',
            [
                'actor'               => $actor,
                'target'              => $target,
                'state'               => $relationship_state,
                'event_uuid'          => $event_uuid,
            ]
        );

        return true;

    } finally {
        bzj_connections_sync_unlock( $lock );
    }
}

// ==================== BLOCK OPERATIONS ====================

function bzj_connections_sync_set_block( $blocker, $blocked, $is_block = true ) {
    global $wpdb;

    $blocker = absint( $blocker );
    $blocked = absint( $blocked );

    $pair = bzj_connections_sync_pair( $blocker, $blocked );

    if ( ! $pair ) {
        return false;
    }

    $lock = bzj_connections_sync_lock( $blocker, $blocked );

    if ( ! $lock ) {
        bzj_connections_sync_log(
            'Cannot acquire lock for block operation',
            [ 'blocker' => $blocker, 'blocked' => $blocked ]
        );
        return false;
    }

    try {

        $existing = bzj_connections_sync_get_relationship( $blocker, $blocked );
        $now = current_time( 'mysql', true );

        // Determine which side is the blocker
        $is_a_blocking_b = ( $pair['user_a'] === $blocker );

        if ( $is_block ) {
            // When blocking, always remove any relationship (BLOCKS DOMINATE)
            $relationship_state = BZJ_REL_NONE;
        } else {
            // When unblocking, preserve the relationship state (but don't restore old relationship)
            $relationship_state = $existing ? $existing->relationship_state : BZJ_REL_NONE;
        }

        $block_a_to_b = isset( $existing->block_a_to_b )
            ? (int) $existing->block_a_to_b
            : 0;

        $block_b_to_a = isset( $existing->block_b_to_a )
            ? (int) $existing->block_b_to_a
            : 0;

        if ( $is_a_blocking_b ) {
            $block_a_to_b = $is_block ? 1 : 0;
        } else {
            $block_b_to_a = $is_block ? 1 : 0;
        }

        $data = [
            'relationship_state' => $relationship_state,
            'block_a_to_b'       => $block_a_to_b,
            'block_b_to_a'       => $block_b_to_a,
            'updated_at'         => $now,
        ];

        if ( $existing ) {
            $wpdb->update(
                bzj_connections_sync_relationships_table(),
                $data,
                [
                    'user_a' => $pair['user_a'],
                    'user_b' => $pair['user_b'],
                ],
                [ '%s', '%d', '%d', '%s' ],
                [ '%d', '%d' ]
            );
        } else {
            $data['user_a']    = $pair['user_a'];
            $data['user_b']    = $pair['user_b'];
            $data['created_at'] = $now;

            $wpdb->insert(
                bzj_connections_sync_relationships_table(),
                $data,
                [ '%d', '%d', '%s', '%d', '%d', '%s', '%s' ]
            );
        }

        bzj_connections_sync_log(
            'Block state updated',
            [
                'blocker'   => $blocker,
                'blocked'   => $blocked,
                'is_block'  => $is_block,
            ]
        );

        return true;

    } finally {
        bzj_connections_sync_unlock( $lock );
    }
}

/**
 * Fires when a new connection request is initiated.
 * Creates a canonical REQUESTED state.
 */
add_action( 'friends_friendship_requested', function( $friendship_id, $initiator_id, $friend_id ) {
    bzj_connections_sync_set_relationship(
        $initiator_id,
        $friend_id,
        BZJ_REL_REQUESTED,
        [
            'requested_by' => $initiator_id,
        ]
    );

    bzj_connections_sync_create_event(
        bzj_connections_sync_uuid(),
        BZJ_ORIGIN_WORDPRESS,
        BZJ_EVENT_TYPE_RELATIONSHIP,
        BZJ_OP_REQUEST,
        $initiator_id,
        $friend_id,
        BZJ_REL_REQUESTED,
        []
    );

    bzj_connections_sync_enqueue_reconciliation(
        $initiator_id,
        $friend_id,
        BZJ_REL_REQUESTED
    );

}, 10, 3 );

/**
 * Fires when a connection request is accepted.
 * Moves to CONNECTED state.
 */
add_action( 'friends_friendship_accepted', function( $friendship_id, $initiator_id, $friend_id ) {
    bzj_connections_sync_set_relationship(
        $initiator_id,
        $friend_id,
        BZJ_REL_CONNECTED
    );

    bzj_connections_sync_create_event(
        bzj_connections_sync_uuid(),
        BZJ_ORIGIN_WORDPRESS,
        BZJ_EVENT_TYPE_RELATIONSHIP,
        BZJ_OP_ACCEPT,
        $friend_id, // The acceptor
        $initiator_id, // The original requester
        BZJ_REL_CONNECTED,
        []
    );

    bzj_connections_sync_enqueue_reconciliation(
        $initiator_id,
        $friend_id,
        BZJ_REL_CONNECTED
    );

}, 10, 3 );

/**

 * Sets block direction and removes any relationship (BLOCKS DOMINATE).

        BZJ_EVENT_TYPE_BLOCK,
        BZJ_OP_BLOCK,
        $blocker_id,
        $blocked_id,
        BZJ_REL_NONE,
        []
    );

    bzj_connections_sync_enqueue_reconciliation(
        $blocker_id,
        $blocked_id,
        BZJ_REL_NONE
    );

}, 10, 2 );

// ==================== EXTERNAL REPORTING ENDPOINT ====================

function bzj_connections_sync_register_rest_endpoints() {
    register_rest_route(
        'bzj/v3',
        '/relationship',
        [
            'methods'             => 'POST',
            'callback'            => 'bzj_connections_sync_handle_external_report',
            'permission_callback' => '__return_true',
        ]
    );
}

function bzj_connections_sync_handle_external_report( $request ) {
    $body = $request->get_body();
    $headers = $request->get_headers();

    $timestamp = isset( $headers['x-bzj-timestamp'] )
        ? sanitize_text_field( $headers['x-bzj-timestamp'][0] )
        : '';

    $signature = isset( $headers['x-bzj-signature'] )
        ? sanitize_text_field( $headers['x-bzj-signature'][0] )
        : '';

    // Validate HMAC signature
    $secret = function_exists( 'bzj_get_sso_secret' )
        ? bzj_get_sso_secret()
        : getenv( 'BUZZ_SSO_SECRET' );

    if ( ! $secret ) {
        bzj_connections_sync_log(
            'External report rejected: missing secret',
            []
        );

        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Missing configuration' ],
            403
        );
    }

    $expected_signature = hash_hmac(
        'sha256',
        $timestamp . '.' . $body,
        $secret
    );

    if ( ! hash_equals( $signature, $expected_signature ) ) {
        bzj_connections_sync_log(
            'External report rejected: signature mismatch',
            [ 'origin' => sanitize_key( $_POST['origin'] ?? 'unknown' ) ]
        );

        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Unauthorized' ],
            401
        );
    }

    // Timestamp within 5 minutes
    $now = time();
    if ( abs( $now - absint( $timestamp ) ) > 300 ) {
        bzj_connections_sync_log(
            'External report rejected: stale timestamp',
            [ 'age_seconds' => abs( $now - absint( $timestamp ) ) ]
        );

        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Request too old' ],
            401
        );
    }

    $payload = json_decode( $body, true );

    if ( ! is_array( $payload ) ) {
        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Invalid JSON' ],
            400
        );
    }

    $origin = sanitize_key( $payload['origin'] ?? '' );
    $origin_user_id = absint( $payload['origin_user_id'] ?? 0 );
    $target_user_id = absint( $payload['target_user_id'] ?? 0 );
    $operation = sanitize_key( $payload['operation'] ?? '' );
    $event_uuid = sanitize_text_field( $payload['event_uuid'] ?? '' );

    if ( ! $origin || ! $origin_user_id || ! $target_user_id || ! $operation || ! $event_uuid ) {
        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Missing required fields' ],
            400
        );
    }

    // Only accept from known platforms
    if ( ! in_array( $origin, [ BZJ_ORIGIN_STREAMS, BZJ_ORIGIN_SOCIALS ], true ) ) {
        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Unknown origin' ],
            400
        );
    }

    // Resolve external user IDs to WordPress IDs
    $actor_wp_id = bzj_connections_sync_resolve_wp_user_id(
        $origin_user_id,
        $origin
    );

    $target_wp_id = bzj_connections_sync_resolve_wp_user_id(
        $target_user_id,
        $origin
    );

    if ( ! $actor_wp_id || ! $target_wp_id ) {
        bzj_connections_sync_log(
            'External report rejected: cannot resolve user IDs',
            [
                'origin'              => $origin,
                'origin_user_id'      => $origin_user_id,
                'target_user_id'      => $target_user_id,
            ]
        );

        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'User not found' ],
            404
        );
    }

    // Process the operation
    $result = bzj_connections_sync_process_external_operation(
        $origin,
        $operation,
        $actor_wp_id,
        $target_wp_id,
        $event_uuid
    );

    if ( ! $result ) {
        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'Operation failed' ],
            500
        );
    }

    bzj_connections_sync_log(
        'External report processed',
        [
            'origin'    => $origin,
            'operation' => $operation,
            'actor_wp'  => $actor_wp_id,
            'target_wp' => $target_wp_id,
        ]
    );

    return new WP_REST_Response(
        [ 'success' => true ],
        200
    );
}

/**
 * Resolve external platform user ID to WordPress user ID.
 */
function bzj_connections_sync_resolve_wp_user_id( $external_id, $origin ) {
    $external_id = absint( $external_id );

    if ( ! $external_id ) {
        return false;
    }

    if ( $origin === BZJ_ORIGIN_STREAMS ) {
        // WoWonder user ID → WordPress user ID via wo_user_id usermeta
        $wp_user = get_users( [
            'meta_key'     => 'wo_user_id',
            'meta_value'   => $external_id,
            'number'       => 1,
            'fields'       => 'ID',
        ] );

        return ! empty( $wp_user ) ? absint( $wp_user[0] ) : false;
    }

    if ( $origin === BZJ_ORIGIN_SOCIALS ) {
        // QuickDate user ID → WordPress user ID via qd_user_id usermeta
        $wp_user = get_users( [
            'meta_key'     => 'qd_user_id',
            'meta_value'   => $external_id,
            'number'       => 1,
            'fields'       => 'ID',
        ] );

        return ! empty( $wp_user ) ? absint( $wp_user[0] ) : false;
    }

    return false;
}

/**
 * Process an external operation (Streams/Socials → WordPress).
 * NO internal flag set; hooks will fire normally.
 */
function bzj_connections_sync_process_external_operation(
    $origin,
    $operation,
    $actor_wp_id,
    $target_wp_id,
    $event_uuid
) {
    $actor_wp_id = absint( $actor_wp_id );
    $target_wp_id = absint( $target_wp_id );

    if ( ! $actor_wp_id || ! $target_wp_id || $actor_wp_id === $target_wp_id ) {
        return false;
    }

    // Log the external event
    $result = bzj_connections_sync_create_event(
        $event_uuid,
        $origin,
        BZJ_EVENT_TYPE_RELATIONSHIP,
        $operation,
        $actor_wp_id,
        $target_wp_id,
        BZJ_REL_NONE,
        []
    );

    if ( ! $result ) {
        return false;
    }

    switch ( $operation ) {

        case BZJ_OP_REQUEST:
            // External follow → BuddyBoss request (respects restrictions)
            $friendship_id = friends_add_friend(
                $actor_wp_id,
                $target_wp_id,
                false // Do not force accept
            );
            return ! empty( $friendship_id );

        case BZJ_OP_ACCEPT:
            // External accept → BuddyBoss accept
            // Find the actual friendship ID
            $friendship = BP_Friends_Friendship::get_friendship(
                $actor_wp_id,
                $target_wp_id
            );

            if ( ! $friendship ) {
                return false;
            }

            return friends_accept_friendship( $friendship->id );

        case BZJ_OP_REMOVE:
            // External remove → BuddyBoss remove
            return friends_remove_friend(
                $actor_wp_id,
                $target_wp_id
            );

        case BZJ_OP_BLOCK:
            // External block → BuddyBoss block via filter adapter
            $handled = apply_filters(
                'bzj_connections_sync_external_block',
                false,
                $actor_wp_id,
                $target_wp_id,
                $origin
            );

            if ( $handled ) {
                return true;
            }

            // If no adapter, log and fail
            bzj_connections_sync_log(
                'External block rejected: no adapter available',
                [
                    'origin' => $origin,
                    'blocker' => $actor_wp_id,
                    'blocked' => $target_wp_id,
                ]
            );

            return false;

        case BZJ_OP_UNBLOCK:
            // External unblock → BuddyBoss unblock via filter adapter
            $handled = apply_filters(
                'bzj_connections_sync_external_unblock',
                false,
                $actor_wp_id,
                $target_wp_id,
                $origin
            );

            if ( $handled ) {
                return true;
            }

            // If no adapter, log and fail
            bzj_connections_sync_log(
                'External unblock rejected: no adapter available',
                [
                    'origin'   => $origin,
                    'blocker'  => $actor_wp_id,
                    'blocked'  => $target_wp_id,
                ]
            );

            return false;

        default:
            return false;
    }
}

function bzj_connections_sync_enqueue_reconciliation( $user_a, $user_b, $state ) {
    $pair = bzj_connections_sync_pair( $user_a, $user_b );

    if ( ! $pair ) {
        return;
    }

    // Get the canonical relationship
    $relationship = bzj_connections_sync_get_relationship( $user_a, $user_b );

    if ( ! $relationship ) {
        return;
    }

    // Queue events for external platforms to reconcile
    bzj_connections_sync_create_event(
        bzj_connections_sync_uuid(),
        BZJ_ORIGIN_WORDPRESS,
        BZJ_EVENT_TYPE_RELATIONSHIP,
        'reconcile_streams',
        $pair['user_a'],
        $pair['user_b'],
        $state,
        [
            'block_a_to_b' => $relationship->block_a_to_b,
            'block_b_to_a' => $relationship->block_b_to_a,
        ]
    );

    bzj_connections_sync_create_event(
        bzj_connections_sync_uuid(),
        BZJ_ORIGIN_WORDPRESS,
        BZJ_EVENT_TYPE_RELATIONSHIP,
        'reconcile_socials',
        $pair['user_a'],
        $pair['user_b'],
        $state,
        [
            'block_a_to_b' => $relationship->block_a_to_b,
            'block_b_to_a' => $relationship->block_b_to_a,
        ]
    );
}

/**
 * Reconcile Streams (WoWonder) follower/block tables with canonical state.
 */
function bzj_connections_sync_reconcile_streams( $user_a, $user_b, $relationship ) {
    global $wpdb;

    // Get external database connections
    $streams_conn = function_exists( 'get_wowonder_db' )
        ? get_wowonder_db()
        : null;

    if ( ! $streams_conn || $streams_conn->connect_errno ) {
        bzj_connections_sync_log(
            'Streams DB connection failed',
            [ 'error' => $streams_conn ? $streams_conn->connect_error : 'No function' ]
        );
        return false;
    }

    // WoWonder follower table
    $followers_table = getenv( 'BUZZ_STREAMS_FOLLOWERS_TABLE' ) ?: 'Wo_Followers';
    $blocks_table = getenv( 'BUZZ_STREAMS_BLOCKS_TABLE' ) ?: 'Wo_Blocks';

    // Resolve WoWonder IDs
    $wo_user_a = get_user_meta( $user_a, 'wo_user_id', true );
    $wo_user_b = get_user_meta( $user_b, 'wo_user_id', true );

    if ( ! $wo_user_a || ! $wo_user_b ) {
        bzj_connections_sync_log(
            'Streams reconciliation skipped: missing wo_user_id',
            [ 'user_a' => $user_a, 'user_b' => $user_b ]
        );
        return false;
    }

    $wo_user_a = absint( $wo_user_a );
    $wo_user_b = absint( $wo_user_b );

    // Determine target relationship state in Streams
    $should_be_connected = (
        (int) $relationship->relationship_state === BZJ_REL_CONNECTED
    );

    $is_blocked = (
        (int) $relationship->block_a_to_b === 1 ||
        (int) $relationship->block_b_to_a === 1
    );

    try {

        // Remove any conflicting blocks
        if ( $is_blocked ) {
            // A blocks B
            if ( (int) $relationship->block_a_to_b === 1 ) {
                $streams_conn->query(
                    $streams_conn->prepare(
                        "DELETE FROM `{$blocks_table}` WHERE blocker = %d AND blocked = %d",
                        $wo_user_a,
                        $wo_user_b
                    )
                );
            }

            // B blocks A
            if ( (int) $relationship->block_b_to_a === 1 ) {
                $streams_conn->query(
                    $streams_conn->prepare(
                        "DELETE FROM `{$blocks_table}` WHERE blocker = %d AND blocked = %d",
                        $wo_user_b,
                        $wo_user_a
                    )
                );
            }

            // Remove all follower relationships (blocks dominate)
            $streams_conn->query(
                $streams_conn->prepare(
                    "DELETE FROM `{$followers_table}` WHERE (following_id = %d AND follower_id = %d) OR (following_id = %d AND follower_id = %d)",
                    $wo_user_a,
                    $wo_user_b,
                    $wo_user_b,
                    $wo_user_a
                )
            );

        } elseif ( $should_be_connected ) {
            // Both directions should be following (mutual connection)
            $streams_conn->query(
                $streams_conn->prepare(
                    "INSERT IGNORE INTO `{$followers_table}` (following_id, follower_id, active, created_at) VALUES (%d, %d, 1, NOW())",
                    $wo_user_a,
                    $wo_user_b
                )
            );

            $streams_conn->query(
                $streams_conn->prepare(
                    "INSERT IGNORE INTO `{$followers_table}` (following_id, follower_id, active, created_at) VALUES (%d, %d, 1, NOW())",
                    $wo_user_b,
                    $wo_user_a
                )
            );

            // Ensure no blocks
            $streams_conn->query(
                $streams_conn->prepare(
                    "DELETE FROM `{$blocks_table}` WHERE (blocker = %d AND blocked = %d) OR (blocker = %d AND blocked = %d)",
                    $wo_user_a,
                    $wo_user_b,
                    $wo_user_b,
                    $wo_user_a
                )
            );

        } else {
            // Remove all relationships
            $streams_conn->query(
                $streams_conn->prepare(
                    "DELETE FROM `{$followers_table}` WHERE (following_id = %d AND follower_id = %d) OR (following_id = %d AND follower_id = %d)",
                    $wo_user_a,
                    $wo_user_b,
                    $wo_user_b,
                    $wo_user_a
                )
            );

            // Ensure no blocks
            $streams_conn->query(
                $streams_conn->prepare(
                    "DELETE FROM `{$blocks_table}` WHERE (blocker = %d AND blocked = %d) OR (blocker = %d AND blocked = %d)",
                    $wo_user_a,
                    $wo_user_b,
                    $wo_user_b,
                    $wo_user_a
                )
            );
        }

        return true;

    } catch ( Exception $e ) {
        bzj_connections_sync_log(
            'Streams reconciliation error',
            [ 'error' => $e->getMessage() ]
        );
        return false;
    }
}

/**
 * Process queued reconciliation events with exponential backoff retry.
 */
function bzj_connections_sync_process_retries() {
    global $wpdb;

    $table = bzj_connections_sync_events_table();
    $now = current_time( 'mysql', true );

    // Get pending and retry events ready for processing
    $events = $wpdb->get_results(
        $wpdb->prepare(
            "SELECT * FROM {$table}
            WHERE status IN (%s, %s)
            AND next_attempt_at <= %s
            AND attempts < %d
            ORDER BY created_at ASC
            LIMIT 25",
            BZJ_EVENT_STATUS_PENDING,
            BZJ_EVENT_STATUS_RETRY,
            $now,
            BZJ_MAX_RETRIES
        )
    );

    if ( empty( $events ) ) {
        return;
    }

    foreach ( $events as $event ) {
        $event_id = absint( $event->id );
        $operation = sanitize_key( $event->operation );
        $actor_wp_id = absint( $event->actor_wp_id );
        $target_wp_id = absint( $event->target_wp_id );

        // Only reconcile events
        if ( ! in_array( $operation, [ 'reconcile_streams', 'reconcile_socials' ], true ) ) {
            $wpdb->update(
                $table,
                [
                    'status'       => BZJ_EVENT_STATUS_PROCESSED,
                    'processed_at' => $now,
                ],
                [ 'id' => $event_id ],
                [ '%s', '%s' ],
                [ '%d' ]
            );
            continue;
        }

        $relationship = bzj_connections_sync_get_relationship( $actor_wp_id, $target_wp_id );

        if ( ! $relationship ) {
            $wpdb->update(
                $table,
                [
                    'status'       => BZJ_EVENT_STATUS_FAILED,
                    'processed_at' => $now,
                    'last_error'   => 'Relationship not found',
                ],
                [ 'id' => $event_id ],
                [ '%s', '%s', '%s' ],
                [ '%d' ]
            );
            continue;
        }

        $success = false;

        if ( $operation === 'reconcile_streams' ) {
            $success = bzj_connections_sync_reconcile_streams(
                $actor_wp_id,
                $target_wp_id,
                $relationship
            );
        } elseif ( $operation === 'reconcile_socials' ) {
            $success = bzj_connections_sync_reconcile_socials(
                $actor_wp_id,
                $target_wp_id,
                $relationship
            );
        }

        if ( $success ) {
            $wpdb->update(
                $table,
                [
                    'status'       => BZJ_EVENT_STATUS_PROCESSED,
                    'processed_at' => $now,
                ],
                [ 'id' => $event_id ],
                [ '%s', '%s' ],
                [ '%d' ]
            );
        } else {
            // Schedule exponential backoff retry
            $attempts = absint( $event->attempts ) + 1;
            $backoff_seconds = BZJ_BASE_RETRY_SECONDS * pow( 2, $attempts - 1 );

            $next_attempt = date(
                'Y-m-d H:i:s',
                strtotime( "+{$backoff_seconds} seconds", strtotime( $now ) )
            );

            $wpdb->update(
                $table,
                [
                    'status'           => BZJ_EVENT_STATUS_RETRY,
                    'attempts'         => $attempts,
                    'next_attempt_at'  => $next_attempt,
                    'last_error'       => 'Reconciliation failed; will retry',
                ],
                [ 'id' => $event_id ],
                [ '%s', '%d', '%s', '%s' ],
                [ '%d' ]
            );

            bzj_connections_sync_log(
                'Event queued for retry',
                [
                    'event_id'   => $event_id,
                    'attempts'   => $attempts,
                    'backoff_s'  => $backoff_seconds,
                ]
            );
        }
    }
}
```

### 1. `/shared/wwqd_bridge.php`

Add this helper function after the existing database/connection helpers:

```php
/**
 * Report a user action from an external platform to WordPress canonical state.
 * Called after local external DB operation succeeds.
 * Uses HMAC-SHA256 signed REST API.
 *
 * @param string $origin 'streams' or 'socials'
 * @param int $origin_user_id External platform user ID (actor)
 * @param int $target_user_id External platform user ID (target)
 * @param string $operation 'request', 'accept', 'remove', 'block', 'unblock'
 * @param string $event_uuid Optional UUID for idempotency
 * @return bool
 */
if ( ! function_exists( 'bzj_connections_sync_report_to_wordpress' ) ) {
    function bzj_connections_sync_report_to_wordpress(
        $origin,
        $origin_user_id,
        $target_user_id,
        $operation,
        $event_uuid = ''
    ) {
        $origin = preg_replace( '/[^a-z_]/', '', strtolower( (string) $origin ) );
        $operation = preg_replace( '/[^a-z_]/', '', strtolower( (string) $operation ) );

        if ( ! in_array( $origin, array( 'streams', 'socials' ), true ) ) {
            return false;
        }

        if ( ! in_array( $operation, array( 'request', 'accept', 'remove', 'block', 'unblock' ), true ) ) {
            return false;
        }

        $secret = function_exists( 'bzj_get_sso_secret' )
            ? bzj_get_sso_secret()
            : getenv( 'BUZZ_SSO_SECRET' );

        $base = getenv( 'WP_BASE_SITE_URL' );

        if ( ! $secret || ! $base ) {
            return false;
        }

        if ( ! $event_uuid ) {
            $event_uuid = function_exists( 'wp_generate_uuid4' )
                ? wp_generate_uuid4()
                : sprintf(
                    '%s-%s-%s-%s-%s',
                    bin2hex( random_bytes( 4 ) ),
                    bin2hex( random_bytes( 2 ) ),
                    bin2hex( random_bytes( 2 ) ),
                    bin2hex( random_bytes( 2 ) ),
                    bin2hex( random_bytes( 6 ) )
                );
        }

        $payload = array(
            'origin'         => $origin,
            'origin_user_id' => absint( $origin_user_id ),
            'target_user_id' => absint( $target_user_id ),
            'operation'      => $operation,
            'event_uuid'     => $event_uuid,
        );

        $body = json_encode( $payload, JSON_UNESCAPED_SLASHES );
        $timestamp = (string) time();
        $signature = hash_hmac( 'sha256', $timestamp . '.' . $body, $secret );

        $url = rtrim( $base, '/' ) . '/wp-json/bzj/v3/relationship';

        $ch = curl_init( $url );
        curl_setopt_array( $ch, array(
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => $body,
            CURLOPT_HTTPHEADER     => array(
                'Content-Type: application/json',
                'X-BZJ-Timestamp: ' . $timestamp,
                'X-BZJ-Signature: ' . $signature,
            ),
            CURLOPT_CONNECTTIMEOUT => 5,
            CURLOPT_TIMEOUT        => 12,
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
        ) );

        $response = curl_exec( $ch );
        $errno = curl_errno( $ch );
        $http = (int) curl_getinfo( $ch, CURLINFO_HTTP_CODE );
        curl_close( $ch );

        if ( $errno || $http < 200 || $http >= 300 ) {
            if ( function_exists( 'wwqd_debug' ) ) {
                wwqd_debug( 'connections_sync_report_failed', array(
                    'origin'    => $origin,
                    'operation' => $operation,
                    'http'      => $http,
                    'errno'     => $errno,
                ) );
            }
            return false;
        }

        $decoded = json_decode( $response, true );
        return ! empty( $decoded['success'] );
    }
}
```

## PART 5: CRITICAL NOTES

1. **WordPress is canonical** — All relationship truth flows from BuddyBoss hooks
2. **Blocks dominate** — Any block removes follower rows on all platforms
3. **Unblock ≠ restore** — Unblocking does not restore old relationships
4. **No wp-load in Streams/Socials** — External platforms remain standalone
5. **External platform changes first** — Local DB operation succeeds before reporting to WordPress
6. **Report failure is logged and retried** — Not a fatal error
7. **BuddyBoss policy remains authoritative** — External requests respect BuddyBoss restrictions
8. **Idempotent by UUID** — Same operation UUID will never double-process
9. **Five-minute retry interval** — Handles transient external platform outages
10. **Audit trail** — All events logged to `/wp-content/bzj-connections-sync.log`
