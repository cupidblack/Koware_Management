**Rule 1: WordPress is source of truth**
- If a sync fails (e.g., user not found in Streams), log and alert but do NOT rollback WordPress change.

- All sync operations check if the relationship already exists before inserting.
- Duplicate API calls do not cause errors.

**Rule 3: Blocking Priority**
- If User A blocks User B, automatically unfriend (if they were friends).
- If User A unfriends User B, do NOT automatically unblock.

**Rule 4: Cascade Enforcement**

/**
 * Generate HMAC signature for API callbacks
 *
 * @param array $payload Data to sign
 * @param string $secret Secret key
 * @return string HMAC-SHA256 signature
 */
function bzj_generate_callback_signature( $payload, $secret = null ) {
    if ( $secret === null ) {
        $secret = bzj_get_sso_secret();
    }

    $json = wp_json_encode( $payload );
    return hash_hmac( 'sha256', $json, $secret );
}

/**
 * Verify HMAC signature for API callbacks
 *
 * @param array $payload Data to verify
 * @param string $signature Provided signature
 * @param string $secret Secret key
 * @return bool True if signature is valid
 */
function bzj_verify_callback_signature( $payload, $signature, $secret = null ) {
    if ( $secret === null ) {
        $secret = bzj_get_sso_secret();
    }

    $expected = bzj_generate_callback_signature( $payload, $secret );
    return hash_equals( $expected, $signature );
}

    $qd_initiator = bzj_get_qd_user_id( $initiator_id );
    $qd_friend = bzj_get_qd_user_id( $friend_id );

    if ( ! $qd_initiator || ! $qd_friend ) {
        bzj_log_sync_event(
            'SYNC_FRIEND_REQUEST_SKIPPED',
            $initiator_id,
            $friend_id,
            'socials',
            'skipped',
            'QuickDate user IDs not found'
        );
        return;
    }

    // Check if already exists
    $existing_status = bzj_qd_friendship_status( $qd_initiator, $qd_friend );
    if ( $existing_status ) {
        bzj_log_sync_event(
            'SYNC_FRIEND_REQUEST_EXISTS',
            $initiator_id,
            $friend_id,
            'socials',
            'skipped',
            'Friendship already exists with status: ' . $existing_status
        );
        return;
    }

    // Insert into QuickDate users_friends table
    $conn = get_qd_db_conn();
    if ( ! $conn ) {
        bzj_log_sync_event(
            'SYNC_FRIEND_REQUEST_FAILED',
            $initiator_id,
            $friend_id,
            'socials',
            'error',
            'Database connection failed'
        );
        return;
    }

    $insert_query = "INSERT INTO `" . QD_DB_NAME . "`.`users_friends` 
                     (from_id, to_id, status, date_created) 
                     VALUES (%d, %d, 'pending', NOW())";

    $query = $conn->prepare( $insert_query, $qd_initiator, $qd_friend );
    
    if ( $conn->query( $query ) ) {
        bzj_log_sync_event(
            'SYNC_FRIEND_REQUEST',
            $initiator_id,
            $friend_id,
            'socials',
            'success',
            'Connection request created'
        );
    } else {
        bzj_log_sync_event(
            'SYNC_FRIEND_REQUEST_FAILED',
            $initiator_id,
            $friend_id,
            'socials',
            'error',
            $conn->error
        );
    }
}
add_action( 'friends_friendship_requested', 'bzj_sync_connection_request', 10, 3 );

/**
 * A6: Block user (with auto-unfriend)
 * Hook: bp_moderation_block_created (custom hook - may need verification)
 *
 * @param int $block_id Block ID
 * @param object $block_object Block object with user_id and item_id
 */
function bzj_sync_user_blocked( $block_id, $block_object ) {
    $blocker_id = $block_object->user_id ?? 0;
    $blocked_id = $block_object->item_id ?? 0;

    if ( ! $blocker_id || ! $blocked_id ) {
        bzj_log_sync_event(
            'SYNC_BLOCK_INVALID',
            $blocker_id,
            $blocked_id,
            'both',
            'error',
            'Invalid block object structure'
        );
        return;
    }

    // Step 1: Auto-unfriend if they are friends
    if ( function_exists( 'friends_check_friendship_status' ) ) {
        $status = friends_check_friendship_status( $blocker_id, $blocked_id );
        
        if ( 'is_friend' === $status ) {
            friends_remove_friend( $blocker_id, $blocked_id );
            bzj_log_sync_event(
                'SYNC_BLOCK_UNFRIEND',
                $blocker_id,
                $blocked_id,
                'both',
                'success',
                'Auto-unfriended on block'
            );
        }
    }

    // Step 2: Get platform user IDs
    $wo_blocker = bzj_get_wo_user_id( $blocker_id );
    $wo_blocked = bzj_get_wo_user_id( $blocked_id );
    $qd_blocker = bzj_get_qd_user_id( $blocker_id );
    $qd_blocked = bzj_get_qd_user_id( $blocked_id );

    // Step 3: Block in WoWonder
    if ( $wo_blocker && $wo_blocked ) {
        // Check if already blocked
        if ( ! bzj_wo_is_blocked( $wo_blocker, $wo_blocked ) ) {
            $conn = get_wowonder_db();
            if ( $conn ) {
                $insert_query = "INSERT INTO `" . WOWONDER_DB_NAME . "`.`Wo_BlockedUsers` 
                                 (user_id, blocked_user_id, date) 
                                 VALUES (%d, %d, NOW())";
                
                $query = $conn->prepare( $insert_query, $wo_blocker, $wo_blocked );
                
                if ( $conn->query( $query ) ) {
                    bzj_log_sync_event(
                        'SYNC_BLOCK_STREAMS',
                        $blocker_id,
                        $blocked_id,
                        'streams',
                        'success',
                        'User blocked in Streams'
                    );
                } else {
                    bzj_log_sync_event(
                        'SYNC_BLOCK_STREAMS_FAILED',
                        $blocker_id,
                        $blocked_id,
                        'streams',
                        'error',
                        $conn->error
                    );
                }
            }
        }
    }

    // Step 4: Block in QuickDate
    if ( $qd_blocker && $qd_blocked ) {
        // Check if already blocked
        if ( ! bzj_qd_is_blocked( $qd_blocker, $qd_blocked ) ) {
            $conn = get_qd_db_conn();
            if ( $conn ) {
                $insert_query = "INSERT INTO `" . QD_DB_NAME . "`.`users_blocked` 
                                 (user_id, blocked_user_id, date) 
                                 VALUES (%d, %d, NOW())";
                
                $query = $conn->prepare( $insert_query, $qd_blocker, $qd_blocked );
                
                if ( $conn->query( $query ) ) {
                    bzj_log_sync_event(
                        'SYNC_BLOCK_SOCIALS',
                        $blocker_id,
                        $blocked_id,
                        'socials',
                        'success',
                        'User blocked in Socials'
                    );
                } else {
                    bzj_log_sync_event(
                        'SYNC_BLOCK_SOCIALS_FAILED',
                        $blocker_id,
                        $blocked_id,
                        'socials',
                        'error',
                        $conn->error
                    );
                }
            }
        }
    }
}

/**

    // Connection Management Endpoint
    register_rest_route(

        '/connection-sync',

/**
 * Verify incoming connection sync request signature
 *
 * @return bool|WP_Error True if valid, WP_Error otherwise
 */
function bzj_verify_connection_sync_request() {
    $signature = isset( $_SERVER['HTTP_X_BZJ_SIGNATURE'] ) ? sanitize_text_field( wp_unslash( $_SERVER['HTTP_X_BZJ_SIGNATURE'] ) ) : '';
    
    if ( empty( $signature ) ) {
        return new WP_Error(
            'bzj_missing_signature',
            __( 'Missing signature header', 'bzj-connections-sync' ),
            array( 'status' => 401 )
        );
    }

    $body = file_get_contents( 'php://input' );
    $payload = json_decode( $body, true );

    if ( ! $payload ) {
        return new WP_Error(
            'bzj_invalid_payload',
            __( 'Invalid JSON payload', 'bzj-connections-sync' ),
            array( 'status' => 400 )
        );
    }

    if ( ! bzj_verify_callback_signature( $payload, $signature ) ) {
        return new WP_Error(
            'bzj_invalid_signature',
            __( 'Invalid signature', 'bzj-connections-sync' ),
            array( 'status' => 401 )
        );
    }

    return true;
}

/**
 * Handle incoming connection sync callback from Streams or Socials
 *
 * @param WP_REST_Request $request REST request object
 * @return WP_REST_Response
 */
function bzj_handle_connection_sync_callback( $request ) {
    $body = $request->get_body();
    $payload = json_decode( $body, true );

    if ( ! $payload || empty( $payload['action'] ) ) {
        return new WP_REST_Response(
            array(
                'success' => false,
                'message' => 'Invalid payload or missing action',
            ),
            400
        );
    }

    $action = sanitize_text_field( $payload['action'] );
    $origin_platform = sanitize_text_field( $payload['origin_platform'] ?? '' );
    $user_id = isset( $payload['user_id'] ) ? (int) $payload['user_id'] : 0;
    $target_id = isset( $payload['target_id'] ) ? (int) $payload['target_id'] : 0;

    // Map platform user IDs to WordPress user IDs
    $wp_user_id = false;
    $wp_target_id = false;

    if ( $origin_platform === 'streams' ) {
        $wp_user_id = bzj_get_wp_user_id_from_wo( $user_id );
        $wp_target_id = bzj_get_wp_user_id_from_wo( $target_id );
    } elseif ( $origin_platform === 'socials' ) {
        $wp_user_id = bzj_get_wp_user_id_from_qd( $user_id );
        $wp_target_id = bzj_get_wp_user_id_from_qd( $target_id );
    }

    if ( ! $wp_user_id || ! $wp_target_id ) {
        bzj_log_sync_event(
            'CALLBACK_USER_NOT_FOUND',
            $wp_user_id ?? 0,
            $wp_target_id ?? 0,
            $origin_platform,
            'error',
            'Failed to map platform user IDs to WordPress'
        );

        return new WP_REST_Response(
            array(
                'success' => false,
                'message' => 'User mapping failed',
            ),
            404
        );
    }

    // Handle different actions
    $result = false;

    switch ( $action ) {
        case 'follow':
            $result = bzj_callback_follow( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'unfollow':
            $result = bzj_callback_unfollow( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'block':
            $result = bzj_callback_block( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'unblock':
            $result = bzj_callback_unblock( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'friend_request':
            $result = bzj_callback_friend_request( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'friend_accept':
            $result = bzj_callback_friend_accept( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'friend_reject':
            $result = bzj_callback_friend_reject( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        case 'friend_delete':
            $result = bzj_callback_friend_delete( $wp_user_id, $wp_target_id, $origin_platform );
            break;

        default:
            return new WP_REST_Response(
                array(
                    'success' => false,
                    'message' => 'Unknown action: ' . $action,
                ),
                400
            );
    }

    $status_code = $result ? 200 : 500;

    return new WP_REST_Response(
        array(
            'success' => $result,
            'action' => $action,
            'timestamp' => current_time( 'mysql' ),
        ),
        $status_code
    );
}

function bzj_send_wp_callback( $action, $user_id, $target_id ) {
