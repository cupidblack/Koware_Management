        KEY updated_at (updated_at)

    CREATE TABLE IF NOT EXISTS {$evt_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

// ==================== UTILITIES ====================

function bzj_connections_sync_uuid() {
    return wp_generate_uuid4();
}

/**
 * Normalize user pair so user_a < user_b.
 * Prevents (A,B) and (B,A) from being stored as separate records.
 */
function bzj_connections_sync_pair( $a, $b ) {
    $a = absint( $a );
    $b = absint( $b );

    if ( ! $a || ! $b || $a === $b ) {
        return false;
    }

    if ( $a > $b ) {
        [ $a, $b ] = [ $b, $a ];
    }

    return [
        'user_a' => $a,
        'user_b' => $b,
    ];
}

/**
 * MySQL named lock for atomic operations.
 */
function bzj_connections_sync_lock( $a, $b, $timeout = 5 ) {
    global $wpdb;

    $pair = bzj_connections_sync_pair( $a, $b );

    if ( ! $pair ) {
        return false;
    }

    $lock_name = sprintf( 'bzj_rel_%d_%d', $pair['user_a'], $pair['user_b'] );

    $result = (int) $wpdb->get_var(
        $wpdb->prepare( 'SELECT GET_LOCK( %s, %d )', $lock_name, $timeout )
    );

    return 1 === $result ? $lock_name : false;
}

function bzj_connections_sync_unlock( $lock_name ) {
    global $wpdb;

    if ( ! $lock_name ) {
        return;
    }

    $wpdb->get_var(
        $wpdb->prepare( 'SELECT RELEASE_LOCK( %s )', $lock_name )
    );
}

// ==================== CANONICAL STATE READER ====================

function bzj_connections_sync_get_relationship( $a, $b ) {
    global $wpdb;

    $pair = bzj_connections_sync_pair( $a, $b );

    if ( ! $pair ) {
        return false;
    }

    return $wpdb->get_row(
        $wpdb->prepare(
            'SELECT * FROM ' . bzj_connections_sync_relationships_table() . '
            WHERE user_a = %d AND user_b = %d LIMIT 1',
            $pair['user_a'],
            $pair['user_b']
        )
    );
}

/**
 * Check if any block exists between users.
 */
function bzj_connections_sync_is_blocked( $a, $b ) {
    $row = bzj_connections_sync_get_relationship( $a, $b );

    if ( ! $row ) {
        return false;
    }

    return (int) $row->block_a_to_b === 1 || (int) $row->block_b_to_a === 1;
}

/**
 * Check if one user blocks another (directional).
 */
function bzj_connections_sync_is_directionally_blocked( $blocker, $blocked ) {
    $pair = bzj_connections_sync_pair( $blocker, $blocked );

    if ( ! $pair ) {
        return false;
    }

    $row = bzj_connections_sync_get_relationship( $blocker, $blocked );

    if ( ! $row ) {
        return false;
    }

    // Determine which side the blocker is on
    if ( $pair['user_a'] === absint( $blocker ) ) {
        return (int) $row->block_a_to_b === 1;
    }

    return (int) $row->block_b_to_a === 1;
}

// ==================== EVENT CREATION ====================

function bzj_connections_sync_create_event(
    $uuid,
    $origin,
    $event_type,
    $operation,
    $actor,
    $target,
    $state,
    $payload
) {
    global $wpdb;

    $table = bzj_connections_sync_events_table();

    // Idempotency check
    $existing = $wpdb->get_var(
        $wpdb->prepare(
            "SELECT id FROM {$table} WHERE event_uuid = %s LIMIT 1",
            $uuid
        )
    );

    if ( $existing ) {
        return absint( $existing );
    }

    $now = current_time( 'mysql', true );

    $ok = $wpdb->insert(
        $table,
        [
            'event_uuid'         => $uuid,
            'origin'             => sanitize_key( $origin ),
            'event_type'         => sanitize_key( $event_type ),
            'operation'          => sanitize_key( $operation ),
            'actor_wp_id'        => absint( $actor ),
            'target_wp_id'       => absint( $target ),
            'relationship_state' => sanitize_key( $state ),
            'payload'            => wp_json_encode( $payload ),
            'status'             => BZJ_EVENT_STATUS_PENDING,
            'attempts'           => 0,
            'next_attempt_at'    => $now,
            'created_at'         => $now,
        ],
        [ '%s', '%s', '%s', '%s', '%d', '%d', '%s', '%s', '%s', '%d', '%s', '%s' ]
    );

    if ( ! $ok ) {
        bzj_connections_sync_log(
            'Event creation failed',
            [
                'error' => $wpdb->last_error,
                'uuid'  => $uuid,
            ]
        );
        return false;
    }

    return $wpdb->insert_id;
}

// ==================== CANONICAL STATE WRITER ====================

function bzj_connections_sync_set_relationship(
    $actor,
    $target,
    $relationship_state,
    $options = []
) {
    global $wpdb;

    $actor  = absint( $actor );
    $target = absint( $target );

    $pair = bzj_connections_sync_pair( $actor, $target );

    if ( ! $pair ) {
        return false;
    }

    $relationship_state = sanitize_key( $relationship_state );

    if ( ! in_array(
        $relationship_state,
        [ BZJ_REL_NONE, BZJ_REL_REQUESTED, BZJ_REL_CONNECTED ],
        true
    ) ) {
        return false;
    }

    $lock = bzj_connections_sync_lock( $actor, $target );

    if ( ! $lock ) {
        bzj_connections_sync_log(
            'Cannot acquire lock',
            [ 'actor' => $actor, 'target' => $target ]
        );
        return false;
    }

    try {

        $existing = bzj_connections_sync_get_relationship( $actor, $target );

        $now = current_time( 'mysql', true );

        $block_a_to_b = isset( $options['block_a_to_b'] )
            ? (int) $options['block_a_to_b']
            : ( $existing ? (int) $existing->block_a_to_b : 0 );

        $block_b_to_a = isset( $options['block_b_to_a'] )
            ? (int) $options['block_b_to_a']
            : ( $existing ? (int) $existing->block_b_to_a : 0 );

        $requested_by = isset( $options['requested_by'] )
            ? absint( $options['requested_by'] )
            : ( $existing ? absint( $existing->requested_by ) : 0 );

        $event_uuid = ! empty( $options['event_uuid'] )
            ? sanitize_text_field( $options['event_uuid'] )
            : bzj_connections_sync_uuid();

        $data = [
            'relationship_state' => $relationship_state,
            'block_a_to_b'       => $block_a_to_b,
            'block_b_to_a'       => $block_b_to_a,
            'requested_by'       => $requested_by ?: null,
            'updated_at'         => $now,
            'last_event_uuid'    => $event_uuid,
        ];

        if ( $existing ) {

            $wpdb->update(
                bzj_connections_sync_relationships_table(),
                $data,
                [
                    'user_a' => $pair['user_a'],
                    'user_b' => $pair['user_b'],
                ],
                [ '%s', '%s', '%s', '%d', '%s', '%s' ],
                [ '%d', '%d' ]
            );

        } else {

            $data['user_a']   = $pair['user_a'];
            $data['user_b']   = $pair['user_b'];
            $data['created_at'] = $now;

            $wpdb->insert(
                bzj_connections_sync_relationships_table(),
                $data,
                [
                    '%d', '%d', '%s', '%d', '%d', '%d', '%s', '%s'
                ]
            );
        }

        return true;

    } finally {
        bzj_connections_sync_unlock( $lock );
    }
}

/**
 * When a BuddyBoss connection request is initiated.
 * Converts to canonical REQUESTED state.
 */
add_action( 'friends_friendship_requested', 'bzj_friends_friendship_requested', 10, 1 );

function bzj_friends_friendship_requested( $friendship ) {
    if ( bzj_connections_sync_internal() ) {
        return;
    }

    $initiator = absint( $friendship->initiator_user_id );
    $receiver  = absint( $friendship->friend_user_id );

    if ( ! $initiator || ! $receiver ) {
        return;
    }

    bzj_connections_sync_internal( true );

    try {

        $uuid = bzj_connections_sync_uuid();

        // Set to REQUESTED
        bzj_connections_sync_set_relationship(
            $initiator,
            $receiver,
            BZJ_REL_REQUESTED,
            [
                'requested_by' => $initiator,
                'event_uuid'   => $uuid,
            ]
        );

        // Create event for external platforms
        bzj_connections_sync_create_event(
            $uuid,
            BZJ_ORIGIN_WORDPRESS,
            BZJ_EVENT_TYPE_RELATIONSHIP,
            BZJ_OP_REQUEST,
            $initiator,
            $receiver,
            BZJ_REL_REQUESTED,
            [ 'initiator' => $initiator ]
        );

        bzj_connections_sync_log(
            'Connection request created',
            [ 'initiator' => $initiator, 'receiver' => $receiver ]
        );

    } finally {
        bzj_connections_sync_internal( false );
    }
}

/**
 * When a BuddyBoss block is created.
 * Force-unfriend if they were connected, then set block flag.
 */
add_action( 'bp_moderation_block_created', 'bzj_bp_moderation_block_created', 10, 2 );

function bzj_bp_moderation_block_created( $block_id, $block_object ) {
    if ( bzj_connections_sync_internal() ) {
        return;
    }

    $blocker = absint( $block_object->user_id );
    $blocked = absint( $block_object->item_id );

    if ( ! $blocker || ! $blocked ) {
        return;
    }

    bzj_connections_sync_internal( true );

    try {

        // Check if friends
        if ( function_exists( 'friends_check_friendship_status' ) ) {
            $status = friends_check_friendship_status( $blocker, $blocked );
            
            if ( in_array( $status, [ 'is_friend', 'pending' ], true ) ) {
                // Force remove friendship
                if ( function_exists( 'friends_remove_friend' ) ) {
                    friends_remove_friend( $blocker, $blocked );
                }
            }
        }

        $uuid = bzj_connections_sync_uuid();

        // Get current pair
        $pair = bzj_connections_sync_pair( $blocker, $blocked );

        // Determine which direction to set
        $options = [];

        if ( $pair['user_a'] === $blocker ) {
            $options['block_a_to_b'] = 1;
        } else {
            $options['block_b_to_a'] = 1;
        }

        $options['event_uuid'] = $uuid;

        // Set to NONE + block flag
        bzj_connections_sync_set_relationship(
            $blocker,
            $blocked,
            BZJ_REL_NONE,
            $options
        );

        // Create event
        bzj_connections_sync_create_event(
            $uuid,
            BZJ_ORIGIN_WORDPRESS,
            BZJ_EVENT_TYPE_BLOCK,
            BZJ_OP_BLOCK,
            $blocker,
            $blocked,
            BZJ_REL_NONE,
            []
        );

        bzj_connections_sync_log(
            'User blocked',
            [ 'blocker' => $blocker, 'blocked' => $blocked ]
        );

    } finally {
        bzj_connections_sync_internal( false );
    }
}

function bzj_connections_sync_register_rest_endpoints() {
    register_rest_route(
        'bzj/v2',
        '/relationship',
        [
            'methods'             => 'POST',
            'callback'            => 'bzj_connections_sync_rest_relationship',
            'permission_callback' => '__return_true', // HMAC auth in callback
            'args'                => [],
        ]
    );
}

function bzj_connections_sync_rest_relationship( $request ) {
    
    // Extract headers
    $timestamp = $request->get_header( 'X-BZJ-Timestamp' );
    $signature = $request->get_header( 'X-BZJ-Signature' );

    if ( ! $timestamp || ! $signature ) {
        return new WP_Error(
            'invalid_request',
            'Missing HMAC headers',
            [ 'status' => 401 ]
        );
    }

    // Get body
    $body = $request->get_body();

    if ( ! $body ) {
        return new WP_Error(
            'invalid_request',
            'Empty body',
            [ 'status' => 400 ]
        );
    }

    // Get secret
    $secret = function_exists( 'bzj_get_sso_secret' )
        ? bzj_get_sso_secret()
        : getenv( 'BUZZ_SSO_SECRET' );

    if ( ! $secret ) {
        bzj_connections_sync_log( 'SSO secret missing' );
        return new WP_Error(
            'auth_failed',
            'Authentication failed',
            [ 'status' => 401 ]
        );
    }

    // Verify HMAC
    $expected_sig = hash_hmac( 'sha256', $timestamp . '.' . $body, $secret );

    if ( ! hash_equals( $signature, $expected_sig ) ) {
        bzj_connections_sync_log(
            'Invalid HMAC signature',
            [ 'origin' => $request->get_param( 'origin' ) ]
        );
        return new WP_Error(
            'auth_failed',
            'Authentication failed',
            [ 'status' => 401 ]
        );
    }

    // Check timestamp freshness (5-minute window)
    $now = time();
    if ( abs( $now - absint( $timestamp ) ) > 300 ) {
        return new WP_Error(
            'auth_failed',
            'Request expired',
            [ 'status' => 401 ]
        );
    }

    // Parse payload
    $payload = json_decode( $body, true );

    if ( ! is_array( $payload ) ) {
        return new WP_Error(
            'invalid_payload',
            'Invalid JSON',
            [ 'status' => 400 ]
        );
    }

    // Validate fields
    $origin         = sanitize_key( $payload['origin'] ?? '' );
    $origin_user_id = absint( $payload['origin_user_id'] ?? 0 );
    $target_user_id = absint( $payload['target_user_id'] ?? 0 );
    $operation      = sanitize_key( $payload['operation'] ?? '' );
    $event_uuid     = sanitize_text_field( $payload['event_uuid'] ?? '' );

    if ( ! $origin || ! $origin_user_id || ! $target_user_id || ! $operation ) {
        return new WP_Error(
            'invalid_payload',
            'Missing required fields',
            [ 'status' => 400 ]
        );
    }

    if ( ! in_array( $origin, [ BZJ_ORIGIN_STREAMS, BZJ_ORIGIN_SOCIALS ], true ) ) {
        return new WP_Error(
            'invalid_origin',
            'Unknown origin',
            [ 'status' => 400 ]
        );
    }

    if ( ! in_array( $operation, [ BZJ_OP_REQUEST, BZJ_OP_ACCEPT, BZJ_OP_REMOVE, BZJ_OP_BLOCK, BZJ_OP_UNBLOCK ], true ) ) {
        return new WP_Error(
            'invalid_operation',
            'Unknown operation',
            [ 'status' => 400 ]
        );
    }

    // Resolve to WordPress user IDs
    $wp_origin_id = bzj_connections_sync_resolve_to_wp_user( $origin, $origin_user_id );
    $wp_target_id = bzj_connections_sync_resolve_to_wp_user( $origin, $target_user_id );

    if ( ! $wp_origin_id || ! $wp_target_id ) {
        bzj_connections_sync_log(
            'Failed to resolve platform user IDs to WordPress',
            [
                'origin'         => $origin,
                'origin_user_id' => $origin_user_id,
                'target_user_id' => $target_user_id,
            ]
        );
        return new WP_REST_Response(
            [ 'success' => false, 'error' => 'user_not_found' ],
            200 // Accept gracefully for idempotency
        );
    }

    // Process the operation
    $success = bzj_connections_sync_process_external_operation(
        $origin,
        $wp_origin_id,
        $wp_target_id,
        $operation,
        $event_uuid
    );

    return new WP_REST_Response(
        [ 'success' => $success ],
        $success ? 200 : 400
    );
}

// ==================== EXTERNAL OPERATION PROCESSOR ====================

function bzj_connections_sync_process_external_operation(
    $origin,
    $origin_user_id,
    $target_user_id,
    $operation,
    $event_uuid = ''
) {
    if ( bzj_connections_sync_internal() ) {
        return true; // Recursive call
    }

    bzj_connections_sync_internal( true );

    try {

        if ( ! $event_uuid ) {
            $event_uuid = bzj_connections_sync_uuid();
        }

        switch ( $operation ) {

            case BZJ_OP_REQUEST:
                return bzj_connections_sync_handle_external_request(
                    $origin_user_id,
                    $target_user_id,
                    $event_uuid,
                    $origin
                );

            case BZJ_OP_ACCEPT:
                return bzj_connections_sync_handle_external_accept(
                    $origin_user_id,
                    $target_user_id,
                    $event_uuid,
                    $origin
                );

            case BZJ_OP_REMOVE:
                return bzj_connections_sync_handle_external_remove(
                    $origin_user_id,
                    $target_user_id,
                    $event_uuid,
                    $origin
                );

            case BZJ_OP_BLOCK:
                return bzj_connections_sync_handle_external_block(
                    $origin_user_id,
                    $target_user_id,
                    $event_uuid,
                    $origin
                );

            case BZJ_OP_UNBLOCK:
                return bzj_connections_sync_handle_external_unblock(
                    $origin_user_id,
                    $target_user_id,
                    $event_uuid,
                    $origin
                );

            default:
                return false;
        }

    } finally {
        bzj_connections_sync_internal( false );
    }
}

/**
 * External platform initiated follow/connection request.
 * Becomes a BuddyBoss connection request (respects BB policy).
 */
function bzj_connections_sync_handle_external_request(
    $initiator_id,
    $receiver_id,
    $event_uuid,
    $origin
) {
    if ( ! function_exists( 'friends_add_friend' ) ) {
        return false;
    }

    // Add connection through BuddyBoss (force_accept=false respects request restrictions)
    $result = friends_add_friend( $initiator_id, $receiver_id, false );

    if ( $result ) {
        bzj_connections_sync_log(
            'External connection request processed',
            [
                'origin'      => $origin,
                'initiator'   => $initiator_id,
                'receiver'    => $receiver_id,
                'event_uuid'  => $event_uuid,
            ]
        );
    }

    return (bool) $result;
}

/**
 * External accept: bypass BuddyBoss and directly accept.
 */
function bzj_connections_sync_handle_external_accept(
    $user_a,
    $user_b,
    $event_uuid,
    $origin
) {
    if ( ! function_exists( 'friends_accept_friendship' ) ) {
        return false;
    }

    // Direct accept
    $result = friends_accept_friendship( 
        max( $user_a, $user_b ),  // friend_user_id (higher ID)
        min( $user_a, $user_b )   // initiator_user_id (lower ID)
    );

    if ( $result ) {
        bzj_connections_sync_log(
            'External connection accept processed',
            [
                'origin'     => $origin,
                'user_a'     => $user_a,
                'user_b'     => $user_b,
                'event_uuid' => $event_uuid,
            ]
        );
    }

    return (bool) $result;
}

/**
 * Resolve platform user ID to WordPress user ID.
 * Returns the WordPress user ID or 0 if not found.
 */
function bzj_connections_sync_resolve_to_wp_user( $origin, $platform_user_id ) {
    
    $platform_user_id = absint( $platform_user_id );

    if ( ! $platform_user_id ) {
        return 0;
    }

    if ( $origin === BZJ_ORIGIN_STREAMS ) {
        $meta_key = 'wo_user_id';
    } elseif ( $origin === BZJ_ORIGIN_SOCIALS ) {
        $meta_key = 'qd_user_id';
    } else {
        return 0;
    }

    $wp_user_id = (int) get_user_meta(
        0, // Not limited to a specific user
        $meta_key,
        false // Get all values
    );

    // Find user with this meta
    $users = get_users( [
        'meta_key'     => $meta_key,
        'meta_value'   => $platform_user_id,
        'number'       => 1,
        'fields'       => 'ids',
    ] );

    return ! empty( $users ) ? absint( $users[0] ) : 0;
}

/**
 * Process pending events every 5 minutes.
 * Only marks as processed when ALL platforms succeed.
 */
function bzj_connections_sync_process_pending_events() {
    global $wpdb;

    $table = bzj_connections_sync_events_table();

    // Get pending/retry events
    $events = $wpdb->get_results(
        "SELECT * FROM {$table}
        WHERE status IN ('" . BZJ_EVENT_STATUS_PENDING . "', '" . BZJ_EVENT_STATUS_RETRY . "')
        AND next_attempt_at <= NOW()
        ORDER BY created_at ASC
        LIMIT 50"
    );

    if ( empty( $events ) ) {
        return;
    }

    foreach ( $events as $event ) {

        // Skip if max retries exceeded
        if ( $event->attempts >= BZJ_MAX_RETRIES ) {
            $wpdb->update(
                $table,
                [
                    'status'       => BZJ_EVENT_STATUS_FAILED,
                    'last_error'   => 'Max retries exceeded',
                    'processed_at' => current_time( 'mysql', true ),
                ],
                [ 'id' => $event->id ],
                [ '%s', '%s', '%s' ],
                [ '%d' ]
            );

            bzj_connections_sync_log(
                'Event max retries exceeded',
                [
                    'event_id' => $event->id,
                    'uuid'     => $event->event_uuid,
                ]
            );

            continue;
        }

        // Process event (fan-out to external platforms)
        $success = bzj_connections_sync_fan_out_event( $event );

        if ( $success ) {

            // Mark as processed
            $wpdb->update(
                $table,
                [
                    'status'       => BZJ_EVENT_STATUS_PROCESSED,
                    'processed_at' => current_time( 'mysql', true ),
                    'attempts'     => $event->attempts + 1,
                ],
                [ 'id' => $event->id ],
                [ '%s', '%s', '%d' ],
                [ '%d' ]
            );

            bzj_connections_sync_log(
                'Event processed successfully',
                [
                    'event_id' => $event->id,
                    'uuid'     => $event->event_uuid,
                ]
            );

        } else {

            // Retry with exponential backoff
            $backoff_seconds = BZJ_BASE_RETRY_SECONDS * pow( 2, $event->attempts );
            $next_attempt = date(
                'Y-m-d H:i:s',
                strtotime( "+ {$backoff_seconds} seconds", strtotime( $event->next_attempt_at ) )
            );

            $wpdb->update(
                $table,
                [
                    'status'         => BZJ_EVENT_STATUS_RETRY,
                    'next_attempt_at' => $next_attempt,
                    'attempts'       => $event->attempts + 1,
                ],
                [ 'id' => $event->id ],
                [ '%s', '%s', '%d' ],
                [ '%d' ]
            );

            bzj_connections_sync_log(
                'Event retry scheduled',
                [
                    'event_id'       => $event->id,
                    'uuid'           => $event->event_uuid,
                    'attempts'       => $event->attempts + 1,
                    'next_attempt'   => $next_attempt,
                ]
            );
        }
    }
}

/**
 * Fan-out event to external platforms.
 * Reconcile canonical state with Streams and Socials.
 */
function bzj_connections_sync_fan_out_event( $event ) {
    
    $actor = absint( $event->actor_wp_id );
    $target = absint( $event->target_wp_id );

    // Get canonical state
    $row = bzj_connections_sync_get_relationship( $actor, $target );

    if ( ! $row ) {
        // No relationship yet
        return true; // Accept as processed
    }

    // Resolve platform IDs
    $streams_origin_id = bzj_connections_sync_resolve_platform_id( $actor, 'streams' );
    $streams_target_id = bzj_connections_sync_resolve_platform_id( $target, 'streams' );

    $socials_origin_id = bzj_connections_sync_resolve_platform_id( $actor, 'socials' );
    $socials_target_id = bzj_connections_sync_resolve_platform_id( $target, 'socials' );

    $streams_ok = true;
    $socials_ok = true;

    // Fan-out to Streams if IDs exist
    if ( $streams_origin_id && $streams_target_id ) {
        $streams_ok = bzj_connections_sync_reconcile_streams(
            $streams_origin_id,
            $streams_target_id,
            $row
        );
    }

    // Fan-out to Socials if IDs exist
    if ( $socials_origin_id && $socials_target_id ) {
        $socials_ok = bzj_connections_sync_reconcile_socials(
            $socials_origin_id,
            $socials_target_id,
            $row
        );
    }

    // Both platforms must succeed for event to be marked processed
    return $streams_ok && $socials_ok;
}

/**
 * Report a completed Streams/Socials action to WordPress canonical endpoint.
 * Called AFTER local operation succeeds on external platform.
 * Uses HMAC-SHA256 signed request for security.
 * 
 * @param string $origin    'streams' or 'socials'
 * @param int    $origin_user_id User ID on the originating platform
 * @param int    $target_user_id User ID of target on the originating platform
 * @param string $operation 'request', 'accept', 'remove', 'block', 'unblock'
 * @param string $event_uuid Optional UUID for idempotency
 * 
 * @return bool True if WordPress accepted the report
 */
if ( ! function_exists( 'bzj_connections_sync_report_to_wordpress' ) ) {
    function bzj_connections_sync_report_to_wordpress( 
        $origin, 
        $origin_user_id, 
        $target_user_id, 
        $operation, 
        $event_uuid = '' 
    ) {
        $origin = preg_replace( '/[^a-z_]/', '', strtolower( (string) $origin ) );
        
        if ( ! in_array( $origin, array( 'streams', 'socials' ), true ) ) {
            return false;
        }

        $operation = preg_replace( '/[^a-z_]/', '', strtolower( (string) $operation ) );
        
        if ( ! in_array( $operation, array( 'request', 'accept', 'remove', 'block', 'unblock' ), true ) ) {
            return false;
        }

        $secret = function_exists( 'bzj_get_sso_secret' ) 
            ? bzj_get_sso_secret() 
            : getenv( 'BUZZ_SSO_SECRET' );
        
        $base = getenv( 'WP_BASE_SITE_URL' );

        if ( ! $secret || ! $base ) {
            return false;
        }

        $payload = array(
            'origin'         => $origin,
            'origin_user_id' => absint( $origin_user_id ),
            'target_user_id' => absint( $target_user_id ),
            'operation'      => $operation,
            'event_uuid'     => $event_uuid ?: (function_exists( 'wp_generate_uuid4' ) 
                ? wp_generate_uuid4() 
                : bin2hex( random_bytes( 16 ) )),
        );

        $body = json_encode( $payload, JSON_UNESCAPED_SLASHES );
        $ts   = (string) time();
        $sig  = hash_hmac( 'sha256', $ts . '.' . $body, $secret );
        $url  = rtrim( $base, '/' ) . '/wp-json/bzj/v2/relationship';

        $ch = curl_init( $url );
        curl_setopt_array( $ch, array(
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_POST           => true,
            CURLOPT_POSTFIELDS     => $body,
            CURLOPT_HTTPHEADER     => array(
                'Content-Type: application/json',
                'X-BZJ-Timestamp: ' . $ts,
                'X-BZJ-Signature: ' . $sig,
            ),
            CURLOPT_CONNECTTIMEOUT => 5,
            CURLOPT_TIMEOUT        => 12,
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
        ) );

        $response = curl_exec( $ch );
        $errno    = curl_errno( $ch );
        $http     = (int) curl_getinfo( $ch, CURLINFO_HTTP_CODE );
        curl_close( $ch );

        if ( $errno || $http < 200 || $http >= 300 ) {
            if ( function_exists( 'wwqd_debug' ) ) {
                wwqd_debug( 'connections_sync_report_failed', array(
                    'origin'    => $origin,
                    'operation' => $operation,
                    'http'      => $http,
                    'errno'     => $errno,
                ) );
            }
            return false;
        }

        $decoded = json_decode( $response, true );
        return ! empty( $decoded['success'] );
    }
}
