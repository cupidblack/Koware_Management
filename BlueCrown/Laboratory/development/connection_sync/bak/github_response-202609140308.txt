function bzj_connections_sync_install() {

    global $wpdb;

    require_once ABSPATH . 'wp-admin/includes/upgrade.php';

    $charset = $wpdb->get_charset_collate();
    $relationships_table = bzj_connections_sync_relationship_table();
    $events_table = bzj_connections_sync_event_table();

    $sql = "
    CREATE TABLE IF NOT EXISTS {$relationships_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        user_a BIGINT UNSIGNED NOT NULL,
        user_b BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL DEFAULT 'none',
        requested_by BIGINT UNSIGNED NULL,
        blocked_by BIGINT UNSIGNED NULL,
        created_at DATETIME NOT NULL,
        updated_at DATETIME NOT NULL,
        last_event_uuid CHAR(36) NULL,
        PRIMARY KEY (id),
        UNIQUE KEY relationship_pair (user_a, user_b),
        KEY state (state),
        KEY requested_by (requested_by),
        KEY blocked_by (blocked_by),
        KEY updated_at (updated_at)
    ) {$charset};

    CREATE TABLE IF NOT EXISTS {$events_table} (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        event_uuid CHAR(36) NOT NULL,
        origin VARCHAR(20) NOT NULL,
        event_type VARCHAR(50) NOT NULL,
        actor_wp_id BIGINT UNSIGNED NOT NULL,
        target_wp_id BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL,
        payload LONGTEXT NULL,
        status VARCHAR(20) NOT NULL DEFAULT 'pending',
        attempts INT UNSIGNED NOT NULL DEFAULT 0,
        next_attempt_at DATETIME NOT NULL,
        last_error TEXT NULL,
        created_at DATETIME NOT NULL,
        processed_at DATETIME NULL,
        PRIMARY KEY (id),
        UNIQUE KEY event_uuid (event_uuid),
        KEY retry_lookup (status, next_attempt_at),
        KEY pair_lookup (actor_wp_id, target_wp_id),
        KEY created_at (created_at)
    ) {$charset};
    ";

    dbDelta($sql);

    update_option(
        'bzj_connections_sync_schema_version',
        BZJ_CONNECTIONS_SYNC_VERSION,
        false
    );

    bzj_connections_sync_log('Database schema installed/updated');
}

| UUID Generation
|--------------------------------------------------------------------------

function bzj_connections_sync_uuid() {

    if (function_exists('wp_generate_uuid4')) {
        return wp_generate_uuid4();
    }

    return sprintf(
        '%04x%04x-%04x-%04x-%04x-%04x%04x%04x',
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0x0fff) | 0x4000,
        mt_rand(0, 0x3fff) | 0x8000,
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0xffff)
    );
}

/*
|--------------------------------------------------------------------------
| Pair Normalization
|--------------------------------------------------------------------------
| 
| Canonicalize user pairs so that user_a < user_b.
| This prevents (A,B) and (B,A) from being separate records.
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_normalize_pair($user_a, $user_b) {

    $user_a = absint($user_a);
    $user_b = absint($user_b);

    if (!$user_a || !$user_b || $user_a === $user_b) {
        return false;
    }

    if ($user_a > $user_b) {
        [$user_a, $user_b] = [$user_b, $user_a];
    }

    return [
        'user_a' => $user_a,
        'user_b' => $user_b,
    ];
}

/*
|--------------------------------------------------------------------------
| MySQL Named Locks
|--------------------------------------------------------------------------
| 
| Use GET_LOCK() and RELEASE_LOCK() for distributed pair-level
| serialization. This avoids creating a permanent lock table.
| 
| Timeout: 5 seconds
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_pair_lock($user_a, $user_b) {

    global $wpdb;

    $pair = bzj_connections_sync_normalize_pair($user_a, $user_b);

    if (!$pair) {
        return false;
    }

    $lock_key = sprintf(
        'bzj_rel_%d_%d',
        $pair['user_a'],
        $pair['user_b']
    );

    $result = (int) $wpdb->get_var(
        $wpdb->prepare("SELECT GET_LOCK(%s, 5)", $lock_key)
    );

    return 1 === $result ? $lock_key : false;
}

function bzj_connections_sync_pair_unlock($lock_key) {

    global $wpdb;

    if (!$lock_key) {
        return;
    }

    $wpdb->get_var(
        $wpdb->prepare("SELECT RELEASE_LOCK(%s)", $lock_key)
    );
}

/*
|--------------------------------------------------------------------------
| Read Canonical State
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_get_state($user_a, $user_b) {

    global $wpdb;

    $pair = bzj_connections_sync_normalize_pair($user_a, $user_b);

    if (!$pair) {
        return false;
    }

    $table = bzj_connections_sync_relationship_table();

    return $wpdb->get_row(
        $wpdb->prepare(
            "SELECT * FROM {$table} WHERE user_a = %d AND user_b = %d LIMIT 1",
            $pair['user_a'],
            $pair['user_b']
        )
    );
}

function bzj_connections_sync_is_blocked($user_a, $user_b) {

    $row = bzj_connections_sync_get_state($user_a, $user_b);

    return $row && $row->state === BZJ_CONNECTIONS_SYNC_STATE_BLOCKED;
}

/*
|--------------------------------------------------------------------------
| Durable Event Creation
|--------------------------------------------------------------------------
| 
| Events are immutable records that serve as:
| - Audit trail
| - Retry queue
| - Loop-prevention mechanism
| - Debugging record
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_create_event(
    $actor_wp_id,
    $target_wp_id,
    $state,
    $event_type,
    array $payload = [],
    $origin = 'wordpress'
) {

    global $wpdb;

    $event_uuid = bzj_connections_sync_uuid();
    $now = current_time('mysql', true);

    $result = $wpdb->insert(
        bzj_connections_sync_event_table(),
        [
            'event_uuid' => $event_uuid,
            'origin' => sanitize_key($origin),
            'event_type' => sanitize_key($event_type),
            'actor_wp_id' => absint($actor_wp_id),
            'target_wp_id' => absint($target_wp_id),
            'state' => sanitize_key($state),
            'payload' => !empty($payload) ? wp_json_encode($payload) : null,
            'status' => 'pending',
            'attempts' => 0,
            'next_attempt_at' => $now,
            'created_at' => $now,
        ],
        [
            '%s', '%s', '%s', '%d', '%d', '%s', '%s',
            '%s', '%d', '%s', '%s',
        ]
    );

    if (!$result) {
        bzj_connections_sync_log(
            'Failed creating relationship event',
            [
                'error' => $wpdb->last_error,
                'actor' => $actor_wp_id,
                'target' => $target_wp_id,
                'event' => $event_type,
            ]
        );
        return false;
    }

    return $event_uuid;
}

/*
|--------------------------------------------------------------------------
| Canonical State Writer
|--------------------------------------------------------------------------
| 
| This is the central function for all state changes.
| All transitions must go through this to ensure:
| - Pair-level serialization
| - Block dominance enforcement
| - Event creation
| - Audit logging
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_set_state(
    $user_a,
    $user_b,
    $state,
    $requested_by = 0,
    $blocked_by = 0,
    $origin = 'wordpress',
    $event_type = '',
    array $payload = []
) {

    global $wpdb;

    $pair = bzj_connections_sync_normalize_pair($user_a, $user_b);

    if (!$pair) {
        return false;
    }

    if (!in_array($state, BZJ_CONNECTIONS_SYNC_STATES, true)) {
        return false;
    }

    $lock = bzj_connections_sync_pair_lock(
        $pair['user_a'],
        $pair['user_b']
    );

    if (!$lock) {
        bzj_connections_sync_log(
            'Could not acquire relationship pair lock',
            $pair
        );
        return false;
    }

    try {

        $table = bzj_connections_sync_relationship_table();

        $existing = bzj_connections_sync_get_state(
            $pair['user_a'],
            $pair['user_b']
        );

        $old_state = $existing
            ? $existing->state
            : BZJ_CONNECTIONS_SYNC_STATE_NONE;

        /*
         * BLOCK DOMINANCE RULE:
         * Block dominates all other states.
         * Once blocked, you cannot transition to requested/connected without unblocking.
         */
        if (
            $old_state === BZJ_CONNECTIONS_SYNC_STATE_BLOCKED &&
            $state !== BZJ_CONNECTIONS_SYNC_STATE_NONE
        ) {
            bzj_connections_sync_log(
                'Blocked state is dominant; cannot transition to ' . $state,
                ['pair' => $pair, 'old_state' => $old_state]
            );
            return true;
        }

        /*
         * No-op check: avoid generating unnecessary events.
         */
        if ($old_state === $state && !$event_type) {
            return true;
        }

        $now = current_time('mysql', true);
        $event_uuid = bzj_connections_sync_uuid();

        if ($existing) {

            $updated = $wpdb->update(
                $table,
                [
                    'state' => $state,
                    'requested_by' => $requested_by ? absint($requested_by) : null,
                    'blocked_by' => $blocked_by ? absint($blocked_by) : null,
                    'updated_at' => $now,
                    'last_event_uuid' => $event_uuid,
                ],
                [
                    'user_a' => $pair['user_a'],
                    'user_b' => $pair['user_b'],
                ],
                [
                    '%s', '%d', '%d', '%s', '%s',
                ],
                [
                    '%d', '%d',
                ]
            );

        } else {

            $updated = $wpdb->insert(
                $table,
                [
                    'user_a' => $pair['user_a'],
                    'user_b' => $pair['user_b'],
                    'state' => $state,
                    'requested_by' => $requested_by ? absint($requested_by) : null,
                    'blocked_by' => $blocked_by ? absint($blocked_by) : null,
                    'created_at' => $now,
                    'updated_at' => $now,
                    'last_event_uuid' => $event_uuid,
                ],
                [
                    '%d', '%d', '%s', '%d', '%d', '%s', '%s', '%s',
                ]
            );
        }

        if ($updated === false) {
            bzj_connections_sync_log(
                'Canonical relationship update failed',
                [
                    'error' => $wpdb->last_error,
                    'pair' => $pair,
                    'state' => $state,
                ]
            );
            return false;
        }

        $event = bzj_connections_sync_create_event(
            $pair['user_a'],
            $pair['user_b'],
            $state,
            $event_type ?: 'state_changed',
            array_merge(
                $payload,
                [
                    'previous_state' => $old_state,
                    'canonical_state' => $state,
                    'event_uuid' => $event_uuid,
                ]
            ),
            $origin
        );

        bzj_connections_sync_log(
            'Canonical relationship state changed',
            [
                'pair' => $pair,
                'from' => $old_state,
                'to' => $state,
                'event_uuid' => $event,
                'origin' => $origin,
            ]
        );

        return $event;

    } finally {
        bzj_connections_sync_pair_unlock($lock);
    }
}

/*
|--------------------------------------------------------------------------
| Remove BuddyBoss Relationships Before Blocking
|--------------------------------------------------------------------------
| 
| When a user is blocked, we must:
| 1. Remove any active friendship
| 2. Cancel pending requests in both directions
| 
| Uses BuddyBoss API when available.
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_remove_wp_relationships($user_a, $user_b) {

    $user_a = absint($user_a);
    $user_b = absint($user_b);

    if (!$user_a || !$user_b || $user_a === $user_b) {
        return;
    }

    /*
     * Remove active friendship in both directions.
     */
    if (
        function_exists('friends_check_friendship_status') &&
        function_exists('friends_remove_friend')
    ) {

        $status_a_b = friends_check_friendship_status($user_a, $user_b);
        if ($status_a_b === 'is_friend') {
            friends_remove_friend($user_a, $user_b);
        }

        $status_b_a = friends_check_friendship_status($user_b, $user_a);
        if ($status_b_a === 'is_friend') {
            friends_remove_friend($user_b, $user_a);
        }
    }

    /*
     * Remove pending requests in both directions.
     */
    if (function_exists('friends_get_friendship_id')) {

        // A → B
        $friendship_id = friends_get_friendship_id($user_a, $user_b);
        if ($friendship_id && function_exists('friends_reject_friendship')) {
            friends_reject_friendship($friendship_id);
        }

        // B → A
        $friendship_id = friends_get_friendship_id($user_b, $user_a);
        if ($friendship_id && function_exists('friends_reject_friendship')) {
            friends_reject_friendship($friendship_id);
        }
    }

    bzj_connections_sync_log(
        'Removed BuddyBoss relationships before blocking',
        ['user_a' => $user_a, 'user_b' => $user_b]
    );
}

/*
|--------------------------------------------------------------------------
| BuddyBoss Friendship Handlers
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_on_friendship_requested(
    $friendship_id,
    $initiator_id,
    $friend_id,
    $friendship
) {

    $initiator_id = absint($initiator_id);
    $friend_id = absint($friend_id);

    if (!$initiator_id || !$friend_id || $initiator_id === $friend_id) {
        return;
    }

    // Check if users are blocked; if so, don't sync
    if (bzj_connections_sync_is_blocked($initiator_id, $friend_id)) {
        bzj_connections_sync_log(
            'Friendship requested but users are blocked; skipping sync',
            ['initiator' => $initiator_id, 'friend' => $friend_id]
        );
        return;
    }

    bzj_connections_sync_set_state(
        $initiator_id,
        $friend_id,
        BZJ_CONNECTIONS_SYNC_STATE_REQUESTED,
        $initiator_id,
        0,
        BZJ_CONNECTIONS_SYNC_ORIGIN_WP,
        'connection_requested'
    );
}

/*
|--------------------------------------------------------------------------
| BuddyBoss Moderation Handlers
|--------------------------------------------------------------------------
| 
| BLOCK ENFORCEMENT:
| When a block is created:
| 1. Remove any existing friendship
| 2. Cancel any pending requests
| 3. Set canonical state to BLOCKED
| 
| Unblocking simply transitions to NONE.
| The previous relationship is NOT restored.
|--------------------------------------------------------------------------
*/

function bzj_connections_sync_on_block_created($block_id, $block_object) {

    if (!is_object($block_object) ||
        empty($block_object->user_id) ||
        empty($block_object->item_id)) {
        return;
    }

    $blocker_id = absint($block_object->user_id);
    $blocked_id = absint($block_object->item_id);

    if (!$blocker_id || !$blocked_id || $blocker_id === $blocked_id) {
        return;
    }

    // Remove any existing connections
    bzj_connections_sync_remove_wp_relationships($blocker_id, $blocked_id);

    // Set canonical state to BLOCKED
    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        BZJ_CONNECTIONS_SYNC_STATE_BLOCKED,
        0,
        $blocker_id,
        BZJ_CONNECTIONS_SYNC_ORIGIN_WP,
        'block_created'
    );
}

function bzj_connections_sync_on_block_deleted($block_id, $block_object) {

    if (!is_object($block_object) ||
        empty($block_object->user_id) ||
        empty($block_object->item_id)) {
        return;
    }

    $blocker_id = absint($block_object->user_id);
    $blocked_id = absint($block_object->item_id);

    if (!$blocker_id || !$blocked_id || $blocker_id === $blocked_id) {
        return;
    }

    // Unblock transitions to NONE (does not restore previous relationship)
    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        BZJ_CONNECTIONS_SYNC_STATE_NONE,
        0,
        0,
        BZJ_CONNECTIONS_SYNC_ORIGIN_WP,
        'block_deleted'
    );
}

/*
|--------------------------------------------------------------------------
| Identity Resolution
|--------------------------------------------------------------------------
| 
| Resolve WordPress user IDs to cross-platform IDs (WoWonder, QuickDate).
|--------------------------------------------------------------------------

function bzj_connections_sync_get_cross_platform_ids($wp_user_id) {

    $wp_user_id = absint($wp_user_id);

    if (!$wp_user_id) {
        return false;
    }

    return [
        'wp' => $wp_user_id,
        'wo' => absint(get_user_meta($wp_user_id, 'wo_user_id', true)),
        'qd' => absint(get_user_meta($wp_user_id, 'qd_user_id', true)),
    ];
}

function bzj_connections_sync_resolve_wp_user_from_external($external_id, $platform) {

    $external_id = absint($external_id);
    $platform = sanitize_key($platform);

    if (!$external_id || !in_array($platform, ['streams', 'socials'], true)) {
        return 0;
    }

    $meta_key = 'streams' === $platform ? 'wo_user_id' : 'qd_user_id';

    $users = get_users([
        'meta_key' => $meta_key,
        'meta_value' => (string) $external_id,
        'number' => 1,
        'fields' => 'ids',
    ]);

    return !empty($users) ? absint($users[0]) : 0;
}

function bzj_connections_sync_rest_verify_request(WP_REST_Request $request) {

    $auth_header = $request->get_header('Authorization');

    if (!$auth_header) {
        return new WP_Error(
            'missing_authorization',
            'Authorization header required',
            ['status' => 401]
        );
    }

    if (strpos($auth_header, 'Bearer ') !== 0) {
        return new WP_Error(
            'invalid_auth_format',
            'Authorization must use Bearer token',
            ['status' => 401]
        );
    }

    $token = substr($auth_header, 7);

    // Get SSO secret from shared db_helpers
    if (function_exists('bzj_get_sso_secret')) {
        $secret = bzj_get_sso_secret();
    } else {
        $secret = defined('BUZZ_SSO_SECRET') ? BUZZ_SSO_SECRET : '';
    }

    if (!$secret || !hash_equals($secret, $token)) {
        return new WP_Error(
            'invalid_token',
            'Authorization token invalid',
            ['status' => 401]
        );
    }

    return true;
}

function bzj_connections_sync_rest_handle_relationship(WP_REST_Request $request) {

    $origin = sanitize_key($request->get_param('origin'));
    $origin_user_id = absint($request->get_param('origin_user_id'));
    $target_user_id = absint($request->get_param('target_user_id'));
    $operation = sanitize_key($request->get_param('operation'));
    $event_uuid = sanitize_text_field($request->get_param('event_uuid'));

    if (!$origin_user_id || !$target_user_id) {
        return new WP_Error(
            'invalid_users',
            'Invalid user IDs',
            ['status' => 400]
        );
    }

    // Resolve external user IDs to WordPress user IDs
    $wp_origin_user = bzj_connections_sync_resolve_wp_user_from_external(
        $origin_user_id,
        $origin
    );
    $wp_target_user = bzj_connections_sync_resolve_wp_user_from_external(
        $target_user_id,
        $origin
    );

    if (!$wp_origin_user || !$wp_target_user) {
        return new WP_Error(
            'user_not_found',
            'Could not resolve one or both users to WordPress',
            ['status' => 404]
        );
    }

    // Handle operation
    switch ($operation) {

        case 'request':
            bzj_connections_sync_set_state(
                $wp_origin_user,
                $wp_target_user,
                BZJ_CONNECTIONS_SYNC_STATE_REQUESTED,
                $wp_origin_user,
                0,
                $origin,
                'external_connection_requested',
                ['event_uuid' => $event_uuid]
            );
            break;

        case 'accept':
            bzj_connections_sync_set_state(
                $wp_origin_user,
                $wp_target_user,
                BZJ_CONNECTIONS_SYNC_STATE_CONNECTED,
                0,
                0,
                $origin,
                'external_connection_accepted',
                ['event_uuid' => $event_uuid]
            );
            break;

        case 'remove':
            bzj_connections_sync_set_state(
                $wp_origin_user,
                $wp_target_user,
                BZJ_CONNECTIONS_SYNC_STATE_NONE,
                0,
                0,
                $origin,
                'external_connection_removed',
                ['event_uuid' => $event_uuid]
            );
            break;

        case 'block':
            // Remove BuddyBoss relationships first
            bzj_connections_sync_remove_wp_relationships(
                $wp_origin_user,
                $wp_target_user
            );
            bzj_connections_sync_set_state(
                $wp_origin_user,
                $wp_target_user,
                BZJ_CONNECTIONS_SYNC_STATE_BLOCKED,
                0,
                $wp_origin_user,
                $origin,
                'external_block_created',
                ['event_uuid' => $event_uuid]
            );
            break;

        case 'unblock':
            bzj_connections_sync_set_state(
                $wp_origin_user,
                $wp_target_user,
                BZJ_CONNECTIONS_SYNC_STATE_NONE,
                0,
                0,
                $origin,
                'external_block_deleted',
                ['event_uuid' => $event_uuid]
            );
            break;

        default:
            return new WP_Error(
                'invalid_operation',
                'Unknown operation: ' . $operation,
                ['status' => 400]
            );
    }

    return [
        'success' => true,
        'message' => 'Relationship updated',
        'origin' => $origin,
        'operation' => $operation,
    ];
}

        return false;

    switch ($state) {

            return wwqd_set_streams_follow_request($wo_user_a, $wo_user_b, $wo_conn);

function wwqd_set_streams_follow_request($wo_user_a, $wo_user_b, $wo_conn) {

    // Get config to check connectivitySystem
    $config_query = "SELECT option_value FROM " . wp_table('options') . " WHERE option_name = 'wo_config' LIMIT 1";
    $config_result = $wo_conn->query($config_query);
    $config_row = $config_result->fetch_assoc();

    if (!$config_row) {
        return false;
    }

    $config = unserialize($config_row['option_value']);

    if (isset($config['connectivitySystem']) && $config['connectivitySystem'] == 1) {
        // Friendship system: create pending request
        $stmt = $wo_conn->prepare(
            "INSERT INTO wo_follow (follower_id, following_id, status) VALUES (?, ?, 'pending') 
             ON DUPLICATE KEY UPDATE status = 'pending'"
        );
        $stmt->bind_param('ii', $wo_user_a, $wo_user_b);
        return $stmt->execute();
    } else {
        // Follow system: direct follow
        $stmt = $wo_conn->prepare(
            "INSERT INTO wo_follow (follower_id, following_id, status) VALUES (?, ?, 'accepted') 
             ON DUPLICATE KEY UPDATE status = 'accepted'"
        );
        $stmt->bind_param('ii', $wo_user_a, $wo_user_b);
        return $stmt->execute();
    }
}

|--------------------------------------------------------------------------
| Master Dispatcher
|--------------------------------------------------------------------------
| 
| Call this from the cron job to process pending events and sync
| to all platforms.
|--------------------------------------------------------------------------
*/

function wwqd_process_pending_relationship_events() {

    global $wpdb;

    $events_table = $wpdb->prefix . 'bzj_relationship_events';

    // Get all pending events
    $pending_events = $wpdb->get_results(
        "SELECT * FROM {$events_table} 
         WHERE status = 'pending' 
         AND attempts < 3 
         ORDER BY created_at ASC 
         LIMIT 100"
    );

    if (empty($pending_events)) {
        return;
    }

    foreach ($pending_events as $event) {

        $actor_wp_id = intval($event->actor_wp_id);
        $target_wp_id = intval($event->target_wp_id);
        $state = $event->state;

        // Get cross-platform IDs
        $wo_a = absint(get_user_meta($actor_wp_id, 'wo_user_id', true));
        $qd_a = absint(get_user_meta($actor_wp_id, 'qd_user_id', true));
        $wo_b = absint(get_user_meta($target_wp_id, 'wo_user_id', true));
        $qd_b = absint(get_user_meta($target_wp_id, 'qd_user_id', true));

        $success = true;

        // Sync to Streams
        if ($wo_a && $wo_b) {
            $success = $success && wwqd_sync_relationship_to_streams($wo_a, $wo_b, $state);
        }

        // Sync to Socials
        if ($qd_a && $qd_b) {
            $success = $success && wwqd_sync_relationship_to_socials($qd_a, $qd_b, $state);
        }

        // Mark event as processed
        $event_id = $event->id;
        $attempts = intval($event->attempts) + 1;
        $now = current_time('mysql', true);

        if ($success) {
            $wpdb->update(
                $events_table,
                [
                    'status' => 'processed',
                    'processed_at' => $now,
                    'attempts' => $attempts,
                ],
                ['id' => $event_id],
                ['%s', '%s', '%d'],
                ['%d']
            );
        } else {
            // Retry later
            $next_retry = date('Y-m-d H:i:s', strtotime($now) + (300 * $attempts)); // 5 min per attempt

            $wpdb->update(
                $events_table,
                [
                    'status' => 'pending',
                    'attempts' => $attempts,
                    'next_attempt_at' => $next_retry,
                    'last_error' => 'Platform sync failed',
                ],
                ['id' => $event_id],
                ['%s', '%d', '%s', '%s'],
                ['%d']
            );
        }
    }
}

// Schedule cron job (runs every 5 minutes)
add_action('bzj_process_relationship_events', 'wwqd_process_pending_relationship_events');

if (!wp_next_scheduled('bzj_process_relationship_events')) {
    wp_schedule_event(current_time('timestamp'), '5min', 'bzj_process_relationship_events');
}

function Wo_CallWordPressRelationshipAPI($origin_user_id, $target_user_id, $operation) {

    global $wo;

    $wp_api_url = defined('WP_BASE_SITE_URL')
        ? WP_BASE_SITE_URL
        : $wo['config']['WP_BASE_SITE_URL'] ?? '';

    if (!$wp_api_url) {
        return false;
    }

    $sso_secret = defined('BUZZ_SSO_SECRET')
        ? BUZZ_SSO_SECRET
        : ($wo['config']['BUZZ_SSO_SECRET'] ?? '');

    if (!$sso_secret) {
        return false;
    }

    $endpoint = rtrim($wp_api_url, '/') . '/wp-json/bzj/v1/relationship';

            'timeout' => 10,
            'sslverify' => true,
        ]
    );

    if (is_wp_error($response)) {
        error_log(
            '[Buzzjuice Sync] WordPress API call failed: ' . $response->get_error_message()
        );

function CallWordPressRelationshipAPI($origin_user_id, $target_user_id, $operation) {

    global $config;

    $wp_api_url = defined('WP_BASE_SITE_URL')
        ? WP_BASE_SITE_URL
        : ($config->WP_BASE_SITE_URL ?? '');

    if (!$wp_api_url) {
        return false;
    }

    $sso_secret = defined('BUZZ_SSO_SECRET')
        ? BUZZ_SSO_SECRET
        : ($config->BUZZ_SSO_SECRET ?? '');

    if (!$sso_secret) {
        return false;
    }

    $endpoint = rtrim($wp_api_url, '/') . '/wp-json/bzj/v1/relationship';

    $payload = [
        'origin' => 'socials',
        'origin_user_id' => intval($origin_user_id),
        'target_user_id' => intval($target_user_id),
        'operation' => $operation,
        'event_uuid' => uniqid('qd_', true),
    ];

    // Use cURL if wp_remote_post is not available
    $ch = curl_init();
    curl_setopt_array($ch, [
        CURLOPT_URL => $endpoint,
        CURLOPT_HTTPHEADER => [
            'Authorization: Bearer ' . $sso_secret,
            'Content-Type: application/json',
        ],
        CURLOPT_POST => 1,
        CURLOPT_POSTFIELDS => json_encode($payload),
        CURLOPT_TIMEOUT => 10,
        CURLOPT_SSL_VERIFYPEER => true,
        CURLOPT_RETURNTRANSFER => true,
    ]);

    $response = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    return 200 <= $http_code && $http_code < 300;
}

## PART 5: STATE MACHINE DIAGRAM

```
                        ┌──────────┐
                        │   NONE   │ ◄──────────────────┐
                        └────┬─────┘                    │
                             │                          │
                       request│                    unblock
                             ▼                          │
                      ┌──────────────┐                  │
                      │  REQUESTED   │                  │
                      └────┬─────────┘                  │
                           │                            │
                       accept│                          │
                           ▼                            │
                      ┌──────────────┐                  │
                      │  CONNECTED   │                  │
                      └────┬─────────┘                  │
                           │                            │
                        remove                          │
                           │                            │
                           └────────────────────────────┘

    ANY STATE ──────── block ──────► ┌──────────┐
                                     │ BLOCKED  │
                                     └──────────┘
```

**Key Invariants:**

1. Only `CONNECTED` and `REQUESTED` are asymmetric (directional)
2. `BLOCKED` can be entered from any state
3. `BLOCKED` can only exit to `NONE` (no prior relationship restoration)
4. Block → Unblock: `BLOCKED` → `NONE` (must reconnect explicitly)
5. All transitions are paired (user_a ↔ user_b, normalized)
