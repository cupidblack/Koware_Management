/**
 * File: wp-content/mu-plugins/bzj-connections-sync-schema.php
 * 
 * Schema migration for Buzzjuice relationship synchronization.
 * Automatically runs during plugin bootstrap if schema version is outdated.

if (!defined('ABSPATH')) exit;

/**
 * Get current schema version from options
 */
function bzj_get_schema_version() {
    return (int) get_option('bzj_connections_sync_db_version', 0);
}

/**
 * Set schema version after successful migration
 */
function bzj_set_schema_version($version) {
    return update_option('bzj_connections_sync_db_version', (int) $version);
}

/**
 * Create required database tables
 * 
 * - wp_bzj_relationships: Canonical relationship state
 * - wp_bzj_relationship_events: Durable event ledger and retry queue
 * - wp_bzj_relationship_locks: Pair-level distributed locks
 */
function bzj_create_schema() {
    global $wpdb;
    
    require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
    
    $charset_collate = $wpdb->get_charset_collate();
    
    $sql = "
    
    CREATE TABLE " . $wpdb->prefix . "bzj_relationships (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        user_a BIGINT UNSIGNED NOT NULL,
        user_b BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL DEFAULT 'none',
        requested_by BIGINT UNSIGNED NULL,
        blocked_by BIGINT UNSIGNED NULL,
        last_event_uuid CHAR(36) NULL,
        created_at DATETIME NOT NULL,
        updated_at DATETIME NOT NULL,
        PRIMARY KEY (id),
        UNIQUE KEY user_pair (user_a, user_b),
        KEY state (state),
        KEY updated_at (updated_at),
        KEY created_at (created_at)
    ) $charset_collate;
    
    CREATE TABLE " . $wpdb->prefix . "bzj_relationship_events (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        event_uuid CHAR(36) NOT NULL,
        origin VARCHAR(20) NOT NULL,
        event_type VARCHAR(50) NOT NULL,
        actor_wp_id BIGINT UNSIGNED NOT NULL,
        target_wp_id BIGINT UNSIGNED NOT NULL,
        state VARCHAR(20) NOT NULL,
        payload LONGTEXT NULL,
        status VARCHAR(20) NOT NULL DEFAULT 'pending',
        attempts INT UNSIGNED NOT NULL DEFAULT 0,
        next_attempt_at DATETIME NULL,
        last_error TEXT NULL,
        created_at DATETIME NOT NULL,
        processed_at DATETIME NULL,
        PRIMARY KEY (id),
        UNIQUE KEY event_uuid (event_uuid),
        KEY status_retry (status, next_attempt_at),
        KEY pair_lookup (actor_wp_id, target_wp_id),
        KEY created_at (created_at),
        KEY origin (origin)
    ) $charset_collate;
    
    CREATE TABLE " . $wpdb->prefix . "bzj_relationship_locks (
        id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
        lock_key VARCHAR(100) NOT NULL,
        acquired_at DATETIME NOT NULL,
        expires_at DATETIME NOT NULL,
        process_id VARCHAR(50) NULL,
        PRIMARY KEY (id),
        UNIQUE KEY lock_key (lock_key),
        KEY expires_at (expires_at)
    ) $charset_collate;
    ";
    
    dbDelta($sql);
    
    bzj_set_schema_version(1);
}

/**
 * Run migrations if schema is outdated
 */
function bzj_check_and_migrate_schema() {
    $current_version = bzj_get_schema_version();
    $target_version = 1;
    
    if ($current_version < $target_version) {
        bzj_create_schema();
    }
}

/**
 * Handle block created
 * 
 * Block always dominates relationship state.
 * Automatically unfriends and removes pending requests.
 * 
 * @param int $block_id
 * @param object $block_object
 */
function bzj_connections_sync_block_created(
    $block_id,
    $block_object
) {
    
    if (!is_object($block_object) || 
        empty($block_object->user_id) || 
        empty($block_object->item_id)) {
        return;
    }
    
    $blocker_id = (int) $block_object->user_id;
    $blocked_id = (int) $block_object->item_id;
    
    if (!$blocker_id || !$blocked_id || $blocker_id === $blocked_id) {
        return;
    }
    
    // Remove any existing connections first
    bzj_connections_sync_remove_wp_relationships($blocker_id, $blocked_id);
    
    // Set canonical state to blocked
    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        'blocked',
        0,
        $blocker_id,
        BZJ_CONNECTIONS_SYNC_ORIGIN,
        'block_created'
    );
}

/**
 * Handle block deleted (unblock)
 * 
 * Unblocking does NOT restore previous connection.
 * State transitions to 'none'.
 * 
 * @param int $block_id
 * @param object $block_object
 */
function bzj_connections_sync_block_deleted(
    $block_id,
    $block_object
) {
    
    if (!is_object($block_object) || 
        empty($block_object->user_id) || 
        empty($block_object->item_id)) {
        return;
    }
    
    $blocker_id = (int) $block_object->user_id;
    $blocked_id = (int) $block_object->item_id;
    
    // Set canonical state to none
    bzj_connections_sync_set_state(
        $blocker_id,
        $blocked_id,
        'none',
        0,
        0,
        BZJ_CONNECTIONS_SYNC_ORIGIN,
        'block_deleted'
    );
}

/**
 * Check if two users are blocked
 * 
 * @param int $user_a
 * @param int $user_b
 * @return bool
 */
function bzj_connections_sync_users_blocked($user_a, $user_b) {
    global $wpdb;
    
    $user_a = (int) $user_a;
    $user_b = (int) $user_b;
    
    if ($user_a > $user_b) {
        $tmp = $user_a;
        $user_a = $user_b;
        $user_b = $tmp;
    }
    
    $table = $wpdb->prefix . 'bzj_relationships';
    
    $result = $wpdb->get_var($wpdb->prepare(
        "SELECT state FROM {$table} 
         WHERE user_a = %d AND user_b = %d 
         LIMIT 1",
        $user_a,
        $user_b
    ));
    
    return 'blocked' === $result;
}

/**
 * Remove WordPress relationships before blocking
 * 
 * Unfriends and removes pending requests in both directions.
 * Uses BuddyBoss API when available.
 * 
 * @param int $user_a
 * @param int $user_b
 */
function bzj_connections_sync_remove_wp_relationships($user_a, $user_b) {
    
    $user_a = (int) $user_a;
    $user_b = (int) $user_b;
    
    if (!$user_a || !$user_b || $user_a === $user_b) {
        return;
    }
    
    // Remove active friendship
    if (function_exists('friends_check_friendship_status') && 
        function_exists('friends_remove_friend')) {
        
        $status = friends_check_friendship_status($user_a, $user_b);
        
        if ('is_friend' === $status) {
            friends_remove_friend($user_a, $user_b);
        }
    }
    
    // Remove pending requests in both directions
    if (class_exists('BP_Friends_Friendship')) {
        
        // User A → User B request
        $friendship_id = BP_Friends_Friendship::get_friendship_id($user_a, $user_b);
        if ($friendship_id) {
            BP_Friends_Friendship::delete($friendship_id);
        }
        
        // User B → User A request
        $friendship_id = BP_Friends_Friendship::get_friendship_id($user_b, $user_a);
        if ($friendship_id) {
            BP_Friends_Friendship::delete($friendship_id);
        }
    }
}

/**
 * Set canonical relationship state
 * 
 * This is the central function for changing relationship state.
 * All state transitions must go through this function to ensure
 * proper locking, auditing, and event propagation.
 * 
 * @param int $user_a
 * @param int $user_b
 * @param string $state (none|requested|connected|blocked)
 * @param int $requested_by
 * @param int $blocked_by
 * @param string $origin
 * @param string $event_type
 * @return bool
 */
function bzj_connections_sync_set_state(
    $user_a,
    $user_b,
    $state,
    $requested_by = 0,
    $blocked_by = 0,
    $origin = 'wordpress',
    $event_type = ''
) {
    
    global $wpdb;
    
    $user_a = (int) $user_a;
    $user_b = (int) $user_b;
    
    if ($user_a === $user_b) {
        return false;
    }
    
    // Normalize pair (smaller ID first)
    if ($user_a > $user_b) {
        $tmp = $user_a;
        $user_a = $user_b;
        $user_b = $tmp;
    }
    
    $state = sanitize_text_field($state);
    $origin = sanitize_text_field($origin);
    $event_type = sanitize_text_field($event_type);
    
    // Acquire pair lock
    $lock_key = "bzj_rel_{$user_a}_{$user_b}";
    $lock_acquired = bzj_relationship_pair_lock($lock_key);
    
    if (!$lock_acquired) {
        bzj_connections_sync_debug_log(
            "Failed to acquire lock for relationship {$user_a} ↔ {$user_b}",
            'warning'
        );
        return false;
    }
    
    try {
        
        $table = $wpdb->prefix . 'bzj_relationships';
        $now = current_time('mysql');
        
        // Check if relationship exists
        $existing = $wpdb->get_row($wpdb->prepare(
            "SELECT * FROM {$table} 
             WHERE user_a = %d AND user_b = %d 
             LIMIT 1",
            $user_a,
            $user_b
        ));
        
        $old_state = $existing ? $existing->state : 'none';
        
        // No-op check: avoid creating duplicate events
        if ($old_state === $state) {
            bzj_connections_sync_debug_log(
                "No-op state transition for {$user_a} ↔ {$user_b}: {$state}",
                'info'
            );
            return true;
        }
        
        // Update or insert relationship record
        if ($existing) {
            
            $wpdb->update(
                $table,
                [
                    'state' => $state,
                    'requested_by' => $requested_by ?: null,
                    'blocked_by' => $blocked_by ?: null,
                    'updated_at' => $now,
                ],
                ['id' => $existing->id],
                ['%s', '%d', '%d', '%s'],
                ['%d']
            );
            
        } else {
            
            $wpdb->insert(
                $table,
                [
                    'user_a' => $user_a,
                    'user_b' => $user_b,
                    'state' => $state,
                    'requested_by' => $requested_by ?: null,
                    'blocked_by' => $blocked_by ?: null,
                    'created_at' => $now,
                    'updated_at' => $now,
                ],
                ['%d', '%d', '%s', '%d', '%d', '%s', '%s']
            );
        }
        
        // Create event for processing
        $event_uuid = bzj_generate_event_uuid();
        
        bzj_connections_sync_create_event(
            $event_type,
            $origin,
            $user_a,
            $user_b,
            $state,
            [
                'previous_state' => $old_state,
                'new_state' => $state,
                'requested_by' => $requested_by,
                'blocked_by' => $blocked_by,
            ],
            $event_uuid
        );
        
        // Update last event UUID on relationship
        if ($existing) {
            $wpdb->update(
                $table,
                ['last_event_uuid' => $event_uuid],
                ['id' => $existing->id],
                ['%s'],
                ['%d']
            );
        }
        
        bzj_connections_sync_debug_log(
            "State set: {$user_a} ↔ {$user_b} = {$state} (origin: {$origin}, event: {$event_type})"
        );
        
        return true;
        
    } catch (Exception $e) {
        
        bzj_connections_sync_debug_log(
            "Error setting state: " . $e->getMessage(),
            'error'
        );
        return false;
        
    } finally {
        
        // Release pair lock
        bzj_relationship_pair_unlock($lock_key);
    }
}

/**
 * Generate a unique event UUID
 * 
 * @return string
 */
function bzj_generate_event_uuid() {
    if (function_exists('wp_generate_uuid4')) {
        return wp_generate_uuid4();
    }
    
    return sprintf(
        '%04x%04x-%04x-%04x-%04x-%04x%04x%04x',
        mt_rand(0, 0xffff), mt_rand(0, 0xffff),
        mt_rand(0, 0xffff),
        mt_rand(0, 0x0fff) | 0x4000,
        mt_rand(0, 0x3fff) | 0x8000,
        mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff)
    );
}

/**
 * Create a relationship event
 * 
 * Events are durable and used for:
 * - Audit trail
 * - Retry queue
 * - Loop prevention
 * - Debugging
 * 
 * @param string $event_type
 * @param string $origin
 * @param int $actor_wp_id
 * @param int $target_wp_id
 * @param string $state
 * @param array $payload
 * @param string $event_uuid
 * @return int|false
 */
function bzj_connections_sync_create_event(
    $event_type,
    $origin,
    $actor_wp_id,
    $target_wp_id,
    $state,
    $payload = [],
    $event_uuid = ''
) {
    
    global $wpdb;
    
    if (!$event_uuid) {
        $event_uuid = bzj_generate_event_uuid();
    }
    
    $table = $wpdb->prefix . 'bzj_relationship_events';
    $now = current_time('mysql');
    
    return $wpdb->insert(
        $table,
        [
            'event_uuid' => $event_uuid,
            'origin' => sanitize_text_field($origin),
            'event_type' => sanitize_text_field($event_type),
            'actor_wp_id' => (int) $actor_wp_id,
            'target_wp_id' => (int) $target_wp_id,
            'state' => sanitize_text_field($state),
            'payload' => !empty($payload) ? wp_json_encode($payload) : null,
            'status' => 'pending',
            'attempts' => 0,
            'next_attempt_at' => $now,
            'created_at' => $now,
        ],
        [
            '%s', '%s', '%s', '%d', '%d', '%s', '%s', '%s', '%d', '%s', '%s'
        ]
    );
}

| Distributed Locking

/**
 * Acquire pair-level lock
 * 
 * Uses database-level mutual exclusion to prevent race conditions
 * during relationship state changes.
 * 
 * Lock timeout: 30 seconds
 * 
 * @param string $lock_key
 * @return bool
 */
function bzj_relationship_pair_lock($lock_key) {
    global $wpdb;
    
    $lock_key = sanitize_text_field($lock_key);
    $table = $wpdb->prefix . 'bzj_relationship_locks';
    $now = current_time('mysql');
    $expires = date('Y-m-d H:i:s', strtotime($now) + 30);
    
    // Clean expired locks
    $wpdb->query("DELETE FROM {$table} WHERE expires_at < '{$now}'");
    
    // Try to insert lock
    $result = $wpdb->query($wpdb->prepare(
        "INSERT INTO {$table} (lock_key, acquired_at, expires_at) 
         VALUES (%s, %s, %s)",
        $lock_key,
        $now,
        $expires
    ));
    
    return (bool) $result;
}

/**
 * Release pair-level lock
 * 
 * @param string $lock_key
 * @return bool
 */
function bzj_relationship_pair_unlock($lock_key) {
    global $wpdb;
    
    $lock_key = sanitize_text_field($lock_key);
    $table = $wpdb->prefix . 'bzj_relationship_locks';
    
    return (bool) $wpdb->delete(
        $table,
        ['lock_key' => $lock_key],
        ['%s']
    );
}

/**
 * Get cross-platform user IDs
 * 
 * Resolves WordPress user ID to their WoWonder and QuickDate IDs.
 * 
 * @param int $wp_user_id
 * @return array
 */
function bzj_get_cross_platform_ids($wp_user_id) {
    
    $wp_user_id = (int) $wp_user_id;
    
    if (!$wp_user_id) {
        return [
            'wp' => null,
            'wo' => null,
            'qd' => null,
        ];
    }
    
    return [
        'wp' => $wp_user_id,
        'wo' => (int) get_user_meta($wp_user_id, 'wo_user_id', true),
        'qd' => (int) get_user_meta($wp_user_id, 'qd_user_id', true),
    ];
}

/**
 * Resolve WordPress user from external platform ID
 * 
 * @param int $external_id
 * @param string $platform (streams|socials)
 * @return int
 */
function bzj_resolve_wp_user_from_external($external_id, $platform) {
    
    $external_id = (int) $external_id;
    $platform = sanitize_text_field($platform);
    
    if (!$external_id || !in_array($platform, ['streams', 'socials'])) {
        return 0;
    }
    
    $meta_key = 'streams' === $platform ? 'wo_user_id' : 'qd_user_id';
    
    $users = get_users([
        'meta_key' => $meta_key,
        'meta_value' => (string) $external_id,
        'number' => 1,
        'fields' => 'ids',
    ]);
    
    return !empty($users) ? (int) $users[0] : 0;
}

            'methods' => 'POST',

            'permission_callback' => 'bzj_verify_request_signature',
            'args' => [
                'origin' => [
                    'required' => true,

                    'enum' => ['streams', 'socials'],
                ],
                'origin_user_id' => [

/**
 * Verify incoming request signature
 * 
 * External platforms must provide Bearer token with BUZZ_SSO_SECRET.
 * 
 * @param WP_REST_Request $request
 * @return bool|WP_Error
 */
function bzj_verify_request_signature(WP_REST_Request $request) {
    
    $auth_header = $request->get_header('Authorization');
    
    if (!$auth_header) {
        return new WP_Error(
            'missing_authorization',
            'Authorization header required',
            ['status' => 401]
        );
    }
    
    if (strpos($auth_header, 'Bearer ') !== 0) {
        return new WP_Error(
            'invalid_auth_format',
            'Authorization must use Bearer token',
            ['status' => 401]
        );
    }
    
    $token = substr($auth_header, 7);
    $secret = bzj_get_sso_secret();
    
    if (!hash_equals($secret, $token)) {
        return new WP_Error(
            'invalid_token',
            'Invalid authentication token',
            ['status' => 403]
        );
    }
    
    return true;
}

/**
 * Get SSO secret
 * 
 * Retrieves from environment or constant fallback.
 * 
 * @return string
 */
function bzj_get_sso_secret() {
    $env = getenv('BUZZ_SSO_SECRET');
    if (!empty($env)) {
        return $env;
    }
    
    if (defined('BUZZ_SSO_SECRET') && BUZZ_SSO_SECRET) {
        return BUZZ_SSO_SECRET;
    }
    
    return 'bzj_fallback_' . hash('sha256', ABSPATH);
}

/**
 * Handle relationship request from external platform
 * 
 * External platforms request relationship changes through this endpoint.
 * All requests are validated and routed through WordPress authority.
 * 
 * @param WP_REST_Request $request
 * @return WP_REST_Response|WP_Error
 */
function bzj_handle_relationship_request(WP_REST_Request $request) {
    
    $origin = $request->get_param('origin');
    $origin_user_id = (int) $request->get_param('origin_user_id');
    $target_user_id = (int) $request->get_param('target_user_id');
    $operation = $request->get_param('operation');
    $event_uuid = $request->get_param('event_uuid');
    
    // Resolve external IDs to WordPress
    $actor_wp_id = bzj_resolve_wp_user_from_external($origin_user_id, $origin);
    $target_wp_id = (int) $target_user_id;
    
    if (!$actor_wp_id) {
        return new WP_Error(
            'invalid_actor',
            "Cannot resolve {$origin} user {$origin_user_id} to WordPress",
            ['status' => 400]
        );
    }
    
    if (!$target_wp_id) {
        return new WP_Error(
            'invalid_target',
            'Target user ID is invalid',
            ['status' => 400]
        );
    }
    
    // Check for idempotency
    if (bzj_event_already_processed($event_uuid)) {
        return rest_ensure_response([
            'success' => true,
            'message' => 'Event already processed (idempotent)',
        ]);
    }
    
    // Route operation
    $result = bzj_handle_external_operation(
        $actor_wp_id,
        $target_wp_id,
        $operation,
        $origin,
        $event_uuid
    );
    
    if (is_wp_error($result)) {
        return $result;
    }
    
    return rest_ensure_response([
        'success' => true,
        'data' => $result,
    ]);
}

/**
 * Check if event was already processed
 * 
 * Uses idempotency key to prevent duplicate operations.
 * 
 * @param string $event_uuid
 * @return bool
 */
function bzj_event_already_processed($event_uuid) {
    global $wpdb;
    
    $event_uuid = sanitize_text_field($event_uuid);
    $table = $wpdb->prefix . 'bzj_relationship_events';
    
    $exists = $wpdb->get_var($wpdb->prepare(
        "SELECT id FROM {$table} WHERE event_uuid = %s LIMIT 1",
        $event_uuid
    ));
    
    return (bool) $exists;
}

/**
 * Handle external platform operation
 * 
 * Routes operations from Streams/Socials to appropriate WordPress handlers.
 * 
 * @param int $actor_wp_id
 * @param int $target_wp_id
 * @param string $operation
 * @param string $origin
 * @param string $event_uuid
 * @return array|WP_Error
 */
function bzj_handle_external_operation(
    $actor_wp_id,
    $target_wp_id,
    $operation,
    $origin,
    $event_uuid
) {
    
    if ($actor_wp_id === $target_wp_id) {
        return new WP_Error(
            'self_operation',
            'Cannot perform relationship operations with self',
            ['status' => 400]
        );
    }
    
    switch ($operation) {
        
        case 'request':
            return bzj_handle_external_request_connection(
                $actor_wp_id,
                $target_wp_id,
                $origin,
                $event_uuid
            );
        
        case 'accept':
            return bzj_handle_external_accept_connection(
                $actor_wp_id,
                $target_wp_id,
                $origin,
                $event_uuid
            );
        
        case 'reject':
        case 'remove':
            return bzj_handle_external_remove_connection(
                $actor_wp_id,
                $target_wp_id,
                $origin,
                $event_uuid,
                $operation
            );
        
        case 'block':
            return bzj_handle_external_block(
                $actor_wp_id,
                $target_wp_id,
                $origin,
                $event_uuid
            );
        
        case 'unblock':
            return bzj_handle_external_unblock(
                $actor_wp_id,
                $target_wp_id,
                $origin,
                $event_uuid
            );
        
        default:
            return new WP_Error(
                'invalid_operation',
                "Operation '{$operation}' not supported",
                ['status' => 400]
            );
    }
}

/**
 * Handle external connection request
 * 
 * @param int $actor_wp_id
 * @param int $target_wp_id
 * @param string $origin
 * @param string $event_uuid
 * @return array|WP_Error
 */
function bzj_handle_external_request_connection(
    $actor_wp_id,
    $target_wp_id,
    $origin,
    $event_uuid
) {
    
    // Check if blocked
    if (bzj_connections_sync_users_blocked($actor_wp_id, $target_wp_id)) {
        return new WP_Error(
            'users_blocked',
            'Users are blocked',
            ['status' => 409]
        );
    }
    
    // Use BuddyBoss API
    if (!function_exists('friends_add_friend')) {
        return new WP_Error(
            'buddyboss_unavailable',
            'BuddyBoss not available',
            ['status' => 503]
        );
    }
    
    $result = friends_add_friend($actor_wp_id, $target_wp_id);
    
    if (!$result) {
        return new WP_Error(
            'request_failed',
            'Failed to create connection request',
            ['status' => 500]
        );
    }
    
    return [
        'action' => 'request_created',
        'actor_id' => $actor_wp_id,
        'target_id' => $target_wp_id,
    ];
}

/**
 * Handle external connection acceptance
 * 
 * @param int $actor_wp_id
 * @param int $target_wp_id
 * @param string $origin
 * @param string $event_uuid
 * @return array|WP_Error
 */
function bzj_handle_external_accept_connection(
    $actor_wp_id,
    $target_wp_id,
    $origin,
    $event_uuid
) {
    
    if (!function_exists('friends_accept_friendship')) {
        return new WP_Error(
            'buddyboss_unavailable',
            'BuddyBoss not available',
            ['status' => 503]
        );
    }
    
    // Find pending friendship
    if (!class_exists('BP_Friends_Friendship')) {
        return new WP_Error(
            'buddyboss_unavailable',
            'BuddyBoss not available',
            ['status' => 503]
        );
    }
    
    $friendship_id = BP_Friends_Friendship::get_friendship_id(
        $target_wp_id,
        $actor_wp_id
    );
    
    if (!$friendship_id) {
        return new WP_Error(
            'no_pending_request',
            'No pending friendship request found',
            ['status' => 409]
        );
    }
    
    $friendship = new BP_Friends_Friendship($friendship_id);
    
    if (!$friendship->accept()) {
        return new WP_Error(
            'accept_failed',
            'Failed to accept friendship',
            ['status' => 500]
        );
    }
    
    return [
        'action' => 'connection_accepted',
        'actor_id' => $actor_wp_id,
        'target_id' => $target_wp_id,
    ];
}

/**
 * Process pending relationship events
 * 
 * Retries failed events with exponential backoff.
 */
function bzj_process_pending_events() {
    global $wpdb;
    
    $table = $wpdb->prefix . 'bzj_relationship_events';
    $now = current_time('mysql');
    
    // Get pending events due for retry
    $events = $wpdb->get_results(
        "SELECT * FROM {$table}
         WHERE status = 'pending'
           AND next_attempt_at <= '{$now}'
         ORDER BY next_attempt_at ASC
         LIMIT 10"
    );
    
    foreach ($events as $event) {
        bzj_process_relationship_event($event);
    }
}

/**
 * Process individual event
 * 
 * @param object $event
 */
function bzj_process_relationship_event($event) {
    global $wpdb;
    
    $table = $wpdb->prefix . 'bzj_relationship_events';
    $max_attempts = 8;
    
    // Check max attempts
    if ($event->attempts >= $max_attempts) {
        $wpdb->update(
            $table,
            [
                'status' => 'failed',
                'last_error' => 'Max retry attempts exceeded',
            ],
            ['id' => $event->id],
            ['%s', '%s'],
            ['%d']
        );
        return;
    }
    
    // Project to external platforms
    $success = true;
    
    if ('streams' === $event->origin || 'socials' === $event->origin) {
        // Event originated externally, project back to same platform
        // and to the other platform
    } else {
        // Event originated from WordPress, project to both platforms
        $success = bzj_project_relationship_event($event);
    }
    
    // Calculate next retry time (exponential backoff)
    $backoff_times = [1, 2, 4, 8, 16, 60, 60, 60]; // minutes
    $next_backoff = $backoff_times[min($event->attempts, count($backoff_times) - 1)];
    $next_attempt = date('Y-m-d H:i:s', strtotime($now) + ($next_backoff * 60));
    
    $status = $success ? 'processed' : 'pending';
    $processed_at = $success ? current_time('mysql') : null;
    
    $wpdb->update(
        $table,
        [
            'status' => $status,
            'attempts' => $event->attempts + 1,
            'next_attempt_at' => $next_attempt,
            'processed_at' => $processed_at,
        ],
        ['id' => $event->id],
        ['%s', '%d', '%s', '%s'],
        ['%d']
    );
}

/**
 * Project relationship event to external platforms
 * 
 * @param object $event
 * @return bool
 */
function bzj_project_relationship_event($event) {
    
    $payload = !empty($event->payload) ? json_decode($event->payload, true) : [];
    
    // Project to WoWonder Streams
    $wo_success = bzj_project_relationship_to_streams(

        $event->actor_wp_id,
        $event->target_wp_id,
        $event->state,
        $payload
    );
    
    return $wo_success && $qd_success;
}

if (!function_exists('_bzj_bridge_get_cross_platform_ids')) {
    /**
     * Get cross-platform user IDs
     * 
     * Resolves WordPress user ID to external platform IDs.
     * When called from external platforms without WordPress, returns null.
     * 
     * @param int $wp_user_id
     * @return array ['wp' => int|null, 'wo' => int|null, 'qd' => int|null]

    function _bzj_bridge_get_cross_platform_ids($wp_user_id) {
        
        $wp_user_id = (int) $wp_user_id;
        
        if (!$wp_user_id) {
            return [
                'wp' => null,
                'wo' => null,
                'qd' => null,
            ];
        }
        
        // If called from WordPress context
        if (function_exists('get_user_meta')) {
            return [
                'wp' => $wp_user_id,
                'wo' => (int) get_user_meta($wp_user_id, 'wo_user_id', true),
                'qd' => (int) get_user_meta($wp_user_id, 'qd_user_id', true),
            ];

        // Fallback for direct bridge calls
        return [
            'wp' => $wp_user_id,
            'wo' => null,
            'qd' => null,
        ];
    }

if (!function_exists('_bzj_bridge_project_relationship_to_streams')) {
    /**
     * Project canonical relationship to WoWonder Streams
     * 
     * Translates WordPress relationship state to Streams database structure.
     * 
     * Streams uses:
     * - T_FOLLOWERS: following_id, follower_id, active, created_at
     * - T_BLOCKS: blocker, blocked
     * 
     * State mapping:
     * - requested: active = 0
     * - connected: active = 1
     * - none: delete relationship
     * - blocked: delete relationship + create block
     * 
     * @param int $wp_actor
     * @param int $wp_target
     * @param string $state
     * @param array $payload
     * @return bool
     */
    function _bzj_bridge_project_relationship_to_streams(
        $wp_actor,
        $wp_target,
        $state,
        $payload = []
    ) {
        
        $wp_actor = (int) $wp_actor;
        $wp_target = (int) $wp_target;
        $state = sanitize_text_field($state);
        
        if (!$wp_actor || !$wp_target) {
            return false;
        }
        
        // Get WoWonder IDs
        $actor_ids = _bzj_bridge_get_cross_platform_ids($wp_actor);
        $target_ids = _bzj_bridge_get_cross_platform_ids($wp_target);
        
        if (empty($actor_ids['wo']) || empty($target_ids['wo'])) {
            // Users don't have WoWonder accounts
            return true;
        }
        
        $wo_actor = (int) $actor_ids['wo'];

/**

 * 
 * External platforms must request through WordPress, which then

 * @param int $actor_id WoWonder user ID (actor)

    function _bzj_streams_request_wp_authority($actor_id, $target_id, $operation) {
        
        $actor_id = (int) $actor_id;
        $target_id = (int) $target_id;
        $operation = sanitize_text_field($operation);
        
        if (!$actor_id || !$target_id || !in_array($operation, 
            ['request', 'accept', 'reject', 'remove', 'block', 'unblock'])) {
            return false;
        }
        
        // Get WordPress authority endpoint
        $wp_base = defined('WP_BASE_SITE_URL') ? WP_BASE_SITE_URL : getenv('WP_BASE_SITE_URL');
        if (!$wp_base) {
            // WordPress URL not configured; fail closed
            return false;
        }
        
        $endpoint = rtrim($wp_base, '/') . '/wp-json/bzj/v1/relationship';
        
        // Get SSO secret
        $secret = defined('BUZZ_SSO_SECRET') ? BUZZ_SSO_SECRET : getenv('BUZZ_SSO_SECRET');
        if (!$secret) {
            return false;
        }
        
        // Generate event UUID for idempotency
        $event_uuid = sprintf(
            '%04x%04x-%04x-%04x-%04x-%04x%04x%04x',

            mt_rand(0, 0xffff),
            mt_rand(0, 0x0fff) | 0x4000,
            mt_rand(0, 0x3fff) | 0x8000,

        $payload = [
            'origin' => 'streams',
            'origin_user_id' => $actor_id,
            'target_user_id' => $target_id,
            'operation' => $operation,
            'event_uuid' => $event_uuid,
        ];
        
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, $endpoint);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
        curl_setopt($ch, CURLOPT_TIMEOUT, 5);
        curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'POST');
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($payload));
        curl_setopt($ch, CURLOPT_HTTPHEADER, [

        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);

        $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);

function bzj_audit_relationships() {
    global $wpdb;
    
    $report = [
        'timestamp' => current_time('mysql'),
        'total_checked' => 0,
        'mismatches' => 0,
        'details' => [],
    ];
    
    $table = $wpdb->prefix . 'bzj_relationships';
    
    $relationships = $wpdb->get_results(
        "SELECT * FROM {$table} WHERE state != 'none' LIMIT 1000"
    );
    
    foreach ($relationships as $rel) {

        // Check Streams state
        $wo_ids = _bzj_bridge_get_cross_platform_ids($rel->user_a);
        $qd_ids = _bzj_bridge_get_cross_platform_ids($rel->user_b);
        
        if (empty($wo_ids['wo']) || empty($qd_ids['wo'])) {
            continue; // Users don't exist on platform
        }
        
        $wo_state = _bzj_audit_get_streams_relationship(
            $wo_ids['wo'],
            $qd_ids['wo']
        );
        
        $qd_state = _bzj_audit_get_socials_relationship(
            $wo_ids['qd'] ?? 0,
            $qd_ids['qd'] ?? 0
        );
        
        if ($wo_state !== $rel->state || $qd_state !== $rel->state) {
            
            $report['mismatches']++;
            
            $report['details'][] = [
                'user_a' => $rel->user_a,
                'user_b' => $rel->user_b,
                'canonical' => $rel->state,
                'streams' => $wo_state,
                'socials' => $qd_state,

    }

/**

function _bzj_audit_get_streams_relationship($wo_user_a, $wo_user_b) {
    
    $db = get_wowonder_db();
    
    if (!$db) {
        return 'none';
    }
    
    // Check if blocked
    $block = $db->query(
        "SELECT 1 FROM `T_BLOCKS` 
         WHERE (blocker = {$wo_user_a} AND blocked = {$wo_user_b})
            OR (blocker = {$wo_user_b} AND blocked = {$wo_user_a})
         LIMIT 1"
    );
    
    if ($block && $block->num_rows > 0) {
        return 'blocked';
    }
    
    // Check follow status
    $follow = $db->query(
        "SELECT `active` FROM `T_FOLLOWERS` 
         WHERE following_id = {$wo_user_b} AND follower_id = {$wo_user_a}
         LIMIT 1"
    );
    
    if ($follow && $row = $follow->fetch_assoc()) {
        return $row['active'] ? 'connected' : 'requested';

            $errors++;
        }
    }
    
    return [
        'success' => true,
        'data' => [
            'message' => "Reconciliation complete: $count synchronized, $errors errors",
            'next_step' => false,
        ],
    ];
}

    check_admin_referer('bzj_migration_nonce', '_wpnonce');
    
    if (!current_user_can('manage_options')) {
        wp_send_json_error('Unauthorized');
    }
    
    $step = isset($_POST['step']) ? (int) $_POST['step'] : 1;
    
    $result = bzj_migration_run_step($step);
    
    wp_send_json($result);
});

        wp_send_json_error('Unauthorized');
    }
    
    $report = bzj_audit_relationships();
